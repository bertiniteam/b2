55 // this solution for *lazy* make shared comes from the SO forum, asked by polytheme, answered by user sehe.
56 // https://stackoverflow.com/questions/21516201/how-to-create-boost-phoenix-make-shared
57 // post found using google search terms `phoenix construct shared_ptr`
58 // the code has been adapted slightly to fit the naming conventions of this project.
59 template <typename T>
60 struct MakeSharedFunctor
61 {
62 template <typename... A>
63 struct result
64 {
65 typedef std::shared_ptr<T> type;
66 };
67
68 template <typename... A>
69 typename result<A...>::type operator()(A&&... a) const
70 {
71 return T::Make(std::forward<A>(a)...);
72 }
73 };
74
75 template <typename T>
76 using make_shared_ = boost::phoenix::function<MakeSharedFunctor<T> >;
77}

152 FunctionParser(qi::symbols<char,std::shared_ptr<Node> > * encountered_symbols) : FunctionParser::base_type(root_rule_,"FunctionParser")
153 {
154 namespace phx = boost::phoenix;
155 using qi::_1;
156 using qi::_2;
157 using qi::_3;
158 using qi::_4;
159 using qi::_val;
160 using qi::eps;
161 using qi::lit;
162
163 using std::pow;
164 using ::pow;
165
166 root_rule_.name("function_");
167 // Return the bare parsed expression (no Function wrapper): the System parser names
168 // it (a NamedExpression for subfunctions) or stores it directly (top-level functions).
169 root_rule_ = expression_ [ _val = _1];
170
171
173 expression_.name("expression_");
174 expression_ =
175 term_ [_val = _1]
176 >> *( (lit('+') > term_ [_val += _1])
177 | (lit('-') > term_ [_val -= _1])
178 )
179 ;
180
181 term_.name("term_");
182 term_ =
183 factor_ [_val = _1]
184 >> *( (lit('*') > factor_ [_val *= _1])
185 | (lit('/') > factor_ [_val /= _1])
186 )
187 ;
188
189 factor_.name("factor_");
190 factor_ =
191 exp_elem_ [_val = _1]
192 >> *(lit('^') // any number of ^somethings
193 > exp_elem_ [ phx::bind( []
194 (std::shared_ptr<Node> & B, std::shared_ptr<Node> P)
195 {
196 B = pow(B,P);
197 },
198 _val,_1)] )
199 ;
200
201 exp_elem_.name("exp_elem_");
202 exp_elem_ =
203 // The negative lookahead keeps a known symbol from matching a
204 // prefix of a longer identifier (e.g. `e` inside `exp`, or `x`
205 // inside `xy`). It must use the UTF-8 continuation predicate so
206 // a following Unicode letter (e.g. `α` after a known `Ω`) also
207 // blocks the match -- a bare ASCII !qi::alnum would not.
208 (symbol_ >> utf8_ident_boundary_parser()) [_val = _1]
209 | ( '(' > expression_ [_val = _1] > ')' ) // using the > expectation here.
210 // unary +/- bind a single factor_, NOT the whole expression_: "-y+x" is
211 // (-y)+x, and "-x^2" is -(x^2). (Binding expression_ here made a leading
212 // minus greedily negate everything after it.)
213 | (lit('-') > factor_ [_val = -_1])
214 | (lit('+') > factor_ [_val = _1])
215 | (lit("sin") > '(' > expression_ [_val = sin_lazy(_1)] > ')' )
216 | (lit("cos") > '(' > expression_ [_val = cos_lazy(_1)] > ')' )
217 | (lit("tan") > '(' > expression_ [_val = tan_lazy(_1)] > ')' )
218 | (lit("exp") > '(' > expression_ [_val = exp_lazy(_1)] > ')' )
219 | (lit("log") > '(' > expression_ [_val = log_lazy(_1)] > ')' )
220 | (lit("sqrt") > '(' > expression_ [_val = sqrt_lazy(_1)] > ')' )
221 ;

231 symbol_ %=

234 number_
