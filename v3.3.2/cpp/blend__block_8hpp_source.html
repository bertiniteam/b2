94 BlendBlock(Var path_variable, std::vector<Nd> coefficients, std::vector<OperandPtr> operands)
95 : path_variable_(std::move(path_variable)),
96 coefficients_(std::move(coefficients)),
97 operands_(std::move(operands)),
98 precision_(DefaultPrecision())
99 {
100 assert(!operands_.empty() && "BlendBlock needs at least one operand");
101 assert(coefficients_.size() == operands_.size() && "one coefficient per operand");
102 derivative_coefficients_.reserve(coefficients_.size());
103 for (auto const& c : coefficients_)
104 derivative_coefficients_.push_back(c->Differentiate(path_variable_));
105 }
106
107 // Memory-isolating copy (ADR-0027): each operand System is deep-copied via the
108 // System copy constructor, which shares its immutable node DAG and compiled SLP Program but
109 // gives it its own per-thread evaluation Memory. The coefficient-system cache is reset (rebuilt
110 // lazily per copy). The coefficient nodes / path variable are shared --- they are never written
111 // during evaluation. This lets path tracking Clone a System into per-thread copies that share
112 // no mutable state.
114 BlendBlock(BlendBlock const& other)
115 : path_variable_(other.path_variable_),
116 coefficients_(other.coefficients_),
117 derivative_coefficients_(other.derivative_coefficients_),
118 precision_(other.precision_)
119 {
120 operands_.reserve(other.operands_.size());
121 for (auto const& op : other.operands_)
122 operands_.push_back(std::make_shared<const SystemT>(*op));
123 // coefficient_system_ deliberately left null: rebuilt lazily per copy
124 }

127 BlendBlock& operator=(BlendBlock const& other)
128 {
129 if (this != &other)
130 {
131 path_variable_ = other.path_variable_;
132 coefficients_ = other.coefficients_;
133 derivative_coefficients_ = other.derivative_coefficients_;
134 precision_ = other.precision_;
135 operands_.clear();
136 operands_.reserve(other.operands_.size());
137 for (auto const& op : other.operands_)
138 operands_.push_back(std::make_shared<const SystemT>(*op));
139 coefficient_system_.reset();
140 }
141 return *this;
142 }

213 // A blend is the coupling homotopy, built from already-prepared (homogenized) operands; its
214 // operands are shared_ptr<const System> and cannot be mutated, so Homogenize is a no-op. It
215 // is homogeneous/polynomial iff all its operands are.
217 void Homogenize(VariableGroup const&, std::shared_ptr<node::Variable> const&) {}

222 return true;

244 void Reintern(node::ReinternMemo& memo)
245 {
246 if (path_variable_)
247 path_variable_ = std::static_pointer_cast<node::Variable>(node::Reintern(path_variable_, memo));
248 for (auto& c : coefficients_)
249 c = node::Reintern(c, memo);
250 derivative_coefficients_.clear();
251 derivative_coefficients_.reserve(coefficients_.size());
252 for (auto const& c : coefficients_)
253 derivative_coefficients_.push_back(c->Differentiate(path_variable_));
254 for (auto& op : operands_)
255 std::const_pointer_cast<SystemT>(op)->ReinternNodes(memo);
256 coefficient_system_.reset(); // rebuilt lazily against the reinterned coefficients
257 }

260 unsigned Precision() const { return precision_; }
261
263 void Precision(unsigned new_precision) const
264 {
265 for (auto const& op : operands_)
266 op->precision(new_precision);
267 // The coefficients are evaluated through the coefficient sub-system's SLP (which carries
268 // its own precision; see EvalCoefficients), so the coefficient nodes and the shared path
269 // variable are no longer evaluated during tracking. Their precision is vestigial and
270 // left untouched, keeping the shared node DAG read-only across threads (ADR-0027).
271 if (coefficient_system_)
272 coefficient_system_->precision(new_precision);
273 precision_ = new_precision;
274 }

277 template <typename T>
278 void EvalInPlace(Eigen::Ref<Vec<T>> result, Vec<T> const& vars, T const& path_value) const
279 {
280 SyncPrecision(vars);
281 result.setZero();
282 const Eigen::Index k = static_cast<Eigen::Index>(NumFunctions());
283 const Vec<T> coeffs = EvalCoefficients<T>(path_value);
284 for (size_t i = 0; i < operands_.size(); ++i)
285 {
286 const Vec<T> fi = operands_[i]->template Eval<T>(vars);
287 result += coeffs(static_cast<Eigen::Index>(i)) * fi.head(k);
288 }
289 }

292 template <typename T>
293 void JacobianInPlace(Eigen::Ref<Mat<T>> J, Vec<T> const& vars, T const& path_value) const
294 {
295 SyncPrecision(vars);
296 J.setZero();
297 const Eigen::Index k = static_cast<Eigen::Index>(NumFunctions());
298 const Vec<T> coeffs = EvalCoefficients<T>(path_value);
299 for (size_t i = 0; i < operands_.size(); ++i)
300 {
301 const Mat<T> Ji = operands_[i]->template Jacobian<T>(vars);
302 J += coeffs(static_cast<Eigen::Index>(i)) * Ji.topRows(k);
303 }
304 }

307 template <typename T>
308 void TimeDerivInPlace(Eigen::Ref<Vec<T>> result, Vec<T> const& vars, T const& path_value) const
309 {
310 SyncPrecision(vars);
311 result.setZero();
312 const Eigen::Index k = static_cast<Eigen::Index>(NumFunctions());
313 const Vec<T> coeffs = EvalCoefficients<T>(path_value);
314 const size_t n = operands_.size();
315 for (size_t i = 0; i < n; ++i)
316 {
317 const Vec<T> fi = operands_[i]->template Eval<T>(vars);
318 result += coeffs(static_cast<Eigen::Index>(n + i)) * fi.head(k);
319 }
320 }

326 template <typename T>
327 void SyncPrecision(Vec<T> const& vars) const
328 {
329 if constexpr (!std::is_same<T, complex_dbl>::value)
330 {
331 if (vars.size() > 0)
332 {
333 const unsigned p = bertini::Precision(vars(0));
334 if (p != precision_)
335 Precision(p);

345 SystemT& EnsureCoefficientSystem() const
346 {
347 if (!coefficient_system_)
348 {
349 auto sys = std::make_shared<SystemT>();
350 for (auto const& c : coefficients_)
351 sys->AddFunction(c);
352 for (auto const& c : derivative_coefficients_)
353 sys->AddFunction(c);
354 sys->AddVariableGroup(VariableGroup{path_variable_});
355 sys->precision(precision_);
356 coefficient_system_ = sys;
357 }
358 return *coefficient_system_;
359 }
360
362 template <typename T>
363 Vec<T> EvalCoefficients(T const& path_value) const
364 {
365 auto& cs = EnsureCoefficientSystem();
366 if constexpr (!std::is_same<T, complex_dbl>::value)
367 cs.precision(bertini::Precision(path_value));
368 Vec<T> t_point(1);
369 t_point(0) = path_value;
370 return cs.template Eval<T>(t_point);
371 }
