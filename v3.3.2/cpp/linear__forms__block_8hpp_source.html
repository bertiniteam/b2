105 void Homogenize(VariableGroup const& /*group*/, std::shared_ptr<node::Variable> const& /*hom_var*/)
106 {
107 if (homogeneous_)
108 throw std::runtime_error("LinearFormsBlock::Homogenize: block is already homogenized "
109 "(multiple affine variable groups are not yet supported for linear-forms blocks)");
110 const auto& M = coefficients_highest_precision_;
111 Mat<complex_mp> Mh(M.rows(), M.cols()); // same #cols: n+1
112 Mh.col(0) = M.col(static_cast<Eigen::Index>(num_vars_)); // constant -> h column (front)
113 Mh.rightCols(static_cast<Eigen::Index>(num_vars_)) =
114 M.leftCols(static_cast<Eigen::Index>(num_vars_)); // original variable columns
115 coefficients_highest_precision_ = Mh;
116 num_vars_ += 1; // h is now a real variable; all n+1 columns are variable columns
117 homogeneous_ = true;
118 BuildWorking();
119 }

135 void Describe(std::ostream& out, size_t& row, VariableGroup const& vars, bool verbose) const
136 {
137 auto const& M = coefficients_highest_precision_;
138 const Eigen::Index n = M.rows();
139 if (n == 0)
140 return;
141 const Eigen::Index cap = static_cast<Eigen::Index>(describe_detail::kTerseRowCap);
142 const Eigen::Index shown = (verbose || n <= cap) ? n : cap;
143
144 // placeholder line per form: f_k = c.[x, y, 1]
145 for (Eigen::Index r = 0; r < shown; ++r)
146 {
147 out << " f_" << row++ << " = c.";
148 describe_detail::PrintAugmentedVars(out, vars, num_vars_, homogeneous_);
149 out << "\n";
150 }
151
152 // the actual coefficients below, as a named-expression-style legend (row r <-> f_k above)
153 const int sig = describe_detail::CoeffSig(verbose);
154 out << " c =\n";
155 for (Eigen::Index r = 0; r < shown; ++r)
156 {
157 out << " [ ";
158 for (Eigen::Index c = 0; c < M.cols(); ++c)
159 {
160 if (c) out << ", ";
161 describe_detail::PrintCoeff(out, M(r, c), sig);
162 }
163 out << " ]\n";
164 }
165
166 if (shown < n)
167 {
168 row += static_cast<size_t>(n - shown); // keep the global row index correct
169 out << " ... (" << (n - shown) << " more form" << (n - shown == 1 ? "" : "s")
170 << "; describe(verbose=True) for all)\n";
171 }
172 }

187 void Precision(unsigned new_precision) const
188 {
189 if (new_precision > DoublePrecision())
190 {
191 auto& wm = std::get<Mat<complex_mp>>(coefficients_working_);
192 for (Eigen::Index r = 0; r < wm.rows(); ++r)
193 for (Eigen::Index c = 0; c < wm.cols(); ++c)
194 {
195 wm(r, c).precision(new_precision);
196 if (new_precision > precision_)
197 wm(r, c) = coefficients_highest_precision_(r, c);
198 }
199 }
200 precision_ = new_precision;
201 }

211 template <typename T>
212 void EvalInPlace(Eigen::Ref<Vec<T>> result, Vec<T> const& vars, T const& /*path_value*/) const
213 {
214 // affine: f(x) = W * [x ; 1] (the trailing 1 carries each row's constant in the last
215 // column). homogeneous (post-Homogenize): every column is a variable column, so it is
216 // just W * vars (the old constant is now the homogenizing variable's coefficient).
217 if (homogeneous_)
218 result.noalias() = Working<T>() * vars;
219 else
220 result.noalias() = Working<T>() * Augment<T>(vars);
221 }
222
231 template <typename T>
232 void JacobianInPlace(Eigen::Ref<Mat<T>> J, Vec<T> const& /*vars*/, T const& /*path_value*/) const
233 {
234 // homogeneous: every column is d f / d x. affine: drop the trailing constant column.
235 if (homogeneous_)
236 J = Working<T>();
237 else
238 J = Working<T>().leftCols(static_cast<Eigen::Index>(num_vars_));
239 }

258 template <typename T>
259 Vec<T> Augment(Vec<T> const& vars) const
260 {
261 Vec<T> aug(static_cast<Eigen::Index>(num_vars_ + 1));
262 aug.head(static_cast<Eigen::Index>(num_vars_)) = vars;
263 T one(1);
264 if constexpr (!std::is_same<T, complex_dbl>::value)
265 one.precision(precision_);
266 aug(static_cast<Eigen::Index>(num_vars_)) = one;
267 return aug;
268 }

270 void BuildWorking() const
271 {
272 const auto& M = coefficients_highest_precision_;
273 auto& wd = std::get<Mat<complex_dbl>>(coefficients_working_);
274 auto& wm = std::get<Mat<complex_mp>>(coefficients_working_);
275 wd.resize(M.rows(), M.cols());
276 wm.resize(M.rows(), M.cols());
277 for (Eigen::Index r = 0; r < M.rows(); ++r)
278 for (Eigen::Index c = 0; c < M.cols(); ++c)
279 {
280 wd(r, c) = complex_dbl(M(r, c));
281 wm(r, c) = M(r, c);
282 }
283 }
