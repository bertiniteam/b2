123 template <unsigned long digits = 50>
124 mpq_rational RandomRat()
125 {
126 // pinned draws (b2rand/1, ADR-0044). A zero denominator is redrawn (deterministically):
127 // the legacy draw could in principle hand mpq a denominator of 0.
128 mpz_int const bound = mpz_int(1) << digits*1000L/301L;
129 auto& stream = records::ThreadDrawStream();
130 mpz_int const num = stream.IntSymmetric(bound);
131 mpz_int den = stream.IntSymmetric(bound);
132 while (den == 0)
133 den = stream.IntSymmetric(bound);
134 return mpq_rational(num, den);
135 }

143 template <unsigned int length_in_digits>
144 real_mp RandomMp()
145 {
146 // Pinned draw (b2rand/1, ADR-0044) from the single per-thread stream shared by
147 // every random type, so SetGlobalSeed()/ReseedThisThread() control them all
148 // uniformly. Uniform on [0,1) at length_in_digits digits, bit-identical on
149 // every platform.
150 real_mp a{records::ThreadDrawStream().UnitRealMp(length_in_digits)};
151 return a;
152 }

271 inline complex rand()
272 {
273 // DRAW ORDER IS A CONTRACT (cross-platform reproducibility, 2026-07-03): the two
274 // component draws are sequenced EXPLICITLY -- real first, then imaginary. Never
275 // put two draws in one full-expression: C++ argument evaluation order is
276 // unspecified, and gcc really does order them differently on x86_64 vs aarch64
277 // (found as an architecture-split seeded-homotopy digest: every (re, im) pair of
278 // every seeded coefficient was transposed between the two).
279 real_mp const re = RandomMp(real_mp(-1),real_mp(1));
280 real_mp const im = RandomMp(real_mp(-1),real_mp(1));
281 return complex( re, im );
282 }

288 inline complex rand_unit()
289 {
290 // draw order is a contract: real first, then imaginary (see rand())
291 real_mp const re = RandomMp(real_mp(-1),real_mp(1));
292 real_mp const im = RandomMp(real_mp(-1),real_mp(1));
293 complex returnme( re, im );
294 return returnme / abs(returnme); // normalize to modulus 1 (NOT sqrt(abs), which left modulus sqrt|z|)

313 inline complex rand_bounded_modulus()
314 {
315 // draw order is a contract: real first, then imaginary (see rand())
316 real_mp re = RandomMp(real_mp(-1),real_mp(1));
317 real_mp im = RandomMp(real_mp(-1),real_mp(1));
318 complex z( re, im );
319 auto m = abs(z);
320 while (m == 0) // measure-zero, but a zero coefficient is degenerate -- redraw
321 {
322 re = RandomMp(real_mp(-1),real_mp(1));
323 im = RandomMp(real_mp(-1),real_mp(1));
324 z = complex( re, im );
325 m = abs(z);
326 }
327 return z / sqrt(m);
328 }

338 void RandomComplexBoundedModulusAssign(complex & a, unsigned num_digits)
339 {
340 auto cached = ThreadPrecision();
341 SetThreadPrecision(num_digits);
342 a.precision(num_digits);
343
344 // draw order is a contract: real first, then imaginary (see rand())
345 real_mp re = RandomMp(real_mp(-1),real_mp(1),num_digits);
346 real_mp im = RandomMp(real_mp(-1),real_mp(1),num_digits);
347 complex z( re, im );

351 re = RandomMp(real_mp(-1),real_mp(1),num_digits);

354 m = abs(z);
355 }

378 inline complex rand_real_bounded_modulus()
379 {
380 complex z( RandomMp(real_mp(-1),real_mp(1)) ); // imaginary part 0
381 auto m = abs(z);

385 m = abs(z);
