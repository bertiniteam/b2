53struct ZeroDimRT
54{
55 // INTERIM default = TotalDegreeBinomial (see policies.hpp): the linear-product TotalDegreeLinearProduct is the
56 // eventual default but currently stalls the Cauchy endgame on harder systems, so the safe
57 // default stays roots of unity until that is fixed.
58 type::Start start = type::Start::TotalDegreeBinomial;

60 type::Endgame endgame = type::Endgame::Cauchy;

77inline type::Start InferStartType(System const& sys)
78{
79 // INTERIM: a single affine group infers TotalDegreeBinomial (the safe default). The eventual choice
80 // here is TotalDegreeLinearProduct (general position), gated on the Cauchy-endgame fix; see policies.hpp.
81 if (sys.NumVariableGroups() == 1 && sys.NumHomVariableGroups() == 0)
82 return type::Start::TotalDegreeBinomial;
83 else
84 return type::Start::MHom;
85}

105template <typename TrackerType, typename EndgameType, typename ... ConstTs>

123template <typename TrackerType, typename ... ConstTs>
124std::unique_ptr<algorithm::AnyZeroDim> ZeroDimSpecifyEndgame(ZeroDimRT const& rt, ConstTs const& ...ts)
125{
126 switch (rt.endgame)
127 {
128 case type::Endgame::PowerSeries:
129 // honor the requested endgame! until 2026-06-12 this hardcoded Cauchy.
130 return ZeroDimSpecifyComplete<TrackerType,
131 typename endgame::EndgameSelector<TrackerType>::PSEG>(ts...);
132
133 case type::Endgame::Cauchy:
134 return ZeroDimSpecifyComplete<TrackerType,
135 typename endgame::EndgameSelector<TrackerType>::Cauchy>(ts...);
136 }
137 throw std::runtime_error("unrecognized endgame type in ZeroDimSpecifyEndgame");
138}

148template <typename ... ConstTs>
149std::unique_ptr<algorithm::AnyZeroDim> ZeroDimSpecifyTracker(ZeroDimRT const& rt, ConstTs const& ...ts)
150{
151 switch (rt.tracker)
152 {
153 case type::Tracker::FixedDouble:
154 return ZeroDimSpecifyEndgame<tracking::DoublePrecisionTracker>(rt, ts...);
155 case type::Tracker::FixedMultiple:
156 return ZeroDimSpecifyEndgame<tracking::MultiplePrecisionTracker>(rt, ts...);
157 case type::Tracker::Adaptive:
158 return ZeroDimSpecifyEndgame<tracking::AMPTracker>(rt, ts...);
159 }
160 throw std::runtime_error("unrecognized tracker type in ZeroDimSpecifyTracker");
161}

174template <typename ... ConstTs>
175std::unique_ptr<algorithm::AnyZeroDim> ZeroDimSpecifyStart(ZeroDimRT const& rt, ConstTs const& ...ts)
176{
177 // append the start-system factory to the argument pack; ZeroDimSolver consumes (target, factory).
178 switch (rt.start)
179 {
180 case type::Start::TotalDegreeLinearProduct:
181 return ZeroDimSpecifyTracker(rt, ts..., start_system::MakeStartFactory<start_system::TotalDegreeLinearProduct>());
182 case type::Start::TotalDegreeBinomial:
183 return ZeroDimSpecifyTracker(rt, ts..., start_system::MakeStartFactory<start_system::TotalDegreeBinomial>());
184 case type::Start::MHom:
185 return ZeroDimSpecifyTracker(rt, ts..., start_system::MakeStartFactory<start_system::MHomogeneous>());
186 case type::Start::User:
187 throw std::runtime_error("trying to use generic zero dim with user homotopy. use the specific UserBlaBla instead");
188 }
189 throw std::runtime_error("unrecognized start system type in ZeroDimSpecifyStart");
190}

203template <typename ... ConstTs>
204std::unique_ptr<algorithm::AnyZeroDim> MakeZeroDim(ZeroDimRT const& rt, ConstTs const& ...ts)
205{
206 return ZeroDimSpecifyStart(rt, ts...);
