472 template<typename ComplexT>
473 SuccessCode Predict(Vec<ComplexT> & next_space,
474 StepMetadata & meta,
475 System const& S,
476 const Vec<ComplexT>& current_space, ComplexT current_time,
477 ComplexT const& delta_t,
478 unsigned & num_steps_since_last_condition_number_computation,
479 unsigned frequency_of_CN_estimation,
480 NumErrorT const& tracking_tolerance,
481 AdaptiveMultiplePrecisionConfig const* AMP_config = nullptr)
482 {
483 auto step_success = FullStep(next_space, S, current_space, current_time, delta_t);
484
485 // Condition estimate (norm_J, norm_J_inverse, condition_number_estimate) is always
486 // computed -- it is reported and used by the fixed-precision path too.
487 SetNormsCond<ComplexT>(meta.norm_J, meta.norm_J_inverse, meta.condition_number_estimate,
488 num_steps_since_last_condition_number_computation, frequency_of_CN_estimation);
489
490 // Fixed-precision (nullptr): just the prediction + condition estimate, no AMP criteria.
491 if (AMP_config == nullptr)
492 return step_success;
493
494 SetSizeProportion<ComplexT>(meta.size_proportion, delta_t);
495 if (predict::HasErrorEstimate(predictor_))
496 SetErrorEstimate<ComplexT>(meta.error_estimate, delta_t);
497
498 if (step_success != SuccessCode::Success)
499 return step_success;
500
501 if (!amp::CriterionA<ComplexT>(meta.norm_J, meta.norm_J_inverse, *AMP_config))
502 return SuccessCode::HigherPrecisionNecessary;
503 if (!amp::CriterionC<ComplexT>(meta.norm_J_inverse, current_space, tracking_tolerance, *AMP_config))
504 return SuccessCode::HigherPrecisionNecessary;
505
506 return step_success;
507 }

578 template<typename ComplexT>
579 SuccessCode FullStep(Vec<ComplexT> & next_space,
580 System const& S,
581 Vec<ComplexT> const& current_space, ComplexT const& current_time,
582 ComplexT const& delta_t)
583 {
584
585 // If using constant predictor
586 if(s_ == 0)
587 {
588 next_space = current_space;
589 return SuccessCode::Success;
590 }
591
592 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
593
594 Mat<ComplexT>& Kref = std::get< Mat<ComplexT> >(K_);
595 Mat<RealT>& aref = std::get< Mat<RealT> >(a_);
596 Vec<RealT>& bref = std::get< Vec<RealT> >(b_);
597 Vec<RealT>& cref = std::get< Vec<RealT> >(c_);
598 Kref.fill(ComplexT(0));
599 Vec<ComplexT>& temp = std::get< Vec<ComplexT> >(step_temp_);
600
601 if(EvalRHS(S, current_space, current_time, Kref, 0) != SuccessCode::Success)
602 {
603 return SuccessCode::MatrixSolveFailureFirstPartOfPrediction;
604 }
605
606 Vec<ComplexT>& stage_pt = std::get< Vec<ComplexT> >(stage_pt_temp_);
607 for(unsigned ii = 1; ii < s_; ++ii)
608 {
609 temp.setZero(); // see https://github.com/bertiniteam/b2/issues/198
610 for(unsigned jj = 0; jj < ii; ++jj)
611 temp += aref(ii,jj)*Kref.col(jj);
612
613 // Evaluate into the preallocated stage-point scratch rather than passing the
614 // expression (current_space + delta_t*temp) to EvalRHS's const Vec& param, which
615 // would materialize a fresh temporary Vec every stage.
616 stage_pt.noalias() = current_space + delta_t*temp;
617 if(EvalRHS<ComplexT>(S, stage_pt, current_time + cref(ii)*delta_t, Kref, ii) != SuccessCode::Success)
618 return SuccessCode::MatrixSolveFailure;
619 }
620
621
622 temp.setZero();
623 for(unsigned ii = 0; ii < s_; ++ii)
624 temp += bref(ii)*Kref.col(ii);
625
626 // next_space is a distinct buffer from current_space/temp, so noalias avoids a
627 // materialized temporary for the axpy.
628 next_space.noalias() = current_space + delta_t*temp;
629
630 return SuccessCode::Success;
631 };

641 template<typename ComplexT>
642 void SetNormsCond(NumErrorT & norm_J, NumErrorT & norm_J_inverse, NumErrorT & condition_number_estimate, unsigned num_steps_since_last_condition_number_computation, unsigned frequency_of_CN_estimation)
643 {
644 // Calculate condition number and update if needed
645 linalg::PartialPivLU<ComplexT>& LUref = std::get< linalg::PartialPivLU<ComplexT> >(LU_);
646 Mat<ComplexT>& dhdxref = std::get< Mat<ComplexT> >(dh_dx_0_);
647
648 Vec<ComplexT> const& randy = std::get< Vec<ComplexT> >(rand_temp_);
649 Vec<ComplexT>& solve_ref = std::get< Vec<ComplexT> >(solve_temp_);
650 LUref.Solve(randy, solve_ref);
651
652 norm_J = NumErrorT(dhdxref.norm());
653 norm_J_inverse = NumErrorT(solve_ref.norm());
654
655
656 if (num_steps_since_last_condition_number_computation >= frequency_of_CN_estimation)
657 {
658 condition_number_estimate = NumErrorT(norm_J * norm_J_inverse);
659 num_steps_since_last_condition_number_computation = 1; // reset the counter to 1
660 }
661 else // no need to compute the condition number
662 num_steps_since_last_condition_number_computation++;
663 }

676 template<typename ComplexT>
677 SuccessCode SetErrorEstimate(NumErrorT & error_estimate, ComplexT const& delta_t)
678 {
679 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
680
681 Mat<ComplexT>& Kref = std::get< Mat<ComplexT> >(K_);
682 Vec<RealT>& b_minus_bstar_ref = std::get< Vec<RealT> >(b_minus_bstar_);
683
684 Vec<ComplexT>& err = std::get< Vec<ComplexT> >(err_temp_); // reused scratch, not a fresh per-step alloc
685
686 err.setZero();
687 for(unsigned ii = 0; ii < s_; ++ii)
688 {
689 err += (b_minus_bstar_ref(ii))*Kref.col(ii);
690 }
691
692 err *= delta_t;
693
694 error_estimate = NumErrorT(err.norm());
695
696 return SuccessCode::Success;
697 };

714 template<typename ComplexT>
715 SuccessCode SetSizeProportion(NumErrorT & size_proportion, ComplexT const& delta_t)
716 {
717 if(predict::HasErrorEstimate(predictor_))
718 {
719 NumErrorT err_est;
720 SetErrorEstimate(err_est, delta_t);
721
722 using std::pow;
723 size_proportion = err_est/NumErrorT(pow(abs(delta_t), p_+1));
724
725 return SuccessCode::Success;
726 }
727 else
728 {
729 // No embedded error estimate (e.g. Euler, RK4). AMP2 (bhswAMP2, Eqs 9-10)
730 // derives the size proportion $a$ from the prediction step itself, written as
731 // the initial Newton residual: ||d|| = a|s|. The predicted step in z is
732 // delta_t*(sum_i b_i K_i), so ||d|| = |delta_t|*||K|| and therefore
733 // a = ||d|| / |s| = ||K|| ~ maxCoeff(|K|),
734 // with NO further division by |delta_t|. The previous
735 // maxCoeff(K)/|delta_t|^p over-divided by the step: it inflated $a$ like
736 // 1/|delta_t|^p as the step shrank, spuriously escalating AMP precision
737 // (DigitsB) on small steps. $a$ is meant to be an O(1), step-independent
738 // proportionality constant -- see SetSizeProportion's error-estimate branch,
739 // which divides err_est by |delta_t|^(p+1) for exactly the same reason
740 // (AMP3 / bhsODEAMP Eq. 6).
741 Mat<ComplexT>& Kref = std::get< Mat<ComplexT> >(K_);
742 size_proportion = NumErrorT(Kref.array().abs().maxCoeff());
743 return SuccessCode::Success;
744 }

761 template<typename ComplexT>
762 SuccessCode EvalRHS(System const& S,
763 const Vec<ComplexT>& space, const ComplexT& time, Mat<ComplexT> & K, unsigned stage)
764 {
765
766
767 if (std::is_same<ComplexT, complex_mp>::value)
768 PrecisionSanityCheck();
769
770 if(stage == 0)
771 {
772 linalg::PartialPivLU<ComplexT>& LUref = std::get< linalg::PartialPivLU<ComplexT> >(LU_);
773 Mat<ComplexT>& dhdxref = std::get< Mat<ComplexT> >(dh_dx_0_);
774
775 if (!std::is_same<ComplexT,complex_dbl>::value)
776 {
777 assert(ThreadPrecision()==current_precision_);
778
779 assert(Precision(space)==current_precision_);
780 assert(Precision(time)==current_precision_);
781 assert(Precision(dhdxref)==current_precision_);
782 assert(Precision(K)==current_precision_);
783 }
784 S.SetAndReset<ComplexT>(space, time);
785 S.JacobianInPlace(dhdxref);
786 // Factor a copy of dh/dx (dh_dx_0_ is read again in SetNormsCond); health check folded in.
787 auto lu_code = LUref.Factor(dhdxref);
788 if (!std::is_same<ComplexT,complex_dbl>::value)
789 {
790 assert(Precision(dhdxref)==current_precision_);
791 assert(Precision(LUref.Factors())==current_precision_);
792 }
793
794 if (lu_code!=MatrixSuccessCode::Success)
795 return SuccessCode::MatrixSolveFailureFirstPartOfPrediction;
796
797 Vec<ComplexT>& dhdtref = std::get< Vec<ComplexT> >(dh_dt_temp_);
798 S.TimeDerivativeInPlace(dhdtref);
799 dhdtref = -dhdtref; // in place; Solve needs materialized rhs
800 LUref.Solve(dhdtref, K.col(stage));
801
802 return SuccessCode::Success;
803
804 }
805 else
806 {
807 S.SetAndReset<ComplexT>(space, time);
808
809 Mat<ComplexT>& dhdxtempref = std::get< Mat<ComplexT> >(dh_dx_temp_);
810 S.JacobianInPlace(dhdxtempref);
811 linalg::PartialPivLU<ComplexT>& LU_temp = std::get< linalg::PartialPivLU<ComplexT> >(LU_);
812 // dh_dx_temp_ is pure scratch here -> factor destructively, skipping the copy.
813 if (LU_temp.FactorDestructive(dhdxtempref)!=MatrixSuccessCode::Success)
814 return SuccessCode::MatrixSolveFailure;
815
816 Vec<ComplexT>& dhdtref = std::get< Vec<ComplexT> >(dh_dt_temp_);
817 S.TimeDerivativeInPlace(dhdtref);
818 dhdtref = -dhdtref; // in place; Solve needs materialized rhs
819 LU_temp.Solve(dhdtref, K.col(stage));
820
821 return SuccessCode::Success;
822 }
823 }
