183 SuccessCode Run(Vec<BCT> const& start_point)
184 {
185 using bertini::Precision;
186 auto prec = Precision(start_point);
187 BCT t = start_time_; Precision(t, prec);
188 BCT t0 = target_time_; Precision(t0, prec);
189 // Fixed precision runs today's straight-line RunImpl<BCT> verbatim (untouched). Adaptive
190 // precision takes the separate double-first driver, RunImplAMP, which lives in the flavor.
191 // RunImplAMP is a member template (template<typename=void>) so the explicit fixed-precision
192 // class instantiations never force-compile it -- hence the .template disambiguator here.
193 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
194 return this->AsFlavor().template RunImplAMP<>(t, start_point, t0);
195 else
196 return this->AsFlavor().RunImpl(t, start_point, t0);
197 }

201 template<typename ComplexT>
202 SuccessCode RefineAllSamples(SampCont<ComplexT> & samples, TimeCont<ComplexT> & times)
203 {
204 for (size_t ii=0; ii<samples.size(); ++ii)
205 {
206 auto refine_success = this->RefineSample(samples[ii], samples[ii], times[ii],
207 this->FinalTolerance() * this->EndgameSettings().sample_point_refinement_factor,
208 this->EndgameSettings().max_num_refinements);
209 if (refine_success != SuccessCode::Success)
210 {
211 // BOOST_LOG_TRIVIAL(severity_level::trace) << "refining failed, code " << int(refine_success);
212 return refine_success;
213 }
214 NotifyObservers(SampleRefined<EmitterType>(AsFlavor()));
215 }
216
217 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec) // known at compile time
218 {
219 auto max_precision = this->EnsureAtUniformPrecision(times, samples);
220 this->GetSystem().precision(max_precision);
221 }
222
223 return SuccessCode::Success;
224 }

258 template<typename ComplexT>
259 SuccessCode EndgameTrackPath(Vec<ComplexT> & result, ComplexT const& start_time, ComplexT const& end_time, Vec<ComplexT> const& start_point) const
260 {
261 if constexpr (std::is_same<ComplexT, BCT>::value)
262 {
263 return this->GetTracker().TrackPath(result, start_time, end_time, start_point);
264 }
265 else
266 {
267 using bertini::Precision;
268 Vec<BCT> mp_start(start_point.size());
269 for (Eigen::Index i = 0; i < start_point.size(); ++i) mp_start(i) = BCT(start_point(i));
270 Precision(mp_start, DoublePrecision());
271
272 BCT mp_t0(start_time); Precision(mp_t0, DoublePrecision());
273 BCT mp_t1(end_time); Precision(mp_t1, DoublePrecision());
274
275 Vec<BCT> mp_result(start_point.size());
276 auto code = this->GetTracker().TrackPath(mp_result, mp_t0, mp_t1, mp_start);
277 if (code != SuccessCode::Success)
278 return code;
279
280 result.resize(mp_result.size());
281 for (Eigen::Index i = 0; i < mp_result.size(); ++i) result(i) = complex_dbl(mp_result(i));
282 return code;
283 }
284 }

287 // EscalateAndMigrate bridges the AMP precision policy (NextEscalatedPrecision and the Cross* helpers,
288 // all in AMPEndgame) with the flavor's container list (its MigrateContainersToPrecision). It lives
289 // here, not in AMPEndgame, because it needs AsFlavor(), which the precision-policy base lacks. Member
290 // template so the explicit fixed-precision class instantiations never force-compile it. Raises the
291 // working precision and migrates every durable container up to it; bounded by a runaway guard.
293 template<typename Dummy = void>
294 SuccessCode EscalateAndMigrate(unsigned guard)
295 {
296 if (guard > 64)
297 return SuccessCode::HigherPrecisionNecessary;
298 unsigned newprec = this->NextEscalatedPrecision();
299 this->AsFlavor().template MigrateContainersToPrecision<>(newprec);
300 this->current_endgame_precision_ = newprec;
301 SetThreadPrecision(newprec);
302 return SuccessCode::Success;
303 }

457 template<typename ComplexT>
458 SuccessCode ComputeInitialSamples(const ComplexT & start_time,const ComplexT & target_time, const Vec<ComplexT> & x_endgame_start, TimeCont<ComplexT> & times, SampCont<ComplexT> & samples) // passed by reference to allow times to be filled as well.
459 {
460 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
461 assert(this->template Get<EndgameConfig>().num_sample_points>0 && "number of sample points must be positive");
462
463 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
464 {
465 assert(Precision(start_time)==Precision(x_endgame_start) && "Computing initial samples requires input time and space with uniform precision");
466 }
467
468 samples.clear();
469 times.clear();
470
471 samples.push_back(x_endgame_start);
472 times.push_back(start_time);
473
474 auto num_vars = this->GetSystem().NumVariables();
475 //start at 1, because the input point is the 0th element.
476 for(unsigned ii=1; ii < this->template Get<EndgameConfig>().num_sample_points; ++ii)
477 {
478 times.emplace_back((times[ii-1] + target_time) * RealT(this->template Get<EndgameConfig>().sample_factor)); // next time is a point between the previous time and target time.
479 samples.emplace_back(Vec<ComplexT>(num_vars)); // sample_factor gives us some point between the two, usually the midpoint.
480
481 auto tracking_success = this->EndgameTrackPath(samples[ii],times[ii-1],times[ii],samples[ii-1]);
482 this->EnsureAtPrecision(times[ii],Precision(samples[ii]));
483
484 if (tracking_success!=SuccessCode::Success)
485 return tracking_success;
486 }
487
488 return SuccessCode::Success;
489 }
