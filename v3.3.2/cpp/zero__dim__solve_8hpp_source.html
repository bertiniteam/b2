391{

393 unsigned long long num_finite_solutions = 0;
394 unsigned long long num_finite_endpoints = 0;
395 unsigned long long num_diverged = 0;
396 unsigned long long num_failed = 0;
397 unsigned long long num_singular = 0;
398 unsigned long long num_real = 0;
399 unsigned long long num_nonsolutions = 0;
400 std::map<SuccessCode, unsigned long long> failures_by_reason;
401 double max_condition_number = 0;
402 unsigned max_precision_used = 0;
403 MidpathCheckReport midpath;
404 bool all_paths_resolved = false;
405};

411template<typename ComplexT>
412SolveReport SummarizeSolve(std::vector<SolutionMetaData<ComplexT>> const& metadata,
413 MidpathCheckReport const& midpath)
414{
415 SolveReport r;
416 r.num_paths_tracked = metadata.size();
417 r.midpath = midpath;
418 double finite_distinct = 0;
419 for (auto const& m : metadata)
420 {
421 if (m.max_precision_used > r.max_precision_used)
422 r.max_precision_used = m.max_precision_used;
423
424 // A path the endgame's security check truncated near infinity (SecurityMaxNormReached) is
425 // diverging, not failing -- it is the security/max-norm heuristic catching a path on its way
426 // to infinity, exactly like a clean GoingToInfinity. (The endgame tests classify the two
427 // together too.) With infinite-path truncation on by default, these are expected, so they
428 // must not inflate num_failed / clear all_paths_resolved.
429 bool diverged = (m.endgame_success_code == SuccessCode::GoingToInfinity
430 || m.endgame_success_code == SuccessCode::SecurityMaxNormReached);
431 if (m.endgame_success_code == SuccessCode::Success)
432 {
433 if (m.is_nonsolution)
434 ++r.num_nonsolutions; // a nonsolution: finite, but not a solution of the target
435 else if (m.is_finite)
436 {
437 ++r.num_finite_endpoints;
438 finite_distinct += 1.0 / m.multiplicity;
439 if (m.is_singular) ++r.num_singular;
440 if (m.is_real) ++r.num_real;
441 if (static_cast<double>(m.condition_number) > r.max_condition_number)
442 r.max_condition_number = static_cast<double>(m.condition_number);
443 }
444 else
445 diverged = true;
446 }
447
448 if (diverged)
449 ++r.num_diverged;
450 else if (m.endgame_success_code != SuccessCode::Success) // neither a solution nor a clean divergence
451 {
452 ++r.num_failed;
453 ++r.failures_by_reason[m.endgame_success_code];
454 }
455 }
456 r.num_finite_solutions = static_cast<unsigned long long>(finite_distinct + 0.5);
457 r.all_paths_resolved = (r.num_failed == 0) && midpath.passed;
458 return r;
459}

664#ifdef BERTINI2_HAVE_MPI
677 void RunParallel(MPI_Comm comm)
678 {
679 using Result = parallel::FullPathResult<BaseComplexT>;
680 using Task = parallel::StartPointTask<BaseComplexT>;
681
682 ScopedSessionState const session_state_guard{}; // a solve leaves the session as it found it
683
684 mpfr_free_cache(); // reproducibility: clear this rank's mpfr constant cache (see Solve())
685
686 solutions_user_coords_fresh_ = false;
687
688 // Each rank built its homotopy independently in the constructor -- with its OWN
689 // random patch, start-system coefficients, and gamma -- so a given path index would
690 // mean a different path on each worker, and the manager would gather a scrambled,
691 // mostly-wrong solution set. Make rank 0 the single authoritative source: broadcast
692 // its actual target system, start system, and homotopy to every rank. These are all
693 // exact (rational coefficients + integer/rational patch), so serialization carries
694 // them bit-for-bit -- there is no per-rank re-derivation that could drift, and the
695 // distributed homotopy is identical to rank 0's serial one. We also broadcast rank
696 // 0's RNG seed so any randomness drawn later (e.g. in the endgame) is consistent; the
697 // per-path tracking RNG is already deterministic in the path index (ReseedThisThread).
698 {
699 unsigned long seed = parallel::IsManager() ? GetGlobalSeed() : 0ul;
700 MPI_Bcast(&seed, 1, MPI_UNSIGNED_LONG, 0, comm);
701 SetGlobalSeed(seed);
702
703 // Install rank 0's authoritative systems on every rank. The plain engine over a
704 // user-supplied homotopy does nothing here (the user owns cross-rank consistency);
705 // ZeroDimSolver, which owns its systems, overrides DistributeSystems to broadcast
706 // rank 0's target / start / homotopy and re-seat the engine's references.
707 DistributeSystems(comm);
708
709 num_start_points_ = StartSystem().NumStartPoints();
710 GetTracker().SetSystem(Homotopy());
711 }
712
713 PreSolveChecks();
714 PreSolveSetup();
715
716 const int n_threads = parallel::WorkerThreadCount();
717
718 // One round = dispatch a set of path indices, each executed as a WHOLE path
719 // (pre-endgame + endgame) by a worker. Round 0 is every path; later rounds re-run
720 // only the crossed paths with escalated settings -- the re-track is path-independent,
721 // so it rides the same demand-driven worker pool (no idle-manager bottleneck).
722 if (parallel::IsManager())
723 {
724 // Rank 0 computes every start point once, authoritatively, and ships each to the
725 // worker that tracks it. Cached so re-track rounds reuse the identical points.
726 std::vector<Vec<BaseComplexT>> start_points(num_start_points_);
727 for (decltype(num_start_points_) ii{0}; ii < num_start_points_; ++ii)
728 start_points[ii] = ComputeStartPoint(static_cast<SolnIndT>(ii));
729
730 auto run_round = [&](std::queue<Task>& q)
731 {
732 parallel::RunManagerLoop<Task, Result>(comm, q,
733 [this](Result const& r){ StoreFullPathResult(r); });
734 };
735
736 // the records seam (ADR-0046): the manager is the sole writer; recall what
737 // is already recorded and dispatch only the rest
738 MaybeAttachAmbientRecords();
739 std::vector<SolnIndT> indices_to_run;
740 for (decltype(num_start_points_) ii{0}; ii < num_start_points_; ++ii)
741 indices_to_run.push_back(static_cast<SolnIndT>(ii));
742 if (records_)
743 {
744 EnsureRunRecorded();
745 indices_to_run = RecallRecordedPaths(indices_to_run);
746 }
747
748 std::queue<Task> queue;
749 for (auto idx : indices_to_run)
750 queue.push(Task{ idx, start_points[idx] });
751 run_round(queue);
752
753 // Midpath/crossing check on the collected boundary points, then bounded parallel
754 // resolve rounds. A continue/stop flag is broadcast to the workers each round so
755 // they know whether to run again (and escalate their settings first).
756 auto passed = midpath_.Check(solutions_at_endgame_boundary_, StartSystem());
757 midpath_report_ = MidpathCheckReport{};
758 midpath_report_.passed = passed;
759 for (auto const& v : midpath_.GetCrossedPaths())
760 midpath_report_.crossed_path_indices.push_back(v.index());
761 midpath_report_.num_crossings_detected =
762 static_cast<unsigned>(midpath_report_.crossed_path_indices.size());
763
764 const auto max_attempts = this->template Get<ZeroDimConf>().max_num_crossed_path_resolve_attempts;
765 unsigned num_resolve_attempts = 0;
766 while (true)
767 {
768 int keep_going = (!passed && num_resolve_attempts < max_attempts) ? 1 : 0;
769 MPI_Bcast(&keep_going, 1, MPI_INT, 0, comm);
770 if (!keep_going)
771 break;
772
773 std::queue<Task> redo;
774 for (auto const& v : midpath_.GetCrossedPaths())
775 if (v.rerun())
776 {
777 auto idx = static_cast<SolnIndT>(v.index());
778 redo.push(Task{ idx, start_points[idx] });
779 }
780 run_round(redo);
781
782 passed = midpath_.Check(solutions_at_endgame_boundary_, StartSystem());
783 ++num_resolve_attempts;
784 }
785 midpath_report_.num_resolve_attempts = num_resolve_attempts;
786 midpath_report_.passed = passed;
787 if (!passed)
788 std::cerr << "warning: " << midpath_report_.num_crossings_detected
789 << " path crossing(s) detected at the endgame boundary remained unresolved "
790 << "after " << num_resolve_attempts << " re-track attempt(s); "
791 << "the affected solutions may be wrong. Consider a higher-order predictor "
792 << "or a tighter tracking tolerance. See EndgameBoundaryMetadata()." << std::endl;
793
794 PostEGAction();
795
796 DeclareFiniteSolutionsResult();
797 }
798 else // worker
799 {
800 auto run_worker_round = [&]()
801 {
802 if (n_threads <= 1)
803 {
804 // Single-threaded worker: execute whole paths on the rank's member
805 // tracker/endgame directly, from rank 0's authoritative start point.
806 parallel::RunWorkerLoop<Task, Result>(comm,
807 [this](Task const& task){ ExecuteOnePath(MemberDuringEGContext(), static_cast<SolnIndT>(task.path_index), task.start_point); },
808 [this](Task const& task) -> Result { return PackFullPathResult(static_cast<SolnIndT>(task.path_index)); });
809 }
810 else
811 {
812 // Threaded worker: each thread owns a self-contained PathThreadState clone
813 // so concurrent whole-path execution shares no mutable state. Cloned per
814 // round so it inherits any escalation (predictor bump) applied to the rank's
815 // member tracker between rounds.
816 auto state_factory = [this]() -> std::unique_ptr<PathThreadState>
817 {
818 // Clone(), not copy: System copies are SHALLOW (shared expression-tree
819 // nodes whose value caches mutate on Eval); Clone() deep-copies the tree.
820 // Trackers/Endgames have no default ctor, so aggregate-init via new.
821 std::unique_ptr<PathThreadState> s(
822 new PathThreadState{ Clone(GetTracker().GetSystem()), Clone(TargetSystem()),
823 GetTracker(), GetEndgame(), {}, {} });
824 s->tracker.SetSystem(s->sys);
825 s->endgame.SetTracker(s->tracker);
826 SetThreadPrecision(this->template Get<ZeroDimConf>().initial_ambient_precision);
827 return s;
828 };
829
830 auto track_fn = [this](std::unique_ptr<PathThreadState>& state, Task const& task) -> Result
831 {
832 auto idx = static_cast<SolnIndT>(task.path_index);
833 ExecuteOnePath(state->Context(), idx, task.start_point);
834 return PackFullPathResult(idx);
835 };
836
837 parallel::RunWorkerLoopThreaded<Task, Result>(
838 comm, state_factory, track_fn, n_threads);
839 }
840 }; // run_worker_round
841
842 while (true)
843 {
844 run_worker_round();
845
846 int keep_going = 0;
847 MPI_Bcast(&keep_going, 1, MPI_INT, 0, comm);
848 if (!keep_going)
849 break;
850
851 // Match the escalation the serial RunMidpathResolution applies between
852 // re-track passes: tighten midpath_retrack_tolerance_ (read by ExecuteOnePath)
853 // and bump a low-order predictor on this rank's member tracker, so the next
854 // round's thread clones inherit it.
855 EscalateRetrackSettings();
856 }
857 }
858 }

1085 void Solve()
1086 {
1087 // Reproducibility: clear this thread's mpfr constant cache (pi, etc.) at the solve
1088 // boundary. mpfr caches transcendental constants at the precision last requested; the
1089 // Cauchy endgame needs pi (roots of unity), so a high-precision solve leaves pi cached
1090 // at high precision. A subsequent lower-precision solve would otherwise reuse that
1091 // cached value rounded down -- differing by ~1 ULP from a freshly-computed
1092 // low-precision pi -- making a solve's behaviour depend on what ran before it in the
1093 // process (non-reproducible). Clearing here makes each solve independent of history.
1094 // Cheap (one constant recompute per solve); thread-local (each worker thread clears its
1095 // own when it starts a solve).
1096 mpfr_free_cache();
1097
1098 ScopedSessionState const session_state_guard{}; // a solve leaves the session as it found it
1099
1100 solutions_user_coords_fresh_ = false;
1101
1102 PreSolveChecks();
1103
1104 PreSolveSetup();
1105
1106 this->NotifyObservers(AlgorithmStarted<AnyZeroDim>(*this));
1107
1108 // Speculative-full-path model: carry every path all the way through (pre-endgame +
1109 // endgame) as one unit, then detect crossings from the collected boundary points and
1110 // re-run any crossed path in full. This is the same shape the distributed solve uses
1111 // (one worker per whole path), so serial and distributed share the per-path primitive
1112 // -- including computing the start point via the same ComputeStartPoint.
1113
1114 // Compute every start point once, up front (deterministic per index, as the MPI
1115 // manager does). The threaded dispatch ships each worker its start point, and the
1116 // serial path / crossed-path re-tracks reuse the identical points from here.
1117 std::vector<Vec<BaseComplexT>> start_points(num_start_points_);
1118 std::vector<SolnIndT> all_indices(num_start_points_);
1119 for (decltype(num_start_points_) ii{0}; ii < num_start_points_; ++ii)
1120 {
1121 auto idx = static_cast<SolnIndT>(ii);
1122 start_points[ii] = ComputeStartPoint(idx);
1123 all_indices[ii] = idx;
1124 }
1125
1126 // the records seam (ADR-0046): ensure the run is on record, recall any paths
1127 // already recorded (solve() is ensure-answered), and compute only the rest
1128 MaybeAttachAmbientRecords();
1129 std::vector<SolnIndT> indices_to_run = all_indices;
1130 if (records_)
1131 {
1132 EnsureRunRecorded();
1133 // recall==false forces a fresh track of every path even when identical results are
1134 // already recorded (e.g. to run path observers or benchmark); the fresh run is still
1135 // recorded. See ZeroDimConfig::recall.
1136 if (this->template Get<ZeroDimConf>().recall)
1137 indices_to_run = RecallRecordedPaths(all_indices);
1138 }
1139
1140 // num_threads: 0 = auto (hardware_concurrency), 1 = serial, N = N threads;
1141 // OMP_NUM_THREADS overrides. n_threads <= 1 takes the pool-free serial path.
1142 const unsigned n_threads =
1143 parallel::EffectiveThreadCount(this->template Get<ZeroDimConf>().num_threads);
1144
1145 if (n_threads <= 1)
1146 {
1147 for (auto idx : indices_to_run)
1148 {
1149 ExecuteOnePath(MemberDuringEGContext(), idx, start_points[idx]);
1150 // the serial path installs results directly (no StoreFullPathResult),
1151 // so emit its record here
1152 RecordCompletedPath(PackFullPathResult(idx));
1153 }
1154 }
1155 else
1156 {
1157 RunPathsThreaded(indices_to_run, start_points, n_threads);
1158 }
1159
1160 // Crossed-path re-tracks run on the main thread against the (escalated) member
1161 // tracker. They are typically rare (often none); running them in parallel too is a
1162 // deferred optimization.
1163 RunMidpathResolution([this, &start_points](SolnIndT idx){
1164 ExecuteOnePath(MemberDuringEGContext(), idx, start_points[idx]);
1165 RecordCompletedPath(PackFullPathResult(idx)); // a re-track appends a fresh record (last wins)
1166 });
1167
1168 PostEGAction();
1169
1170 DeclareFiniteSolutionsResult();
1171
1172 this->NotifyObservers(AlgorithmComplete<AnyZeroDim>(*this));
1173 }

1377 SolutionMetaDataT MetadataForPoint(Vec<BaseComplexT> const& point, NumErrorT tol,
1378 bool user_coords = true, bool representatives_only = true) const
1379 {
1380 auto const& sols = user_coords ? SolutionsUserCoords() : SolutionsInternalCoords();
1381 auto const& md = SolutionMetadata();
1382
1383 std::vector<size_t> matches;
1384 for (size_t i = 0; i < md.size() && i < sols.size(); ++i)
1385 {
1386 if (representatives_only && !md[i].multiplicity_representative)
1387 continue;
1388 if (!bertini::IsDistinct(sols[i], point, tol))
1389 matches.push_back(i);
1390 }
1391
1392 if (matches.empty())
1393 throw std::runtime_error("metadata_for: no solution matches the given point within tol");
1394
1395 // More than one representative among the matches => tol grouped two distinct clusters.
1396 // With representatives_only (the default) and the default tol this cannot happen -- distinct
1397 // representatives are at least SamePointTolerance apart and the window is smaller -- so it
1398 // only fires for a deliberately coarse user-supplied tol, where it is the honest answer.
1399 std::vector<size_t> reps;
1400 for (auto i : matches)
1401 if (md[i].multiplicity_representative)
1402 reps.push_back(i);
1403 if (reps.size() > 1)
1404 throw std::runtime_error("metadata_for: the point matches more than one distinct solution cluster; reduce tol");
1405 if (reps.size() == 1)
1406 return md[reps.front()];
1407 // No representative matched (only reachable with representatives_only=false, when the point
1408 // is a non-representative copy whose representative fell outside tol); return the matched
1409 // copy's own record (it still carries the cluster's multiplicity).
1410 return md[matches.front()];
1411 }

1429 std::vector<SolutionMetaDataT> CoincidentMetadataForPoint(Vec<BaseComplexT> const& point, NumErrorT tol, bool user_coords = true) const
1430 {
1431 auto const& sols = user_coords ? SolutionsUserCoords() : SolutionsInternalCoords();
1432 auto const& md = SolutionMetadata();
1433
1434 // Find the representative NEAREST the point (within tol). Nearest, not "all within tol", so
1435 // the cluster is identified unambiguously even for a coarse tol.
1436 size_t rep = md.size();
1437 NumErrorT best = std::numeric_limits<NumErrorT>::infinity();
1438 for (size_t i = 0; i < md.size() && i < sols.size(); ++i)
1439 {
1440 if (!md[i].multiplicity_representative)
1441 continue;
1442 NumErrorT d = static_cast<NumErrorT>((sols[i] - point).template lpNorm<Eigen::Infinity>());
1443 if (d <= tol && d < best) { best = d; rep = i; }
1444 }
1445 if (rep == md.size())
1446 throw std::runtime_error("metadata_for: no solution matches the given point within tol");
1447
1448 // Gather the whole cluster: every endpoint the solver deemed the same solution as the
1449 // representative (within the same-point clustering tolerance).
1450 const NumErrorT same_tol = SamePointTolerance();
1451 std::vector<SolutionMetaDataT> out;
1452 for (size_t i = 0; i < md.size() && i < sols.size(); ++i)
1453 if (!bertini::IsDistinct(sols[i], sols[rep], same_tol))
1454 out.push_back(md[i]);
1455 return out;
1456 }

1501 void PreSolveSetup()
1502 {
1503 // Fixed-multiple precision: FixedPrecisionConfig.precision (on the tracker) is the
1504 // authoritative precision for the whole solve. Lift the tracker, the ambient/thread
1505 // precision, the start-point precision (via initial_ambient_precision), and the systems
1506 // all to that one value -- TrackerLoopInitialization requires them to agree. (Double is
1507 // fixed at 16 and adaptive manages its own precision, so neither enters here.)
1508 if constexpr (!tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1509 {
1510 auto const& fp = GetTracker().template Get<PrecisionConfig>();
1511 // Double validates (its PrecisionSetup throws on any precision other than 16);
1512 // fixed-multiple adopts the value into precision_.
1513 GetTracker().PrecisionSetup(fp);
1514 if constexpr (!std::is_same<BaseComplexT, complex_dbl>::value)
1515 if (fp.precision != 0)
1516 {
1517 ZeroDimConf zdc = this->template Get<ZeroDimConf>();
1518 zdc.initial_ambient_precision = fp.precision; // -> thread + start-point precision
1519 this->template Set<ZeroDimConf>(zdc);
1520 TargetSystem().precision(fp.precision);
1521 Homotopy().precision(fp.precision);
1522 }
1523 }
1524
1525 auto num_as_size_t = static_cast<SolnIndT>(num_start_points_);
1526
1527 solution_final_metadata_.resize(num_as_size_t);
1528 solutions_at_endgame_boundary_.resize(num_as_size_t);
1529 solutions_post_endgame_.resize(num_as_size_t);
1530
1531 SetMidpathRetrackTol(this->template Get<Tolerances>().newton_before_endgame);
1532
1533 // The solver's Tolerances.final_tolerance is the single source of truth for how tightly
1534 // a solve converges its endpoints. Flow it into the endgame here (the endgame keeps its
1535 // own EndgameConfig.final_tolerance for standalone, solver-independent use, but under a
1536 // solve the solver wins). Without this the endgame would silently ignore the solver's
1537 // final_tolerance -- e.g. a classic-input FinalTol -- and use its own default instead.
1538 endgame_.SetFinalTolerance(this->template Get<Tolerances>().final_tolerance);
1539
1540 // Default the start-point precision to the initial ambient precision. A caller can
1541 // override it via SetStartPointPrecision (e.g. to carry a higher precision forward
1542 // from a previous solve whose output feeds this one).
1543 if (!start_point_precision_set_by_user_)
1544 start_point_precision_ = this->template Get<ZeroDimConf>().initial_ambient_precision;
1545 }

1606 void ExecuteBeforeEG(BeforeEGContext ctx, SolnIndT soln_ind, Vec<BaseComplexT> const& start_point)
1607 {
1608 ReseedThisThread(static_cast<uint64_t>(soln_ind));
1609
1610 if (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1611 {
1612 ctx.tracker.AddObserver(ctx.first_prec_rec);
1613 ctx.tracker.AddObserver(ctx.min_max_prec);
1614 }
1615
1616 auto& smd = solution_final_metadata_[soln_ind];
1617 smd.path_index = soln_ind;
1618 smd.solution_index = soln_ind;
1619
1620 auto initial_prec = this->template Get<ZeroDimConf>().initial_ambient_precision;
1621 // SetThreadPrecision: writes thread-local only, safe from concurrent threads.
1622 SetThreadPrecision(initial_prec);
1623
1624 // Draw the condition-number probe direction ONCE here, for the whole track of this
1625 // point (pre-endgame AND endgame). We have just reseeded this thread's RNG to a value
1626 // determined solely by the path index, so the probe is deterministic per path and
1627 // identical whether the path is tracked in a serial loop or on a distributed worker --
1628 // regardless of how the paths were split across ranks. It is NOT refreshed for the
1629 // endgame (which would churn the RNG across its sample-circle sub-tracks), so the same
1630 // direction persists start to finish.
1631 ctx.tracker.RefreshConditionDirection();
1632
1633 // The times are stored precision-free (mpq_rational); materialize them as the tracking
1634 // complex type at the current working precision.
1635 BaseComplexT t_start ( BaseRealT(this->template Get<ZeroDimConf>().start_time) );
1636 BaseComplexT t_endgame_boundary( BaseRealT(this->template Get<ZeroDimConf>().endgame_boundary) );
1637
1638 // The start point is supplied by the caller (computed once, authoritatively, via
1639 // ComputeStartPoint) rather than regenerated here. In a distributed solve rank 0
1640 // computes it and sends it with the task, so a worker never derives its own -- that,
1641 // plus authoritative pi (issue #156), is what makes the start point identical across
1642 // serial and distributed runs.
1643
1644 // Reset to the configured initial step size for this fresh path. In the
1645 // speculative-full-path model a path's endgame runs immediately before the next
1646 // path's pre-endgame tracking on the same tracker, and the endgame leaves
1647 // ReinitializeInitialStepSize(false) with a tiny step; without restoring it here the
1648 // next path would crawl from the start time with that tiny step (effectively a hang).
1649 ctx.tracker.ReinitializeInitialStepSize(true);
1650
1651 // Begin every AMP path in hardware double precision; the AMP step criteria escalate to
1652 // multiprecision only where a path actually needs it (and the endgame, which clears
1653 // this override below, manages its own precision). Total-degree start points are born
1654 // at LowestMultiplePrecision (the generator's arithmetic widens past the requested
1655 // digits, see issue #308), and initial_ambient_precision is itself a multiprecision
1656 // value, so without forcing double here the tracker starts -- and, in practice, stays
1657 // -- in multiprecision on every path, even trivially well-conditioned ones.
1658 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1659 ctx.tracker.SetStartPrecision(DoublePrecision());
1660
1661 Vec<BaseComplexT> result;
1662 auto tracking_success = ctx.tracker.TrackPath(result, t_start, t_endgame_boundary, start_point);
1663
1664 solutions_at_endgame_boundary_[soln_ind] =
1665 EGBoundaryMetaDataT({ result, tracking_success, ctx.tracker.CurrentStepsize(), ctx.tracker.CurrentPrecision() });
1666
1667 // Clear the start-precision override so it does not leak into the endgame, which
1668 // shares this tracker instance for its sample circles.
1669 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1670 ctx.tracker.SetStartPrecision(std::nullopt);
1671
1672 smd.pre_endgame_success_code = tracking_success;
1673
1674 if (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1675 {
1676 if (ctx.first_prec_rec.DidPrecisionIncrease())
1677 {
1678 smd.precision_changed = true;
1679 smd.time_of_first_prec_increase = ctx.first_prec_rec.TimeOfIncrease();
1680 }
1681 else
1682 ctx.tracker.RemoveObserver(ctx.first_prec_rec);
1683 ctx.tracker.RemoveObserver(ctx.min_max_prec);
1684 using std::max;
1685 smd.max_precision_used =
1686 max(smd.max_precision_used, ctx.min_max_prec.MaxPrecision());
1687 }

1829 void ExecuteOnePath(DuringEGContext ctx, SolnIndT soln_ind, Vec<BaseComplexT> const& start_point)
1830 {
1831 // Carry the executing tracker on the path events: &ctx.tracker is the member tracker
1832 // in a serial solve and the thread-local clone in a threaded one, so a meta-observer
1833 // attaches its per-path sub-observer to the tracker that actually runs this path.
1834 Observable const* exec_tracker = &ctx.tracker;
1835
1836 // wall-clock the whole path (pre-endgame + endgame). steady_clock: monotonic, safe on
1837 // the worker thread; written to this path's own metadata slot, so no cross-path race.
1838 auto const path_start_clock = std::chrono::steady_clock::now();
1839 auto stamp_path_time = [&]{
1840 solution_final_metadata_[soln_ind].path_time_seconds =
1841 std::chrono::duration<double>(std::chrono::steady_clock::now() - path_start_clock).count();
1842 };
1843
1844 this->NotifyObservers(PathStarted<AnyZeroDim>(*this, static_cast<std::size_t>(soln_ind), exec_tracker));
1845
1846 ctx.tracker.SetTrackingTolerance(midpath_retrack_tolerance_);
1847 ExecuteBeforeEG(BeforeEGContext{ ctx.tracker, ctx.first_prec_rec, ctx.min_max_prec }, soln_ind, start_point);
1848
1849 if (solution_final_metadata_[soln_ind].pre_endgame_success_code != SuccessCode::Success)
1850 {
1851 stamp_path_time();
1852 this->NotifyObservers(PathComplete<AnyZeroDim>(*this, static_cast<std::size_t>(soln_ind), exec_tracker));
1853 return;
1854 }
1855
1856 ctx.tracker.SetTrackingTolerance(this->template Get<Tolerances>().newton_during_endgame);
1857 ExecuteDuringEG(ctx, soln_ind);
1858
1859 stamp_path_time();
1860 this->NotifyObservers(PathComplete<AnyZeroDim>(*this, static_cast<std::size_t>(soln_ind), exec_tracker));
1861 }

1930

1950 // metadata cannot be computed.

1970 }

1995 // stored solution point (DoublePrecision() for a fast-lane path, the mpfr precision for an
1996 // escalated one). accuracy_digits is how many of those digits the convergence agreement
1997 // supports: floor(-log10(accuracy_estimate)), clamped to [0, precision_digits].
1998 smd.precision_digits = static_cast<unsigned>(Precision(solutions_post_endgame_[soln_ind]));

2004 : smd.precision_digits;

2683 template<typename SystemType>
2684 struct OwnedHomotopy
2685 {
2686 using StartSystemBaseT = bertini::start_system::StartSystem;
2687 using FactoryT = bertini::start_system::StartSystemFactory<SystemType>;
2688
2690 OwnedHomotopy(SystemType const& target, FactoryT factory, std::string const& path_variable_name)
2691 : user_target_(Clone(target)), owned_target_(Clone(target)), owned_factory_(std::move(factory))
2692 {
2693 ConsistencyCheck(); // feasibility (no path var; not under-constrained; polynomial)
2694 SquareUp(); // randomize an over-determined system down to square
2695 RankCheck(); // square, but can isolated solutions even exist?
2696 PrepareTarget(owned_target_); // homogenize + auto-patch the (now square) target
2697 owned_start_ = owned_factory_(owned_target_); // start system over the prepared target
2698 // Treat the configured name as a BASE: mangle it only if it would collide with a
2699 // user variable, so the injected path variable can never overlap a user symbol.
2700 owned_homotopy_ = MakeHomotopy(owned_target_, *owned_start_,
2701 UniquePathVariableName(owned_target_, path_variable_name));
2702 }
2703
2705 SystemType const& BuiltTarget() const { return owned_target_; }
2710 SystemType const& UserTarget() const { return user_target_; }
2712 StartSystemBaseT const& BuiltStart() const { return *owned_start_; }
2714 SystemType const& BuiltHomotopy() const { return owned_homotopy_; }
2715
2716 protected:
2717 SystemType user_target_;
2718 SystemType owned_target_;
2719 std::shared_ptr<StartSystemBaseT> owned_start_;
2720 SystemType owned_homotopy_;
2721 FactoryT owned_factory_;
2722
2723 // When the user's system was over-determined, SquareUp randomizes it down to square for
2724 // tracking and keeps the ORIGINAL (natural, un-homogenized) system here so ZeroDimSolver
2725 // can discard the extraneous solutions the squaring introduces. was_randomized_ gates
2726 // that filter; randomization_matrix_ records the (exact) coefficient matrix used.
2727 bool was_randomized_ = false;
2728 SystemType original_natural_target_;
2729 Mat<complex_mp> randomization_matrix_;
2730
2734 void ConsistencyCheck() const
2735 {
2736 if (owned_target_.HavePathVariable())
2737 throw std::runtime_error("unable to perform zero dim solve on target system -- has path variable, use a HomotopySolver instead.");
2738
2739 // A square zero-dim system needs one equation per dimension. Each projective
2740 // (homogeneous) variable group of size k spans P^{k-1}: its k coordinates carry only
2741 // k-1 dimensions because scale is free, so it needs one fewer equation than it has
2742 // variables. Subtract that free scale per projective group before comparing. An
2743 // UNDER-determined system has a positive-dimensional solution set, so no isolated
2744 // solutions to compute -- raise a helpful error rather than tracking garbage.
2745 if (owned_target_.NumVariables() - owned_target_.NumHomVariableGroups() > owned_target_.NumTotalFunctions())
2746 throw std::runtime_error("unable to perform zero dim solve on target system -- it is under-determined (fewer equations than variables), so its solution set is positive-dimensional, not zero-dimensional. ZeroDimSolver computes isolated solutions only; add equations, or use a positive-dimensional method.");
2747
2748 if (!owned_target_.IsPolynomial())
2749 throw std::runtime_error("unable to perform zero dim solve on target system -- system is non-polynomial, use a HomotopySolver instead.");
2750 }
2751
2758 void SquareUp()
2759 {
2760 auto const dimension = owned_target_.NumVariables() - owned_target_.NumHomVariableGroups();
2761 if (owned_target_.NumTotalFunctions() <= dimension)
2762 return; // already square (or under-determined, which ConsistencyCheck already rejected)
2763
2764 original_natural_target_ = Clone(owned_target_); // keep the original N-function system
2765 owned_target_ = owned_target_.Randomize();
2766 randomization_matrix_ = owned_target_.RandomizationMatrix();
2767 was_randomized_ = true;
2768 }
2769
2779 void RankCheck() const
2780 {
2781 if (owned_target_.HavePathVariable())
2782 return;
2783 auto const n = static_cast<Eigen::Index>(owned_target_.NumVariables());
2784 if (static_cast<Eigen::Index>(owned_target_.NumTotalFunctions()) != n || n == 0)
2785 return; // not a square map: ConsistencyCheck governs feasibility here
2786
2787 // a generic complex sample point: a zero-dimensional variety misses it almost surely,
2788 // where the Jacobian attains its generic (full) rank.
2789 Vec<complex_dbl> pt = Vec<complex_dbl>::Random(n);
2790 Mat<complex_dbl> J = owned_target_.template Jacobian<complex_dbl>(pt);
2791 Eigen::FullPivLU<Mat<complex_dbl>> lu(J);
2792 lu.setThreshold(1e-10);
2793 if (lu.rank() < n)
2794 throw std::runtime_error("unable to perform zero dim solve on target system -- the Jacobian is rank-deficient at a generic point, so the solution set is positive-dimensional, not zero-dimensional. ZeroDimSolver computes isolated solutions only.");
2795 }

2861 void PostEGAction() override
2862 {
2863 EngineT::PostEGAction(); // the engine's finite/real/multiplicity/singular classification
2864
2865 if (!this->was_randomized_)
2866 return;
2867
2868 // genuine roots satisfy the original system to ~solve accuracy; spurious ones miss it
2869 // by O(1). A generous multiple of the endgame tolerance separates the two cleanly.
2870 const double threshold =
2871 1e3 * static_cast<double>(this->template Get<TolerancesConfig>().newton_during_endgame);
2872
2873 for (decltype(this->num_start_points_) ii{0}; ii < this->num_start_points_; ++ii)
2874 {
2875 auto& smd = this->solution_final_metadata_[ii];
2876 if (smd.endgame_success_code != SuccessCode::Success || !smd.is_finite)
2877 continue;
2878
2879 auto user_pt = this->TargetSystem().DehomogenizePoint(this->solutions_post_endgame_[ii]);
2880 // Evaluate the original system in DOUBLE precision: the filter only needs to tell a
2881 // genuine root (tiny residual) from an extraneous one (O(1)), and a double residual
2882 // does that robustly without a precision mismatch between the (possibly AMP/high
2883 // precision) solution point and the original system's working precision.
2884 Vec<complex_dbl> pt_d(user_pt.size());
2885 for (Eigen::Index k = 0; k < user_pt.size(); ++k)
2886 pt_d(k) = complex_dbl(user_pt(k));
2887 auto residual = static_cast<NumErrorT>(
2888 this->original_natural_target_.template Eval<complex_dbl>(pt_d).template lpNorm<Eigen::Infinity>());
2889 smd.function_residual = residual; // report residual against the ORIGINAL system
2890 if (static_cast<double>(residual) > threshold)
2891 smd.is_nonsolution = true; // extraneous: finite, but not a solution of the original system
2892 }
