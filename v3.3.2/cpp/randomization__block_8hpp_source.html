112 // Memory-isolating copy (ADR-0027): the operand System is deep-copied via the
113 // System copy constructor, which shares its immutable node DAG and compiled SLP Program but
114 // gives it its own per-thread evaluation Memory. Everything else (the coefficient matrices,
115 // multidegrees, working buffers) is value state, copied per copy. This lets path tracking
116 // Clone a System into per-thread copies that share no mutable state.

200 void Homogenize(VariableGroup const& /*group*/, Var const& hom_var)
201 {
202 if (homogenized_)
203 return;
204 hom_vars_.push_back(hom_var);
205 if (hom_vars_.size() == num_groups_)
206 {
207 operand_->Homogenize(VariableGroup(hom_vars_));
208 homogenized_ = true;
209 // the hom var of group g now sits at a fixed position of the (shared) variable ordering;
210 // the block reads h_g from there during evaluation.
211 hom_var_index_.assign(num_groups_, -1);
212 auto const& ordering = operand_->Variables();
213 for (size_t g = 0; g < num_groups_; ++g)
214 for (size_t k = 0; k < ordering.size(); ++k)
215 if (ordering[k] == hom_vars_[g]) { hom_var_index_[g] = static_cast<int>(k); break; }
216 }
217 }

305 void Precision(unsigned new_precision) const
306 {
307 if (new_precision > DoublePrecision())
308 {
309 auto& wm = std::get<Mat<complex_mp>>(coefficients_working_);
310 for (Eigen::Index r = 0; r < wm.rows(); ++r)

327 template <typename T>
328 void EvalInPlace(Eigen::Ref<Vec<T>> result, Vec<T> const& vars, T const& /*path_value*/) const
329 {
330 SyncPrecision(vars);
331 const size_t N = static_cast<size_t>(coefficients_highest_precision_.cols());
332 const Vec<T> f = operand_->template Eval<T>(vars).head(static_cast<Eigen::Index>(N));
333 const auto& C = Working<T>();
334
335 result.setZero();
336 for (size_t i = 0; i < NumFunctions(); ++i)
337 {
338 T acc = Zero<T>();
339 for (size_t j = 0; j < N; ++j)
340 {
341 const T& c = C(static_cast<Eigen::Index>(i), static_cast<Eigen::Index>(j));
342 if (IsZero(c))
343 continue;
344 acc += c * f(static_cast<Eigen::Index>(j)) * HPower<T>(i, j, vars);
345 }
346 result(static_cast<Eigen::Index>(i)) = acc;
347 }
348 }

357 template <typename T>
358 void JacobianInPlace(Eigen::Ref<Mat<T>> J, Vec<T> const& vars, T const& /*path_value*/) const
359 {
360 SyncPrecision(vars);
361 const size_t N = static_cast<size_t>(coefficients_highest_precision_.cols());
362 const Vec<T> f = operand_->template Eval<T>(vars).head(static_cast<Eigen::Index>(N));
363 const Mat<T> Jf = operand_->template Jacobian<T>(vars).topRows(static_cast<Eigen::Index>(N));
364 const auto& C = Working<T>();
365
366 J.setZero();
367 for (size_t i = 0; i < NumFunctions(); ++i)
368 {
369 for (size_t j = 0; j < N; ++j)
370 {
371 const T& c = C(static_cast<Eigen::Index>(i), static_cast<Eigen::Index>(j));
372 if (IsZero(c))
373 continue;
374
375 const T W = HPower<T>(i, j, vars); // W_ij = prod_g h_g^{e_g}
376 // operand-Jacobian term: c_ij * W_ij * (d f_j / d x), across all columns.
377 J.row(static_cast<Eigen::Index>(i)) += (c * W) * Jf.row(static_cast<Eigen::Index>(j));
378
379 // h-power terms: only the homogenizing-variable columns pick up f_j * dW/dh_{g'}.
380 if (homogenized_)
381 {
382 for (size_t g = 0; g < num_groups_; ++g)
383 {
384 const int e = Deficit(i, j, g);
385 if (e <= 0 || hom_var_index_[g] < 0)
386 continue; // e<0 means c_ij==0 (already skipped); e==0 -> no dependence
387 // dW/dh_g = e * h_g^{e-1} * prod_{g2 != g} h_{g2}^{e_{g2}}
388 const T dW = MixedPartial<T>(i, j, g, vars);
389 J(static_cast<Eigen::Index>(i), hom_var_index_[g]) += c * f(static_cast<Eigen::Index>(j)) * dW;
390 }
391 }
392 }
393 }

422 int Deficit(size_t i, size_t j, size_t g) const
423 {
424 return target_multidegrees_[i][g] - operand_multidegrees_[j][g];
425 }

428 template <typename T>
429 T HPower(size_t i, size_t j, Vec<T> const& vars) const
430 {
431 T w(1);
432 if constexpr (!std::is_same<T, complex_dbl>::value) w.precision(precision_);
433 if (!homogenized_)
434 return w;
435 for (size_t g = 0; g < num_groups_; ++g)
436 {
437 const int e = Deficit(i, j, g);
438 if (e == 0 || hom_var_index_[g] < 0)
439 continue;
440 w *= IntPow<T>(vars(hom_var_index_[g]), e);
441 }
442 return w;
443 }
444
446 template <typename T>
447 T MixedPartial(size_t i, size_t j, size_t g, Vec<T> const& vars) const
448 {
449 const int eg = Deficit(i, j, g);
450 T d(eg);
451 if constexpr (!std::is_same<T, complex_dbl>::value) d.precision(precision_);
452 d *= IntPow<T>(vars(hom_var_index_[g]), eg - 1);
453 for (size_t g2 = 0; g2 < num_groups_; ++g2)
454 {
455 if (g2 == g)
456 continue;
457 const int e2 = Deficit(i, j, g2);
458 if (e2 == 0 || hom_var_index_[g2] < 0)
459 continue;
460 d *= IntPow<T>(vars(hom_var_index_[g2]), e2);
461 }
462 return d;
463 }

466 template <typename T>
467 static T IntPow(T const& base, int e)
468 {
469 T result(1);
470 if constexpr (!std::is_same<T, complex_dbl>::value) result.precision(bertini::Precision(base));
471 for (int k = 0; k < e; ++k)
472 result *= base;
473 return result;
474 }

477 int GroupIndex(VariableGroup const& vars) const
478 {
479 auto const& groups = operand_->VariableGroups();
480 for (size_t g = 0; g < groups.size(); ++g)
481 if (SameGroup(groups[g], vars))
482 return static_cast<int>(g);
483 return -1;
484 }
485 static bool SameGroup(VariableGroup const& a, VariableGroup const& b)
486 {
487 if (a.size() != b.size())
488 return false;
489 for (size_t k = 0; k < a.size(); ++k)
490 if (a[k] != b[k])
491 return false;
492 return true;
493 }

509 void BuildWorking() const
510 {
511 const auto& M = coefficients_highest_precision_;
512 auto& wd = std::get<Mat<complex_dbl>>(coefficients_working_);
513 auto& wm = std::get<Mat<complex_mp>>(coefficients_working_);
514 wd.resize(M.rows(), M.cols());
515 wm.resize(M.rows(), M.cols());
516 for (Eigen::Index r = 0; r < M.rows(); ++r)
517 for (Eigen::Index c = 0; c < M.cols(); ++c)
518 {
519 wd(r, c) = complex_dbl(M(r, c));
520 wm(r, c) = M(r, c);
521 }
522 }
