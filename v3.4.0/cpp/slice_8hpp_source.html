131 static Slice RandomReal(VariableGroup const& v, unsigned dim, bool homogeneous = false, bool orthogonal = true, Vec<complex_mp> const* through_point = nullptr)
132 {
133 typedef void (*funtype) (complex_mp&, unsigned); // the type for number generation
134 // bounded-modulus draw (away from 0 and infinity), matching patches and the start systems;
135 // kept REAL so a real slice stays real. The orthogonal path is real too (real=true below):
136 // it QR-factors a matrix of REAL units, yielding a real orthogonal coefficient block (issue
137 // #294 -- previously the orthogonal path hardcoded the complex orthonormal matrix, so a
138 // "real" slice came out complex).
139 funtype gen = bertini::multiprecision::RandomRealBoundedModulusAssign;
140 return Make(v, dim, homogeneous, orthogonal, /*real=*/true, gen, through_point);
141 }

175 static Slice Make(VariableGroup const& v, unsigned dim, bool homogeneous, bool orthogonal, bool real, std::function<void(complex_mp&, unsigned)> gen, Vec<complex_mp> const* through_point = nullptr)
176 {
177 const unsigned num_vars = static_cast<unsigned>(v.size());
178
179 if (through_point && static_cast<unsigned>(through_point->size()) != num_vars)
180 throw std::runtime_error("Slice: through_point must have num_variables entries");
181
182 Mat<complex_mp> coeffs(dim, num_vars); // the variable coefficients (one row per form)
183
184 if (orthogonal)
185 {
186 // conjugate-orthonormal coefficient matrix (orthonormal linear forms), drawn the b1 way
187 // via RandomConjugateOrthonormalMatrix (ADR-0041) -- it generates square and truncates,
188 // so the old transpose dance is gone. Built at max precision, like the rest of the slice.
189 // A real slice QR-factors a matrix of REAL units (a real orthogonal block); a complex
190 // slice uses complex units. (Issue #294: the real path must stay real.)
191 auto prev_precision = DefaultPrecision();
192 DefaultPrecision(MaxPrecisionAllowed());
193 if (real)
194 coeffs = bertini::RandomConjugateOrthonormalMatrix<real_mp>(dim, num_vars)
195 .unaryExpr([](real_mp const& r){ return complex_mp(r); });
196 else
197 coeffs = bertini::RandomConjugateOrthonormalMatrix<complex_mp>(dim, num_vars);
198 DefaultPrecision(prev_precision);
199 }
200 else
201 {
202 for (unsigned ii(0); ii < dim; ++ii)
203 for (unsigned jj(0); jj < num_vars; ++jj)
204 gen(coeffs(ii, jj), MaxPrecisionAllowed());
205 }
206
207 assert(static_cast<unsigned>(coeffs.rows()) == dim);
208 assert(static_cast<unsigned>(coeffs.cols()) == num_vars);
209
210 // Through a point. Homogeneous: project the rows into p's orthogonal complement (a.p = 0),
211 // zero constant column. Affine: the constant column is -A*p, so every form vanishes at p.
212 if (through_point && homogeneous)
213 return FromCoefficients(v, AugmentHomogeneousThroughPoint(coeffs, *through_point, orthogonal), /*homogeneous=*/true);
214 if (through_point)
215 return FromCoefficients(v, AugmentAffineThroughPoint(coeffs, *through_point), /*homogeneous=*/false);
216
217 // Assemble the augmented matrix: [ coeffs | constants ]. A homogeneous slice's constant
218 // column is zero; otherwise it is freshly generated.
219 Mat<complex_mp> augmented(dim, num_vars + 1);
220 augmented.leftCols(num_vars) = coeffs;
221 if (homogeneous)
222 augmented.col(num_vars).setZero();
223 else
224 for (unsigned ii(0); ii < dim; ++ii)
225 gen(augmented(ii, num_vars), MaxPrecisionAllowed());
226
227 return FromCoefficients(v, augmented, homogeneous);
228 }

309 Slice Concatenate(Slice const& other) const
310 {
311 if (NumVariables() != other.NumVariables())
312 throw std::runtime_error("Slice::Concatenate requires both slices to be on the same number of variables");
313
314 Mat<complex_mp> const& A = Coefficients();
315 Mat<complex_mp> const& B = other.Coefficients();
316 Mat<complex_mp> stacked(A.rows() + B.rows(), A.cols());
317 stacked.topRows(A.rows()) = A;
318 stacked.bottomRows(B.rows()) = B;
319
320 return FromCoefficients(sliced_vars_, stacked, is_homogeneous_ && other.is_homogeneous_);
321 }

422 static Mat<complex_mp> AugmentAffineThroughPoint(Mat<complex_mp> const& coeffs, Vec<complex_mp> const& point)
423 {
424 const auto num_vars = coeffs.cols();
425
426 auto prev_precision = DefaultPrecision();
427 DefaultPrecision(MaxPrecisionAllowed());
428
429 Mat<complex_mp> A = coeffs; // copies, re-materialized at max precision below
430 Vec<complex_mp> p = point;
431 bertini::Precision(A, MaxPrecisionAllowed()); // qualify: class has member Precision overloads
432 bertini::Precision(p, MaxPrecisionAllowed());
433
434 Mat<complex_mp> augmented(coeffs.rows(), num_vars + 1);
435 augmented.leftCols(num_vars) = A;
436 augmented.col(num_vars) = -(A * p); // b = -A*p, so A*p + b = 0
437
438 DefaultPrecision(prev_precision);
439 return augmented;
440 }

456 static Mat<complex_mp> AugmentHomogeneousThroughPoint(Mat<complex_mp> const& coeffs, Vec<complex_mp> const& point, bool orthogonal)
457 {
458 const auto dim = coeffs.rows();
459 const auto num_vars = coeffs.cols();
460
461 auto prev_precision = DefaultPrecision();
462 DefaultPrecision(MaxPrecisionAllowed());
463
464 Mat<complex_mp> A = coeffs;
465 Vec<complex_mp> p = point;
466 bertini::Precision(A, MaxPrecisionAllowed()); // qualify: class has member Precision overloads
467 bertini::Precision(p, MaxPrecisionAllowed());
468
469 real_mp p_norm_sq = p.squaredNorm(); // pbar.p = ||p||^2, real-positive for p != 0
470 if (p_norm_sq == real_mp(0))
471 throw std::runtime_error("Slice: a homogeneous slice through a point needs a nonzero point");
472
473 // project every row into p's orthogonal complement (bilinear a.p = 0): subtract the
474 // rank-one correction (A*p) pbar^T / ||p||^2.
475 Vec<complex_mp> Ap = A * p; // bilinear a_i . p, one per row
476 A = A - (Ap / complex_mp(p_norm_sq)) * p.conjugate().transpose();
477
478 if (orthogonal)
479 {
480 // orthonormalize the projected rows via a QR of A^T: the columns of Q span the row
481 // space of A (which lies in p's complement), so Q^T's rows stay orthogonal to p.
482 Eigen::HouseholderQR<Mat<complex_mp>> qr(A.transpose());
483 Mat<complex_mp> Q = qr.householderQ() * Mat<complex_mp>::Identity(num_vars, num_vars);
484 A = Q.leftCols(dim).transpose();
485 }
486
487 Mat<complex_mp> augmented(dim, num_vars + 1);
488 augmented.leftCols(num_vars) = A;
489 augmented.col(num_vars).setZero(); // homogeneous: no constant term
490
491 DefaultPrecision(prev_precision);
492 return augmented;
493 }
