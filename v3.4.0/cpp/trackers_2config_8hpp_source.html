193 struct AdaptiveMultiplePrecisionConfig
194 {
195 NumErrorT coefficient_bound;
196 NumErrorT degree_bound;
197
198 NumErrorT epsilon;
199 // rename to linear_solve_error_bound.
200
201 NumErrorT Phi;
202 // \f$\Phi\f$ is error in Jacobian evaluation divided by the unit roundoff error, \f$10^{-P}\f$
203 // rename to jacobian_eval_error_bound
204
205 NumErrorT Psi;
206 // Error in function evaluation, divided by the precision-dependent unit roundoff error.
207 // rename to function_eval_error_bound
208
209 int safety_digits_1 = 1;
210 int safety_digits_2 = 1;
211 unsigned int maximum_precision = 300;
212
213 // Note: a single setting -- Bertini 1's StepsForIncrease, i.e.
214 // SteppingConfig::consecutive_successful_steps_before_stepsize_increase -- gates BOTH stepsize
215 // increase AND precision decrease (the required number of consecutive successful steps).
216 // Precision decrease is additionally subject to B1's digits-margin hysteresis (see
217 // ExtraDigitsBeforePrecisionDecrease in amp_tracker.hpp). The old, duplicate AMP-config setting
218 // `consecutive_successful_steps_before_precision_decrease` (a B2 deviation) was removed.
219
220 unsigned max_num_precision_decreases = 10;
221

230 void SetBoundsAndEpsilonFrom(System const& sys)
231 {
232 using std::pow;
233
234 epsilon = pow(NumErrorT(sys.NumVariables()),2);
235 degree_bound = sys.DegreeBound();
236 coefficient_bound = sys.CoefficientBound<complex_dbl>();
237 }
238
239
247 void SetPhiPsiFromBounds()
248 {
249 Phi = degree_bound*(degree_bound-NumErrorT(1))*coefficient_bound;
250 Psi = degree_bound*coefficient_bound; //Psi from the AMP paper.
251 }
252
254 void SetAMPConfigFrom(System const& sys)
255 {
256 SetBoundsAndEpsilonFrom(sys);
257 SetPhiPsiFromBounds();
258 }
259
268 AdaptiveMultiplePrecisionConfig() : coefficient_bound(1000), degree_bound(5), epsilon(1), safety_digits_1(1), safety_digits_2(1), maximum_precision(300)
269 {
270 SetPhiPsiFromBounds();
271 }
