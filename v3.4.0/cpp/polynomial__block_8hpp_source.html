156 void Precision(unsigned new_precision) const
157 {
158 // The SLP (its per-thread Memory) is the sole evaluator and carries its own precision; the
159 // function / derivative / variable nodes are no longer evaluated during tracking, so their
160 // precision is vestigial and left untouched (keeps the shared node DAG read-only).
161 slp_.precision(new_precision);
162 precision_ = new_precision;
163 }

226 void Differentiate() const
227 {

235 void SimplifyFunctions() const
236 {
237 // Intentionally inert on the top-level functions, preserving long-standing behavior:
238 // when functions were Function-wrapped, Handle::Simplified() returned self (an opaque
239 // boundary), so top-level simplification never happened. Now that functions are bare
240 // roots, actually simplifying them here would be a behavior change, deferred to its own step.
241 Invalidate();
242 }

247 void SimplifyDerivatives() const
248 {
249 using bertini::Simplify;
250 for (auto& n : space_derivatives_) n = Simplify(n);
251 for (auto& n : time_derivatives_) n = Simplify(n);
252 }

258 void Reintern(node::ReinternMemo& memo)
259 {
260 for (auto& f : functions_)
261 f = node::Reintern(f, memo);
262 for (auto& c : constant_subfunctions_)
263 c = std::static_pointer_cast<node::NamedExpression>(node::Reintern(c, memo));
264 for (auto& v : variables_)
265 v = std::static_pointer_cast<node::Variable>(node::Reintern(v, memo));
266 if (path_variable_)
267 path_variable_ = std::static_pointer_cast<node::Variable>(node::Reintern(path_variable_, memo));
268 space_derivatives_.clear();
269 time_derivatives_.clear();
270 Invalidate();
271 }

273private:
274 void EnsureDifferentiated() const { if (!is_differentiated_) Differentiate(); }

276 template <typename T>
277 void SetValues(Vec<T> const& vars, T const& path_value) const
278 {
279 slp_.SetVariableValues(vars);
280 if (path_variable_) slp_.template SetPathVariable<T>(path_value);
281 }
282
283 void DifferentiateUsingDerivatives() const
284 {
285 const size_t n = functions_.size();
286 space_derivatives_.resize(n * variables_.size());
287 for (size_t jj = 0; jj < variables_.size(); ++jj)
288 for (size_t ii = 0; ii < n; ++ii)
289 space_derivatives_[ii + jj * n] = functions_[ii]->Differentiate(variables_[jj]);
290 if (path_variable_)
291 {
292 time_derivatives_.resize(n);
293 for (size_t ii = 0; ii < n; ++ii)
294 time_derivatives_[ii] = functions_[ii]->Differentiate(path_variable_);
295 }
296 }
