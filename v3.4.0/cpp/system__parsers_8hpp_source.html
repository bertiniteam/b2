46 template <typename Iterator>
47 static bool parse(Iterator first, Iterator last, System& sys)
48 {
49 using boost::spirit::qi::double_;
50 using boost::spirit::qi::_1;
51 using boost::spirit::qi::phrase_parse;
52 using boost::spirit::ascii::space;
53 using boost::phoenix::ref;
54
55 SystemParser<Iterator> S;
56
57 System s{};
58 bool r = phrase_parse(first, last,
59 S,
60 space,
61 s);
62
63 if (!r || first != last) // fail if we did not get a full match
64 return false;
65
66 // every definition has now filled its function box: emit the bare entries.
67 S.EmitDeclaredFunctions(s);
68 sys = s;
69 return r;
70 }

76 inline
77 System::System(std::string const& input)
78 {
79 System sys;
80
81 parsing::classic::SystemParser<std::string::const_iterator> S;
82
83 // Treat the input as UTF-8; drop a leading BOM so it is not seen as a
84 // stray leading character by the grammar.
85 std::string cleaned = input;
86 parsing::classic::StripUTF8BOM(cleaned);
87
88 std::string::const_iterator iter = cleaned.begin();
89 std::string::const_iterator end = cleaned.end();
90
91 bool s = phrase_parse(iter, end, S,boost::spirit::ascii::space, sys);
92
93 if (!s || iter!=end)
94 {
95 std::string remaining(iter, end);
96 if (remaining.size() > 60)
97 remaining = remaining.substr(0, 60) + "...";
98 if (remaining.empty())
99 remaining = "<end of input>";
100 throw std::runtime_error(
101 "[SystemParser] parser did not consume entire input; "
102 "unparsed remainder: \"" + remaining + "\"");
103 }
104
105 // every definition has now filled its function box: emit the bare entries.
106 S.EmitDeclaredFunctions(sys);
107
108 using std::swap;
109 swap(sys,*this);
110 }
