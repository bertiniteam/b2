125inline
126unsigned EnsureAtUniformPrecision(TimeCont<complex_mp> & times, SampCont<complex_mp> & samples)
127{
128 auto def_prec = ThreadPrecision();
129 if (std::any_of(begin(times),end(times),[=](auto const& p){return Precision(p)!=def_prec;})
130 ||
131 std::any_of(begin(samples),end(samples),[=](auto const& p){return Precision(p)!=def_prec;}))
132 {
133 auto max_precision = max(MaxPrecision(samples), MaxPrecision(times));
134
135 SetThreadPrecision(max_precision);
136 SetPrecision(times, max_precision);
137 SetPrecision(samples, max_precision);
138 return max_precision;
139 }
140 return def_prec;
141
142}
