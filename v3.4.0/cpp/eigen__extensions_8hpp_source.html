80 // ThreadPrecision (thread-local), not DefaultPrecision (global): these
81 // tolerances are consumed inside tracking (LU solves, norms, convergence
82 // checks), which may run on std::thread workers whose precision is set
83 // thread-locally. On the main thread the two agree.
84 // epsilon() and dummy_precision() are pow(10, -precision): expensive (transcendental,
85 // heap-allocating) yet a pure function of the thread precision. They are called per
86 // LU-diagonal element, per Newton iteration, per predictor stage, per step, so recomputing
87 // them dominated the multiprecision path. Memoize per thread, keyed on the current
88 // precision -- the cached value is bit-identical to a fresh pow at that precision.
89 inline static Real dummy_precision()
90 {
91 using bertini::ThreadPrecision;
92 thread_local unsigned cached_prec = 0;
93 thread_local Real cached_val;
94 const unsigned p = ThreadPrecision();
95 if (p != cached_prec)
96 {
97 cached_val = pow(mpfr_real(10), -int(p-3));
98 cached_prec = p;
99 }
100 return cached_val;
101 }
102
103 inline static Real epsilon()
104 {
105 using bertini::ThreadPrecision;
106 thread_local unsigned cached_prec = 0;
107 thread_local Real cached_val;
108 const unsigned p = ThreadPrecision();
109 if (p != cached_prec)
110 {
111 cached_val = pow(mpfr_real(10), -int(p));
112 cached_prec = p;
113 }
114 return cached_val;
115 }

403 template<typename T>
404 inline
405 bool IsSmallValue(T const& testme)
406 {
407 // |testme| <= eps*100 <=> abs2(testme) <= (eps*100)^2. Comparing squared magnitudes uses
408 // abs2 (re^2+im^2 for complex; see abs2_impl<complex_mp>) and avoids the allocating, sqrt-based
409 // std::abs/hypot on the multiprecision complex type. eps*100 is non-negative, so the squared
410 // comparison is exactly equivalent.
411 using Real = typename Eigen::NumTraits<T>::Real;
412 const Real thresh = Eigen::NumTraits<T>::epsilon() * 100;
413 return Eigen::numext::abs2(testme) <= thresh * thresh;
414 }

450 template<typename T>
451 inline
452 bool IsLargeChange(T const& numerator, T const& denomenator)
453 {
454 static_assert(!Eigen::NumTraits<T>::IsInteger, "IsLargeChange cannot be used safely on non-integral types");
455 // |num/den| >= 1/dummy <=> |num|*dummy >= |den| <=> abs2(num)*dummy^2 >= abs2(den).
456 // The squared form drops BOTH the complex division (num/den) and the sqrt-based abs/hypot --
457 // each of which allocates multiprecision temporaries -- and is exact since all quantities are
458 // non-negative. (In its hot caller, LUPartialPivotDecompositionSuccessful, the denominator has
459 // already passed IsSmallValue, so it is non-tiny here.)
460 using Real = typename Eigen::NumTraits<T>::Real;
461 const Real d = Eigen::NumTraits<T>::dummy_precision();
462 return Eigen::numext::abs2(numerator) * (d * d) >= Eigen::numext::abs2(denomenator);
463 }

482 template <typename Derived>
483 MatrixSuccessCode LUPartialPivotDecompositionSuccessful(Eigen::MatrixBase<Derived> const& LU)
484 {
485 #ifndef BERTINI_DISABLE_ASSERTS
486 assert(LU.rows()==LU.cols() && "non-square matrix in LUPartialPivotDecompositionSuccessful");
487 assert(LU.rows()>0 && "empty matrix in LUPartialPivotDecompositionSuccessful");
488 #endif
489
490 // this loop won't test entry (0,0). it's tested separately after.
491 for (Eigen::Index ii = LU.rows()-1; ii > 0; ii--)
492 {
493 if (IsSmallValue(LU(ii,ii)))
494 {
495 return MatrixSuccessCode::SmallValue;
496 }
497
498 if (IsLargeChange(LU(ii-1,ii-1),LU(ii,ii)))
499 {
500 return MatrixSuccessCode::LargeChange;
501 }
502 }
503
504 // this line is the reason for the above assert on non-empty matrix.
505 if (IsSmallValue(LU(0,0)))
506 {
507 return MatrixSuccessCode::SmallValue;
508 }
509
510 return MatrixSuccessCode::Success;
511 }

516 template <typename NumberType>
517 Eigen::Matrix<NumberType, Eigen::Dynamic, Eigen::Dynamic> KahanMatrix(unsigned int mat_size, NumberType c)
518 {
519 using std::sqrt;
520 NumberType s, scale(1.0);
521 s = sqrt( (NumberType(1.0)-c) * (NumberType(1.0)+c) );
522
523
524 Eigen::Matrix<NumberType, Eigen::Dynamic, Eigen::Dynamic> A(mat_size,mat_size);
525
526
527 for (unsigned int ii=0; ii<mat_size; ii++) {
528 for (unsigned int jj=0; jj<ii; jj++) {
529 A(ii,jj) = NumberType(0.0);
530 }
531 for (unsigned int jj=ii; jj<mat_size; jj++) {
532 A(ii,jj) = -c * NumberType(1.0);
533 }
534 }
535
536
537 for (unsigned int ii=0; ii<mat_size; ii++) {
538 A(ii,ii) += NumberType(1)+c;
539 }
540
541 for (unsigned int jj=0; jj<mat_size; jj++){
542 for (unsigned int kk=0;kk<mat_size;kk++){
543 A(jj,kk) *= scale;
544 }
545 scale *= s;
546 }
547
548 for (unsigned int jj=0;jj<mat_size;jj++){
549 for (unsigned int kk=0;kk<mat_size;kk++){
550 A(kk,jj)/= NumberType(jj) + NumberType(1);
551 }
552 }
553 return A;
554 }

590 template <typename NumberType>
591 inline
592 Mat<NumberType> RandomConjugateOrthonormalMatrix(unsigned int rows, unsigned int cols)
593 {
594 using std::max;
595 const unsigned int dim = max(rows, cols); // generate square, then truncate -- Bertini 1's recipe
596 Mat<NumberType> seed = RandomOfUnits<NumberType>(dim, dim);
597 Eigen::HouseholderQR<Mat<NumberType> > qr(seed);
598 Mat<NumberType> Q = qr.householderQ() * Mat<NumberType>::Identity(dim, dim);
599 return Q.topLeftCorner(rows, cols);
600 }
