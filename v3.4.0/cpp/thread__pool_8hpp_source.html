108template<typename T>
109class ThreadSafeQueue
110{
111 std::deque<T> queue_;
112 std::mutex mutex_;
113 std::condition_variable cv_;
114
115public:
117 void push(T item)
118 {
119 {
120 std::lock_guard<std::mutex> lock(mutex_);
121 queue_.push_back(std::move(item));
122 }
123 cv_.notify_one();
124 }
125
127 T pop()
128 {
129 std::unique_lock<std::mutex> lock(mutex_);
130 cv_.wait(lock, [this]{ return !queue_.empty(); });
131 T item = std::move(queue_.front());
132 queue_.pop_front();
133 return item;
134 }
135
137 std::optional<T> try_pop()
138 {
139 std::lock_guard<std::mutex> lock(mutex_);
140 if (queue_.empty())
141 return std::nullopt;
142 T item = std::move(queue_.front());
143 queue_.pop_front();
144 return item;
145 }
146};

165template<typename TaskT, typename ResultT, typename StateFactory, typename TrackFn>
166class WorkerThreadPool
167{
168 using StateT = std::invoke_result_t<StateFactory>;
169 using WorkItem = std::variant<TaskT, PoolShutdownSentinel>;
170
171 ThreadSafeQueue<WorkItem> task_queue_;
172 ThreadSafeQueue<ResultT> result_queue_;
173 std::vector<std::thread> threads_;
174 int n_threads_;
175
176public:
181 WorkerThreadPool(int n_threads, StateFactory state_factory, TrackFn track_fn)
182 : n_threads_(n_threads)
183 {
184 threads_.reserve(static_cast<size_t>(n_threads));
185 for (int i = 0; i < n_threads; ++i)
186 {
187 threads_.emplace_back([this, state_factory, track_fn]() mutable
188 {
189 // Build thread-local state once (System copy + Tracker copy).
190 StateT state = state_factory();
191
192 while (true)
193 {
194 WorkItem item = task_queue_.pop();
195
196 if (std::holds_alternative<PoolShutdownSentinel>(item))
197 break;
198
199 TaskT const& task = std::get<TaskT>(item);
200 ResultT result = track_fn(state, task);
201 result_queue_.push(std::move(result));
202 }
203 });
204 }
205 }
206
208 void submit(TaskT task)
209 {
210 task_queue_.push(WorkItem{std::move(task)});
211 }
212
214 ResultT collect()
215 {
216 return result_queue_.pop();
217 }
218
220 std::optional<ResultT> try_collect()
221 {
222 return result_queue_.try_pop();
223 }
224
226 void shutdown()
227 {
228 for (int i = 0; i < n_threads_; ++i)
229 task_queue_.push(WorkItem{PoolShutdownSentinel{}});
230 for (auto& t : threads_)
231 t.join();
232 threads_.clear();
233 }
234};
