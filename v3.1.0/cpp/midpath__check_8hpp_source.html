164 template <typename StartSystemT>
165 bool Check(BoundaryData const& boundary_data, StartSystemT const& start_system)
166 {
167 // Each Check is independent: the algorithm calls this repeatedly (once before any
168 // resolve, then again after each MidpathResolve). Reset the accumulated state so a
169 // later call reports *this* call's crossings, not the union with prior calls.
170 // Without this, passed_ (only ever set false below) would stay false forever once
171 // any crossing was seen, and crossed_paths_ would grow without bound -- the resolve
172 // loop in EGBoundaryAction would then never see a clean pass and would re-track
173 // stale indices.
174 passed_ = true;
175 crossed_paths_.clear();
176
177 for (PathIndT ii = 0; ii < boundary_data.size(); ++ii)
178 {
179 if ( boundary_data[ii].success_code != SuccessCode::Success)
180 continue;
181
182 const Vec<ComplexT>& solution_ii = boundary_data[ii].path_point;
183 const auto start_ii = start_system.template StartPoint<ComplexT>(ii);
184
185 for (PathIndT jj = ii+1; jj < boundary_data.size(); ++jj)
186 {
187 if ( boundary_data[jj].success_code != SuccessCode::Success)
188 continue;
189
190 const Vec<ComplexT>& solution_jj = boundary_data[jj].path_point;
191 const Vec<ComplexT> diff_sol = solution_ii - solution_jj;
192
193 if ((diff_sol.template lpNorm<Eigen::Infinity>()/solution_ii.template lpNorm<Eigen::Infinity>()) < SamePointTol())
194 {
195 bool i_already_stored = false;
196 bool j_already_stored = false;
197 // Check if start points are the same
198
199 const auto start_jj = start_system.template StartPoint<ComplexT>(jj);
200 auto diff_start = start_ii - start_jj;
201 // same_start is true when the two paths *began at the same start point*
202 // (their start points differ by less than the tolerance). That governs
203 // whether this coincidence is a genuine path crossing worth re-tracking:
204 // CrossedPath sets rerun_ = !same_start, so two *distinct* starts that
205 // have collided (same_start == false) get re-tracked, while two paths
206 // that legitimately share a start point do not. NOTE: this comparison
207 // was historically inverted (>), which flipped rerun and meant genuine
208 // crossings were never re-tracked.
209 bool same_start = (diff_start.template lpNorm<Eigen::Infinity>() < SamePointTol());
210
211
212 // Check if path has already been stored in crossed_paths_
213 for(auto& v : crossed_paths_)
214 {
215 if(v.index() == ii)
216 {
217 v.crossed_with(std::make_pair(jj,same_start));
218 i_already_stored = true;
219 v.rerun(same_start);
220 }
221 if(v.index() == jj)
222 {
223 v.crossed_with(std::make_pair(ii,same_start));
224 j_already_stored = true;
225 v.rerun(same_start);
226 }
227 }
228
229 // If not already stored, create CrossedPath object and add to crossed_paths_
230 if(!i_already_stored)
231 {
232 CrossedPath tempPath(ii, jj, same_start);
233 crossed_paths_.push_back(tempPath);
234 }
235 if(!j_already_stored)
236 {
237 CrossedPath tempPath(jj, ii, same_start);
238 crossed_paths_.push_back(tempPath);
239 }
240
241 passed_ = false;
242
243 }
244 }
245 }
246 return passed_;
247 };
