63template<typename ComplexT>
64 Vec<ComplexT> HermiteInterpolateAndSolve(ComplexT const& target_time, const unsigned int num_sample_points, const TimeCont<ComplexT> & times, const SampCont<ComplexT> & samples, const SampCont<ComplexT> & derivatives, ContStart shift_from = ContStart::Back)
65{
66 assert((times.size() >= num_sample_points) && "must have sufficient number of sample times");
67 assert((samples.size() >= num_sample_points) && "must have sufficient number of sample points");
68 assert((derivatives.size() >= num_sample_points) && "must have sufficient number of derivatives");
69
70
71 unsigned num_t, num_s, num_d;
72 if (shift_from == ContStart::Back)
73 {
74 num_t = static_cast<unsigned>(times.size()-1);
75 num_s = static_cast<unsigned>(samples.size()-1);
76 num_d = static_cast<unsigned>(derivatives.size()-1);
77 }
78 else
79 {
80 num_t = num_s = num_d = num_sample_points-1;
81 }
82
83 Mat< Vec<ComplexT> > space_differences(2*num_sample_points,2*num_sample_points);
84 Vec<ComplexT> time_differences(2*num_sample_points);
85
86
87 for(unsigned int ii=0; ii<num_sample_points; ++ii)
88 {
89 space_differences(2*ii,0) = samples[ num_s-ii]; /* F[2*i][0] = samples[i]; */
90 space_differences(2*ii+1,0) = samples[ num_s-ii]; /* F[2*i+1][0] = samples[i]; */
91 space_differences(2*ii+1,1) = derivatives[num_d-ii]; /* F[2*i+1][1] = derivatives[i]; */
92 time_differences(2*ii) = times[ num_t-ii]; /* z[2*i] = times[i]; */
93 time_differences(2*ii+1) = times[ num_t-ii]; /* z[2*i+1] = times[i]; */
94 }
95
96 //Add first round of finite differences to fill out rest of matrix.
97 for(unsigned int ii=1; ii< num_sample_points; ++ii)
98 {
99 space_differences(2*ii,1) = (space_differences(2*ii,0) - space_differences(2*ii-1,0)) / (time_differences(2*ii) - time_differences(2*ii-1));
100
101 }
102
103 //Filling out finite difference matrix to get the diagonal for hermite interpolation polyonomial.
104 for(unsigned int ii=2; ii < 2*num_sample_points; ++ii)
105 {
106 for(unsigned int jj=2; jj <=ii; ++jj)
107 {
108 space_differences(ii,jj) =
109 (space_differences(ii,jj-1) - space_differences(ii-1,jj-1))
110 /
111 (time_differences(ii) - time_differences(ii-jj));
112 }
113 }
114
115 //Start of Result from Hermite polynomial, this is using the diagonal of the
116 //finite difference matrix.
117 Vec<ComplexT> Result = space_differences(2*num_sample_points - 1,2*num_sample_points - 1);
118
119
120 //This builds the hermite polynomial from the highest term down.
121 //As we multiply the previous result we will construct the highest term down to the last term.
122 for (unsigned ii=num_sample_points-1; ii >= 1; --ii)
123 {
124 Result = ((Result*(target_time - time_differences(ii)) + space_differences(2*ii, 2*ii)) * (target_time - time_differences(ii-1)) + space_differences(2*ii-1, 2*ii-1)).eval();
125 }
126
127 // Last term in hermite polynomial.
128 return (Result * (target_time - time_differences(0)) + space_differences(0,0)).eval();
129} //re: HermiteInterpolateAndSolve
