146 inline void DefaultPrecision(unsigned prec)
147 {
148 real_mp::default_precision(prec);
149 complex_mp::default_precision(prec);
150 // Boost.Multiprecision >= 1.87 reads thread_default_precision when
151 // default-constructing mpfr temporaries (including those created by
152 // boost::python const& extractors). If left at 0, mpfr_init2 aborts.
153 // Set both so that calls to default_precision() align thread-local
154 // with the static default. See commit 3111255b for original context.
155 real_mp::thread_default_precision(prec);
156 complex_mp::thread_default_precision(prec);
157#ifdef BMP_EXPRESSION_TEMPLATES
158 // With ET on, the default global policy for mpc_complex_backend is preserve_related_precision
159 // and for mpfr_float_backend it is preserve_target_precision. Both differ from what we want.
160 // Setting preserve_related_precision per-thread mirrors the mpc_complex_backend global default
161 // and ensures copy/construction semantics match the et_off behavior. Specifically:
162 // - mpc_complex copy ctor uses preserve_related_precision() (>= 3) to preserve source precision
163 // - assign_components_set_precision uses preserve_component_precision() (>= 2) to resize
164 // a complex from real components at higher-than-default precision
165 // - mpc_complex = real_mp uses preserve_component_precision() (>= 2) to resize
166 // preserve_related_precision satisfies all three thresholds.
167 // For real_mp, preserve_related_precision also enables source-precision-preserving copies.
168 real_mp::thread_default_variable_precision_options(
169 bmp::variable_precision_options::preserve_related_precision);
170 complex_mp::thread_default_variable_precision_options(
171 bmp::variable_precision_options::preserve_related_precision);
172#endif
173 }
174
175 // Sets thread-local precision only — does NOT write the global default_precision.
176 // Safe to call concurrently from multiple std::thread workers tracking at different
177 // precisions. Use instead of DefaultPrecision() inside per-thread tracking loops.
179 inline void SetThreadPrecision(unsigned prec)
180 {
181 real_mp::thread_default_precision(prec);
182 complex_mp::thread_default_precision(prec);
183 }

186 inline unsigned ThreadPrecision()
187 {
188 return static_cast<unsigned>(real_mp::thread_default_precision());
189 }
