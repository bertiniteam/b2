252 SuccessCode TrackPath(Vec<ComplexT> & solution_at_endtime,
253 ComplexT const& start_time, ComplexT const& endtime,
254 Vec<ComplexT> const& start_point
255 ) const
256 {
257 if (start_point.size()!=static_cast<Eigen::Index>(GetSystem().NumVariables()))
258 throw std::runtime_error("start point size must match the number of variables in the system to be tracked");
259
260
261
262 SuccessCode initialization_code = TrackerLoopInitialization(start_time, endtime, start_point);
263 if (initialization_code!=SuccessCode::Success)
264 {
265 PostTrackCleanup();
266 return initialization_code;
267 }
268
269 NotifyObservers(TrackingStarted<typename TrackerTraits<D>::EventEmitterType>(static_cast<typename TrackerTraits<D>::EventEmitterType const&>(*this)));
270
271 // as precondition to this while loop, the correct container, either complex_dbl or mpfr, must have the correct data.
272 while (!IsSymmRelDiffSmall(current_time_,endtime_, Eigen::NumTraits<ComplexT>::epsilon()))
273 {
274 SuccessCode pre_iteration_code = PreIterationCheck();
275 if (pre_iteration_code!=SuccessCode::Success)
276 {
277 PostTrackCleanup();
278 return pre_iteration_code;
279 }
280
281 using std::abs;
282 // compute the next delta_t
283 if (abs(endtime_-current_time_) < abs(current_stepsize_))
284 delta_t_ = endtime_-current_time_;
285 else
286 delta_t_ = current_stepsize_ * (endtime_ - current_time_)/abs(endtime_ - current_time_);
287
288 step_success_code_ = TrackerIteration();
289
290 if (infinite_path_truncation_ && (CheckGoingToInfinity()==SuccessCode::GoingToInfinity))
291 {
292 OnInfiniteTruncation();
293 PostTrackCleanup();
294 return SuccessCode::GoingToInfinity;
295 }
296 else if (step_success_code_==SuccessCode::Success)
297 OnStepSuccess();
298 else
299 OnStepFail();
300
301 }// re: while
302
303
304 CopySolution(solution_at_endtime);
305 PostTrackCleanup();
306 return SuccessCode::Success;
307 }
