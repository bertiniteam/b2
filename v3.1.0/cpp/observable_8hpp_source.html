249 void EnqueueOrApplyAdd(Watcher w) const
250 {
251 if (dispatch_depth_ > 0)
252 {
253 PendingOp op;
254 op.kind = PendingOp::Add;
255 op.watcher = std::move(w); // carries `owned`, so the observer stays alive until drained
256 pending_ops_.push_back(std::move(op));
257 }
258 else
259 AddWatcher(std::move(w));
260 }

262 void AddWatcher(Watcher const& w) const
263 {
264 auto present_in = [&](ObserverList const& c) {
265 return std::find_if(c.begin(), c.end(),
266 [&](Watcher const& x){ return x.raw == w.raw; }) != c.end();
267 };
268
269 auto types = w.raw->SubscribedEventTypes();
270 if (types.empty())
271 {
272 if (!present_in(untyped_watchers_))
273 untyped_watchers_.push_back(w);
274 }
275 else
276 {
277 for (auto& ti : types)
278 {
279 auto& bucket = typed_watchers_[ti];
280 if (std::find_if(bucket.begin(), bucket.end(),
281 [&](Watcher const& x){ return x.raw == w.raw; }) == bucket.end())
282 bucket.push_back(w);
283 }
284 }
285 }
286
287 void RemoveWatcher(AnyObserver* who) const
288 {
289 auto erase_from = [&](ObserverList& c) {
290 c.erase(std::remove_if(c.begin(), c.end(),
291 [&](Watcher const& x){ return x.raw == who; }),
292 c.end());
293 };
294 erase_from(untyped_watchers_);
295 for (auto& [ti, bucket] : typed_watchers_)
296 erase_from(bucket);
297 }

299 void DrainPendingOps() const
300 {
301 // Apply in request order so that, within one notification, a remove
302 // followed by a re-add (or vice versa) lands on the intended state.
303 for (auto& op : pending_ops_)
304 {
305 if (op.kind == PendingOp::Add)
306 AddWatcher(op.watcher);
307 else
308 RemoveWatcher(op.watcher.raw);
309 }
310 pending_ops_.clear();
311 }

313 void DispatchEvent(AnyEvent const& e) const
314 {
315 std::lock_guard<std::recursive_mutex> lock(notify_mutex_);
316 ++dispatch_depth_;
317
318 auto run = [&](ObserverList& bucket) {
319 for (auto& w : bucket)
320 {
321 // w.raw stays valid for the whole call: owning entries hold a
322 // shared_ptr, non-owning ones are the caller's responsibility.
323 if (w.raw->Observe(e) == ObserveResult::Unsubscribe)
324 {
325 PendingOp op;
326 op.kind = PendingOp::Remove;
327 op.watcher.raw = w.raw;
328 pending_ops_.push_back(std::move(op));
329 }
330 }
331 };
332
333 auto it = typed_watchers_.find(std::type_index(typeid(e)));
334 if (it != typed_watchers_.end())
335 run(it->second);
336 run(untyped_watchers_);
337
338 --dispatch_depth_;
339 if (dispatch_depth_ == 0)
340 DrainPendingOps();
341 }
342
343 mutable std::unordered_map<std::type_index, ObserverList> typed_watchers_;
344 mutable ObserverList untyped_watchers_;
345
346 mutable unsigned dispatch_depth_ = 0;
347 mutable std::vector<PendingOp> pending_ops_;
348
349 // Serializes notification and watcher-list mutation so the same observable can be safely
350 // notified from several worker threads at once (the MPI-less threaded solve fires per-path
351 // events from each tracking thread onto one shared solver). Recursive because an observer's
352 // Observe() may itself Add/RemoveObserver on this same observable during a dispatch (the
353 // "meta-observer" pattern) -- that re-entrant call re-locks on the same thread and is routed
354 // through the deferred pending_ops_ path via dispatch_depth_. Held across Observe() calls,
355 // so observer callbacks are serialized; the heavy numerical work runs outside this lock.
356 // Single-threaded callers pay one uncontended recursive lock per event -- negligible.
357 mutable std::recursive_mutex notify_mutex_;
358 };
