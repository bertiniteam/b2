49 class FirstPrecisionRecorder
50 : public TypedObserver< FirstPrecisionRecorder<TrackerT>, TrackerT,
51 TrackingStarted<typename TrackerTraits<TrackerT>::EventEmitterType>,
52 PrecisionChanged<typename TrackerTraits<TrackerT>::EventEmitterType> >
53 { BOOST_TYPE_INDEX_REGISTER_CLASS
54
55 using EmitterT = typename TrackerTraits<TrackerT>::EventEmitterType;
56
57 public:
58
60 ObserveResult OnEvent(TrackingStarted<EmitterT> const& e)
61 {
62 precision_increased_ = false;
63 starting_precision_ = e.Get().CurrentPrecision();
64 return ObserveResult::KeepObserving;
65 }
66
68 ObserveResult OnEvent(PrecisionChanged<EmitterT> const& e)
69 {
70 auto next = e.Next();
71 if (next > e.Previous())
72 {
73 precision_increased_ = true;
74 next_precision_ = next;
75 time_of_first_increase_ = e.Get().CurrentTime();
76 // done: ask to be dropped instead of mutating the list mid-dispatch.
77 return ObserveResult::Unsubscribe;
78 }
79 return ObserveResult::KeepObserving;
80 }
81
83 unsigned StartPrecision() const
84 {
85 return starting_precision_;
86 }
87
89 unsigned NextPrecision() const
90 {
91 return next_precision_;
92 }
93
95 bool DidPrecisionIncrease() const
96 {
97 return precision_increased_;
98 }
99
101 typename TrackerTraits<TrackerT>::BaseComplexT TimeOfIncrease() const
102 {
103 return time_of_first_increase_;
104 }
105
106 virtual ~FirstPrecisionRecorder() = default;
107
108 private:
109
110 unsigned starting_precision_;
111 unsigned next_precision_;
112 bool precision_increased_;
113 typename TrackerTraits<TrackerT>::BaseComplexT time_of_first_increase_;
114 };

120 : public TypedObserver< MinMaxPrecisionRecorder<TrackerT>, TrackerT,
121 TrackingStarted<typename TrackerTraits<TrackerT>::EventEmitterType>,
122 PrecisionChanged<typename TrackerTraits<TrackerT>::EventEmitterType> >
123 { BOOST_TYPE_INDEX_REGISTER_CLASS
124
125 using EmitterT = typename TrackerTraits<TrackerT>::EventEmitterType;
126
127 public:
128
130 ObserveResult OnEvent(TrackingStarted<EmitterT> const& e)
131 {
132 min_precision_ = e.Get().CurrentPrecision();
133 max_precision_ = e.Get().CurrentPrecision();
134 return ObserveResult::KeepObserving;
135 }
136
138 ObserveResult OnEvent(PrecisionChanged<EmitterT> const& e)
139 {
140 auto next_precision = e.Next();
141 if (next_precision < min_precision_)
142 min_precision_ = next_precision;
143 if (next_precision > max_precision_)
144 max_precision_ = next_precision;
145 return ObserveResult::KeepObserving;
146 }
147
149 unsigned MinPrecision() const
150 {
151 return min_precision_;
152 }
153
155 unsigned MaxPrecision() const
156 {
157 return max_precision_;
158 }
159
161 void MinPrecision(unsigned m)
162 { min_precision_ = m;}
163
165 void MaxPrecision(unsigned m)
166 { max_precision_ = m;}
167
168 virtual ~MinMaxPrecisionRecorder() = default;
169
170 private:
171
172 unsigned min_precision_ = std::numeric_limits<unsigned>::max();
173 unsigned max_precision_ = 0;
174 };

245 template<class TrackerT>
246 class GoryDetailLogger : public Observer<TrackerT>
247 { BOOST_TYPE_INDEX_REGISTER_CLASS
248 public:
249
250 using EmitterT = typename TrackerTraits<TrackerT>::EventEmitterType;
251
252 virtual ~GoryDetailLogger() = default;
253
255 virtual ObserveResult Observe(AnyEvent const& e) override
256 {
257
258
259 if (auto p = dynamic_cast<const Initializing<EmitterT,complex_dbl>*>(&e))
260 {
261 BOOST_LOG_TRIVIAL(severity_level::debug) << std::setprecision(static_cast<int>(p->Get().GetSystem().precision()))
262 << "initializing in double, tracking path\nfrom\tt = "
263 << p->StartTime() << "\nto\tt = " << p->EndTime()
264 << "\n from\tx = \n" << p->StartPoint()
265 << "\n tracking system " << p->Get().GetSystem() << "\n\n";
266 }
267 else if (auto p = dynamic_cast<const Initializing<EmitterT,complex_mp>*>(&e))
268 {
269 BOOST_LOG_TRIVIAL(severity_level::debug) << std::setprecision(static_cast<int>(p->Get().GetSystem().precision()))
270 << "initializing in multiprecision, tracking path\nfrom\tt = " << p->StartTime() << "\nto\tt = " << p->EndTime() << "\n from\tx = \n" << p->StartPoint()
271 << "\n tracking system " << p->Get().GetSystem() << "\n\n";
272 }
273
274 else if(auto p = dynamic_cast<const TrackingEnded<EmitterT>*>(&e))
275 BOOST_LOG_TRIVIAL(severity_level::trace) << "tracking ended";
276
277 else if (auto p = dynamic_cast<const NewStep<EmitterT>*>(&e))
278 {
279 auto& t = p->Get();
280 BOOST_LOG_TRIVIAL(severity_level::trace) << "Tracker iteration " << t.NumTotalStepsTaken() << "\ncurrent precision: " << t.CurrentPrecision();
281
282
283 BOOST_LOG_TRIVIAL(severity_level::trace) << std::setprecision(static_cast<int>(t.CurrentPrecision()))
284 << "t = " << t.CurrentTime()
285 << "\ncurrent stepsize: " << t.CurrentStepsize()
286 << "\ndelta_t = " << t.DeltaT()
287 << "\ncurrent x size = " << t.CurrentPoint().size()
288 << "\ncurrent x = " << t.CurrentPoint();
289 }
290
291
292
293 else if (auto p = dynamic_cast<const SingularStartPoint<EmitterT>*>(&e))
294 BOOST_LOG_TRIVIAL(severity_level::trace) << "singular start point";
295 else if (auto p = dynamic_cast<const InfinitePathTruncation<EmitterT>*>(&e))
296 BOOST_LOG_TRIVIAL(severity_level::trace) << "tracker iteration indicated going to infinity, truncated path";
297
298
299
300
301 else if (auto p = dynamic_cast<const SuccessfulStep<EmitterT>*>(&e))
302 {
303 BOOST_LOG_TRIVIAL(severity_level::trace) << "tracker iteration successful\n\n\n";
304 }
305
306 else if (auto p = dynamic_cast<const FailedStep<EmitterT>*>(&e))
307 {
308 BOOST_LOG_TRIVIAL(severity_level::trace) << "tracker iteration unsuccessful\n\n\n";
309 }
310
311
312
313
314
315
316 else if (auto p = dynamic_cast<const SuccessfulPredict<EmitterT,complex_mp>*>(&e))
317 {
318 BOOST_LOG_TRIVIAL(severity_level::trace) << std::setprecision(static_cast<int>(Precision(p->ResultingPoint()))) << "prediction successful (complex_mp), result:\n" << p->ResultingPoint();
319 }
320 else if (auto p = dynamic_cast<const SuccessfulPredict<EmitterT,complex_dbl>*>(&e))
321 {
322 BOOST_LOG_TRIVIAL(severity_level::trace) << std::setprecision(static_cast<int>(Precision(p->ResultingPoint()))) << "prediction successful (complex_dbl), result:\n" << p->ResultingPoint();
323 }
324
325 else if (auto p = dynamic_cast<const SuccessfulCorrect<EmitterT,complex_mp>*>(&e))
326 {
327 BOOST_LOG_TRIVIAL(severity_level::trace) << std::setprecision(static_cast<int>(Precision(p->ResultingPoint()))) << "correction successful (complex_mp), result:\n" << p->ResultingPoint();
328 }
329 else if (auto p = dynamic_cast<const SuccessfulCorrect<EmitterT,complex_dbl>*>(&e))
330 {
331 BOOST_LOG_TRIVIAL(severity_level::trace) << std::setprecision(static_cast<int>(Precision(p->ResultingPoint()))) << "correction successful (complex_dbl), result:\n" << p->ResultingPoint();
332 }
333
334
335 else if (auto p = dynamic_cast<const PredictorHigherPrecisionNecessary<EmitterT>*>(&e))
336 BOOST_LOG_TRIVIAL(severity_level::trace) << "Predictor, higher precision necessary";
337 else if (auto p = dynamic_cast<const CorrectorHigherPrecisionNecessary<EmitterT>*>(&e))
338 BOOST_LOG_TRIVIAL(severity_level::trace) << "corrector, higher precision necessary";
339
340
341
342 else if (auto p = dynamic_cast<const CorrectorMatrixSolveFailure<EmitterT>*>(&e))
343 BOOST_LOG_TRIVIAL(severity_level::trace) << "corrector, matrix solve failure or failure to converge";
344 else if (auto p = dynamic_cast<const PredictorMatrixSolveFailure<EmitterT>*>(&e))
345 BOOST_LOG_TRIVIAL(severity_level::trace) << "predictor, matrix solve failure or failure to converge";
346 else if (auto p = dynamic_cast<const FirstStepPredictorMatrixSolveFailure<EmitterT>*>(&e))
347 BOOST_LOG_TRIVIAL(severity_level::trace) << "Predictor, matrix solve failure in initial solve of prediction";
348
349
350 else if (auto p = dynamic_cast<const PrecisionChanged<EmitterT>*>(&e))
351 BOOST_LOG_TRIVIAL(severity_level::debug) << "changing precision from " << p->Previous() << " to " << p->Next();
352
353 else
354 BOOST_LOG_TRIVIAL(severity_level::debug) << "unlogged event, of type: " << boost::typeindex::type_id_runtime(e).pretty_name();
355
356 return ObserveResult::KeepObserving;
357 }
