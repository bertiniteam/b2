102 Patch(Patch const& other)
103 {
104 variable_group_sizes_ = other.variable_group_sizes_;
105 precision_ = DefaultPrecision();
106
107 // a little shorthand unpacking the tuple
108 std::vector<Vec<complex_mp> >& coefficients_mpfr = std::get<std::vector<Vec<complex_mp> > >(this->coefficients_working_);
109 std::vector<Vec<complex_dbl> >& coefficients_dbl = std::get<std::vector<Vec<complex_dbl> > >(this->coefficients_working_);
110
111 coefficients_highest_precision_.resize(other.NumVariableGroups());
112 coefficients_mpfr.resize(variable_group_sizes_.size());
113 coefficients_dbl.resize(variable_group_sizes_.size());
114
115 for (unsigned ii(0); ii<other.NumVariableGroups(); ++ii)
116 {
117 auto curr_size = variable_group_sizes_[ii];
118
119 coefficients_highest_precision_[ii].resize(curr_size);
120 coefficients_dbl[ii].resize(curr_size);
121 coefficients_mpfr[ii].resize(curr_size);
122
123 for (unsigned jj(0); jj<curr_size; jj++)
124 {
125 coefficients_highest_precision_[ii](jj).precision(other.coefficients_highest_precision_[ii](jj).precision());
126
127 coefficients_highest_precision_[ii](jj) = other.coefficients_highest_precision_[ii](jj);
128
129 coefficients_dbl[ii](jj) = complex_dbl(coefficients_highest_precision_[ii](jj));
130 coefficients_mpfr[ii](jj) = complex_mp(coefficients_highest_precision_[ii](jj));
131
132 assert(coefficients_highest_precision_[ii](jj) == other.coefficients_highest_precision_[ii](jj));
133 }
134 }
135 }

153 Patch(std::vector<unsigned> const& sizes) : coefficients_highest_precision_(sizes.size()), variable_group_sizes_(sizes), precision_(DefaultPrecision())
154 {
155 using bertini::Precision;
156 using bertini::multiprecision::RandomComplex;
157
158 std::vector<Vec<complex_mp> >& coefficients_mpfr = std::get<std::vector<Vec<complex_mp> > >(coefficients_working_);
159 std::vector<Vec<complex_dbl> >& coefficients_dbl = std::get<std::vector<Vec<complex_dbl> > >(coefficients_working_);
160
161 coefficients_highest_precision_.resize(sizes.size());
162 coefficients_dbl.resize(sizes.size());
163 coefficients_mpfr.resize(sizes.size());
164
165 for (size_t ii=0; ii<sizes.size(); ++ii)
166 {
167 // this produces coefficients at maximum precision.
168 coefficients_highest_precision_[ii].resize(sizes[ii]);
169 for (unsigned jj=0; jj<sizes[ii]; ++jj)
170 {
171 // bounded-modulus draw (away from 0 and infinity), matching the linear-product /
172 // mhom / binomial start-system coefficients -- a heavy-tailed coefficient here scales
173 // the patch equation badly and feeds the same near-t=0 conditioning trouble.
174 multiprecision::RandomComplexBoundedModulusAssign(coefficients_highest_precision_[ii](jj), MaxPrecisionAllowed());
175 }
176
177 coefficients_dbl[ii].resize(sizes[ii]);
178 for (unsigned jj=0; jj<sizes[ii]; ++jj)
179 {
180 coefficients_dbl[ii](jj) = complex_dbl(coefficients_highest_precision_[ii](jj));
181 }
182
183 // assignment preserves precision of source.
184 // https://github.com/boostorg/multiprecision/issues/75
185 coefficients_mpfr[ii].resize(sizes[ii]);
186 for (unsigned jj=0; jj<sizes[ii]; ++jj)
187 {
188 coefficients_mpfr[ii](jj) = coefficients_highest_precision_[ii](jj);
189 coefficients_mpfr[ii](jj).precision(precision_);
190 }
191
192 assert(Precision(coefficients_mpfr[ii](0))==precision_);
193 }
194 }

209 static Patch RandomReal(std::vector<unsigned> const& sizes)
210 {
211 using bertini::multiprecision::RandomReal;
212 using bertini::Precision;
213
214 Patch p;
215
216 p.variable_group_sizes_ = sizes;
217
218
219
220 std::vector<Vec<complex_mp> >& coefficients_mpfr = std::get<std::vector<Vec<complex_mp> > >(p.coefficients_working_);
221 std::vector<Vec<complex_dbl> >& coefficients_dbl = std::get<std::vector<Vec<complex_dbl> > >(p.coefficients_working_);
222
223 p.coefficients_highest_precision_.resize(sizes.size());
224 coefficients_mpfr.resize(sizes.size());
225 coefficients_dbl.resize(sizes.size());
226
227 for (size_t ii=0; ii<sizes.size(); ++ii)
228 {
229 p.coefficients_highest_precision_[ii].resize(sizes[ii]);
230 for (unsigned jj=0; jj<sizes[ii]; ++jj)
231 // real bounded-modulus draw: away from 0 and infinity (same recipe as the complex
232 // patch / start systems) but kept REAL, so a real patch keeps a real path real.
233 multiprecision::RandomRealBoundedModulusAssign(p.coefficients_highest_precision_[ii](jj), MaxPrecisionAllowed());
234
235 coefficients_mpfr[ii] = p.coefficients_highest_precision_[ii];
236 Precision(coefficients_mpfr[ii],DefaultPrecision());
237 assert(Precision(coefficients_mpfr[ii](0))==DefaultPrecision());
238
239 coefficients_dbl[ii].resize(sizes[ii]);
240 for (unsigned jj=0; jj<sizes[ii]; ++jj)
241 coefficients_dbl[ii](jj) = complex_dbl(p.coefficients_highest_precision_[ii](jj));
242 }
243
244 return p;
245 }

265 void Precision(unsigned new_precision) const
266 {
267 if (precision_==new_precision)
268 return;
269
270 using bertini::Precision;
271 std::vector<Vec<complex_mp> >& coefficients_mpfr = std::get<std::vector<Vec<complex_mp> > >(coefficients_working_);
272
273 for (unsigned ii = 0; ii < NumVariableGroups(); ++ii)
274 {
275 for (unsigned jj=0; jj<variable_group_sizes_[ii]; ++jj)
276 {
277 if (new_precision>precision_)
278 {
279 coefficients_mpfr[ii](jj) = coefficients_highest_precision_[ii](jj);
280 coefficients_mpfr[ii](jj).precision(new_precision);
281 }
282 else
283 coefficients_mpfr[ii](jj).precision(new_precision);
284
285 assert(Precision(coefficients_mpfr[ii](jj))==new_precision);
286 }
287 }
288
289 precision_ = new_precision;
290 }

307 template<typename Derived, typename T>
308 void EvalInPlace(Eigen::MatrixBase<Derived> & function_values, Vec<T> const& x) const
309 {
310 static_assert(std::is_same<typename Derived::Scalar,T>::value,"scalar types must match");
311
312 #ifndef BERTINI_DISABLE_ASSERTS
313 if (! (function_values.size()>=NumVariableGroups()) )
314 {
315 std::stringstream ss;
316 ss << "container for function values must be of length at least as long as the number of variable groups. the input vector into which to write is of length " << function_values.size();
317 throw std::runtime_error(ss.str());
318 }
319// assert((bertini::Precision(x(0))==DoublePrecision() || bertini::Precision(x(0)) == Precision())
320// && "precision of input vector must match current working precision of patch during evaluation"
321// );
322 #endif
323
324 // unpack from the tuple of working coefficients
325 const std::vector<Vec<T> >& coefficients = std::get<std::vector<Vec<T> > >(coefficients_working_);
326
327 unsigned offset(static_cast<unsigned>(function_values.size() - NumVariableGroups())); // by precondition this number is at least 0. the precondition is ensured by the public wrapper
328 unsigned counter(0);
329 for (unsigned ii = 0; ii < NumVariableGroups(); ++ii)
330 {
331 T& value = function_values(ii+offset);
332 value = T(-1);
333 for (unsigned jj=0; jj<variable_group_sizes_[ii]; ++jj)
334 {
335 value += x(counter)*coefficients[ii](jj);
336 counter++;
337 }
338 }
339 }

365 template<typename Derived, typename T>
366 void JacobianInPlace(Eigen::MatrixBase<Derived> & jacobian, Vec<T> const& x) const
367 {
368 static_assert(std::is_same<typename Derived::Scalar,T>::value,"scalar types must match");
369 (void)x; // only used in asserts; the jacobian of a patch is constant
370
371
372 #ifndef BERTINI_DISABLE_ASSERTS
373 assert(jacobian.rows()>=NumVariableGroups() && "input jacobian must have at least as many rows as variable groups");
374 assert(jacobian.cols()==NumVariables() && "input jacobian must have as many columns as the patch has variables");
375 assert(
376 (bertini::Precision(x(0))==DoublePrecision() || bertini::Precision(x(0)) == Precision())
377 && "precision of input vector must match current working precision of patch during evaluation"
378 );
379 #endif
380
381 const std::vector<Vec<T> >& coefficients = std::get<std::vector<Vec<T> > >(coefficients_working_);
382
383 unsigned offset(static_cast<unsigned>(jacobian.rows() - NumVariableGroups())); // by precondition this number is at least 0. the precondition is ensured by the public wrapper
384
385 // A patch row is sparse -- one block of coefficients per variable group, zero elsewhere.
386 // Zero the rows the patch owns before writing those coefficients, so the patch FULLY
387 // defines its own rows and the caller need not pre-zero the matrix. The block-composed
388 // Jacobian path allocates J uninitialized and assigns only the block (function) rows;
389 // without this the patch rows' off-coefficient entries are read uninitialized -- benign
390 // (zeroed pages) on Linux/macOS, but garbage on Windows, where a degree-2 homotopy's
391 // Jacobian "evaluated" to ~1e252 and wrecked the AMP condition-number estimate.
392 jacobian.block(offset, 0, NumVariableGroups(), jacobian.cols()).setZero();
393
394 unsigned counter(0);
395 for (unsigned ii = 0; ii < NumVariableGroups(); ++ii)
396 for (unsigned jj=0; jj<variable_group_sizes_[ii]; ++jj)
397 jacobian(ii+offset,counter++) = coefficients[ii](jj);
398 }

410 JacobianInPlace(jacobian, x);

425 template<typename T>
426 void RescalePointToFitInPlace(Vec<T> & x) const
427 {
428 #ifndef BERTINI_DISABLE_ASSERTS
429 assert(x.size() == NumVariables() && "input point for rescaling to fit a patch must have same length as total number of variables being patched, in all variable groups.");
430 assert((bertini::Precision(x(0))==DoublePrecision() || bertini::Precision(x(0)) == Precision())
431 && "precision of input vector must match current working precision of patch during rescaling"
432 );
433 #endif
434
435 const std::vector<Vec<T> >& coefficients = std::get<std::vector<Vec<T> > >(coefficients_working_);
436
437 unsigned starting_index_counter(0);
438 for (unsigned ii=0; ii<NumVariableGroups(); ii++)
439 {
440 auto subvec = x.segment(starting_index_counter,variable_group_sizes_[ii]);
441 subvec /= subvec.transpose() * coefficients[ii];
442 starting_index_counter += variable_group_sizes_[ii];
443 }
444 }

522 bool operator==(Patch const& rhs) const
523 {
524 if (NumVariableGroups()!=rhs.NumVariableGroups())
525 return false;
526 if (NumVariables()!=rhs.NumVariables())
527 return false;
528 for (unsigned ii=0; ii<NumVariableGroups(); ii++)
529 if (variable_group_sizes_[ii]!=rhs.variable_group_sizes_[ii])

531 for (unsigned ii=0; ii<NumVariableGroups(); ii++)
