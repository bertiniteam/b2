136 line_.name("line_of_settings_input");

139 last_line_.name("line_of_settings_input_with_no_eol");

142 colon_.name("Set of characters with no : or eol");
143 colon_ = lexeme[*(char_ - ":") >> omit[":"] ];
144
145 yes_eol_.name("Set of characters with eol");
146 yes_eol_ = lexeme[*(char_ - eol - ";") >> omit[-(char_(';') >> *(char_ - eol) )] >> eol];
147
148 no_eol_.name("Set of characters with no eol");
149 no_eol_ = lexeme[*(char_ - ";") >> omit[-(char_(';') >> *(char_ ) )]];
