69template<typename TaskT, typename ResultT>
70void RunManagerLoop(
71 MPI_Comm comm,
72 std::queue<TaskT>& tasks,
73 std::function<void(ResultT const&)> store_fn)
74{
75 int world_size = 0;
76 MPI_Comm_size(comm, &world_size);
77 const int num_workers = world_size - 1;
78 if (num_workers <= 0)
79 return;
80
81 TaskT sentinel_value = detail::make_sentinel(TaskT{});
82
83 // Each worker announces how many tasks it can have in flight at once
84 // (1 for a serial worker, n_threads for a threaded one).
85 std::map<int, int> capacity; // rank -> max tasks in flight
86 for (int ii = 0; ii < num_workers; ++ii)
87 {
88 int cap = 0;
89 MPI_Status status;
90 MPI_Recv(&cap, 1, MPI_INT, MPI_ANY_SOURCE, TAG_CAPACITY, comm, &status);
91 capacity[status.MPI_SOURCE] = (cap >= 1) ? cap : 1;
92 }
93
94 std::map<int, int> outstanding; // rank -> tasks dispatched but not yet returned
95 int active_workers = num_workers;
96
97 // Seed each worker up to its capacity. A rank seeded with zero tasks
98 // (queue exhausted) is immediately sent the sentinel.
99 for (auto const& [rank, cap] : capacity)
100 {
101 outstanding[rank] = 0;
102 for (int ii = 0; ii < cap && !tasks.empty(); ++ii)
103 {
104 TaskT task = tasks.front(); tasks.pop();
105 mpi_send_serialized(comm, rank, TAG_WORK_ITEM, task);
106 ++outstanding[rank];
107 }
108 if (outstanding[rank] == 0)
109 {
110 mpi_send_serialized(comm, rank, TAG_WORK_ITEM, sentinel_value);
111 --active_workers;
112 }
113 }
114
115 // Drain: refill a rank as each result arrives; retire it with a sentinel
116 // once the queue is empty and its last outstanding task has come back.
117 while (active_workers > 0)
118 {
119 ResultT result;
120 int worker = mpi_recv_serialized_any(comm, TAG_RESULT, result);
121
122 store_fn(result);
123 --outstanding[worker];
124
125 if (!tasks.empty())
126 {
127 TaskT task = tasks.front(); tasks.pop();
128 mpi_send_serialized(comm, worker, TAG_WORK_ITEM, task);
129 ++outstanding[worker];
130 }
131 else if (outstanding[worker] == 0)
132 {
133 mpi_send_serialized(comm, worker, TAG_WORK_ITEM, sentinel_value);
134 --active_workers;
135 }
136 }
137}
