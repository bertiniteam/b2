102 static Slice RandomReal(VariableGroup const& v, unsigned dim, bool homogeneous = false, bool orthogonal = true)
103 {
104 typedef void (*funtype) (complex_mp&, unsigned); // the type for number generation
105 // bounded-modulus draw (away from 0 and infinity), matching patches and the start systems;
106 // kept REAL so a real slice stays real. The orthogonal path is real too (real=true below):
107 // it QR-factors a matrix of REAL units, yielding a real orthogonal coefficient block (issue
108 // #294 -- previously the orthogonal path hardcoded the complex orthonormal matrix, so a
109 // "real" slice came out complex).
110 funtype gen = bertini::multiprecision::RandomRealBoundedModulusAssign;
111 return Make(v, dim, homogeneous, orthogonal, /*real=*/true, gen);
112 }

138 static Slice Make(VariableGroup const& v, unsigned dim, bool homogeneous, bool orthogonal, bool real, std::function<void(complex_mp&, unsigned)> gen)
139 {
140 const unsigned num_vars = static_cast<unsigned>(v.size());
141
142 Mat<complex_mp> coeffs(dim, num_vars); // the variable coefficients (one row per form)
143
144 if (orthogonal)
145 {
146 // conjugate-orthonormal coefficient matrix (orthonormal linear forms), drawn the b1 way
147 // via RandomConjugateOrthonormalMatrix (ADR-0041) -- it generates square and truncates,
148 // so the old transpose dance is gone. Built at max precision, like the rest of the slice.
149 // A real slice QR-factors a matrix of REAL units (a real orthogonal block); a complex
150 // slice uses complex units. (Issue #294: the real path must stay real.)
151 auto prev_precision = DefaultPrecision();
152 DefaultPrecision(MaxPrecisionAllowed());
153 if (real)
154 coeffs = bertini::RandomConjugateOrthonormalMatrix<real_mp>(dim, num_vars)
155 .unaryExpr([](real_mp const& r){ return complex_mp(r); });
156 else
157 coeffs = bertini::RandomConjugateOrthonormalMatrix<complex_mp>(dim, num_vars);
158 DefaultPrecision(prev_precision);
159 }
160 else
161 {
162 for (unsigned ii(0); ii < dim; ++ii)
163 for (unsigned jj(0); jj < num_vars; ++jj)
164 gen(coeffs(ii, jj), MaxPrecisionAllowed());
165 }
166
167 assert(static_cast<unsigned>(coeffs.rows()) == dim);
168 assert(static_cast<unsigned>(coeffs.cols()) == num_vars);
169
170 // Assemble the augmented matrix: [ coeffs | constants ]. A homogeneous slice's constant
171 // column is zero; otherwise it is freshly generated.
172 Mat<complex_mp> augmented(dim, num_vars + 1);
173 augmented.leftCols(num_vars) = coeffs;
174 if (homogeneous)
175 augmented.col(num_vars).setZero();
176 else
177 for (unsigned ii(0); ii < dim; ++ii)
178 gen(augmented(ii, num_vars), MaxPrecisionAllowed());
179
180 return FromCoefficients(v, augmented, homogeneous);
181 }

262 Slice Concatenate(Slice const& other) const
263 {
264 if (NumVariables() != other.NumVariables())
265 throw std::runtime_error("Slice::Concatenate requires both slices to be on the same number of variables");
266
267 Mat<complex_mp> const& A = Coefficients();
268 Mat<complex_mp> const& B = other.Coefficients();
269 Mat<complex_mp> stacked(A.rows() + B.rows(), A.cols());
270 stacked.topRows(A.rows()) = A;
271 stacked.bottomRows(B.rows()) = B;
272
273 return FromCoefficients(sliced_vars_, stacked, is_homogeneous_ && other.is_homogeneous_);
274 }

300 Slice Rows(std::vector<unsigned> const& indices) const
301 {
302 Mat<complex_mp> const& C = Coefficients();
303 Mat<complex_mp> sub(static_cast<Eigen::Index>(indices.size()), C.cols());
304 for (size_t ii = 0; ii < indices.size(); ++ii)
305 {
306 if (indices[ii] >= Dimension())
307 throw std::runtime_error("Slice::Rows asked for a form index outside the slice");
308 sub.row(static_cast<Eigen::Index>(ii)) = C.row(indices[ii]);
309 }
310 return FromCoefficients(sliced_vars_, sub, is_homogeneous_);
311 }
