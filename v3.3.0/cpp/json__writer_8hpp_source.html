65 inline boost::json::object SystemPartsJson(System const& sys)
66 {
67 namespace json = boost::json;
68 json::object parts;
69
70 auto groups_as_json = [](auto const& groups) {
71 json::array out;
72 for (auto const& grp : groups)
73 {
74 json::array names;
75 for (auto const& v : grp)
76 names.push_back(json::value(v->name()));
77 out.push_back(names);
78 }
79 return out;
80 };
81 parts["variable_groups"] = groups_as_json(sys.VariableGroups());
82 parts["hom_variable_groups"] = groups_as_json(sys.HomVariableGroups());
83
84 {
85 json::array homogenizers;
86 for (auto const& v : sys.HomogenizingVariables())
87 homogenizers.push_back(json::value(v->name()));
88 parts["homogenizing_variables"] = homogenizers;
89 }
90
91 if (sys.HavePathVariable())
92 parts["path_variable"] = sys.GetPathVariable()->name();
93 else
94 parts["path_variable"] = nullptr;
95
96 auto const functions = sys.NaturalFunctionsAsNodes();
97 {
98 // named subexpressions are discovered inside the trees (nested included)
99 json::object named;
100 std::vector<std::shared_ptr<const node::Node>> roots(functions.begin(),
101 functions.end());
102 for (auto const& ne : node::Find<node::NamedExpression>(roots))
103 {
104 std::ostringstream expr;
105 expr << *(ne->EntryNode());
106 named[ne->name()] = expr.str();
107 }
108 if (!named.empty())
109 parts["named_subexpressions"] = named;
110 }
111 {
112 json::array function_texts;
113 for (auto const& f : functions)
114 {
115 std::ostringstream expr;
116 expr << *f;
117 function_texts.push_back(json::value(expr.str()));
118 }
119 parts["functions"] = function_texts;
120 }
121
122 {
123 // one summary object per evaluation block, in evaluation order --
124 // a randomized, sliced, or blended system says so here (the exact
125 // coefficients live in the canonical encoding)
126 struct BlockKindName
127 {
128 std::string operator()(blocks::PolynomialBlock const&) const { return "polynomial"; }
129 std::string operator()(blocks::LinearFormsBlock const&) const { return "linear_forms"; }
130 std::string operator()(blocks::ProductsOfLinearsBlock const&) const { return "products_of_linears"; }
131 std::string operator()(blocks::BlendBlock<System> const&) const { return "blend"; }
132 std::string operator()(blocks::RandomizationBlock<System> const&) const { return "randomization"; }
133 };
134 json::array block_summaries;
135 for (auto const& b : sys.Blocks())
136 {
137 json::object summary;
138 summary["kind"] = std::visit(BlockKindName{}, b);
139 summary["num_functions"] = static_cast<std::int64_t>(
140 std::visit([](auto const& blk) { return blk.NumFunctions(); }, b));
141 block_summaries.push_back(summary);
142 }
143 parts["blocks"] = block_summaries;
144 }
145
146 parts["is_patched"] = sys.IsPatched();
147 if (sys.IsPatched())
148 parts["num_patches"] = static_cast<std::int64_t>(sys.NumPatches());
149
150 return parts;
151 }
