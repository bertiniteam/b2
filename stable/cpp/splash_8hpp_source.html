190inline
191std::string MPIVersion()
192{
193#ifdef BERTINI2_HAVE_MPI
194 char buf[MPI_MAX_LIBRARY_VERSION_STRING];
195 int len = 0;
196 MPI_Get_library_version(buf, &len);
197 return std::string(buf, static_cast<size_t>(len));
198#else
199 return "not available (serial build)";
200#endif
201}
