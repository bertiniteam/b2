218 void EvalInPlace(Vec<T> & function_values) const
219 {
220
221 if (function_values.size() != static_cast<Eigen::Index>(NumTotalFunctions()))
222 {
223 std::stringstream ss;
224 ss << "trying to evaluate system in-place, but number length of vector into which to write the values (" << function_values.size() << ") doesn't match number of system user-defined functions plus patches ( " << NumNaturalFunctions() << "+" << NumPatches() << ") = " << NumTotalFunctions() << "). Use System.NumTotalFunctions() to make the container for in-place evaluation";
225 throw std::runtime_error(ss.str());
226 }
227
228 if (!is_differentiated_)
229 Differentiate(); // syncs + (for the polynomial block) compiles the SLP
230
231 EvalBlocksInPlace<T>(function_values);
232 if (IsPatched())
233 patch_.EvalInPlace(function_values,
234 std::get<Vec<T> >(current_variable_values_));
235 CoerceBlockOutputPrecision(function_values);
236 }

697 {

717 CoerceBlockOutputPrecision(ds_dt);

892 template<typename T>
893 void SetVariables(const Vec<T> & new_values) const
894 {
895 if (new_values.size()!= static_cast<Eigen::Index>(NumVariables()))
896 {
897 throw std::runtime_error("variable vector of different length from system-owned variables in SetVariables");
898 }
899
900 #ifndef BERTINI_DISABLE_PRECISION_CHECKS
901 // A system with no variables (a constant) has an empty point: there is no
902 // precision to read from it, so skip the check.
903 if (new_values.size() > 0)
904 {
905 if constexpr (!std::is_same<T,complex_dbl>::value) {
906 if (Precision(new_values) != this->precision())
907 throw std::runtime_error("precision of input point in SetVariables (" + std::to_string(Precision(new_values)) + ") must match the precision of the system (" + std::to_string(this->precision()) + ").");
908 }
909 }
910 #endif
911
912 // Blocks are value-in: the polynomial block feeds this stored vector into its SLP, the
913 // structured blocks compute on it directly, and the patch reads it too (see
914 // EvalBlocksInPlace). The shared Variable nodes are no longer written during evaluation
915 // (ADR-0027), so the node DAG stays read-only across threads.
916 std::get<Vec<T> >(current_variable_values_) = new_values;
917 }
