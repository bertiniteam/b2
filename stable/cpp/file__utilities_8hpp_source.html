47 inline
48 void OpenInFileThrowIfFail(ifstream & in,
49 Path const& input_path)
50 {
51 using namespace fs;
52 try{
53 if (exists(input_path))
54 {
55 if (is_directory(input_path))
56 {
57 std::stringstream err_msg;
58 err_msg << "attempting to open file but is a directory, of name '" << input_path.string() << "'";
59 throw std::runtime_error(err_msg.str());
60 }
61 else
62 {
63 in.open(input_path.c_str());
64 if (!in.is_open())
65 {
66 std::stringstream err_msg;
67 err_msg << "file '" << input_path.string() << "' hypothetically exists, but failed to open correctly";
68 throw std::runtime_error(err_msg.str());
69 }
70 }
71 }
72 else
73 {
74 std::stringstream err_msg;
75 err_msg << "attempting to open file which doesn't exist, of name '" << input_path.string() << "'";
76 throw std::runtime_error(err_msg.str());
77 }
78 }
79 catch (const filesystem_error& ex)
80 {
81 std::stringstream err_msg;
82 err_msg << "boost::filesystem throw while attempting to open file '" << input_path.string() << "': '" << ex.what() << "'";
83 throw std::runtime_error(err_msg.str());
84 }
85 }

93 inline
94 std::string FileToString(Path const& input_path)
95 {
96 ifstream infile;
97 OpenInFileThrowIfFail(infile, input_path);
98 std::istreambuf_iterator<char> file_begin(infile), file_end;
99 std::string contents(file_begin, file_end);
100
101 // Treat file contents as UTF-8; drop a leading byte-order mark (EF BB BF)
102 // so it is not parsed as a stray leading character.
103 if (contents.size() >= 3 &&
104 static_cast<unsigned char>(contents[0]) == 0xEF &&
105 static_cast<unsigned char>(contents[1]) == 0xBB &&
106 static_cast<unsigned char>(contents[2]) == 0xBF)
107 {
108 contents.erase(0, 3);
109 }
110
111 return contents;
112 }
