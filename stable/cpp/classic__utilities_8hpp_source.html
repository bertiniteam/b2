372 SplitInputFile ParseInputFile(std::string str_input_file)
373 {
374 std::string::const_iterator iter = str_input_file.begin();
375 std::string::const_iterator end = str_input_file.end();
376
377 std::string uncommented_str_input;
378 CommentStripper<std::string::const_iterator> commentparser;
379 phrase_parse(iter, end, commentparser,boost::spirit::ascii::space, uncommented_str_input);
380
381 iter = uncommented_str_input.begin();
382 end = uncommented_str_input.end();
383
384 SplitInputFileParser<std::string::const_iterator> spliter;
385
386 SplitInputFile input_file;
387 phrase_parse(iter, end, spliter,boost::spirit::ascii::space, input_file);
388
389 return input_file;
390 }

397 std::tuple<std::string, std::string> SplitIntoConfigAndInput(Path const& input_file)
398 {
399 // Route through ParseInputFile, which runs the CommentStripper pass BEFORE splitting.
400 // Bertini 1 uses '%' as its comment marker; calling the raw SplitInputFileParser here --
401 // as this used to -- left '%' comments in the text, and the downstream system/settings
402 // parsers then choked ("parser did not consume entire input"). A classic input file
403 // with comments must parse, for backwards compatibility.
404 auto config_and_input = ParseInputFile(FileToString(input_file));
405 return std::make_tuple(config_and_input.Config(), config_and_input.Input());
406 }
