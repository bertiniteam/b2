59 template<typename ConfigT>
60 ConfigT FillConfigStruct(std::string const& config_str)
61 {
62 std::string::const_iterator iter = config_str.begin();
63 std::string::const_iterator end = config_str.end();
64 ConfigT settings;
65 ConfigSettingParser<std::string::const_iterator, ConfigT> parser;
66 auto parse_success = phrase_parse(iter, end, parser,boost::spirit::ascii::space, settings);
67 if (!parse_success || iter!=end)
68 {
69 std::string remaining(iter, end);
70 if (remaining.size() > 60)
71 remaining = remaining.substr(0, 60) + "...";
72 if (remaining.empty())
73 remaining = "<end of input>";
74 throw std::runtime_error(
75 std::string("[config] parser did not consume entire input; "
76 "unparsed remainder: \"") + remaining + "\"");
77 }
78
79 return settings;
80 }
