110 template<typename OutDerived>
111 void Solve(Vec<NumT> const& b, Eigen::MatrixBase<OutDerived> const& x_out) const
112 {
113 // Eigen idiom for a write-to output that may be a temporary Block (e.g. K.col(stage)).
114 Eigen::MatrixBase<OutDerived>& x = const_cast<Eigen::MatrixBase<OutDerived>&>(x_out);
115
116 // y_ = P b : copy then replay the row swaps recorded during factorization.
117 y_ = b;
118 for (unsigned k = 0; k < n_; ++k)
119 if (piv_[k] != k)
120 std::swap(y_[k], y_[piv_[k]]);
121
122 // forward substitution: L y = P b, L unit lower-triangular (implicit unit diagonal).
123 for (unsigned i = 1; i < n_; ++i)
124 for (unsigned j = 0; j < i; ++j)
125 {
126 t_ = lu_(i, j);
127 t_ *= y_[j];
128 y_[i] -= t_; // y_[i] -= lu_(i,j) * y_[j]
129 }
130
131 // back substitution: U x = y, U upper-triangular.
132 for (unsigned ii = n_; ii-- > 0; )
133 {
134 for (unsigned j = ii + 1; j < n_; ++j)
135 {
136 t_ = lu_(ii, j);
137 t_ *= x(j);
138 y_[ii] -= t_; // y_[ii] -= lu_(ii,j) * x(j)
139 }
140 x(ii) = y_[ii];
141 x(ii) /= lu_(ii, ii);
142 }
143 }

147 Mat<NumT> const& Factors() const { return lu_; }

149private:
151 MatrixSuccessCode FactorInPlace()
152 {
153 for (unsigned k = 0; k < n_; ++k)
154 {
155 // partial pivot: pick the row p >= k with the largest |lu_(p,k)|.
156 unsigned p = k;
157 amax_ = abs(lu_(k, k));
158 for (unsigned i = k + 1; i < n_; ++i)
159 {
160 acur_ = abs(lu_(i, k));
161 if (acur_ > amax_) { amax_ = acur_; p = i; }
162 }
163 piv_[k] = p;
164 if (p != k)
165 lu_.row(k).swap(lu_.row(p));
166
167 // scale the sub-column below the pivot by the reciprocal of the pivot -- one (costly)
168 // mpc division per column instead of one per sub-row, then cheap multiplies. Pin recip_
169 // to the pivot's own precision first: during AMP the working precision changes, and a
170 // reciprocal left at a stale precision would contaminate the factorization (and, through
171 // the solution, trip the endgame's uniform-precision checks).
172 if constexpr (is_mp)
173 Precision(recip_, Precision(lu_(k, k)));
174 recip_ = 1;
175 recip_ /= lu_(k, k); // recip_ = 1 / a_kk, at the pivot's precision
176 for (unsigned i = k + 1; i < n_; ++i)
177 lu_(i, k) *= recip_; // l_ik = a_ik * (1 / a_kk)
178
179 // rank-1 trailing-submatrix update, reusing t_ for every multiply-subtract so the
180 // whole O(N^3) sweep allocates nothing beyond the scratch scalar.
181 for (unsigned j = k + 1; j < n_; ++j)
182 for (unsigned i = k + 1; i < n_; ++i)
183 {
184 t_ = lu_(i, k);
185 t_ *= lu_(k, j);
186 lu_(i, j) -= t_; // lu_(i,j) -= lu_(i,k) * lu_(k,j)
187 }
188 }
189 return LUPartialPivotDecompositionSuccessful(lu_);
190 }
191
193 void ApplyPrecision()
194 {
195 if constexpr (is_mp)
196 {
197 Precision(lu_, precision_);
198 Precision(y_, precision_);
199 Precision(t_, precision_);
200 Precision(recip_, precision_); // re-pinned per pivot in FactorInPlace, but keep it valid
201 Precision(amax_, precision_);
202 Precision(acur_, precision_);
203 }
204 }

214 RealT acur_;
