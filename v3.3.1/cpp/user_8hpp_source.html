138inline void load_construct_data(
139 Archive & ar, bertini::start_system::User * t, const unsigned int /*file_version*/
140){
141 // retrieve data from archive required to construct new instance
142 bertini::System sys;
143 ar >> sys;
144
145 bool solns_in_dbl;
146 ar >> solns_in_dbl;
147
148 if (solns_in_dbl)
149 {
150 bertini::SampCont<bertini::complex_dbl> solns;
151 ar >> solns;
152 ::new(t)bertini::start_system::User(sys, solns);
153 }
154 else
155 {
156 bertini::SampCont<bertini::complex_mp> solns;
