64inline bool IsEmojiBase(char32_t cp)
65{
66 return (cp >= 0x1F000 && cp <= 0x1FAFF) // emoticons, pictographs, transport, supplemental & extended-A (incl. skin tones, regional indicators)
67 || (cp >= 0x2600 && cp <= 0x27BF) // miscellaneous symbols + dingbats (☀ ❤ ✨ ✅ ...)
68 || (cp >= 0x2B00 && cp <= 0x2BFF); // miscellaneous symbols and arrows (⭐ ⬅ ...)
69}

73inline bool IsEmojiGlue(char32_t cp)
74{
75 return cp == 0x200D // ZERO WIDTH JOINER (👩‍👩‍👧)
76 || cp == 0xFE0E || cp == 0xFE0F // variation selectors 15 / 16 (❤️)
77 || cp == 0x20E3; // combining enclosing keycap
78 // skin-tone modifiers (U+1F3FB..FF) and regional indicators (U+1F1E6..FF) are
79 // already covered by IsEmojiBase, so they continue a name via that predicate.
80}

86inline bool IsIdentStart(char32_t cp)
87{
88 return boost::spirit::char_encoding::unicode::isalpha(cp)
89 || detail::IsEmojiBase(cp);
90}

95inline bool IsIdentCont(char32_t cp)
96{
97 return boost::spirit::char_encoding::unicode::isalnum(cp)
98 || cp == U'[' || cp == U']' || cp == U'_'
99 || detail::IsEmojiBase(cp) || detail::IsEmojiGlue(cp);
100}

106inline bool IsValidVariableName(std::string const& s)
107{
108 if (s.empty())
109 return false;
110 try
111 {
112 boost::u8_to_u32_iterator<std::string::const_iterator> it(s.begin(), s.begin(), s.end());
113 boost::u8_to_u32_iterator<std::string::const_iterator> end(s.end(), s.begin(), s.end());
114 if (it == end || !IsIdentStart(*it))
115 return false;
116 for (++it; it != end; ++it)
117 if (!IsIdentCont(*it))
118 return false;
119 return true;
120 }
121 catch (...)
122 {
123 return false; // malformed UTF-8
124 }
125}

129inline void ThrowIfInvalidVariableName(std::string const& s)
130{
131 if (!IsValidVariableName(s))
132 throw std::runtime_error(
133 "invalid variable name \"" + s + "\": a name must be a nonempty identifier -- "
134 "it starts with a letter (ASCII or Unicode, e.g. x or \xCE\xA9) and continues with "
135 "letters, digits, or [ ] _ . Expressions, operators, whitespace, and leading "
136 "digits are not allowed.");
137}
