200 template <typename OutputIterator, typename T>
201 static bool generate(OutputIterator sink, Vec<T> const& c)
202 {
203 auto s = c.size();
204 for (decltype(s) ii=0; ii<s; ++ii)
205 {
206 bool b = generate(sink, c(ii));
207 if (!b)
208 return b;
209 boost::spirit::karma::generate(sink, (boost::spirit::karma::char_), '\n');
210 }
211 return true;
212 }
