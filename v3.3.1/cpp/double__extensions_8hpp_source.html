95 inline complex_dbl pow(const complex_dbl & z, int power)
96 {
97 if (power < 0) {
98 return pow(1./z, -power);
99 }
100 else if (power==0)
101 return complex_dbl(1,0);
102 else if(power==1)
103 return z;
104 else if(power==2)
105 return z*z;
106 else if(power==3)
107 return z*z*z;
108 else
109 {
110 unsigned int p(static_cast<unsigned int>(power));
111 complex_dbl result(1,0), z_to_the_current_power_of_two = z;
112 // have copy of p in memory, can freely modify it.
113 do {
114 if ( (p & 1) == 1 ) { // get the lowest bit of the number
115 result *= z_to_the_current_power_of_two;
116 }
117 z_to_the_current_power_of_two *= z_to_the_current_power_of_two; // square z_to_the_current_power_of_two
118 } while (p >>= 1);
119
120 return result;
121 }
122 }
