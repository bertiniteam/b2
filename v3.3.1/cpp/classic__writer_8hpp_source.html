59 inline void EmitSystem(std::ostream& out, System const& sys)
60 {
61 auto emit_groups = [&out](char const* keyword, auto const& groups){
62 for (auto const& grp : groups)
63 {
64 out << keyword << " ";
65 for (size_t i = 0; i < grp.size(); ++i)
66 out << (i ? ", " : "") << *grp[i];
67 out << ";\n";
68 }
69 };
70 emit_groups("variable_group", sys.VariableGroups());
71 emit_groups("hom_variable_group", sys.HomVariableGroups());
72
73 auto functions = sys.NaturalFunctionsAsNodes();
74
75 // Named subexpressions are not stored separately; they are discovered inside the function
76 // trees (nested ones included). Emit each as `name = expr;` before the functions, the way
77 // the classic parser expects a subfunction to be defined ahead of its use.
78 {
79 std::vector<std::shared_ptr<const node::Node>> roots(functions.begin(), functions.end());
80 for (auto const& ne : node::Find<node::NamedExpression>(roots))
81 out << ne->name() << " = " << ne->EntryNode() << ";\n";
82 }
83
84 out << "function ";
85 for (size_t i = 0; i < functions.size(); ++i)
86 out << (i ? ", " : "") << "f" << i;
87 out << ";\n";
88 for (size_t i = 0; i < functions.size(); ++i)
89 out << "f" << i << " = " << functions[i] << ";\n";
90 }

93 inline std::string SystemToClassic(System const& sys)
94 {
95 std::ostringstream ss;
96 EmitSystem(ss, sys);
97 return ss.str();
98 }

132 inline int PredictorToClassic(tracking::Predictor p)
133 {
134 using P = tracking::Predictor;
135 switch (p)
136 {
137 case P::Constant: return -1;
138 case P::Euler: return 0;
139 case P::Heun: return 1;
140 case P::RK4: return 2;
141 case P::HeunEuler: return 1;
142 case P::RKNorsett34: return 4;
143 case P::RKF45: return 5;
144 case P::RKCashKarp45: return 6;
145 case P::RKDormandPrince56: return 7;
146 case P::RKVerner67: return 8;
147 }
148 return 5; // RKF45 -- the Bertini 2 default, a safe fallback
149 }

155 inline void EmitConfig(std::ostream& out, System const& sys, ClassicWriteOptions const& opt)
156 {

176 inline void WriteClassicInput(std::ostream& out, System const& sys,
177 ClassicWriteOptions const& opt = ClassicWriteOptions{})
178 {
179 out << "CONFIG\n";
180 EmitConfig(out, sys, opt);
181 out << "END;\n\nINPUT\n";
182 EmitSystem(out, sys);
183 out << "END;\n";
184 }
