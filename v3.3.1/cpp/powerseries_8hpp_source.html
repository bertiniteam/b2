352 template<typename ComplexT>
353 unsigned ComputeBoundOnCycleNumber()
354 {
355 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
356 using std::log; using std::abs;
357
358 const auto& samples = std::get<SampCont<ComplexT> >(samples_);
359
360 AssertSizesTimeSpace<ComplexT>();
361
362 auto num_samples = samples.size();
363 const Vec<ComplexT> & sample0 = samples[num_samples-3];
364 const Vec<ComplexT> & sample1 = samples[num_samples-2];
365 const Vec<ComplexT> & sample2 = samples[num_samples-1]; // most recent sample. oldest samples at front of the container
366
367
368// should this only be if the system is homogenized?
369 const auto& rand_vector = std::get<Vec<ComplexT> >(rand_vector_);
370 ComplexT rand_sum1 = ((sample1 - sample0).transpose()*rand_vector).sum();
371 ComplexT rand_sum2 = ((sample2 - sample1).transpose()*rand_vector).sum();
372
373 if ( abs(rand_sum1)==0 || abs(rand_sum2)==0) // avoid division by 0
374 {
375 upper_bound_on_cycle_number_ = 1;
376 return upper_bound_on_cycle_number_;
377 }
378
379 RealT estimate = log(static_cast<RealT>(this->EndgameSettings().sample_factor))/log(abs(rand_sum2/rand_sum1));
380
381 if (estimate < 1) // would be nan if sample points are same as each other
382 upper_bound_on_cycle_number_ = 1;
383 else
384 {
385 using std::max;
386 auto upper_bound = unsigned(round(estimate)*this->template Get<PowerSeriesConfig>().cycle_number_amplification);
387 upper_bound_on_cycle_number_ = max(upper_bound,this->template Get<PowerSeriesConfig>().max_cycle_number);
388 }
389
390 return upper_bound_on_cycle_number_;
391 }//end ComputeBoundOnCycleNumber

411 template<typename ComplexT>
412 unsigned ComputeCycleNumber(ComplexT const& t0)
413 {
414 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
415
416 const auto& samples = std::get<SampCont<ComplexT> >(samples_);
417 const auto& times = std::get<TimeCont<ComplexT> >(times_);
418
419 AssertSizesTimeSpaceDeriv<ComplexT>();
420
421 const Vec<ComplexT> &most_recent_sample = samples.back();
422 const ComplexT& most_recent_time = times.back();
423
424 //Compute upper bound for cycle number.
425 ComputeBoundOnCycleNumber<ComplexT>();
426
427
428 unsigned num_pts;
429 if (samples.size() > this->EndgameSettings().num_sample_points)
430 num_pts = this->EndgameSettings().num_sample_points;
431 else
432 num_pts = this->EndgameSettings().num_sample_points-1;
433
434
435 auto min_found_difference = Eigen::NumTraits<RealT>::highest();
436
437 TimeCont<ComplexT> s_times(num_pts);
438 SampCont<ComplexT> s_derivatives(num_pts);
439
440 for(unsigned int candidate = 1; candidate <= upper_bound_on_cycle_number_; ++candidate)
441 {
442 using std::pow;
443
444 std::tie(s_times, s_derivatives) = TransformToSPlane(static_cast<int>(candidate), t0, num_pts, ContStart::Front);
445 RealT cand_power{1/static_cast<RealT>(candidate)};
446 RealT curr_diff = (HermiteInterpolateAndSolve<ComplexT>(
447 pow((most_recent_time-t0)/(times[0]-t0),cand_power), // the target time
448 num_pts,s_times,samples,s_derivatives, ContStart::Front) // the input data
449 -
450 most_recent_sample).template lpNorm<Eigen::Infinity>();
451
452 if (curr_diff < min_found_difference)
453 {
454 min_found_difference = curr_diff;
455 this->cycle_number_ = candidate;
456 }
457
458 }// end cc loop over cycle number possibilities
459
460 return this->cycle_number_;
461 }//end ComputeCycleNumber

477 template<typename ComplexT>
478 void ComputeAllDerivatives()
479 {
480 auto& samples = std::get<SampCont<ComplexT> >(samples_);
481 auto& times = std::get<TimeCont<ComplexT> >(times_);
482 auto& derivatives = std::get<SampCont<ComplexT> >(derivatives_);
483
484 assert((samples.size() == times.size()) && "must have same number of times and samples");
485
486 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec) // known at compile time
487 {
488 auto max_precision = this->EnsureAtUniformPrecision(times, samples);
489 this->GetSystem().precision(max_precision);
490 }
491
492 //Compute dx_dt for each sample.
493 derivatives.clear(); derivatives.resize(samples.size());
494 for(unsigned ii = 0; ii < samples.size(); ++ii)
495 {
496 derivatives[ii] = -this->GetSystem().Jacobian(samples[ii],times[ii]).lu().solve(this->GetSystem().TimeDerivative(samples[ii],times[ii]));
497 }
498 }

511 template <typename ComplexT>
512 std::tuple<TimeCont<ComplexT>, SampCont<ComplexT>> TransformToSPlane(int cycle_num, ComplexT const& t0, unsigned num_pts, ContStart shift_from)
513 {
514 if (cycle_num==0)
515 throw std::runtime_error("cannot transform to s plane with cycle number 0");
516 AssertSizesTimeSpaceDeriv<ComplexT>();
517
518
519 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
520
521 const auto& times = std::get<TimeCont<ComplexT> >(times_);
522 const auto& derivatives = std::get<SampCont<ComplexT> >(derivatives_);
523
524 RealT c = static_cast<RealT>(cycle_num);
525 RealT one_over_c = 1/c;
526
527 size_t offset_t, offset_d;
528 if (shift_from == ContStart::Back)
529 {
530 offset_t = times.size()-num_pts;
531 offset_d = derivatives.size()-num_pts;
532 }
533 else
534 offset_t = offset_d = 0;
535
536
537 TimeCont<ComplexT> s_times(num_pts);
538 SampCont<ComplexT> s_derivatives(num_pts);
539
540 ComplexT time_shift = times[offset_t] - t0;
541
542 for(unsigned ii = 0; ii < num_pts; ++ii){
543 s_times[ii] = pow((times[ii+offset_t]-t0)/time_shift, one_over_c);
544 s_derivatives[ii] = derivatives[ii+offset_d]*( c*pow(s_times[ii],cycle_num-1))*time_shift;
545 }
546
547 return std::make_tuple(s_times, s_derivatives);
548 }

568 template<typename ComplexT>
569 SuccessCode ComputeApproximationOfXAtT0(Vec<ComplexT>& result, const ComplexT & t0)
570 {
571 const auto c = ComputeCycleNumber<ComplexT>(t0);
572
573 auto num_pts = this->EndgameSettings().num_sample_points;
574
575 TimeCont<ComplexT> s_times;
576 SampCont<ComplexT> s_derivatives;
577
578 std::tie(s_times, s_derivatives) = TransformToSPlane(static_cast<int>(c), t0, num_pts, ContStart::Back);
579 // the data was transformed to be on the interval [0 1] so we can hard-code the time-to-solve as 0 here.
580
581 Precision(result, Precision(s_derivatives.back()));
582 result = HermiteInterpolateAndSolve(ComplexT(0), num_pts, s_times, std::get<SampCont<ComplexT> >(samples_), s_derivatives, ContStart::Back);
583 return SuccessCode::Success;
584 }//end ComputeApproximationOfXAtT0

603 template<typename ComplexT>
604 SuccessCode AdvanceTime(const ComplexT & target_time)
605 {
606 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
607
608 auto& samples = std::get<SampCont<ComplexT> >(samples_);
609 auto& times = std::get<TimeCont<ComplexT> >(times_);
610
611 AssertSizesTimeSpaceDeriv<ComplexT>();
612
613 Vec<ComplexT> next_sample;
614 ComplexT next_time = (times.back() + target_time) * static_cast<RealT>(this->EndgameSettings().sample_factor); //setting up next time value using the midpoint formula, sample_factor will give us some
615
616 if (abs(next_time - target_time) < this->EndgameSettings().min_track_time) // generalized for target_time not equal to 0.
617 {
618 NotifyObservers(MinTrackTimeReached<EmitterType>(*this));
619 return SuccessCode::MinTrackTimeReached;
620 }
621
622 SuccessCode tracking_success = this->EndgameTrackPath(next_sample,times.back(),next_time,samples.back());
623 if (tracking_success != SuccessCode::Success)
624 return tracking_success;
625
626 // Pure-(i) escalation: return BEFORE pushing the new sample so the window stays an untouched
627 // checkpoint; the driver migrates to mpfr and retries this advance. Elided for fixed precision.
628 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
629 if (this->adaptive_numeric_type_active_ &&
630 this->GetTracker().GetCurrentPrecision() > this->current_endgame_precision_)
631 return SuccessCode::HigherPrecisionNecessary;
632
633 NotifyObservers(InEGOperatingZone<EmitterType>(*this));
634
635 this->EnsureAtPrecision(next_time,Precision(next_sample));
636
637
638 times.push_back(next_time);
639 samples.push_back(next_sample);
640
641
642
643
644 auto refine_success = this->RefineSample(samples.back(), next_sample, times.back(),
645 this->FinalTolerance() * this->EndgameSettings().sample_point_refinement_factor,
646 this->EndgameSettings().max_num_refinements);
647 if (refine_success != SuccessCode::Success)
648 {
649 // Adaptive-numeric-type: a refine that double cannot satisfy is a request to cross to mpfr.
650 // Roll back the just-pushed sample so the window is a clean checkpoint, then signal the driver.
651 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
652 if (this->adaptive_numeric_type_active_ &&
653 (refine_success == SuccessCode::HigherPrecisionNecessary || refine_success == SuccessCode::FailedToConverge))
654 {
655 times.pop_back();
656 samples.pop_back();
657 return SuccessCode::HigherPrecisionNecessary;
658 }
659 NotifyObservers(RefiningFailed<EmitterType>(*this));
660 return refine_success;
661 }
662
663 this->EnsureAtPrecision(times.back(),Precision(samples.back()));
664
665 NotifyObservers(SampleRefined<EmitterType>(*this));
666
667 // we keep one more samplepoint than needed around, for estimating the cycle number
668 if (times.size() > this->EndgameSettings().num_sample_points+1)
669 {
670 times.pop_front();
671 samples.pop_front();
672 }
673
674 return SuccessCode::Success;
675 }

694 template<typename ComplexT>
695 SuccessCode RunImpl(const ComplexT & start_time, const Vec<ComplexT> & start_point, ComplexT const& target_time)
696 {
697 if (start_point.size()!=static_cast<Eigen::Index>(this->GetSystem().NumVariables()))
698 {
699 std::stringstream err_msg;
700 err_msg << "number of variables in start point for PSEG, " << start_point.size() << ", must match the number of variables in the system, " << this->GetSystem().NumVariables();
701 throw std::runtime_error(err_msg.str());
702 }
703
704 // thread-local only: the endgame may run on a std::thread worker.
705 SetThreadPrecision(Precision(start_point));
706
707 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
708 //Set up for the endgame.
709 ClearTimesAndSamples<ComplexT>();
710
711
712 // unpack some references for easy use
713 auto& samples = std::get<SampCont<ComplexT> >(samples_);
714 auto& times = std::get<TimeCont<ComplexT> >(times_);
715 Vec<ComplexT>& latest_approx = this->final_approximation_;
716 Vec<ComplexT>& prev_approx = this->previous_approximation_;
717
718 // this is for estimating a ... norm?
719 SetRandVec<ComplexT>(static_cast<int>(start_point.size()));
720
721
722 auto initial_sample_success = this->ComputeInitialSamples(start_time, target_time, start_point, times, samples);
723
724 if (initial_sample_success!=SuccessCode::Success)
725 {
726 NotifyObservers(EndgameFailure<EmitterType>(*this));
727 return initial_sample_success;
728 }
729
730 this->template RefineAllSamples<ComplexT>(samples, times);
731 ComputeAllDerivatives<ComplexT>();
732
733
734
735 auto extrapolation_code = ComputeApproximationOfXAtT0(prev_approx, target_time);
736 latest_approx = prev_approx;
737
738 if (extrapolation_code != SuccessCode::Success)
739 return extrapolation_code;
740
741 RealT norm_of_dehom_of_latest_approx(0); // initialized to 0 so the security check never reads indeterminate values
742 RealT norm_of_dehom_of_prev_approx(0);
743 if (this->SecuritySettings().level <= 0)
744 norm_of_dehom_of_prev_approx = this->GetSystem().InfinityNormOfDehomogenized(prev_approx);
745
746
747 NumErrorT& approx_error = this->approximate_error_;
748 approx_error = 1;
749
750 while (approx_error > this->FinalTolerance())
751 {
752 auto advance_code = AdvanceTime<ComplexT>(target_time);
753 if (advance_code!=SuccessCode::Success)
754 {
755 NotifyObservers(EndgameFailure<EmitterType>(*this));
756 return advance_code;
757 }
758
759 // this code is what bertini1 does... it refines all samples, like, all the time.
760 this->template RefineAllSamples<ComplexT>(samples, times);
761 ComputeAllDerivatives<ComplexT>();
762
763 extrapolation_code = ComputeApproximationOfXAtT0(latest_approx, target_time);
764 if (extrapolation_code!=SuccessCode::Success)
765 {
766 NotifyObservers(EndgameFailure<EmitterType>(*this));
767 return extrapolation_code;
768 }
769
770 approx_error = static_cast<NumErrorT>((latest_approx - prev_approx).template lpNorm<Eigen::Infinity>());
771 NotifyObservers(ApproximatedRoot<EmitterType>(*this));
772
773
774 if(this->SecuritySettings().level <= 0)
775 {
776 norm_of_dehom_of_latest_approx = this->GetSystem().InfinityNormOfDehomogenized(latest_approx);
777 if(norm_of_dehom_of_latest_approx > this->SecuritySettings().max_norm && norm_of_dehom_of_prev_approx > this->SecuritySettings().max_norm)
778 {
779 NotifyObservers(SecurityMaxNormReached<EmitterType>(*this));
780 return SuccessCode::SecurityMaxNormReached;
781 }
782 norm_of_dehom_of_prev_approx = norm_of_dehom_of_latest_approx;
783 }
784
785
786
787
788 Precision(prev_approx, Precision(latest_approx));
789 prev_approx = latest_approx;
790 } //end while
791
792 NotifyObservers(Converged<EmitterType>(*this));
793 return SuccessCode::Success;
794
795 } //end PSEG

798 // ================================================================================================
799 // Adaptive-numeric-type (double-first) PowerSeries endgame. Mirrors the Cauchy flavor: compute in
800 // the hardware complex_dbl fast lane while the tracker's authoritative precision stays double, and
801 // cross to complex_mp only when the tracker escalates (pure-(i)). Member templates so the explicit
802 // fixed-precision class instantiations never force-compile them. Fixed precision uses RunImpl<BCT>.
803 // ================================================================================================
804
806 template<typename Dummy = void>
807 SuccessCode RunImplAMP(BCT const& start_time, Vec<BCT> const& start_point, BCT const& target_time)
808 {
809 using bertini::Precision;
810 using RealT = typename Eigen::NumTraits<BCT>::Real;
811
812 if (start_point.size()!=static_cast<Eigen::Index>(this->GetSystem().NumVariables()))
813 {
814 std::stringstream err_msg;
815 err_msg << "number of variables in start point for PSEG, " << start_point.size() << ", must match the number of variables in the system, " << this->GetSystem().NumVariables();
816 throw std::runtime_error(err_msg.str());
817 }
818
819 this->adaptive_numeric_type_active_ = true;
820 struct Disarmer { bool& flag; ~Disarmer(){ flag = false; } } disarm{this->adaptive_numeric_type_active_};
821
822 this->current_endgame_precision_ = std::max(DoublePrecision(), Precision(start_point));
823
824 // ---- SETUP (restart-in-mpfr on escalation; bounded) ----
825 while (true)
826 {
827 SuccessCode code = (this->current_endgame_precision_==DoublePrecision())
828 ? SetupSegmentT<complex_dbl>(complex_dbl(start_time), this->DowncastToDouble(start_point), complex_dbl(target_time))
829 : SetupSegmentT<complex_mp>(this->AtActivePrecisionScalar(start_time), this->AtActivePrecisionVec(start_point), this->AtActivePrecisionScalar(target_time));
830 if (code == SuccessCode::HigherPrecisionNecessary)
831 {
832 this->current_endgame_precision_ = this->NextEscalatedPrecision();
833 SetThreadPrecision(this->current_endgame_precision_);
834 this->GetSystem().precision(this->current_endgame_precision_);
835 continue;
836 }
837 if (code != SuccessCode::Success)
838 return code;
839 break;
840 }
841
842 // previous_approximation_ (BCT) holds the first extrapolation, set by SetupSegmentT.
843 RealT norm_prev(0), norm_latest(0);
844 if (this->SecuritySettings().level <= 0)
845 norm_prev = this->GetSystem().InfinityNormOfDehomogenized(this->previous_approximation_);
846
847 this->approximate_error_ = 1;
848
849 // ---- MAIN LOOP: each tracker-touching phase self-heals (migrate-and-retry in mpfr) ----
850 while (this->approximate_error_ > this->FinalTolerance())
851 {
852 auto adv = AdvanceTimeAMP(target_time);
853 if (adv != SuccessCode::Success) { NotifyObservers(EndgameFailure<EmitterType>(*this)); return adv; }
854
855 auto ref = RefineAllSamplesAMP();
856 if (ref != SuccessCode::Success) return ref;
857
858 ComputeAllDerivativesAMP();
859
860 auto ext = ComputeApproxAMP(target_time); // -> final_approximation_ (BCT)
861 if (ext != SuccessCode::Success) { NotifyObservers(EndgameFailure<EmitterType>(*this)); return ext; }
862
863 Precision(this->previous_approximation_, Precision(this->final_approximation_));
864 this->approximate_error_ = static_cast<NumErrorT>((this->final_approximation_ - this->previous_approximation_).template lpNorm<Eigen::Infinity>());
865 NotifyObservers(ApproximatedRoot<EmitterType>(*this));
866
867 if (this->SecuritySettings().level <= 0)
868 {
869 norm_latest = this->GetSystem().InfinityNormOfDehomogenized(this->final_approximation_);
870 if (norm_latest > this->SecuritySettings().max_norm && norm_prev > this->SecuritySettings().max_norm)
871 {
872 NotifyObservers(SecurityMaxNormReached<EmitterType>(*this));
873 return SuccessCode::SecurityMaxNormReached;
874 }
875 norm_prev = norm_latest;
876 }
877
878 this->previous_approximation_ = this->final_approximation_;
879 }
880
881 NotifyObservers(Converged<EmitterType>(*this));
882 return SuccessCode::Success;
883 }

886 // Pre-loop work + first extrapolation, at one numeric type. Reports HigherPrecisionNecessary up to
887 // the driver (which restarts setup in mpfr) on escalation.
889 template<typename ComplexT>
890 SuccessCode SetupSegmentT(ComplexT const& start_time, Vec<ComplexT> const& start_point, ComplexT const& target_time)
891 {
892 SetThreadPrecision(Precision(start_point));
893 ClearTimesAndSamples<ComplexT>();
894
895 auto& samples = std::get<SampCont<ComplexT> >(samples_);
896 auto& times = std::get<TimeCont<ComplexT> >(times_);
897
898 SetRandVec<ComplexT>(static_cast<int>(start_point.size()));
899
900 auto init = this->ComputeInitialSamples(start_time, target_time, start_point, times, samples);
901 if (init != SuccessCode::Success) { NotifyObservers(EndgameFailure<EmitterType>(*this)); return init; }
902 if (this->GetTracker().GetCurrentPrecision() > this->current_endgame_precision_)
903 return SuccessCode::HigherPrecisionNecessary;
904
905 auto ref = this->template RefineAllSamples<ComplexT>(samples, times);
906 if (ref == SuccessCode::HigherPrecisionNecessary || ref == SuccessCode::FailedToConverge)
907 return SuccessCode::HigherPrecisionNecessary;
908 if (ref != SuccessCode::Success) return ref;
909
910 ComputeAllDerivatives<ComplexT>();
911
912 Vec<ComplexT> first_approx;
913 auto ext = ComputeApproximationOfXAtT0<ComplexT>(first_approx, target_time);
914 if (ext != SuccessCode::Success) return ext;
915 this->previous_approximation_ = this->ToBCT(first_approx, this->current_endgame_precision_);
916 return SuccessCode::Success;
917 }

1007 auto& rv = std::get<Vec<complex_mp> >(rand_vector_);
1008 if (rv.size() > 0) Precision(rv, newprec);
1009 }
1010 if (this->final_approximation_.size() > 0) Precision(this->final_approximation_, newprec);
1011 if (this->previous_approximation_.size() > 0) Precision(this->previous_approximation_, newprec);
