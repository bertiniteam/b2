165template<typename ComplexT>
166inline unsigned DefaultInitialAmbientPrecision()
167{
168 if constexpr (std::is_same<ComplexT, complex_dbl>::value)
169 return DoublePrecision();
170 else
171 return DefaultPrecision();
172}
173
174// Not templated on the complex type: the three times are stored as exact, precision-free
175// mpq_rational (real) and converted to the tracking complex type at the current working precision at
176// use -- the same pattern SteppingConfig uses for its step sizes. This keeps the config precision-
177// agnostic (one struct, not a DoublePrec/Multiprec pair), and is in fact more correct under adaptive
178// precision: a value like the endgame boundary 1/10 is materialized at full working precision rather
179// than frozen at whatever precision the config happened to be constructed at.
180//
181// The times are real because a zero-dim solve tracks along the real t-axis (1 -> 0). Complex
182// homotopy times remain available at the endgame level (set directly), just not through this config.
184struct ZeroDimConfig
185{
186 // Per-complex-type default; the ZeroDim algorithm overwrites this in DefaultSettingsSetup with
187 // DefaultInitialAmbientPrecision<BaseComplexT>() (it knows its tracking type, this struct does not).
188 unsigned initial_ambient_precision = DefaultPrecision();
189 unsigned max_num_crossed_path_resolve_attempts = 2;
190
195 unsigned num_threads = 0;
196
202 bool recall = true;

205 mpq_rational endgame_boundary{1, 10};

208 std::string path_variable_name = "ZERO_DIM_PATH_VARIABLE";
209};
