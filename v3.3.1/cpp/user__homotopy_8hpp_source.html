47// User homotopies use the RefToGiven policy (the user owns target/start/homotopy), so they
48// get their own tracker/endgame dispatch chain -- distinct from the generic ZeroDimSpecify*
49// ladder, which always clones. ts... = (target, start_system, homotopy).
50
51template <typename TrackerType, typename EndgameType, typename ... ConstTs>
52std::unique_ptr<algorithm::AnyZeroDim> UserHomSpecifyComplete(ConstTs const& ...ts)
53{
54 return std::make_unique<
55 algorithm::HomotopySolver<TrackerType, EndgameType, System>
56 >(ts...);
57}
58
59template <typename TrackerType, typename ... ConstTs>
60std::unique_ptr<algorithm::AnyZeroDim> UserHomSpecifyEndgame(ZeroDimRT const& rt, ConstTs const& ...ts)
61{
62 switch (rt.endgame)
63 {
64 case type::Endgame::PowerSeries:
65 return UserHomSpecifyComplete<TrackerType, typename endgame::EndgameSelector<TrackerType>::PSEG>(ts...);
66 case type::Endgame::Cauchy:
67 return UserHomSpecifyComplete<TrackerType, typename endgame::EndgameSelector<TrackerType>::Cauchy>(ts...);
68 }
69 throw std::runtime_error("unrecognized endgame type in UserHomSpecifyEndgame");
70}
71
72template <typename ... ConstTs>
73std::unique_ptr<algorithm::AnyZeroDim> UserHomSpecifyTracker(ZeroDimRT const& rt, ConstTs const& ...ts)
74{
75 switch (rt.tracker)
76 {
77 case type::Tracker::FixedDouble:
78 return UserHomSpecifyEndgame<tracking::DoublePrecisionTracker>(rt, ts...);
79 case type::Tracker::FixedMultiple:
80 return UserHomSpecifyEndgame<tracking::MultiplePrecisionTracker>(rt, ts...);
81 case type::Tracker::Adaptive:
82 return UserHomSpecifyEndgame<tracking::AMPTracker>(rt, ts...);
83 }
84 throw std::runtime_error("unrecognized tracker type in UserHomSpecifyTracker");
85}
86
87template <typename ... ConstTs>
88std::unique_ptr<algorithm::AnyZeroDim> MakeUserHom(ZeroDimRT const& rt, ConstTs const& ...ts)
89{
90 return UserHomSpecifyTracker(rt, ts...);
91}
