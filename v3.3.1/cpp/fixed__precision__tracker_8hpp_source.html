167 SuccessCode TrackerIteration() const override
168 {
169 static_assert(std::is_same< typename Eigen::NumTraits<RealT>::Real,
170 typename Eigen::NumTraits<ComplexT>::Real>::value,
171 "underlying complex type and the type for comparisons must match");
172
173 this->NotifyObservers(NewStep<EmitterType >(*this));
174
175 Vec<ComplexT>& predicted_space = std::get<Vec<ComplexT> >(this->temporary_space_); // this will be populated in the Predict step
176 Vec<ComplexT>& current_space = std::get<Vec<ComplexT> >(this->current_space_); // the thing we ultimately wish to update
177 ComplexT current_time = ComplexT(this->current_time_);
178 ComplexT delta_t = ComplexT(this->delta_t_);
179
180 SuccessCode predictor_code = Predict(predicted_space, current_space, current_time, delta_t);
181
182 if (predictor_code!=SuccessCode::Success)
183 {
184 this->NotifyObservers(FirstStepPredictorMatrixSolveFailure<EmitterType >(*this));
185
186 this->next_stepsize_ = RealT(Get<Stepping>().step_size_fail_factor)*this->current_stepsize_;
187
188 UpdateStepsize();
189
190 return predictor_code;
191 }
192
193 this->NotifyObservers(SuccessfulPredict<EmitterType , ComplexT>(*this, predicted_space));
194
195 Vec<ComplexT>& tentative_next_space = std::get<Vec<ComplexT> >(this->tentative_space_); // this will be populated in the Correct step
196
197 ComplexT tentative_next_time = current_time + delta_t;
198
199 SuccessCode corrector_code = Correct(tentative_next_space,
200 predicted_space,
201 tentative_next_time);
202
203 if (corrector_code == SuccessCode::GoingToInfinity)
204 {
205 // there is no corrective action possible...
206 return corrector_code;
207 }
208 else if (corrector_code!=SuccessCode::Success)
209 {
210 this->NotifyObservers(CorrectorMatrixSolveFailure<EmitterType >(*this));
211
212 this->next_stepsize_ = RealT(Get<Stepping>().step_size_fail_factor)*this->current_stepsize_;
213 UpdateStepsize();
214
215 return corrector_code;
216 }
217
218
219 this->NotifyObservers(SuccessfulCorrect<EmitterType , ComplexT>(*this, tentative_next_space));
220
221 // copy the tentative vector into the current space vector;
222 current_space = tentative_next_space;
223 return SuccessCode::Success;
224 }

568 SuccessCode TrackerLoopInitialization(BaseComplexT const& start_time,
569 BaseComplexT const& end_time,
570 Vec<BaseComplexT> const& start_point) const override
571 {
572
573 // ThreadPrecision: thread-local read, correct both on the main thread
574 // (where it equals the global) and on std::thread workers.
575 if (start_point(0).precision()!=ThreadPrecision())
576 {
577 std::stringstream err_msg;
578 err_msg << "start point for fixed multiple precision tracker has differing precision from default (" << start_point(0).precision() << "!=" << ThreadPrecision() << "), tracking cannot start";
579 throw std::runtime_error(err_msg.str());
580 }
581
582 if (start_point(0).precision()!=this->CurrentPrecision())
583 {
584 std::stringstream err_msg;
585 err_msg << "start point for fixed multiple precision tracker has differing precision from tracker's precision (" << start_point(0).precision() << "!=" << this->CurrentPrecision() << "), tracking cannot start";
586 throw std::runtime_error(err_msg.str());
587 }
588
589 if (ThreadPrecision()!=this->CurrentPrecision())
590 {
591 std::stringstream err_msg;
592 err_msg << "current default precision differs from tracker's precision (" << ThreadPrecision() << "!=" << this->CurrentPrecision() << "), tracking cannot start";
593 throw std::runtime_error(err_msg.str());
594 }
595
596
597 this->NotifyObservers(Initializing<EmitterType,BaseComplexT>(*this,start_time, end_time, start_point));
598
599 // Reset precision of all persistent members before assignment so that
600 // BMP preserve_related_precision doesn't propagate stale high precision
601 // from a previous TrackPath call into the new one.
602 this->current_time_.precision(precision_);
603 this->endtime_.precision(precision_);
604 this->delta_t_.precision(precision_);
605 this->current_stepsize_.precision(precision_);
606 this->next_stepsize_.precision(precision_);
607 Precision(std::get<Vec<BaseComplexT>>(this->current_space_), precision_);
608 Precision(std::get<Vec<BaseComplexT>>(this->temporary_space_), precision_);
609 Precision(std::get<Vec<BaseComplexT>>(this->tentative_space_), precision_);
610 this->predictor_.ChangePrecision(precision_);
611 this->corrector_.ChangePrecision(precision_);
612
613 // set up the master current time and the current step size
614 this->current_time_ = start_time;
615 this->endtime_ = end_time;
616 std::get<Vec<BaseComplexT> >(this->current_space_) = start_point;
617 if (this->reinitialize_stepsize_)
618 this->SetStepSize(min(real_mp(Get<Stepping>().initial_step_size),real_mp(abs(start_time-end_time)/Get<Stepping>().min_num_steps)));
619
620 ResetCounters();
621
622 return SuccessCode::Success;
623 }
624
626 bool PrecisionSanityCheck() const
627 {
628 return GetSystem().precision() == precision_ &&
629 ThreadPrecision()==precision_ &&
630 std::get<Vec<complex_mp> >(current_space_)(0).precision() == precision_ &&
631 std::get<Vec<complex_mp> >(tentative_space_)(0).precision() == precision_ &&
632 std::get<Vec<complex_mp> >(temporary_space_)(0).precision() == precision_ &&
633 Precision(this->endtime_)==precision_
634 ;
635 }

640 void SyncPrecisionToConfig()
641 {
642 FixedPrecisionConfig c = this->template Get<FixedPrecisionConfig>();
643 c.precision = precision_;
644 this->template Set<FixedPrecisionConfig>(c);
645 }
