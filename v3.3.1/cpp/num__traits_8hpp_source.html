225 inline
226 complex_dbl rand_complex()
227 {
228 using std::abs;
229 using std::sqrt;
230 // DRAW ORDER IS A CONTRACT (cross-platform reproducibility, 2026-07-03): the two
231 // component draws are sequenced EXPLICITLY -- real first, then imaginary. Never
232 // put two draws in one full-expression: C++ argument evaluation order is
233 // unspecified, and gcc really does order them differently on x86_64 vs aarch64
234 // (found as an architecture-split seeded-homotopy digest: every (re, im) pair of
235 // every seeded coefficient was transposed between the two).
236 double const re = records::DrawSymmetricDouble();
237 double const im = records::DrawSymmetricDouble();
238 complex_dbl returnme(re, im);
239 return returnme / sqrt( abs(returnme));
240 }

377 inline static
378 mpq_rational FromString(std::string const& str)
379 {
380 std::string s = str;
381 bool negative = false;
382 if (!s.empty() && s[0] == '-') { negative = true; s = s.substr(1); }
383 else if (!s.empty() && s[0] == '+') { s = s.substr(1); }
384
385 int exp_shift = 0;
386 auto e_pos = s.find_first_of("eE");
387 if (e_pos != std::string::npos) {
388 exp_shift = std::stoi(s.substr(e_pos + 1));
389 s = s.substr(0, e_pos);
390 }
391
392 auto dot_pos = s.find('.');
393 int decimal_places = 0;
394 if (dot_pos != std::string::npos) {
395 decimal_places = static_cast<int>(s.size()) - static_cast<int>(dot_pos) - 1;
396 s.erase(dot_pos, 1);
397 }
398
399 if (s.empty() || s.find_first_not_of('0') == std::string::npos) {
400 s = "0";
401 } else {
402 s = s.substr(s.find_first_not_of('0'));
403 }
404 mpz_int numer(s);
405 if (negative) numer = -numer;
406
407 int net_exp = decimal_places - exp_shift;
408 if (net_exp > 0) {
409 mpz_int denom = 1;
410 for (int i = 0; i < net_exp; ++i) denom *= 10;
411 return mpq_rational(numer, denom);
412 } else if (net_exp < 0) {
413 mpz_int mult = 1;
414 for (int i = 0; i < -net_exp; ++i) mult *= 10;
415 return mpq_rational(numer * mult, 1);
416 } else {
417 return mpq_rational(numer, 1);
418 }
419 }
420 };
