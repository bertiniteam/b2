64 using blros = blog::record_ostream;
66 template<typename T>
67 blros& operator<<(typename std::enable_if<std::is_enum<T>::value, blros>::type& stream, const T& e)
68 {
69 return stream << static_cast<typename std::underlying_type<T>::type>(e);
70 }
