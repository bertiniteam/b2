139 // Declaration blocks may appear in any number and any order, which is what
140 // Bertini 1 input looks like. What IS enforced: a second `pathvariable` is
141 // refused here, and a name declared but never defined is reported by name from
142 // EmitDeclaredFunctions once the whole input has been read (it cannot be caught
143 // earlier, since the defining line may come anywhere after the declaration).
144 // Deliberately not enforced: a definition for a name that was never declared is
145 // absorbed as a subfunction, which is what Bertini 1 does.
146 root_rule_.name("system_parsing");
147 root_rule_ =
148 *(
149 variable_group_ [phx::bind(&System::AddVariableGroup, _val, _1)]
150 |
151 hom_variable_group_ [phx::bind(&System::AddHomVariableGroup, _val, _1)]
152 |
153 variables_ [phx::bind(&System::AddUngroupedVariables, _val, _1)]
154 |
155 functions_ [phx::bind([this](std::vector<std::string> const& names){ this->CollectFunctionNames(names); }, _1)]
156 |
157 constants_ [phx::bind([this](std::vector<std::string> const& names){ this->CollectConstantNames(names); }, _1)]
158 |
159 parameters_ [phx::bind([this](std::vector<std::string> const& names){ this->CollectParameterNames(names); }, _1)]
160 |
161 implicit_parameters_ [phx::bind(&System::AddImplicitParameters, _val, _1)]
162 |
163 path_variable_ [phx::bind([this](System& s, Var const& v){ this->AddPathVariableChecked(s, v); }, _val, _1)]
164 |
165 // definition_ before subfunction_: a declared name (function/constant/
166 // parameter) lives only in encountered_functions_, not encountered_symbols_,
167 // so its definition line would otherwise be misread as a fresh subfunction.
168 // definition_ matches declared LHS names; subfunction_ catches the rest.

354 void MakeAndDeclareName(std::string & name, std::string str)
355 {
356 // A declare-then-define name (function / constant / parameter) is just a
357 // marker until its definition is parsed: record the name in
358 // encountered_functions_ so the definition_ rule can match its LHS. We do
359 // NOT add it to encountered_symbols_ yet -- the name binds to its
360 // NamedExpression only once defined (DefineNamed), at which point references
361 // to it resolve. (Define-before-use is assumed.)
362 name = str;
363 encountered_functions_.add(str, str);
364 }

370 void MakeAndAddVariable(Var & V, std::string str)
371 {
372 V = Variable::Make(str);
373 encountered_symbols_.add(str, V);
374 }

376 // Bertini-1 input declares the declare-then-define kinds first ("function f, g;",
377 // "constant c;", "parameter p;"), then defines them (and any inline subfunctions)
378 // later. We record the declared names per kind, and bind each name to an immutable
379 // NamedExpression when its definition is parsed (DefineNamed). After the whole input
380 // is parsed we emit them to the System (EmitDeclaredFunctions): the System receives
381 // bare, fully-built expressions -- no Function declaration box is ever created.
382 public:
384 void CollectFunctionNames(std::vector<std::string> const& names)
385 { for (auto const& n : names) declared_function_names_.push_back(n); }
386
387 void CollectConstantNames(std::vector<std::string> const& names)
388 { for (auto const& n : names) declared_constant_names_.push_back(n); }
389
390 void CollectParameterNames(std::vector<std::string> const& names)
391 { for (auto const& n : names) declared_parameter_names_.push_back(n); }
392
393 // Bind a declared name to its (immutable) NamedExpression: makes references to the
394 // name resolve, and records the definition for emission to the System.
396 void DefineNamed(std::string const& name, Nd const& expr)
397 {
398 auto ne = node::NamedExpression::Make(expr, name);
399 encountered_symbols_.add(name, ne);
400 definitions_[name] = ne;
401 }

411 void AddPathVariableChecked(System& s, Var const& v)
412 {
413 if (have_path_variable_declaration_)
414 throw std::runtime_error(
415 "[SystemParser] more than one `pathvariable` declared: a system has "
416 "exactly one path variable. Declare only the one you mean.");
417 have_path_variable_declaration_ = true;
418 s.AddPathVariable(v);
419 }
420
428 void EmitDeclaredFunctions(System& s) const
429 {
430 for (auto const& n : declared_function_names_)
431 s.AddFunction(DefinitionOf(n, "function")->EntryNode());
432 for (auto const& n : declared_constant_names_)
433 s.AddConstant(DefinitionOf(n, "constant"));
434 for (auto const& n : declared_parameter_names_)
435 s.AddParameter(DefinitionOf(n, "parameter"));
436 }

438 private:
443 std::shared_ptr<node::NamedExpression> const& DefinitionOf(std::string const& name,
444 std::string const& kind) const
445 {
446 auto const found = definitions_.find(name);
447 if (found == definitions_.end())
448 throw std::runtime_error(
449 "[SystemParser] " + kind + " `" + name + "` was declared but never defined. "
450 "Every declared name needs a defining line, as in `" + name + " = ...;`.");
451 return found->second;
452 }
453
454 bool have_path_variable_declaration_ = false;
455 std::vector<std::string> declared_function_names_, declared_constant_names_, declared_parameter_names_;
456 std::map<std::string, std::shared_ptr<node::NamedExpression>> definitions_;
457 };
