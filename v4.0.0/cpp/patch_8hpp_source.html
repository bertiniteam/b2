101 Patch(Patch const& other)
102 {
103 variable_group_sizes_ = other.variable_group_sizes_;
104 precision_ = DefaultPrecision();
105
106 // a little shorthand unpacking the tuple
107 std::vector<Vec<complex_mp> >& coefficients_mpfr = std::get<std::vector<Vec<complex_mp> > >(this->coefficients_working_);
108 std::vector<Vec<complex_dbl> >& coefficients_dbl = std::get<std::vector<Vec<complex_dbl> > >(this->coefficients_working_);
109
110 coefficients_highest_precision_.resize(other.NumVariableGroups());
111 coefficients_mpfr.resize(variable_group_sizes_.size());
112 coefficients_dbl.resize(variable_group_sizes_.size());
113
114 for (unsigned ii(0); ii<other.NumVariableGroups(); ++ii)
115 {
116 auto curr_size = variable_group_sizes_[ii];
117
118 coefficients_highest_precision_[ii].resize(curr_size);
119 coefficients_dbl[ii].resize(curr_size);
120 coefficients_mpfr[ii].resize(curr_size);
121
122 for (unsigned jj(0); jj<curr_size; jj++)
123 {
124 coefficients_highest_precision_[ii](jj).precision(other.coefficients_highest_precision_[ii](jj).precision());
125
126 coefficients_highest_precision_[ii](jj) = other.coefficients_highest_precision_[ii](jj);
127
128 coefficients_dbl[ii](jj) = complex_dbl(coefficients_highest_precision_[ii](jj));
129 coefficients_mpfr[ii](jj) = complex_mp(coefficients_highest_precision_[ii](jj));
130
131 assert(coefficients_highest_precision_[ii](jj) == other.coefficients_highest_precision_[ii](jj));
132 }
133 }
134 }

152 Patch(std::vector<unsigned> const& sizes) : coefficients_highest_precision_(sizes.size()), variable_group_sizes_(sizes), precision_(DefaultPrecision())
153 {
154 using bertini::Precision;
155 using bertini::multiprecision::RandomComplex;
156
157 std::vector<Vec<complex_mp> >& coefficients_mpfr = std::get<std::vector<Vec<complex_mp> > >(coefficients_working_);
158 std::vector<Vec<complex_dbl> >& coefficients_dbl = std::get<std::vector<Vec<complex_dbl> > >(coefficients_working_);
159
160 coefficients_highest_precision_.resize(sizes.size());
161 coefficients_dbl.resize(sizes.size());
162 coefficients_mpfr.resize(sizes.size());
163
164 for (size_t ii=0; ii<sizes.size(); ++ii)
165 {
166 // this produces coefficients at maximum precision.
167 coefficients_highest_precision_[ii].resize(sizes[ii]);
168 for (unsigned jj=0; jj<sizes[ii]; ++jj)
169 {
170 // bounded-modulus draw (away from 0 and infinity), matching the linear-product /
171 // mhom / binomial start-system coefficients -- a heavy-tailed coefficient here scales
172 // the patch equation badly and feeds the same near-t=0 conditioning trouble.
173 multiprecision::RandomComplexBoundedModulusAssign(coefficients_highest_precision_[ii](jj), MaxPrecisionAllowed());
174 }
175
176 coefficients_dbl[ii].resize(sizes[ii]);
177 for (unsigned jj=0; jj<sizes[ii]; ++jj)
178 {
179 coefficients_dbl[ii](jj) = complex_dbl(coefficients_highest_precision_[ii](jj));
180 }
181
182 // assignment preserves precision of source.
183 // https://github.com/boostorg/multiprecision/issues/75
184 coefficients_mpfr[ii].resize(sizes[ii]);
185 for (unsigned jj=0; jj<sizes[ii]; ++jj)
186 {
187 coefficients_mpfr[ii](jj) = coefficients_highest_precision_[ii](jj);
188 coefficients_mpfr[ii](jj).precision(precision_);
189 }
190
191 assert(Precision(coefficients_mpfr[ii](0))==precision_);
192 }
193 }

204 Patch(std::vector<unsigned> const& sizes, std::vector<Vec<complex_mp>> const& coefficients)
205 : variable_group_sizes_(sizes), precision_(DefaultPrecision())
206 {
207 if (coefficients.size() != sizes.size())
208 throw std::invalid_argument("Patch: " + std::to_string(coefficients.size())
209 + " coefficient vectors for " + std::to_string(sizes.size()) + " variable groups");
210
211 std::vector<Vec<complex_mp> >& coefficients_mpfr = std::get<std::vector<Vec<complex_mp> > >(coefficients_working_);
212 std::vector<Vec<complex_dbl> >& coefficients_dbl = std::get<std::vector<Vec<complex_dbl> > >(coefficients_working_);
213
214 coefficients_highest_precision_.resize(sizes.size());
215 coefficients_dbl.resize(sizes.size());
216 coefficients_mpfr.resize(sizes.size());
217
218 for (size_t ii=0; ii<sizes.size(); ++ii)
219 {
220 if (static_cast<size_t>(coefficients[ii].size()) != sizes[ii])
221 throw std::invalid_argument("Patch: group " + std::to_string(ii) + " has "
222 + std::to_string(coefficients[ii].size()) + " coefficients for "
223 + std::to_string(sizes[ii]) + " variables");
224
225 coefficients_highest_precision_[ii].resize(sizes[ii]);
226 coefficients_dbl[ii].resize(sizes[ii]);
227 coefficients_mpfr[ii].resize(sizes[ii]);
228 for (unsigned jj=0; jj<sizes[ii]; ++jj)
229 {
230 // set the precision first so the copy is exact at the source's precision
231 coefficients_highest_precision_[ii](jj).precision(coefficients[ii](jj).precision());
232 coefficients_highest_precision_[ii](jj) = coefficients[ii](jj);
233 coefficients_dbl[ii](jj) = complex_dbl(coefficients_highest_precision_[ii](jj));
234 coefficients_mpfr[ii](jj) = coefficients_highest_precision_[ii](jj);
235 coefficients_mpfr[ii](jj).precision(precision_);
236 }
237 }

253 static Patch RandomReal(std::vector<unsigned> const& sizes)
254 {
255 using bertini::multiprecision::RandomReal;
256 using bertini::Precision;
257
258 Patch p;
259
260 p.variable_group_sizes_ = sizes;
261
262
263
264 std::vector<Vec<complex_mp> >& coefficients_mpfr = std::get<std::vector<Vec<complex_mp> > >(p.coefficients_working_);
265 std::vector<Vec<complex_dbl> >& coefficients_dbl = std::get<std::vector<Vec<complex_dbl> > >(p.coefficients_working_);
266
267 p.coefficients_highest_precision_.resize(sizes.size());
268 coefficients_mpfr.resize(sizes.size());
269 coefficients_dbl.resize(sizes.size());
270
271 for (size_t ii=0; ii<sizes.size(); ++ii)
272 {
273 p.coefficients_highest_precision_[ii].resize(sizes[ii]);
274 for (unsigned jj=0; jj<sizes[ii]; ++jj)
275 // real bounded-modulus draw: away from 0 and infinity (same recipe as the complex
276 // patch / start systems) but kept REAL, so a real patch keeps a real path real.
277 multiprecision::RandomRealBoundedModulusAssign(p.coefficients_highest_precision_[ii](jj), MaxPrecisionAllowed());
278
279 coefficients_mpfr[ii] = p.coefficients_highest_precision_[ii];
280 Precision(coefficients_mpfr[ii],DefaultPrecision());
281 assert(Precision(coefficients_mpfr[ii](0))==DefaultPrecision());
282
283 coefficients_dbl[ii].resize(sizes[ii]);
284 for (unsigned jj=0; jj<sizes[ii]; ++jj)
285 coefficients_dbl[ii](jj) = complex_dbl(p.coefficients_highest_precision_[ii](jj));
286 }
287
288 return p;
289 }

327 void Precision(unsigned new_precision) const
328 {
329 if (precision_==new_precision)
330 return;
331
332 using bertini::Precision;
333 std::vector<Vec<complex_mp> >& coefficients_mpfr = std::get<std::vector<Vec<complex_mp> > >(coefficients_working_);
334
335 for (unsigned ii = 0; ii < NumVariableGroups(); ++ii)
336 {
337 for (unsigned jj=0; jj<variable_group_sizes_[ii]; ++jj)
338 {
339 if (new_precision>precision_)
340 {
341 coefficients_mpfr[ii](jj) = coefficients_highest_precision_[ii](jj);
342 coefficients_mpfr[ii](jj).precision(new_precision);
343 }
344 else
345 coefficients_mpfr[ii](jj).precision(new_precision);
346
347 assert(Precision(coefficients_mpfr[ii](jj))==new_precision);
348 }
349 }
350
351 precision_ = new_precision;
352 }

369 template<typename Derived, typename T>
370 void EvalInPlace(Eigen::MatrixBase<Derived> & function_values, Vec<T> const& x) const
371 {
372 static_assert(std::is_same<typename Derived::Scalar,T>::value,"scalar types must match");
373
374 SyncPrecision(x);
375
376 #ifndef BERTINI_DISABLE_ASSERTS
377 if (! (function_values.size()>=NumVariableGroups()) )
378 {
379 std::stringstream ss;
380 ss << "container for function values must be of length at least as long as the number of variable groups. the input vector into which to write is of length " << function_values.size();
381 throw std::runtime_error(ss.str());
382 }
383 // Formerly an assert that the input's precision MATCHED the patch's -- disabled long
384 // ago because it could not be honoured. A patch is an evaluable type, so it now
385 // self-aligns instead of demanding to be aligned for: ADR-0057.
386// );
387 #endif
388
389 // unpack from the tuple of working coefficients
390 const std::vector<Vec<T> >& coefficients = std::get<std::vector<Vec<T> > >(coefficients_working_);
391
392 unsigned offset(static_cast<unsigned>(function_values.size() - NumVariableGroups())); // by precondition this number is at least 0. the precondition is ensured by the public wrapper
393 unsigned counter(0);
394 for (unsigned ii = 0; ii < NumVariableGroups(); ++ii)
395 {
396 T& value = function_values(ii+offset);
397 value = T(-1);
398 for (unsigned jj=0; jj<variable_group_sizes_[ii]; ++jj)
399 {
400 value += x(counter)*coefficients[ii](jj);
401 counter++;
402 }
403 }
404 }

416 EvalInPlace(function_values, x);

430 template<typename Derived, typename T>
431 void JacobianInPlace(Eigen::MatrixBase<Derived> & jacobian, Vec<T> const& x) const
432 {
433 static_assert(std::is_same<typename Derived::Scalar,T>::value,"scalar types must match");
434
435 SyncPrecision(x); // the jacobian is constant in x, but it is written FROM the
436 // working coefficients, so they must be at x's precision
437
438
439 #ifndef BERTINI_DISABLE_ASSERTS
440 assert(jacobian.rows()>=NumVariableGroups() && "input jacobian must have at least as many rows as variable groups");
441 assert(jacobian.cols()==NumVariables() && "input jacobian must have as many columns as the patch has variables");
442 assert(
443 (bertini::Precision(x(0))==DoublePrecision() || bertini::Precision(x(0)) == Precision())
444 && "precision of input vector must match current working precision of patch during evaluation"
445 );
446 #endif
447
448 const std::vector<Vec<T> >& coefficients = std::get<std::vector<Vec<T> > >(coefficients_working_);
449
450 unsigned offset(static_cast<unsigned>(jacobian.rows() - NumVariableGroups())); // by precondition this number is at least 0. the precondition is ensured by the public wrapper
451
452 // A patch row is sparse -- one block of coefficients per variable group, zero elsewhere.
453 // Zero the rows the patch owns before writing those coefficients, so the patch FULLY
454 // defines its own rows and the caller need not pre-zero the matrix. The block-composed
455 // Jacobian path allocates J uninitialized and assigns only the block (function) rows;
456 // without this the patch rows' off-coefficient entries are read uninitialized -- benign
457 // (zeroed pages) on Linux/macOS, but garbage on Windows, where a degree-2 homotopy's
458 // Jacobian "evaluated" to ~1e252 and wrecked the AMP condition-number estimate.
459 jacobian.block(offset, 0, NumVariableGroups(), jacobian.cols()).setZero();
460
461 unsigned counter(0);
462 for (unsigned ii = 0; ii < NumVariableGroups(); ++ii)
463 for (unsigned jj=0; jj<variable_group_sizes_[ii]; ++jj)
464 jacobian(ii+offset,counter++) = coefficients[ii](jj);
465 }

492 template<typename T>
493 void RescalePointToFitInPlace(Vec<T> & x) const
494 {
495 // Self-align to the point, exactly as EvalInPlace and the Jacobian do -- ADR-0057.
496 // Without this, rescaling read whatever precision the working coefficients happened to
497 // be left at by the last evaluation, and the rescaled point came back REPORTING the
498 // requested precision while carrying only the older one's correct digits. Nothing
499 // downstream could see it: the assert that used to stand here demanded the caller align
500 // the precisions beforehand, and asserts are compiled out of a release build, so the
501 // one check that could have caught it was absent exactly where it mattered (b2#461).
502 SyncPrecision(x);
503
504 #ifndef BERTINI_DISABLE_ASSERTS
505 assert(x.size() == NumVariables() && "input point for rescaling to fit a patch must have same length as total number of variables being patched, in all variable groups.");
506 #endif
507
508 const std::vector<Vec<T> >& coefficients = std::get<std::vector<Vec<T> > >(coefficients_working_);
509
510 unsigned starting_index_counter(0);
511 for (unsigned ii=0; ii<NumVariableGroups(); ii++)
512 {
513 auto subvec = x.segment(starting_index_counter,variable_group_sizes_[ii]);
514 subvec /= subvec.transpose() * coefficients[ii];
515 starting_index_counter += variable_group_sizes_[ii];
516 }
517 }
