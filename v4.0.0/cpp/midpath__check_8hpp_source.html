159 template <typename StartSystemT>
160 bool Check(BoundaryData const& boundary_data, StartSystemT const& start_system)
161 {
162 // Each Check is independent: the algorithm calls this repeatedly (once before any
163 // resolve, then again after each MidpathResolve). Reset the accumulated state so a
164 // later call reports *this* call's crossings, not the union with prior calls.
165 // Without this, passed_ (only ever set false below) would stay false forever once
166 // any crossing was seen, and crossed_paths_ would grow without bound -- the resolve
167 // loop in EGBoundaryAction would then never see a clean pass and would re-track
168 // stale indices.
169 passed_ = true;
170 crossed_paths_.clear();
171
172 for (PathIndT ii = 0; ii < boundary_data.size(); ++ii)
173 {
174 if ( boundary_data[ii].success_code != SuccessCode::Success)
175 continue;
176
177 const Vec<ComplexT>& solution_ii = boundary_data[ii].path_point;
178 const auto start_ii = start_system.template StartPoint<ComplexT>(ii);
179
180 for (PathIndT jj = ii+1; jj < boundary_data.size(); ++jj)
181 {
182 if ( boundary_data[jj].success_code != SuccessCode::Success)
183 continue;
184
185 const Vec<ComplexT>& solution_jj = boundary_data[jj].path_point;
186 const Vec<ComplexT> diff_sol = solution_ii - solution_jj;
187
188 if ((diff_sol.template lpNorm<Eigen::Infinity>()/solution_ii.template lpNorm<Eigen::Infinity>()) < SamePointTol())
189 {
190 bool i_already_stored = false;
191 bool j_already_stored = false;
192 // Check if start points are the same
193
194 const auto start_jj = start_system.template StartPoint<ComplexT>(jj);
195 auto diff_start = start_ii - start_jj;
196 // same_start is true when the two paths *began at the same start point*
197 // (their start points differ by less than the tolerance). That governs
198 // whether this coincidence is a genuine path crossing worth re-tracking:
199 // CrossedPath sets rerun_ = !same_start, so two *distinct* starts that
200 // have collided (same_start == false) get re-tracked, while two paths
201 // that legitimately share a start point do not. NOTE: this comparison
202 // was historically inverted (>), which flipped rerun and meant genuine
203 // crossings were never re-tracked.
204 bool same_start = (diff_start.template lpNorm<Eigen::Infinity>() < SamePointTol());
205
206
207 // Check if path has already been stored in crossed_paths_
208 for(auto& v : crossed_paths_)
209 {
210 if(v.index() == ii)
211 {
212 v.crossed_with(std::make_pair(jj,same_start));
213 i_already_stored = true;
214 v.rerun(same_start);
215 }
216 if(v.index() == jj)
217 {
218 v.crossed_with(std::make_pair(ii,same_start));
219 j_already_stored = true;
220 v.rerun(same_start);
221 }
222 }
223
224 // If not already stored, create CrossedPath object and add to crossed_paths_
225 if(!i_already_stored)
226 {
227 CrossedPath tempPath(ii, jj, same_start);
228 crossed_paths_.push_back(tempPath);
229 }
230 if(!j_already_stored)
231 {
232 CrossedPath tempPath(jj, ii, same_start);
233 crossed_paths_.push_back(tempPath);
234 }
235
236 passed_ = false;
237
238 }
239 }
240 }
241 return passed_;
242 };
