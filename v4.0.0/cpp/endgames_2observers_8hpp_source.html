48template <typename EndgameT>
49struct GoryDetailLogger : public Observer<EndgameT>
50{BOOST_TYPE_INDEX_REGISTER_CLASS
51
52using EmitterT = EndgameT;
53using BCT = typename EndgameT::BaseComplexT;
54
55virtual ~GoryDetailLogger() = default;
56
57virtual ObserveResult Observe(AnyEvent const& e) override
58{
59 if(auto p = dynamic_cast<const TimeAdvanced<EmitterT>*>(&e))
60 {
61 BOOST_LOG_TRIVIAL(severity_level::debug) << "time advanced " << p->Get().LatestTime();
62 }
63
64 else if (auto p = dynamic_cast<const SampleRefined<EmitterT>*>(&e))
65 {
66 BOOST_LOG_TRIVIAL(severity_level::debug) << "refined a sample, huzzah";
67 }
68
69 else if (auto p = dynamic_cast<const CircleAdvanced<EmitterT>*>(&e))
70 {
71 BOOST_LOG_TRIVIAL(severity_level::debug) << "advanced around the circle, to " << p->NewSample()<< " at time " << p->NewTime();
72 }
73
74 else if (auto p = dynamic_cast<const ClosedLoop<EmitterT>*>(&e))
75 {
76 BOOST_LOG_TRIVIAL(severity_level::debug) << "closed a loop, cycle number " << p->Get().CycleNumber();
77 }
78 else if (auto p = dynamic_cast<const ApproximatedRoot<EmitterT>*>(&e))
79 {
80 BOOST_LOG_TRIVIAL(severity_level::debug) << "approximated the target root. approximation " << p->Get().template FinalApproximation<BCT>() << " with error " << p->Get().ApproximateError();
81 }
82
83 else if (auto p = dynamic_cast<const PrecisionChanged<AMPEndgame>*>(&e))
84 {
85 BOOST_LOG_TRIVIAL(severity_level::debug) << "precision changed from " << p->Previous() << " to " << p->Next();
86 }
87
88 else if (auto p = dynamic_cast<const InEGOperatingZone<EmitterT>*>(&e))
89 {
90 BOOST_LOG_TRIVIAL(severity_level::debug) << "made it to the endgame operating zone at time " << p->Get().LatestTime();
91 }
92
93 else if(auto p = dynamic_cast<const Converged<EmitterT>*>(&e))
94 {
95 BOOST_LOG_TRIVIAL(severity_level::debug) << "converged at time " << p->Get().LatestTime() << " with result " << p->Get().template FinalApproximation<BCT>() << " and residual " << p->Get().ApproximateError();
96 }
97 else if (auto p = dynamic_cast<const Initializing<EmitterT>*>(&e))
98 {
99 BOOST_LOG_TRIVIAL(severity_level::debug) << "starting running " << boost::typeindex::type_id<EmitterT>().pretty_name();
100 }
101 else
102 {
103 BOOST_LOG_TRIVIAL(severity_level::debug) << "unprogrammed response for event of type " << boost::typeindex::type_id_runtime(e).pretty_name();
104 }
105
106 return ObserveResult::KeepObserving;
107}
108
109}; // gory detail

124template <typename EndgameT>
125struct EventRecorder : public Observer<EndgameT>
126{BOOST_TYPE_INDEX_REGISTER_CLASS
127
128using EmitterT = EndgameT;
129using BCT = typename EndgameT::BaseComplexT;
130
131unsigned num_events = 0;
132unsigned num_time_advanced = 0;
133unsigned num_sample_refined = 0;
134unsigned num_circle_advanced = 0;
135unsigned num_closed_loop = 0;
136unsigned num_approximated_root = 0;
137unsigned num_in_eg_zone = 0;
138unsigned num_converged = 0;
139unsigned num_precision_changed = 0;
140
141Vec<BCT> last_circle_point;
142BCT last_circle_time;
143Vec<BCT> converged_point;
144
145virtual ObserveResult Observe(AnyEvent const& e) override
146{
147 ++num_events;
148 if (auto p = dynamic_cast<const TimeAdvanced<EmitterT>*>(&e))
149 { ++num_time_advanced; (void)p->Get().LatestTime(); }
150
151 else if (dynamic_cast<const SampleRefined<EmitterT>*>(&e))
152 { ++num_sample_refined; }
153
154 else if (auto p = dynamic_cast<const CircleAdvanced<EmitterT>*>(&e))
155 { ++num_circle_advanced; last_circle_point = p->NewSample(); last_circle_time = p->NewTime(); }
156
157 else if (dynamic_cast<const ClosedLoop<EmitterT>*>(&e))
158 { ++num_closed_loop; }
159
160 else if (auto p = dynamic_cast<const ApproximatedRoot<EmitterT>*>(&e))
161 { ++num_approximated_root; (void)p->Get().template FinalApproximation<BCT>(); (void)p->Get().ApproximateError(); }
162
163 else if (dynamic_cast<const InEGOperatingZone<EmitterT>*>(&e))
164 { ++num_in_eg_zone; }
165
166 else if (auto p = dynamic_cast<const Converged<EmitterT>*>(&e))
167 { ++num_converged; converged_point = p->Get().template FinalApproximation<BCT>(); (void)p->Get().LatestTime(); }
168
169 else if (dynamic_cast<const PrecisionChanged<AMPEndgame>*>(&e))
170 { ++num_precision_changed; }
171
172 return ObserveResult::KeepObserving;
173}
174
175}; // EventRecorder

211template <typename EndgameT>
212struct SampleSequenceCollector : public Observer<EndgameT>
213{BOOST_TYPE_INDEX_REGISTER_CLASS
214
215using EmitterT = EndgameT;
216using BCT = typename EndgameT::BaseComplexT;
217
218std::vector<Vec<BCT>> path_samples;
219std::vector<BCT> path_times;
220
221std::vector<Vec<BCT>> circle_samples;
222std::vector<BCT> circle_times;
223
224std::vector<Vec<BCT>> approximations;
225std::vector<BCT> approximation_times;
226std::vector<NumErrorT> approximation_errors;
227std::vector<unsigned> cycle_numbers;
228
229std::vector<BCT> advance_times;
230
231// Run boundaries. One entry per endgame Run, holding the size each bucket had when that
232// run began -- so a collector attached to a whole solve can still tell one path from the
233// next. Samples of run j are path_samples[run_path_starts[j] .. run_path_starts[j+1]).
234std::vector<size_t> run_path_starts;
235std::vector<size_t> run_circle_starts;
236std::vector<size_t> run_approx_starts;
237
238size_t num_precision_increases = 0;
239
241void Clear()
242{
243 path_samples.clear(); path_times.clear();
244 circle_samples.clear(); circle_times.clear();
245 approximations.clear(); approximation_times.clear();
246 approximation_errors.clear(); cycle_numbers.clear();
247 advance_times.clear();
248 run_path_starts.clear(); run_circle_starts.clear();
249 run_approx_starts.clear();
250 num_precision_increases = 0;
251}
252
254size_t NumRuns() const { return run_path_starts.size(); }
255
257size_t NumSamples() const { return path_samples.size(); }
258
259virtual ObserveResult Observe(AnyEvent const& e) override
260{
261 if (auto p = dynamic_cast<const ComputedSamplePoint<EmitterT>*>(&e))
262 {
263 path_samples.push_back(p->NewSample());
264 path_times.push_back(p->NewTime());
265 }
266
267 else if (auto p = dynamic_cast<const CircleAdvanced<EmitterT>*>(&e))
268 {
269 circle_samples.push_back(p->NewSample());
270 circle_times.push_back(p->NewTime());
271 }
272
273 else if (auto p = dynamic_cast<const ApproximatedRoot<EmitterT>*>(&e))
274 {
275 approximations.push_back(p->Get().template FinalApproximation<BCT>());
276 approximation_times.push_back(p->Get().LatestTime());
277 approximation_errors.push_back(p->Get().ApproximateError());
278 cycle_numbers.push_back(p->Get().CycleNumber());
279 }
280
281 else if (auto p = dynamic_cast<const TimeAdvanced<EmitterT>*>(&e))
282 {
283 advance_times.push_back(p->Get().LatestTime());
284 }
285
286 else if (dynamic_cast<const Initializing<EmitterT>*>(&e))
287 {
288 run_path_starts.push_back(path_samples.size());
289 run_circle_starts.push_back(circle_samples.size());
290 run_approx_starts.push_back(approximations.size());
291 }
292
293 else if (dynamic_cast<const PrecisionChanged<EmitterT>*>(&e))
294 {
295 ++num_precision_increases;
296 }
297
298 else if (dynamic_cast<const SamplesRecomputedAtHigherPrecision<EmitterT>*>(&e))
299 {
300 // The endgame needed a higher precision before it had its first approximation, and
301 // tracks its sample window again at the new precision. The samples it announced at
302 // the lower precision are superseded: drop them as the endgame does, so the sequence
303 // holds one approach at one precision and its times keep marching toward the target.
304 if (!run_path_starts.empty())
305 {
306 path_samples.resize(run_path_starts.back());
307 path_times.resize(run_path_starts.back());
308 circle_samples.resize(run_circle_starts.back());
309 circle_times.resize(run_circle_starts.back());
310 approximations.resize(run_approx_starts.back());
311 approximation_times.resize(run_approx_starts.back());
312 approximation_errors.resize(run_approx_starts.back());
313 cycle_numbers.resize(run_approx_starts.back());
314 }
315 }
316
317 return ObserveResult::KeepObserving;
318}
319
320}; // SampleSequenceCollector
