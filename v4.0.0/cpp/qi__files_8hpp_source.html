91inline void StripClassicComments(std::string& s)
92{
93 std::string out;
94 out.reserve(s.size());
95 bool in_comment = false;
96 for (char c : s)
97 {
98 if (c == '\n')
99 {
100 in_comment = false;
101 out.push_back(c);
102 }
103 else if (in_comment)
104 continue;
105 else if (c == '%')
106 in_comment = true;
107 else
108 out.push_back(c);
109 }
110 s.swap(out);
111}

125inline void StripClassicFileWrappers(std::string& s)
126{
127 auto skip_ws = [](std::string const& t, size_t i) {
128 while (i < t.size() && std::isspace(static_cast<unsigned char>(t[i]))) ++i;
129 return i;
130 };
131 // case-insensitive check for `word` at position i, followed by a non-identifier char
132 auto keyword_at = [](std::string const& t, size_t i, char const* word) {
133 size_t n = std::strlen(word);
134 if (i + n > t.size()) return false;
135 for (size_t k = 0; k < n; ++k)
136 if (std::toupper(static_cast<unsigned char>(t[i+k])) != word[k]) return false;
137 if (i + n < t.size())
138 {
139 unsigned char c = static_cast<unsigned char>(t[i+n]);
140 if (std::isalnum(c) || c=='_') return false;
141 }
142 return true;
143 };
144
145 size_t i = skip_ws(s, 0);
146
147 // a leading CONFIG section, if present, runs to its first END;
148 if (keyword_at(s, i, "CONFIG"))
149 {
150 size_t e = s.find("END;", i);
151 if (e == std::string::npos)
152 return; // malformed; let the grammar report it
153 i = skip_ws(s, e + 4);
154 }
155
156 // an INPUT section wrapper, if present: drop the keyword and the matching trailing END;
157 if (keyword_at(s, i, "INPUT"))
158 {
159 size_t body = skip_ws(s, i + 5);
160 size_t e = s.rfind("END;");
161 if (e == std::string::npos || e < body)
162 return; // malformed; let the grammar report it
163 s = s.substr(body, e - body);
164 return;
165 }
166
167 if (i != 0)
168 s = s.substr(i);
169}

172inline std::string FormatParseError(
173 std::string::const_iterator begin,
174 std::string::const_iterator end,
175 std::string::const_iterator err_pos,
176 boost::spirit::info const& what,
177 std::string const& parser_name)
178{
179 int line = 1 + (int)std::count(begin, err_pos, '\n');
180 auto rev = std::find(std::make_reverse_iterator(err_pos),
181 std::make_reverse_iterator(begin), '\n');
182 int col = 1 + (int)std::distance(rev.base(), err_pos);
183
184 std::string remaining(err_pos, end);
185 if (remaining.size() > 60)
186 remaining = remaining.substr(0, 60) + "...";
187 if (remaining.empty())
188 remaining = "<end of input>";
189
190 std::ostringstream oss;
191 oss << "[" << parser_name << "] parse error at line " << line
192 << ", col " << col << ":\n"
193 << " expected: " << what << "\n"
194 << " found: \"" << remaining << "\"\n";
195 return oss.str();
196}
