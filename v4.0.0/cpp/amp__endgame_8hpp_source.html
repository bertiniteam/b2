94 SuccessCode RefineSampleImpl(Vec<complex_mp> & result, Vec<complex_mp> const& current_sample, complex_mp const& current_time, NumErrorT tol, unsigned max_iterations) const
95 {
96
97 using bertini::Precision;
98 assert(Precision(current_time)==Precision(current_sample) && "precision of sample and time to be refined in AMP endgame must match");
99
100 using std::max;
101 auto& TR = this->GetTracker();
102 TR.ChangePrecision(Precision(current_time));
103
104 auto refinement_success = this->GetTracker().Refine(result,current_sample,current_time,
105 tol,
106 max_iterations);
107
108
109 if (refinement_success==SuccessCode::HigherPrecisionNecessary ||
110 refinement_success==SuccessCode::FailedToConverge)
111 {
112
113 using bertini::Precision;
114
115 // thread-local precision: endgames may run on std::thread workers, so
116 // neither read nor write the global default precision.
117 auto prev_precision = ThreadPrecision();
118 auto higher_precision = max(prev_precision,LowestMultiplePrecision())+ PrecisionIncrement();
119 SetThreadPrecision(higher_precision);
120 this->GetTracker().ChangePrecision(higher_precision);
121
122 NotifyObservers(PrecisionChanged<EmitterType>(*this, prev_precision, higher_precision));
123
124 auto next_sample_higher_prec = current_sample;
125 Precision(next_sample_higher_prec, higher_precision);
126
127 auto result_higher_prec = Vec<complex_mp>(current_sample.size());
128
129 auto time_higher_precision = current_time;
130 Precision(time_higher_precision,higher_precision);
131
132 assert(time_higher_precision.precision()==ThreadPrecision());
133 refinement_success = this->GetTracker().Refine(result_higher_prec,
134 next_sample_higher_prec,
135 time_higher_precision,
136 tol,
137 max_iterations);
138
139 Precision(result, higher_precision);
140 result = result_higher_prec;
141
142 assert(Precision(result)==ThreadPrecision());
143 }
144 return refinement_success;
145 }

164 // ================================================================================================
165 // Adaptive-numeric-type endgame state + the precision/numeric-type helpers shared by both AMP
166 // endgame flavors (Cauchy and PowerSeries). These live here, in the AMP precision policy, because
167 // they are entirely about the complex_dbl <-> complex_mp crossing and the AMP tracker's authority --
168 // not generic endgame concerns. The flavors inherit them (AMPEndgame is the precision-policy base
169 // of EndgameBase) and call them through this->.
170 // ================================================================================================
171
172 // current_endgame_precision_ is the precision the endgame is *currently computing in*: DoublePrecision()
173 // is the hardware-complex_dbl fast lane, higher is the mpfr slot. adaptive_numeric_type_active_ gates
174 // the escalation hooks in the flavors' shared phase methods so they fire only while the double-first
175 // driver is orchestrating (never for the AMP-PowerSeries forwarding path, were one to exist).
176 mutable unsigned current_endgame_precision_ = DoublePrecision();
177 mutable bool adaptive_numeric_type_active_ = false;
178
179 // Choose the next (higher) working precision: at least LowestMultiplePrecision(), following the
180 // tracker's authority if it climbed higher, and strictly above the current precision.
182 unsigned NextEscalatedPrecision() const
183 {
184 using std::max;
185 unsigned from_tracker = this->GetTracker().GetCurrentPrecision();
186 unsigned newprec = max(static_cast<unsigned>(LowestMultiplePrecision()), from_tracker);
187 if (newprec <= current_endgame_precision_)
188 newprec = current_endgame_precision_ + PrecisionIncrement();
189 return newprec;
190 }
191
192 // Cross a deque of times / samples, or a single vector, from the complex_dbl slot to the complex_mp
193 // slot, then set the mpfr precision via the existing SetPrecision helper. Templated on the tuple type
194 // so this header need not name the endgames' TupleOfTimes / TupleOfSamps / TupOfVec aliases. The
195 // element crossing is the same one the AMP tracker performs; no library Vec-cast exists for these types.
197 template<typename TimesTuple>
198 void CrossTimesUp(TimesTuple& times, unsigned newprec) const
199 {
200 auto& from = std::get<TimeCont<complex_dbl> >(times);
201 auto& to = std::get<TimeCont<complex_mp> >(times);
202 to.clear();
203 for (auto const& t : from) to.push_back(complex_mp(t));
204 from.clear();
205 tracking::adaptive::SetPrecision(to, newprec);
206 }
207
209 template<typename SampsTuple>
210 void CrossSampsUp(SampsTuple& samps, unsigned newprec) const
211 {
212 auto& from = std::get<SampCont<complex_dbl> >(samps);
213 auto& to = std::get<SampCont<complex_mp> >(samps);
214 to.clear();
215 for (auto const& v : from)
216 {
217 Vec<complex_mp> w(v.size());
218 for (Eigen::Index i = 0; i < v.size(); ++i) w(i) = complex_mp(v(i));
219 to.push_back(std::move(w));
220 }
221 from.clear();
222 tracking::adaptive::SetPrecision(to, newprec);
223 }
224
226 template<typename VecTuple>
227 void CrossVecUp(VecTuple& vec, unsigned newprec) const
228 {
229 using bertini::Precision;
230 auto& from = std::get<Vec<complex_dbl> >(vec);
231 auto& to = std::get<Vec<complex_mp> >(vec);
232 to.resize(from.size());
233 for (Eigen::Index i = 0; i < from.size(); ++i) to(i) = complex_mp(from(i));
234 if (to.size() > 0) Precision(to, newprec);
235 from.resize(0);
236 }

242 Vec<complex_dbl> out(v.size());

244 return out;

256 // Widen an active-type vector (a fast-lane extrapolation result) to complex_mp at the given precision:
257 // the single boundary conversion of an approximation back to the ambient type.
259 template<typename ComplexT>
260 Vec<complex_mp> ToBCT(Vec<ComplexT> const& v, unsigned prec) const
261 {
262 using bertini::Precision;
263 Vec<complex_mp> out(v.size());
264 for (Eigen::Index i = 0; i < v.size(); ++i) out(i) = complex_mp(v(i));
265 if (out.size() > 0) Precision(out, prec);
266 return out;
267 }
