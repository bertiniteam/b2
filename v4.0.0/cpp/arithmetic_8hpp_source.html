108 // Build a complete sum from a full (term, sign) list. The node is fully constructed
109 // before Make() interns it -- so callers never AddOperand AFTER Make (which, with interning,
110 // could mutate a shared interned node).
111 explicit SumOperator(std::vector<std::pair<std::shared_ptr<Node>, bool>> const& terms)
112 {
113 for (auto const& t : terms)
114 AddOperand(t.first, t.second);
115 CanonicalizeNaryOperands(operands_, signs_, false);
116 }
