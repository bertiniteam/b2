304 SuccessCode TrackPath(Vec<ComplexT> & solution_at_endtime,
305 ComplexT const& start_time, ComplexT const& endtime,
306 Vec<ComplexT> const& start_point
307 ) const
308 {
309 if (start_point.size()!=static_cast<Eigen::Index>(GetSystem().NumVariables()))
310 throw std::runtime_error("start point size must match the number of variables in the system to be tracked");
311
312 // whatever route the settings were set by, they take effect here
313 SyncSettings();
314
315 SuccessCode initialization_code = TrackerLoopInitialization(start_time, endtime, start_point);
316 if (initialization_code!=SuccessCode::Success)
317 {
318 PostTrackCleanup();
319 return initialization_code;
320 }
321
322 NotifyObservers(TrackingStarted<typename TrackerTraits<D>::EventEmitterType>(static_cast<typename TrackerTraits<D>::EventEmitterType const&>(*this)));
323
324 // as precondition to this while loop, the correct container, either complex_dbl or mpfr, must have the correct data.
325 while (!IsSymmRelDiffSmall(current_time_,endtime_, Eigen::NumTraits<ComplexT>::epsilon()))
326 {
327 // Has somebody outside asked us to stop? Checked FIRST, before the
328 // tracker's own budgets, because it is the only one whose answer is not
329 // about this path: a keyboard interrupt is a statement about the process.
330 // Between steps is the only place we can honour it without threads, so a
331 // stop overruns by at most the step in flight -- which at high precision
332 // is not free, and is the price of not preempting mid-step.
333 if (StopRequested())
334 {
335 PostTrackCleanup();
336 return SuccessCode::ExternallyTerminated;
337 }
338
339 // Has our wall-clock deadline passed? A deadline rather than a duration,
340 // because it must survive across TrackPath calls: an endgame issues hundreds
341 // of them for one path, and a per-call budget would restart with each.
342 // Same overrun as the stop above: at most the step in flight.
343 if (wall_clock_deadline_ && std::chrono::steady_clock::now() >= *wall_clock_deadline_)
344 {
345 PostTrackCleanup();
346 return SuccessCode::WallClockLimitReached;
347 }
348
349 SuccessCode pre_iteration_code = PreIterationCheck();
350 if (pre_iteration_code!=SuccessCode::Success)
351 {
352 PostTrackCleanup();
353 return pre_iteration_code;
354 }
355
356 using std::abs;
357 // compute the next delta_t
358 if (abs(endtime_-current_time_) < abs(current_stepsize_))
359 delta_t_ = endtime_-current_time_;
360 else
361 delta_t_ = current_stepsize_ * (endtime_ - current_time_)/abs(endtime_ - current_time_);
362
363 step_success_code_ = TrackerIteration();
364
365 if (infinite_path_truncation_ && (CheckGoingToInfinity()==SuccessCode::GoingToInfinity))
366 {
367 OnInfiniteTruncation();
368 PostTrackCleanup();
369 return SuccessCode::GoingToInfinity;
370 }
371 else if (step_success_code_==SuccessCode::Success)
372 OnStepSuccess();
373 else
374 {
375 OnStepFail();
376 // A terminal failure adjusted nothing, so the next iteration would be
377 // bit-identical to the one that just failed -- an infinite loop that
378 // burns a core forever instead of failing the path.
379 if (StepFailureIsTerminal(step_success_code_))
380 {
381 PostTrackCleanup();
382 return step_success_code_;
383 }
384 }
385
386 }// re: while
387
388
389 CopySolution(solution_at_endtime);
390 PostTrackCleanup();
391 return SuccessCode::Success;
392 }
