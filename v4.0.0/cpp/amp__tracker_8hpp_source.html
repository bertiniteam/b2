136 inline RealT StepsizeSatisfyingCriterionB(unsigned precision,
137 unsigned digits_B,
138 unsigned num_newton_iterations,
139 unsigned predictor_order = 0)
140 {
141 return pow(RealT(10), -(static_cast<int>(digits_B) - static_cast<int>(precision))
142 * static_cast<int>(num_newton_iterations) / (predictor_order+1.0));
143 }

236 template<typename RealT>
237 void MinimizeTrackingCost(unsigned & new_precision, RealT & new_stepsize,
238 unsigned min_precision, RealT const& min_stepsize,
239 unsigned max_precision, RealT const& max_stepsize,
240 unsigned digits_B,
241 unsigned num_newton_iterations,
242 unsigned predictor_order = 0)
243 {
244 double min_cost = Eigen::NumTraits<double>::highest();
245
246 unsigned minimizing_precision = 0; // initialize to an impossible value.
247
248 // a few casts so that we can work in double precision, because doing this in multiprec sucks
249 double min_stepsize_lowprec = static_cast<double>(min_stepsize);
250 double max_stepsize_lowprec = static_cast<double>(max_stepsize);
251
252 auto minimizer_routine =
253 [&min_cost, &minimizing_precision, digits_B, num_newton_iterations, predictor_order, min_stepsize_lowprec, max_stepsize_lowprec](unsigned p)
254 {
255 double criterion_b_stepsize = StepsizeSatisfyingCriterionB<double>(p, digits_B, num_newton_iterations, predictor_order);
256 if (criterion_b_stepsize < min_stepsize_lowprec)
257 return; // precision too low to satisfy min_stepsize constraint
258 double candidate_stepsize = min(criterion_b_stepsize, max_stepsize_lowprec);
259 using std::abs;
260 double current_cost = ArithmeticCost(p) / abs(candidate_stepsize);
261
262 if (current_cost < min_cost)
263 {
264 min_cost = current_cost;
265 minimizing_precision = p;
266 }
267 };
268
269 unsigned lowest_mp_precision_to_test = min_precision;
270
271 if (min_precision<=DoublePrecision())
272 minimizer_routine(DoublePrecision());
273
274
275 if (lowest_mp_precision_to_test < LowestMultiplePrecision())
276 lowest_mp_precision_to_test = LowestMultiplePrecision();
277 else
278 lowest_mp_precision_to_test = (lowest_mp_precision_to_test/PrecisionIncrement()) * PrecisionIncrement(); // use integer arithmetic to round.
279
280
281 if (max_precision < LowestMultiplePrecision())
282 max_precision = LowestMultiplePrecision();
283 else
284 max_precision = (max_precision/PrecisionIncrement()) * PrecisionIncrement(); // use integer arithmetic to round.
285
286 for (unsigned p = lowest_mp_precision_to_test; p <= max_precision; p+=PrecisionIncrement())
287 minimizer_routine(p);
288
289 if (minimizing_precision==0){
290 throw std::runtime_error("MinimizeTrackingCost failed to find a suitable stepsize and precision");
291 }
292
293 new_precision = minimizing_precision; // copy the computed value
294 // next, because the above computed the new stepsize in double precision, which may be lowprec, we compute in full precision
295 new_stepsize = max(
296 min(
297 StepsizeSatisfyingCriterionB<RealT>(new_precision, digits_B, num_newton_iterations, predictor_order),
298 max_stepsize
299 ),
300 min_stepsize
301 );
302 }

511 SuccessCode TrackerLoopInitialization(complex_mp const& start_time,
512 complex_mp const& end_time,
513 Vec<complex_mp> const& start_point) const override
514 {
515 initial_precision_ = override_start_precision_.value_or(Precision(start_point(0)));
516
517 #ifndef BERTINI_DISABLE_ASSERTS
518 assert(
519 (!preserve_precision_
520 ||
521 -log10(this->TrackingTolerance()) <= initial_precision_)
522 && "when tracking a path, either preservation of precision must be turned off (so precision can be higher at the end of tracking), or the initial precision must be high enough to support the resulting points to the desired tolerance"
523 );
524 #endif
525
526 NotifyObservers(Initializing<AMPTracker,complex_mp>(*this,start_time, end_time, start_point));
527 SetThreadPrecision(initial_precision_);
528 // set up the master current time and the current step size
529
530 current_time_ = start_time;
531 current_time_.precision(initial_precision_);
532
533
534 endtime_highest_precision_ = end_time;
535 endtime_highest_precision_.precision(initial_precision_);
536
537 endtime_ = end_time;
538 endtime_.precision(initial_precision_);
539
540 // The first step is built at THIS track's start precision, and the precision is set
541 // after the assignment, never before it. current_precision_ still holds wherever the
542 // previous track ended, and an assignment takes its source's precision
543 // (preserve_related_precision), so building it at current_precision_ rounded the
544 // exact initial_step_size at the previous track's precision. A track that starts in
545 // double then rounded that a second time, to a double, and the last bit of its first
546 // step -- and so its whole step sequence -- depended on which path the tracker had
547 // tracked before: thread scheduling made the threaded solve irreproducible, and path
548 // order made a serial solve depend on it (#378).
549 if (reinitialize_stepsize_)
550 {
551 real_mp segment_length = abs(start_time-end_time)/Get<Stepping>().min_num_steps;
552 SetStepSize(min(real_mp(Get<Stepping>().initial_step_size, initial_precision_),segment_length));
553 }
554 current_stepsize_.precision(initial_precision_);
555
556 // populate the current space value with the start point, in appropriate precision
557 if (initial_precision_==DoublePrecision())
558 MultipleToDouble(start_point);
559 else
560 MultipleToMultiple(initial_precision_, start_point);
561
562 ChangePrecision<upsample_refine_off>(initial_precision_);
563
564 ResetCounters();
565
566 auto initial_refinement_code = InitialRefinement();
567
568 #ifndef BERTINI_DISABLE_ASSERTS
569 if (initial_precision_==DoublePrecision()){
570 PrecisionSanityCheck<complex_dbl>();
571 }
572 else{
573 PrecisionSanityCheck<complex_mp>();
574 }
575 #endif
576
577 return initial_refinement_code;
578 }

599 SuccessCode InitialRefinement() const
600 {
601 SuccessCode initial_refinement_code = RefineStoredPoint();
602 if (initial_refinement_code!=SuccessCode::Success)
603 {
604 do {
605 if (current_precision_ > Get<PrecConf>().maximum_precision)
606 {
607 NotifyObservers(SingularStartPoint<EmitterType>(*this));
608 return SuccessCode::SingularStartPoint;
609 }
610
611 if (current_precision_==DoublePrecision())
612 initial_refinement_code = ChangePrecision<upsample_refine_on>(LowestMultiplePrecision());
613 else
614 initial_refinement_code = ChangePrecision<upsample_refine_on>(current_precision_+PrecisionIncrement());
615 }
616 while (initial_refinement_code!=SuccessCode::Success);
617 }
618 return SuccessCode::Success;
619 }

713 template <typename ComplexT>
714 SuccessCode TrackerIteration() const // not an override, because it is templated
715 {
716
717 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
718
719 #ifndef BERTINI_DISABLE_ASSERTS
720 assert(PrecisionSanityCheck<ComplexT>() && "precision sanity check failed. some internal variable is not in correct precision");
721 #endif
722
723 NotifyObservers(NewStep<EmitterType>(*this));
724
725 Vec<ComplexT>& predicted_space = std::get<Vec<ComplexT> >(temporary_space_); // this will be populated in the Predict step
726 Vec<ComplexT>& current_space = std::get<Vec<ComplexT> >(current_space_); // the thing we ultimately wish to update
727 ComplexT current_time = ComplexT(current_time_);
728 ComplexT delta_t = ComplexT(delta_t_);
729
730 #ifndef BERTINI_DISABLE_ASSERTS
731 PrecisionSanityCheck<ComplexT>();
732 #endif
733
734 SuccessCode predictor_code = Predict<ComplexT, RealT>(predicted_space, current_space, current_time, delta_t);
735 if (predictor_code==SuccessCode::MatrixSolveFailureFirstPartOfPrediction)
736 {
737 NotifyObservers(FirstStepPredictorMatrixSolveFailure<EmitterType>(*this));
738 InitialMatrixSolveError();
739 return predictor_code;
740 }
741 else if (predictor_code==SuccessCode::MatrixSolveFailure)
742 {
743 NotifyObservers(PredictorMatrixSolveFailure<EmitterType>(*this));
744 NewtonConvergenceError();// decrease stepsize, and adjust precision as necessary
745 return predictor_code;
746 }
747 else if (predictor_code==SuccessCode::HigherPrecisionNecessary)
748 {
749 NotifyObservers(PredictorHigherPrecisionNecessary<EmitterType>(*this));
750 auto adj_code = AMPCriterionError<ComplexT>();
751 if (adj_code != SuccessCode::Success)
752 return adj_code;
753 return predictor_code;
754 }
755
756
757 NotifyObservers(SuccessfulPredict<AMPTracker, ComplexT>(*this, predicted_space));
758
759 Vec<ComplexT>& tentative_next_space = std::get<Vec<ComplexT> >(tentative_space_); // this will be populated in the Correct step
760
761 ComplexT tentative_next_time = current_time + delta_t;
762
763 SuccessCode corrector_code = Correct<ComplexT, RealT>(tentative_next_space,
764 predicted_space,
765 tentative_next_time);
766
767 #ifndef BERTINI_DISABLE_ASSERTS
768 assert(PrecisionSanityCheck<ComplexT>() && "precision sanity check failed. some internal variable is not in correct precision");
769 #endif
770
771 if (corrector_code==SuccessCode::MatrixSolveFailure || corrector_code==SuccessCode::FailedToConverge)
772 {
773 NotifyObservers(CorrectorMatrixSolveFailure<EmitterType>(*this));
774 NewtonConvergenceError();
775 return corrector_code;
776 }
777 else if (corrector_code == SuccessCode::HigherPrecisionNecessary)
778 {
779 NotifyObservers(CorrectorHigherPrecisionNecessary<EmitterType>(*this));
780 auto adj_code = AMPCriterionError<ComplexT>();
781 if (adj_code != SuccessCode::Success)
782 return adj_code;
783 return corrector_code;
784 }
785 else if (corrector_code == SuccessCode::GoingToInfinity)
786 {
787 // there is no corrective action possible...
788 return corrector_code;
789 }
790
791 NotifyObservers(SuccessfulCorrect<AMPTracker, ComplexT>(*this, tentative_next_space));
792
793 // copy the tentative vector into the current space vector;
794 current_space = tentative_next_space;
795 return AdjustAMPStepSuccess<ComplexT>();
796 }

864 template <typename ComplexT>
865 SuccessCode AdjustAMPStepSuccess() const
866 {
867 // This mirrors Bertini 1's AMP2_update: pick the cost-minimizing (precision, stepsize)
868 // after a successful step. Faithful-to-B1 points, two of which were previously wrong:
869 // (1) The rule-B precision requirement CREDITS THE STEP. digits_B is the right-hand side
870 // of P + (p+1)*xi/N > digits_B (xi = -log10(stepsize)); the floor is therefore
871 // digits_B - (p+1)*(-log10(min_stepsize))/N, crediting the smallest allowed step, NOT
872 // the raw digits_B. Without this credit a tiny step (large size_proportion in the
873 // roundoff regime) spuriously forced precision up. B1: digits_B2 = ceil(P0 - eta_minSS/N).
874 // (2) Precision decrease (and stepsize increase) is gated by BOTH B1's StepsForIncrease
875 // consecutive-successful-steps counter AND the digits-margin hysteresis
876 // (ExtraDigitsBeforePrecisionDecrease). The single StepsForIncrease setting
877 // (SteppingConfig::consecutive_successful_steps_before_stepsize_increase) governs both
878 // gates -- the old, duplicate AMP-config precision-decrease setting was removed.
879 real_mp min_stepsize = current_stepsize_ * real_mp(Get<Stepping>().step_size_fail_factor, current_precision_);
880 real_mp max_stepsize = min( current_stepsize_ * real_mp(Get<Stepping>().step_size_success_factor, current_precision_), real_mp(Get<Stepping>().max_step_size, current_precision_));
881
882 const unsigned steps_for_increase = Get<Stepping>().consecutive_successful_steps_before_stepsize_increase; // B1 StepsForIncrease
883
884 if (num_successful_steps_since_stepsize_increase_ < steps_for_increase)
885 max_stepsize = current_stepsize_; // not enough successes in a row: disallow stepsize increase
886
887 const unsigned N = Get<NewtonConfig>().max_num_newton_iterations;
888 const unsigned curr_digits = current_precision_;
889 const unsigned extra = ExtraDigitsBeforePrecisionDecrease(curr_digits);
890
891 // rule B, step-credited (credit the smallest allowed step = the most credit)
892 int digits_B = static_cast<int>(ceil( double(B_RHS<ComplexT>())
893 - (predictor_order_ + 1.0) * double(-log10(min_stepsize)) / N ));
894 int digits_C = static_cast<int>(DigitsC<ComplexT>());
895 int digits_stepsize = static_cast<int>(MinDigitsForStepsizeInterval(min_stepsize, max_stepsize, abs(current_time_ - endtime_)));
896
897 // each requirement may only lower precision if it clears the hysteresis margin
898 ApplyPrecisionDecreaseMargin(digits_B, curr_digits, extra);
899 ApplyPrecisionDecreaseMargin(digits_C, curr_digits, extra);
900 ApplyPrecisionDecreaseMargin(digits_stepsize, curr_digits, extra);
901
902 int min_digits = std::max({ static_cast<int>(this->DigitsTrackingTolerance()),
903 static_cast<int>(digits_final_),
904 digits_B, digits_C, digits_stepsize,
905 static_cast<int>(DoublePrecision()) });
906 unsigned min_precision = static_cast<unsigned>(min_digits); // >= DoublePrecision(), so non-negative
907
908 // StepsForIncrease gate + safety cap: precision may only be lowered after enough successful
909 // steps in a row and while under the decrease budget; otherwise hold the current precision.
910 const bool decrease_allowed =
911 (num_successful_steps_since_precision_decrease_ >= steps_for_increase)
912 &&
913 (num_precision_decreases_ < Get<PrecConf>().max_num_precision_decreases);
914 if (!decrease_allowed)
915 min_precision = max(min_precision, current_precision_);
916
917 unsigned max_precision = Get<PrecConf>().maximum_precision;
918
919 try {
920 MinimizeTrackingCost(next_precision_, next_stepsize_,
921 min_precision, min_stepsize,
922 max_precision, max_stepsize,
923 DigitsB<ComplexT>(), // RAW digits_B (=P0) for the stepsize relation, per B1 minimize_cost
924 N,
925 predictor_order_);
926 } catch (std::runtime_error const&) {
927 return SuccessCode::FailedToSelectPrecisionAndStepsize;
928 }
929
930 if ( (next_stepsize_ > current_stepsize_) || (next_precision_ < current_precision_) )
931 num_successful_steps_since_stepsize_increase_ = 0;
932 else
933 ++num_successful_steps_since_stepsize_increase_;
934
935 if (next_precision_ < current_precision_)
936 {
937 ++num_precision_decreases_;
938 num_successful_steps_since_precision_decrease_ = 0;
939 }
940 else
941 ++num_successful_steps_since_precision_decrease_;
942
943 return UpdatePrecisionAndStepsize();
944 }

972 {
973 next_precision_ = current_precision_;
974
975 next_stepsize_ = real_mp(Get<Stepping>().step_size_fail_factor, CurrentPrecision())*current_stepsize_;

1002 template<typename ComplexT>
1003 SuccessCode AMPCriterionError() const
1004 {
1005
1006 unsigned min_next_precision; // sure, i could use a trigraph here, but it'd be terrible
1007 if (current_precision_==DoublePrecision())
1008 min_next_precision = LowestMultiplePrecision(); // precision increases
1009 else
1010 min_next_precision = current_precision_ + (1+num_consecutive_failed_steps_)*PrecisionIncrement(); // precision increases
1011
1012
1013 real_mp min_stepsize = MinStepSizeForPrecision(current_precision_, abs(current_time_ - endtime_));
1014 real_mp max_stepsize = current_stepsize_ * real_mp(Get<Stepping>().step_size_fail_factor, current_precision_); // Stepsize decreases.
1015
1016 if (min_stepsize > max_stepsize)
1017 {
1018 // stepsizes are incompatible, must increase precision
1019 next_precision_ = min_next_precision;
1020 // decrease stepsize somewhat less than the fail factor
1021 next_stepsize_ = max(current_stepsize_ * (1+real_mp(Get<Stepping>().step_size_fail_factor, current_precision_))/2, min_stepsize);
1022 }
1023 else
1024 {
1025 unsigned digits_B = DigitsB<ComplexT>(); // RAW digits_B (=P0); the stepsize relation in MinimizeTrackingCost needs it raw
1026
1027 // rule-B precision floor CREDITS THE STEP (B1 AMP2_update: digits_B2 = ceil(P0 - eta_minSS/N)):
1028 // floor = digits_B - (p+1)*(-log10(min_stepsize))/N, crediting the smallest allowed step.
1029 // This keeps a small step (large size_proportion in the roundoff regime) from forcing
1030 // precision up by itself; genuine ill-conditioning still escalates via D / rules A,C.
1031 const unsigned N = Get<NewtonConfig>().max_num_newton_iterations;
1032 int digits_B_credited = static_cast<int>(ceil( double(digits_B)
1033 - (predictor_order_ + 1.0) * double(-log10(min_stepsize)) / N ));
1034 if (digits_B_credited < 0)
1035 digits_B_credited = 0;
1036
1037 unsigned min_precision = max(min_next_precision,
1038 static_cast<unsigned>(digits_B_credited),
1039 DigitsC<ComplexT>(),
1040 MinDigitsForStepsizeInterval(min_stepsize, max_stepsize, abs(current_time_ - endtime_)),
1041 digits_final_
1042 );
1043
1044 try {
1045 MinimizeTrackingCost(next_precision_, next_stepsize_,
1046 min_precision, min_stepsize,
1047 Get<PrecConf>().maximum_precision, max_stepsize,
1048 digits_B,
1049 N,
1050 predictor_order_);
1051 } catch (std::runtime_error const&) {
1052 return SuccessCode::FailedToSelectPrecisionAndStepsize;
1053 }
1054
1055 }
1056
1057 UpdatePrecisionAndStepsize();
1058 return SuccessCode::Success;
1059 }

1420 {
1421 if (new_precision==current_precision_) // no op
1422 return SuccessCode::Success;
1423
1424
1425 NotifyObservers(PrecisionChanged<EmitterType>(*this,current_precision_,new_precision));
1426

1429 // reset the counter for estimating the condition number.
1430 num_steps_since_last_condition_number_computation_ = this->Get<Stepping>().frequency_of_CN_estimation;
1431
1432 if (new_precision==DoublePrecision() && current_precision_>DoublePrecision())
1433 {
1434 // convert from multiple precision to double precision
1435 MultipleToDouble();

1437 else if(new_precision > DoublePrecision() && current_precision_ == DoublePrecision())
