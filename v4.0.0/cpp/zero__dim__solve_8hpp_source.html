438 unsigned long long num_finite_solutions = 0;

440 unsigned long long num_diverged = 0;

442 unsigned long long num_singular = 0;
443 unsigned long long num_real = 0;
444 unsigned long long num_nonsolutions = 0;
445 std::map<SuccessCode, unsigned long long> failures_by_reason;
446 double max_condition_number = 0;
447 unsigned max_precision_used = 0;
448 MidpathCheckReport midpath;
449 bool all_paths_resolved = false;

456template<typename ComplexT>
457SolveReport SummarizeSolve(std::vector<SolutionMetaData<ComplexT>> const& metadata,
458 MidpathCheckReport const& midpath)
459{
460 SolveReport r;
461 r.num_paths_tracked = metadata.size();
462 r.midpath = midpath;
463 double finite_distinct = 0;
464 for (auto const& m : metadata)
465 {
466 if (m.max_precision_used > r.max_precision_used)
467 r.max_precision_used = m.max_precision_used;
468
469 // A path the endgame's security check truncated near infinity (SecurityMaxNormReached) is
470 // diverging, not failing -- it is the security/max-norm heuristic catching a path on its way
471 // to infinity, exactly like a clean GoingToInfinity. (The endgame tests classify the two
472 // together too.) With infinite-path truncation on by default, these are expected, so they
473 // must not inflate num_failed / clear all_paths_resolved.
474 bool diverged = (m.endgame_success_code == SuccessCode::GoingToInfinity
475 || m.endgame_success_code == SuccessCode::SecurityMaxNormReached);
476 if (m.endgame_success_code == SuccessCode::Success)
477 {
478 if (m.is_nonsolution)
479 ++r.num_nonsolutions; // a nonsolution: finite, but not a solution of the target
480 else if (m.is_finite)
481 {
482 ++r.num_finite_endpoints;
483 finite_distinct += 1.0 / m.multiplicity;
484 if (m.is_singular) ++r.num_singular;
485 if (m.is_real) ++r.num_real;
486 if (static_cast<double>(m.condition_number) > r.max_condition_number)
487 r.max_condition_number = static_cast<double>(m.condition_number);
488 }
489 else
490 diverged = true;
491 }
492
493 if (diverged)
494 ++r.num_diverged;
495 else if (m.endgame_success_code != SuccessCode::Success) // neither a solution nor a clean divergence
496 {
497 ++r.num_failed;
498 ++r.failures_by_reason[m.endgame_success_code];
499 }
500 }
501 r.num_finite_solutions = static_cast<unsigned long long>(finite_distinct + 0.5);
502 r.all_paths_resolved = (r.num_failed == 0) && midpath.passed;
503 return r;
504}

708#ifdef BERTINI2_HAVE_MPI
721 void RunParallel(MPI_Comm comm)
722 {
723 using Result = parallel::FullPathResult<BaseComplexT>;
724 using Task = parallel::StartPointTask<BaseComplexT>;
725
726 ScopedSessionState const session_state_guard{}; // a solve leaves the session as it found it
727
728 mpfr_free_cache(); // reproducibility: clear this rank's mpfr constant cache (see Solve())
729
730 solutions_user_coords_fresh_ = false;
731
732 // Each rank built its homotopy independently in the constructor -- with its OWN
733 // random patch, start-system coefficients, and gamma -- so a given path index would
734 // mean a different path on each worker, and the manager would gather a scrambled,
735 // mostly-wrong solution set. Make rank 0 the single authoritative source: broadcast
736 // its actual target system, start system, and homotopy to every rank. These are all
737 // exact (rational coefficients + integer/rational patch), so serialization carries
738 // them bit-for-bit -- there is no per-rank re-derivation that could drift, and the
739 // distributed homotopy is identical to rank 0's serial one. We also broadcast rank
740 // 0's RNG seed so any randomness drawn later (e.g. in the endgame) is consistent; the
741 // per-path tracking RNG is already deterministic in the path index (ReseedThisThread).
742 {
743 unsigned long seed = parallel::IsManager() ? GetGlobalSeed() : 0ul;
744 MPI_Bcast(&seed, 1, MPI_UNSIGNED_LONG, 0, comm);
745 SetGlobalSeed(seed);
746
747 // Install rank 0's authoritative systems on every rank. The plain engine over a
748 // user-supplied homotopy does nothing here (the user owns cross-rank consistency);
749 // ZeroDimSolver, which owns its systems, overrides DistributeSystems to broadcast
750 // rank 0's target / start / homotopy and re-seat the engine's references.
751 DistributeSystems(comm);
752
753 num_start_points_ = StartSystem().NumStartPoints();
754 GetTracker().SetSystem(Homotopy());
755 }
756
757 PreSolveChecks();
758 PreSolveSetup();
759
760 // threads per rank: the same rule as the standalone solve (BERTINI_NUM_THREADS,
761 // then num_threads, then every CPU this rank may run on)
762 const int n_threads = static_cast<int>(
763 parallel::EffectiveThreadCount(this->template Get<ZeroDimConf>().num_threads));
764
765 // One round = dispatch a set of path indices, each executed as a WHOLE path
766 // (pre-endgame + endgame) by a worker. Round 0 is every path; later rounds re-run
767 // only the crossed paths with escalated settings -- the re-track is path-independent,
768 // so it rides the same demand-driven worker pool (no idle-manager bottleneck).
769 if (parallel::IsManager())
770 {
771 // Rank 0 computes every start point once, authoritatively, and ships each to the
772 // worker that tracks it. Cached so re-track rounds reuse the identical points.
773 std::vector<Vec<BaseComplexT>> start_points(num_start_points_);
774 for (decltype(num_start_points_) ii{0}; ii < num_start_points_; ++ii)
775 start_points[ii] = ComputeStartPoint(static_cast<SolnIndT>(ii));
776
777 auto run_round = [&](std::queue<Task>& q)
778 {
779 parallel::RunManagerLoop<Task, Result>(comm, q,
780 [this](Result const& r){ StoreFullPathResult(r); });
781 };
782
783 // the records seam (ADR-0046): the manager is the sole writer; recall what
784 // is already recorded and dispatch only the rest
785 MaybeAttachAmbientRecords();
786 std::vector<SolnIndT> indices_to_run;
787 for (decltype(num_start_points_) ii{0}; ii < num_start_points_; ++ii)
788 indices_to_run.push_back(static_cast<SolnIndT>(ii));
789 if (records_)
790 {
791 EnsureRunRecorded();
792 indices_to_run = RecallRecordedPaths(indices_to_run);
793 }
794
795 std::queue<Task> queue;
796 for (auto idx : indices_to_run)
797 queue.push(Task{ idx, start_points[idx] });
798 run_round(queue);
799
800 // Midpath/crossing check on the collected boundary points, then bounded parallel
801 // resolve rounds. A continue/stop flag is broadcast to the workers each round so
802 // they know whether to run again (and escalate their settings first).
803 auto passed = midpath_.Check(solutions_at_endgame_boundary_, StartSystem());
804 midpath_report_ = MidpathCheckReport{};
805 midpath_report_.passed = passed;
806 for (auto const& v : midpath_.GetCrossedPaths())
807 midpath_report_.crossed_path_indices.push_back(v.index());
808 midpath_report_.num_crossings_detected =
809 static_cast<unsigned>(midpath_report_.crossed_path_indices.size());
810
811 const auto max_attempts = this->template Get<ZeroDimConf>().max_num_crossed_path_resolve_attempts;
812 unsigned num_resolve_attempts = 0;
813 while (true)
814 {
815 int keep_going = (!passed && num_resolve_attempts < max_attempts) ? 1 : 0;
816 MPI_Bcast(&keep_going, 1, MPI_INT, 0, comm);
817 if (!keep_going)
818 break;
819
820 std::queue<Task> redo;
821 for (auto const& v : midpath_.GetCrossedPaths())
822 if (v.rerun())
823 {
824 auto idx = static_cast<SolnIndT>(v.index());
825 redo.push(Task{ idx, start_points[idx] });
826 }
827 run_round(redo);
828
829 passed = midpath_.Check(solutions_at_endgame_boundary_, StartSystem());
830 ++num_resolve_attempts;
831 }
832 midpath_report_.num_resolve_attempts = num_resolve_attempts;
833 midpath_report_.passed = passed;
834 FlagUnresolvedCrossings(passed);
835 RecordMidpathOutcome();
836 if (!passed)
837 std::cerr << "warning: " << midpath_report_.num_crossings_detected
838 << " path crossing(s) detected at the endgame boundary remained unresolved "
839 << "after " << num_resolve_attempts << " re-track attempt(s); "
840 << "the affected solutions may be wrong. Consider a higher-order predictor "
841 << "or a tighter tracking tolerance. See EndgameBoundaryMetadata()." << std::endl;
842
843 PostEGAction();
844
845 DeclareFiniteSolutionsResult();
846 }
847 else // worker
848 {
849 auto run_worker_round = [&]()
850 {
851 if (n_threads <= 1)
852 {
853 // Single-threaded worker: execute whole paths on the rank's member
854 // tracker/endgame directly, from rank 0's authoritative start point.
855 parallel::RunWorkerLoop<Task, Result>(comm,
856 [this](Task const& task){ ExecuteOnePath(MemberDuringEGContext(), static_cast<SolnIndT>(task.path_index), task.start_point); },
857 [this](Task const& task) -> Result { return PackFullPathResult(static_cast<SolnIndT>(task.path_index)); });
858 }
859 else
860 {
861 // Threaded worker: each thread owns a self-contained PathThreadState clone
862 // so concurrent whole-path execution shares no mutable state. Cloned per
863 // round so it inherits any escalation (predictor bump) applied to the rank's
864 // member tracker between rounds.
865 auto state_factory = [this]() -> std::unique_ptr<PathThreadState>
866 {
867 // Clone(), not copy: System copies are SHALLOW (shared expression-tree
868 // nodes whose value caches mutate on Eval); Clone() deep-copies the tree.
869 // Trackers/Endgames have no default ctor, so aggregate-init via new.
870 std::unique_ptr<PathThreadState> s(
871 new PathThreadState{ Clone(GetTracker().GetSystem()), Clone(TargetSystem()),
872 GetTracker(), GetEndgame(), {} });
873 s->tracker.SetSystem(s->sys);
874 s->endgame.SetTracker(s->tracker);
875 SetThreadPrecision(this->template Get<ZeroDimConf>().initial_ambient_precision);
876 return s;
877 };
878
879 auto track_fn = [this](std::unique_ptr<PathThreadState>& state, Task const& task) -> Result
880 {
881 auto idx = static_cast<SolnIndT>(task.path_index);
882 ExecuteOnePath(state->Context(), idx, task.start_point);
883 return PackFullPathResult(idx);
884 };
885
886 parallel::RunWorkerLoopThreaded<Task, Result>(
887 comm, state_factory, track_fn, n_threads);
888 }
889 }; // run_worker_round
890
891 while (true)
892 {
893 run_worker_round();
894
895 int keep_going = 0;
896 MPI_Bcast(&keep_going, 1, MPI_INT, 0, comm);
897 if (!keep_going)
898 break;
899
900 // Match the escalation the serial RunMidpathResolution applies between
901 // re-track passes: tighten midpath_retrack_tolerance_ (read by ExecuteOnePath)
902 // and bump a low-order predictor on this rank's member tracker, so the next
903 // round's thread clones inherit it.
904 EscalateRetrackSettings();
905 }
906 }
907 }

1145 void Solve()
1146 {
1147 // Reproducibility: clear this thread's mpfr constant cache (pi, etc.) at the solve
1148 // boundary. mpfr caches transcendental constants at the precision last requested; the
1149 // Cauchy endgame needs pi (roots of unity), so a high-precision solve leaves pi cached
1150 // at high precision. A subsequent lower-precision solve would otherwise reuse that
1151 // cached value rounded down -- differing by ~1 ULP from a freshly-computed
1152 // low-precision pi -- making a solve's behaviour depend on what ran before it in the
1153 // process (non-reproducible). Clearing here makes each solve independent of history.
1154 // Cheap (one constant recompute per solve); thread-local (each worker thread clears its
1155 // own when it starts a solve).
1156 mpfr_free_cache();
1157
1158 ScopedSessionState const session_state_guard{}; // a solve leaves the session as it found it
1159
1160 solutions_user_coords_fresh_ = false;
1161
1162 PreSolveChecks();
1163
1164 PreSolveSetup();
1165
1166 // the whole-solve wall-clock budget, counted from here; none unless asked for
1167 solve_deadline_.reset();
1168 if (auto const limit = this->template Get<ZeroDimConf>().max_solve_wall_clock_duration; limit > 0)
1169 solve_deadline_ = std::chrono::steady_clock::now()
1170 + std::chrono::duration_cast<std::chrono::steady_clock::duration>(std::chrono::duration<double>(limit));
1171
1172 this->NotifyObservers(AlgorithmStarted<AnyZeroDim>(*this));
1173
1174 // Speculative-full-path model: carry every path all the way through (pre-endgame +
1175 // endgame) as one unit, then detect crossings from the collected boundary points and
1176 // re-run any crossed path in full. This is the same shape the distributed solve uses
1177 // (one worker per whole path), so serial and distributed share the per-path primitive
1178 // -- including computing the start point via the same ComputeStartPoint.
1179
1180 // Compute every start point once, up front (deterministic per index, as the MPI
1181 // manager does). The threaded dispatch ships each worker its start point, and the
1182 // serial path / crossed-path re-tracks reuse the identical points from here.
1183 std::vector<Vec<BaseComplexT>> start_points(num_start_points_);
1184 std::vector<SolnIndT> all_indices(num_start_points_);
1185 for (decltype(num_start_points_) ii{0}; ii < num_start_points_; ++ii)
1186 {
1187 auto idx = static_cast<SolnIndT>(ii);
1188 start_points[ii] = ComputeStartPoint(idx);
1189 all_indices[ii] = idx;
1190 }
1191
1192 // the records seam (ADR-0046): ensure the run is on record, recall any paths
1193 // already recorded (solve() is ensure-answered), and compute only the rest
1194 MaybeAttachAmbientRecords();
1195 std::vector<SolnIndT> indices_to_run = all_indices;
1196 if (records_)
1197 {
1198 EnsureRunRecorded();
1199 // which recorded outcomes are reused is RecallPolicy's business, applied inside
1200 indices_to_run = RecallRecordedPaths(all_indices);
1201 }
1202
1203 // num_threads: 0 = auto (every CPU this process may run on), 1 = serial, N = N
1204 // threads; BERTINI_NUM_THREADS overrides. n_threads <= 1 takes the pool-free
1205 // serial path.
1206 const unsigned n_threads =
1207 parallel::EffectiveThreadCount(this->template Get<ZeroDimConf>().num_threads);
1208
1209 if (n_threads <= 1)
1210 {
1211 for (auto idx : indices_to_run)
1212 {
1213 if (StopRequested() || SolveDeadlinePassed())
1214 break; // ExecuteOnePath would no-op; this just stops asking
1215 ExecuteOnePath(MemberDuringEGContext(), idx, start_points[idx]);
1216 // the serial path installs results directly (no StoreFullPathResult),
1217 // so emit its record here
1218 RecordCompletedPath(PackFullPathResult(idx));
1219 }
1220 }
1221 else
1222 {
1223 RunPathsThreaded(indices_to_run, start_points, n_threads);
1224 }
1225
1226 TallyStoppedSolve(indices_to_run);
1227
1228 // Crossed-path re-tracks run on the main thread against the (escalated) member
1229 // tracker. They are typically rare (often none); running them in parallel too is a
1230 // deferred optimization.
1231 //
1232 // Skipped entirely when the solve was stopped: the midpath check compares paths
1233 // against each other at the endgame boundary, and a set where an arbitrary
1234 // number of paths never ran has nothing meaningful to compare. Re-tracking
1235 // under a standing stop request would also just abandon each path again.
1236 if (!stopped_early_)
1237 RunMidpathResolution([this, &start_points](SolnIndT idx){
1238 ExecuteOnePath(MemberDuringEGContext(), idx, start_points[idx]);
1239 RecordCompletedPath(PackFullPathResult(idx)); // a re-track appends a fresh record (last wins)
1240 });
1241
1242 PostEGAction();
1243
1244 DeclareFiniteSolutionsResult();
1245
1246 this->NotifyObservers(AlgorithmComplete<AnyZeroDim>(*this));
1247 }

1451 SolutionMetaDataT MetadataForPoint(Vec<BaseComplexT> const& point, NumErrorT tol,
1452 bool user_coords = true, bool representatives_only = true) const
1453 {
1454 auto const& sols = user_coords ? SolutionsUserCoords() : SolutionsInternalCoords();
1455 auto const& md = SolutionMetadata();
1456
1457 std::vector<size_t> matches;
1458 for (size_t i = 0; i < md.size() && i < sols.size(); ++i)
1459 {
1460 if (representatives_only && !md[i].multiplicity_representative)
1461 continue;
1462 if (!bertini::IsDistinct(sols[i], point, tol))
1463 matches.push_back(i);
1464 }
1465
1466 if (matches.empty())
1467 throw std::runtime_error("metadata_for: no solution matches the given point within tol");
1468
1469 // More than one representative among the matches => tol grouped two distinct clusters.
1470 // With representatives_only (the default) and the default tol this cannot happen -- distinct
1471 // representatives are at least SamePointTolerance apart and the window is smaller -- so it
1472 // only fires for a deliberately coarse user-supplied tol, where it is the honest answer.
1473 std::vector<size_t> reps;
1474 for (auto i : matches)
1475 if (md[i].multiplicity_representative)
1476 reps.push_back(i);
1477 if (reps.size() > 1)
1478 throw std::runtime_error("metadata_for: the point matches more than one distinct solution cluster; reduce tol");
1479 if (reps.size() == 1)
1480 return md[reps.front()];
1481 // No representative matched (only reachable with representatives_only=false, when the point
1482 // is a non-representative copy whose representative fell outside tol); return the matched
1483 // copy's own record (it still carries the cluster's multiplicity).
1484 return md[matches.front()];
1485 }

1503 std::vector<SolutionMetaDataT> CoincidentMetadataForPoint(Vec<BaseComplexT> const& point, NumErrorT tol, bool user_coords = true) const
1504 {
1505 auto const& sols = user_coords ? SolutionsUserCoords() : SolutionsInternalCoords();
1506 auto const& md = SolutionMetadata();
1507
1508 // Find the representative NEAREST the point (within tol). Nearest, not "all within tol", so
1509 // the cluster is identified unambiguously even for a coarse tol.
1510 size_t rep = md.size();
1511 NumErrorT best = std::numeric_limits<NumErrorT>::infinity();
1512 for (size_t i = 0; i < md.size() && i < sols.size(); ++i)
1513 {
1514 if (!md[i].multiplicity_representative)
1515 continue;
1516 NumErrorT d = static_cast<NumErrorT>((sols[i] - point).template lpNorm<Eigen::Infinity>());
1517 if (d <= tol && d < best) { best = d; rep = i; }
1518 }
1519 if (rep == md.size())
1520 throw std::runtime_error("metadata_for: no solution matches the given point within tol");
1521
1522 // Gather the whole cluster: every endpoint the solver deemed the same solution as the
1523 // representative (within the same-point clustering tolerance).
1524 const NumErrorT same_tol = SamePointTolerance();
1525 std::vector<SolutionMetaDataT> out;
1526 for (size_t i = 0; i < md.size() && i < sols.size(); ++i)
1527 if (!bertini::IsDistinct(sols[i], sols[rep], same_tol))
1528 out.push_back(md[i]);
1529 return out;
1530 }

1577 // The midpath checker is a consumer of the solver's MidPathConfig, not a second
1578 // home for it, so whatever the caller last set on the solver takes effect here --
1579 // whether they named the config or reached the field through the flat surface.

1687 void ExecuteBeforeEG(BeforeEGContext ctx, SolnIndT soln_ind, Vec<BaseComplexT> const& start_point)
1688 {
1689 ReseedThisThread(static_cast<uint64_t>(soln_ind));
1690
1691 auto& smd = solution_final_metadata_[soln_ind];
1692 smd.path_index = soln_ind;
1693 smd.solution_index = soln_ind;
1694
1695 // The step tally spans the whole path -- this track and every sub-track the
1696 // endgame will issue -- so it starts here and is read after each stage.
1697 ctx.tracker.ResetCumulativeStepCounts();
1698
1699 // The wall-clock budget, likewise per path and spanning both stages: armed here as
1700 // a deadline (so the endgame's many sub-tracks share it), cleared when the path is
1701 // over. Recorded with the path whether or not it bites.
1702 auto const wall_clock_limit = this->template Get<ZeroDimConf>().max_path_wall_clock_duration;
1703 smd.wall_clock_limit_seconds = wall_clock_limit;
1704 ctx.tracker.ClearMaxWallClockTime();
1705 if (wall_clock_limit > 0)
1706 ctx.tracker.SetMaxWallClockDuration(std::chrono::duration<double>(wall_clock_limit));
1707 // the whole-solve deadline, when there is one and it comes first, is the one that binds
1708 if (solve_deadline_ && (!ctx.tracker.MaxWallClockTime() || *solve_deadline_ < *ctx.tracker.MaxWallClockTime()))
1709 ctx.tracker.SetMaxWallClockTime(*solve_deadline_);
1710
1711 auto initial_prec = this->template Get<ZeroDimConf>().initial_ambient_precision;
1712 // SetThreadPrecision: writes thread-local only, safe from concurrent threads.
1713 SetThreadPrecision(initial_prec);
1714
1715 // Draw the condition-number probe direction ONCE here, for the whole track of this
1716 // point (pre-endgame AND endgame). We have just reseeded this thread's RNG to a value
1717 // determined solely by the path index, so the probe is deterministic per path and
1718 // identical whether the path is tracked in a serial loop or on a distributed worker --
1719 // regardless of how the paths were split across ranks. It is NOT refreshed for the
1720 // endgame (which would churn the RNG across its sample-circle sub-tracks), so the same
1721 // direction persists start to finish.
1722 ctx.tracker.RefreshConditionDirection();
1723
1724 // The times are stored precision-free (mpq_rational); materialize them as the tracking
1725 // complex type at the current working precision.
1726 BaseComplexT t_start ( BaseRealT(this->template Get<ZeroDimConf>().start_time) );
1727 BaseComplexT t_endgame_boundary( BaseRealT(this->template Get<ZeroDimConf>().endgame_boundary) );
1728
1729 // The start point is supplied by the caller (computed once, authoritatively, via
1730 // ComputeStartPoint) rather than regenerated here. In a distributed solve rank 0
1731 // computes it and sends it with the task, so a worker never derives its own -- that,
1732 // plus authoritative pi (issue #156), is what makes the start point identical across
1733 // serial and distributed runs.
1734
1735 // Reset to the configured initial step size for this fresh path. In the
1736 // speculative-full-path model a path's endgame runs immediately before the next
1737 // path's pre-endgame tracking on the same tracker, and the endgame leaves
1738 // ReinitializeInitialStepSize(false) with a tiny step; without restoring it here the
1739 // next path would crawl from the start time with that tiny step (effectively a hang).
1740 ctx.tracker.ReinitializeInitialStepSize(true);
1741
1742 // Begin every AMP path in hardware double precision; the AMP step criteria escalate to
1743 // multiprecision only where a path actually needs it (and the endgame, which clears
1744 // this override below, manages its own precision). Total-degree start points are born
1745 // at LowestMultiplePrecision (the generator's arithmetic widens past the requested
1746 // digits, see issue #308), and initial_ambient_precision is itself a multiprecision
1747 // value, so without forcing double here the tracker starts -- and, in practice, stays
1748 // -- in multiprecision on every path, even trivially well-conditioned ones.
1749 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1750 ctx.tracker.SetStartPrecision(DoublePrecision());
1751
1752 Vec<BaseComplexT> result;
1753 auto tracking_success = ctx.tracker.TrackPath(result, t_start, t_endgame_boundary, start_point);
1754
1755 solutions_at_endgame_boundary_[soln_ind] =
1756 EGBoundaryMetaDataT({ result, tracking_success, ctx.tracker.CurrentStepsize(), ctx.tracker.CurrentPrecision() });
1757
1758 // Clear the start-precision override so it does not leak into the endgame, which
1759 // shares this tracker instance for its sample circles.
1760 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1761 ctx.tracker.SetStartPrecision(std::nullopt);
1762
1763 smd.pre_endgame_success_code = tracking_success;
1764
1765 if (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1766 HarvestPathPrecision(smd, ctx.path_prec);
1767
1768 StampWhereThePathGotTo(smd, ctx.tracker, tracking_success, ctx.tracker.CurrentTime());
1769 }

1940 void ExecuteOnePath(DuringEGContext ctx, SolnIndT soln_ind, Vec<BaseComplexT> const& start_point)
1941 {
1942 // Carry the executing tracker on the path events: &ctx.tracker is the member tracker
1943 // in a serial solve and the thread-local clone in a threaded one, so a meta-observer
1944 // attaches its per-path sub-observer to the tracker that actually runs this path.
1945 Observable const* exec_tracker = &ctx.tracker;
1946
1947 // Somebody asked us to stop before this path began, so do not begin it. Its
1948 // metadata keeps the default NeverStarted, which is the honest description and
1949 // is already distinguishable from a path we started and abandoned. Checked
1950 // here rather than only in the dispatch loops so the threaded pool, which has
1951 // every task submitted up front, drains without doing work.
1952 if (StopRequested() || SolveDeadlinePassed())
1953 return;
1954
1955 // wall-clock the whole path (pre-endgame + endgame). steady_clock: monotonic, safe on
1956 // the worker thread; written to this path's own metadata slot, so no cross-path race.
1957 auto const path_start_clock = std::chrono::steady_clock::now();
1958 auto stamp_path_time = [&]{
1959 solution_final_metadata_[soln_ind].path_time_seconds =
1960 std::chrono::duration<double>(std::chrono::steady_clock::now() - path_start_clock).count();
1961 };
1962
1963 this->NotifyObservers(PathStarted<AnyZeroDim>(*this, static_cast<std::size_t>(soln_ind), exec_tracker));
1964
1965 // One precision record for the whole path, across the track to the boundary and
1966 // every endgame sub-track: started here, read after each stage, detached at the end.
1967 constexpr bool adaptive = tracking::TrackerTraits<TrackerType>::IsAdaptivePrec;
1968 if (adaptive)
1969 {
1970 ctx.path_prec.Reset();
1971 ctx.tracker.AddObserver(ctx.path_prec);
1972 }
1973
1974 ctx.tracker.SetTrackingTolerance(midpath_retrack_tolerance_);
1975 ExecuteBeforeEG(BeforeEGContext{ ctx.tracker, ctx.path_prec }, soln_ind, start_point);
1976
1977 ReadSolveTimeoutAsAnInterrupt(solution_final_metadata_[soln_ind].pre_endgame_success_code, path_start_clock);
1978 if (solution_final_metadata_[soln_ind].pre_endgame_success_code != SuccessCode::Success)
1979 {
1980 if (adaptive)
1981 ctx.tracker.RemoveObserver(ctx.path_prec);
1982 ctx.tracker.ClearMaxWallClockTime();
1983 stamp_path_time();
1984 this->NotifyObservers(PathComplete<AnyZeroDim>(*this, static_cast<std::size_t>(soln_ind), exec_tracker,
1985 solution_final_metadata_[soln_ind].pre_endgame_success_code));
1986 return;
1987 }
1988
1989 ctx.tracker.SetTrackingTolerance(this->template Get<Tolerances>().newton_during_endgame);
1990 ExecuteDuringEG(ctx, soln_ind);
1991 if (adaptive)
1992 ctx.tracker.RemoveObserver(ctx.path_prec);
1993
1994 ReadSolveTimeoutAsAnInterrupt(solution_final_metadata_[soln_ind].endgame_success_code, path_start_clock);
1995 ctx.tracker.ClearMaxWallClockTime(); // the budget was this path's; do not carry it to the next
1996 stamp_path_time();
1997 this->NotifyObservers(PathComplete<AnyZeroDim>(*this, static_cast<std::size_t>(soln_ind), exec_tracker,
1998 solution_final_metadata_[soln_ind].endgame_success_code));
1999 }

2876 std::vector<SolnIndT> RecallRecordedPaths(std::vector<SolnIndT> const& all_indices)
2877 {
2878 std::map<std::size_t, boost::json::object> recorded; // last record per index wins
2879 for (auto const& rec : records_->ResultsOf(records_run_id_))
2880 {
2881 auto const* k = rec.if_contains("kind");
2882 if (!k || !k->is_string() || k->get_string() != "path")
2883 continue;
2884 if (auto const* idx = rec.if_contains("index"); idx && idx->is_int64())
2885 recorded[static_cast<std::size_t>(idx->as_int64())] = rec;
2886 }
2887
2888 auto const policy = this->template Get<RecordsConfig>().recall;
2889 auto const current_limit = this->template Get<ZeroDimConf>().max_path_wall_clock_duration;
2890
2891 num_recalled_ = 0;
2892 recalling_ = true;
2893 std::vector<SolnIndT> missing;
2894 for (auto const idx : all_indices)
2895 {
2896 auto const found = recorded.find(static_cast<std::size_t>(idx));
2897 if (found == recorded.end() || policy == RecallPolicy::Nothing)
2898 {
2899 missing.push_back(idx);
2900 continue;
2901 }
2902
2903 auto decoded = records::DecodeFullPathResult<BaseComplexT>(found->second, idx);
2904
2905 // The rule for reusing a record: a path record is reusable when the run
2906 // that would reuse it asks no more of that path than the run that produced
2907 // it. A completed path -- success, divergence, a failure under settings
2908 // that are part of the ask -- always qualifies, since nothing more is being
2909 // asked. An ABANDONED path was ended for a reason deliberately kept out of
2910 // the ask, so the record says "we gave up here", which is true and is not
2911 // an answer; whether to reuse it is the user's call (RecallPolicy). Under
2912 // the default: a path somebody stopped is always re-tracked, since an
2913 // interrupt has no budget to compare; one a wall-clock limit cut off is
2914 // reused only while the current budget asks no more patience than the one
2915 // that abandoned it, and re-tracked once the budget goes up -- which is
2916 // exactly when paying again is the point. Either way its record stays, so
2917 // the store still says what happened; it is simply not mistaken for a
2918 // conclusion.
2919 if (policy == RecallPolicy::Completed && !AbandonmentIsReusable(decoded, current_limit))
2920 {
2921 missing.push_back(idx);
2922 continue;
2923 }
2924
2925 StoreFullPathResult(std::move(decoded));
2926 ++num_recalled_;
2927 }
2928 recalling_ = false;
2929
2930 // the narrative stays complete: a recalled ask WAS asked. One small
2931 // history line per re-ask -- when, how much came from the store, how
2932 // much this session still computed (the kill-and-rerun story, in the
2933 // record). Fresh runs (no prior header) narrate via their run header.
2934 if (records_run_resumed_)
2935 records_->Append({{"kind", "recall"},
2936 {"run", records_run_id_},
2937 {"when", records::TimeStampNow()},
2938 {"num_recalled", static_cast<std::int64_t>(num_recalled_)},
2939 {"num_computed", static_cast<std::int64_t>(missing.size())},
2940 {"producer", records::ProducerInfo()}});
2941 return missing;
2942 }

3167 void ConsistencyCheck() const
3168 {
3169 if (owned_target_.HavePathVariable())
3170 throw std::runtime_error("unable to perform zero dim solve on target system -- has path variable, use a HomotopySolver instead.");
3171
3172 // A square zero-dim system needs one equation per dimension. Each projective
3173 // (homogeneous) variable group of size k spans P^{k-1}: its k coordinates carry only
3174 // k-1 dimensions because scale is free, so it needs one fewer equation than it has
3175 // variables. Subtract that free scale per projective group before comparing. An
3176 // UNDER-determined system has a positive-dimensional solution set, so no isolated
3177 // solutions to compute -- raise a helpful error rather than tracking garbage.
3178 if (owned_target_.NumVariables() - owned_target_.NumHomVariableGroups() > owned_target_.NumTotalFunctions())
