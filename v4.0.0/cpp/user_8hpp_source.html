134template<class Archive>
135inline void load_construct_data(
136 Archive & ar, bertini::start_system::User * t, const unsigned int /*file_version*/
137){
138 // retrieve data from archive required to construct new instance
139 bertini::System sys;
140 ar >> sys;
141
142 bool solns_in_dbl;
143 ar >> solns_in_dbl;
144
145 if (solns_in_dbl)
146 {
147 bertini::SampCont<bertini::complex_dbl> solns;
148 ar >> solns;
149 ::new(t)bertini::start_system::User(sys, solns);
150 }
151 else
152 {
153 bertini::SampCont<bertini::complex_mp> solns;
