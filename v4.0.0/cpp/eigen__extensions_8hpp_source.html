77 // ThreadPrecision (thread-local), not DefaultPrecision (global): these
78 // tolerances are consumed inside tracking (LU solves, norms, convergence
79 // checks), which may run on std::thread workers whose precision is set
80 // thread-locally. On the main thread the two agree.
81 // epsilon() and dummy_precision() are pow(10, -precision): expensive (transcendental,
82 // heap-allocating) yet a pure function of the thread precision. They are called per
83 // LU-diagonal element, per Newton iteration, per predictor stage, per step, so recomputing
84 // them dominated the multiprecision path. Memoize per thread, keyed on the current
85 // precision -- the cached value is bit-identical to a fresh pow at that precision.
86 inline static Real dummy_precision()
87 {
88 using bertini::ThreadPrecision;
89 thread_local unsigned cached_prec = 0;
90 thread_local Real cached_val;
91 const unsigned p = ThreadPrecision();
92 if (p != cached_prec)
93 {
94 cached_val = pow(mpfr_real(10), -int(p-3));
95 cached_prec = p;
96 }
97 return cached_val;
98 }
99
100 inline static Real epsilon()
101 {
102 using bertini::ThreadPrecision;
103 thread_local unsigned cached_prec = 0;
104 thread_local Real cached_val;
105 const unsigned p = ThreadPrecision();
106 if (p != cached_prec)
107 {
108 cached_val = pow(mpfr_real(10), -int(p));
109 cached_prec = p;
110 }
111 return cached_val;
112 }

400 template<typename T>
401 inline
402 bool IsSmallValue(T const& testme)
403 {
404 // |testme| <= eps*100 <=> abs2(testme) <= (eps*100)^2. Comparing squared magnitudes uses
405 // abs2 (re^2+im^2 for complex; see abs2_impl<complex_mp>) and avoids the allocating, sqrt-based
406 // std::abs/hypot on the multiprecision complex type. eps*100 is non-negative, so the squared
407 // comparison is exactly equivalent.
408 using Real = typename Eigen::NumTraits<T>::Real;
409 const Real thresh = Eigen::NumTraits<T>::epsilon() * 100;
410 return Eigen::numext::abs2(testme) <= thresh * thresh;
411 }

447 template<typename T>
448 inline
449 bool IsLargeChange(T const& numerator, T const& denomenator)
450 {
451 static_assert(!Eigen::NumTraits<T>::IsInteger, "IsLargeChange cannot be used safely on non-integral types");
452 // |num/den| >= 1/dummy <=> |num|*dummy >= |den| <=> abs2(num)*dummy^2 >= abs2(den).
453 // The squared form drops BOTH the complex division (num/den) and the sqrt-based abs/hypot --
454 // each of which allocates multiprecision temporaries -- and is exact since all quantities are
455 // non-negative. (In its hot caller, LUPartialPivotDecompositionSuccessful, the denominator has
456 // already passed IsSmallValue, so it is non-tiny here.)
457 using Real = typename Eigen::NumTraits<T>::Real;
458 const Real d = Eigen::NumTraits<T>::dummy_precision();
459 return Eigen::numext::abs2(numerator) * (d * d) >= Eigen::numext::abs2(denomenator);
460 }

479 template <typename Derived>
480 MatrixSuccessCode LUPartialPivotDecompositionSuccessful(Eigen::MatrixBase<Derived> const& LU)
481 {
482 #ifndef BERTINI_DISABLE_ASSERTS
483 assert(LU.rows()==LU.cols() && "non-square matrix in LUPartialPivotDecompositionSuccessful");
484 assert(LU.rows()>0 && "empty matrix in LUPartialPivotDecompositionSuccessful");
485 #endif
486
487 // this loop won't test entry (0,0). it's tested separately after.
488 for (Eigen::Index ii = LU.rows()-1; ii > 0; ii--)
489 {
490 if (IsSmallValue(LU(ii,ii)))
491 {
492 return MatrixSuccessCode::SmallValue;
493 }
494
495 if (IsLargeChange(LU(ii-1,ii-1),LU(ii,ii)))
496 {
497 return MatrixSuccessCode::LargeChange;
498 }
499 }
500
501 // this line is the reason for the above assert on non-empty matrix.
502 if (IsSmallValue(LU(0,0)))
503 {
504 return MatrixSuccessCode::SmallValue;
505 }
506
507 return MatrixSuccessCode::Success;
508 }

513 template <typename NumberType>
514 Eigen::Matrix<NumberType, Eigen::Dynamic, Eigen::Dynamic> KahanMatrix(unsigned int mat_size, NumberType c)
515 {
516 using std::sqrt;
517 NumberType s, scale(1.0);
518 s = sqrt( (NumberType(1.0)-c) * (NumberType(1.0)+c) );
519
520
521 Eigen::Matrix<NumberType, Eigen::Dynamic, Eigen::Dynamic> A(mat_size,mat_size);
522
523
524 for (unsigned int ii=0; ii<mat_size; ii++) {
525 for (unsigned int jj=0; jj<ii; jj++) {
526 A(ii,jj) = NumberType(0.0);
527 }
528 for (unsigned int jj=ii; jj<mat_size; jj++) {
529 A(ii,jj) = -c * NumberType(1.0);
530 }
531 }
532
533
534 for (unsigned int ii=0; ii<mat_size; ii++) {
535 A(ii,ii) += NumberType(1)+c;
536 }
537
538 for (unsigned int jj=0; jj<mat_size; jj++){
539 for (unsigned int kk=0;kk<mat_size;kk++){
540 A(jj,kk) *= scale;
541 }
542 scale *= s;
543 }
544
545 for (unsigned int jj=0;jj<mat_size;jj++){
546 for (unsigned int kk=0;kk<mat_size;kk++){
547 A(kk,jj)/= NumberType(jj) + NumberType(1);
548 }
549 }
550 return A;
551 }

600 template <typename NumberType>
601 inline
602 Mat<NumberType> RandomConjugateOrthonormalMatrix(unsigned int rows, unsigned int cols)
603 {
604 using std::max;
605 using std::min;
606 const unsigned int tall = max(rows, cols);
607 const unsigned int thin = min(rows, cols);
608 Mat<NumberType> seed = Mat<NumberType>(tall, thin).unaryExpr(
609 [](NumberType const& /*x*/) { return RandomOrthonormalSeed<NumberType>(); });
610 Eigen::HouseholderQR<Mat<NumberType> > qr(seed);
611 Mat<NumberType> Q = qr.householderQ() * Mat<NumberType>::Identity(tall, thin); // tall x thin, orthonormal columns
612 if (rows >= cols)
613 return Q;
614 return Q.transpose(); // rows x cols with orthonormal rows: M M^H = (Q^H Q)^T = I
615 }
