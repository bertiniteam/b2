217 // Compile-time bank selectors (ADR-0034). After NumType inference, each instruction's opcode word
218 // gets these high bits set to record which bank (real or complex) each operand and the result live
219 // in, so the hot eval loop reads the banks inline from the instruction it has already loaded instead
220 // of looking up slot_numtype_[slot] per operand. The base Operation occupies bits 0..16, so these

866 template<typename Derived>
867 void SetVariableValues(Eigen::MatrixBase<Derived> const& variable_values) const{
868 using NumT = typename Derived::Scalar;
869
870 // The Memory's precision is an ARTIFACT OF THE CURRENT EVALUATION, never an
871 // invariant to defend. The compiled Program is a precision-independent tape of
872 // operations; only the Memory holding values carries digits. So evaluating at
873 // whatever precision the caller brings is always meaningful, and the right
874 // response to a mismatch is to RE-TAG THE MEMORY, not to refuse.
875 //
876 // Refusing was a genuine trap, not merely unergonomic. A Memory takes its
877 // precision from the ambient DefaultPrecision() when the program is lazily
878 // compiled, while the owning System keeps whatever it was told, so the two
879 // diverge the moment anything moves the ambient default -- which an AMP tracker
880 // or endgame does as a matter of course. The System was then WEDGED with no way
881 // out: evaluating at its own reported precision raised here, and
882 // System::precision(n) could not repair it, because both setters short-circuit
883 // when the value they are handed already equals the one they hold. See #377.
884 //
885 // An empty variable vector (a constant program with no variables) has no
886 // precision to read. Re-tagging refills the constants from their exact recipes
887 // at the new precision, so accuracy is rebuilt rather than padded with zeros.
888 SyncPrecision(variable_values);
889
890 auto& memory = memory_.Get<NumT>(); // unpack for local reference
891
892 for (size_t ii = 0; ii < program_->number_of_.Variables; ++ii) {
893 //assign to memory
894 memory[ii + program_->input_locations_.Variables] = variable_values(static_cast<Eigen::Index>(ii));
895 }
896 memory_.is_evaluated_ = false;
897 }

907 template<typename ComplexT>
908 void SetPathVariable(ComplexT const& time) const{
909
910 // Same doctrine as SetVariableValues, but this one only ever raises: the
911 // variables have already been written into memory by the time the path variable
912 // arrives, so re-tagging DOWNWARD here would truncate them. Memory therefore
913 // ends an evaluation at the max of its arguments' precisions.
914 if constexpr (!std::is_same<ComplexT,complex_dbl>::value)
915 {
916 if (Precision(time) > memory_.precision_)
917 this->precision(Precision(time));
918 }
919
920 if (!this->HavePathVariable())
921 throw std::runtime_error("calling Eval with path variable, but this StraightLineProgram doesn't have one.");
922 // then actually copy the path variable into where it goes in memory
923
924 auto& memory = memory_.Get<ComplexT>(); // unpack for local reference
925
926 memory[program_->input_locations_.Time] = time;
927 // assigning an mp value adopts the SOURCE's precision, so a lower-precision time
928 // would otherwise leave one slot out of step with the rest of memory
929 if constexpr (!std::is_same<ComplexT,complex_dbl>::value)
930 Precision(memory[program_->input_locations_.Time], memory_.precision_);
931 memory_.is_evaluated_ = false;
932 }
