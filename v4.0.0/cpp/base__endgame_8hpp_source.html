171 // Drawn per run, never lazily on first use: a lazy draw lands in whichever path first needs
172 // the probe, so the direction a path used, and the draws it consumed, would depend on the
173 // paths the same endgame ran before it, and on which thread ran them (#378, ADR-0071).
174 void RefreshCOverKProbe(unsigned size) const
175 {
176 c_over_k_probe_as_drawn_.resize(size);
177 for (unsigned ii = 0; ii < size; ++ii)
178 c_over_k_probe_as_drawn_(ii) = RandomUnit<complex_dbl>();
179 std::apply([](auto&... working){ (working.resize(0), ...); }, c_over_k_probe_);
180 }

195 template<typename ComplexT>
196 Vec<ComplexT> const& GetCOverKProbe(unsigned size, unsigned prec) const
197 {
198 using bertini::Precision;
199 if (static_cast<unsigned>(c_over_k_probe_as_drawn_.size()) != size)
200 RefreshCOverKProbe(size); // samples of another size than this run began with
201
202 auto& probe = std::get<Vec<ComplexT> >(c_over_k_probe_);
203 bool stale = static_cast<unsigned>(probe.size()) != size;
204 if constexpr (!std::is_same<ComplexT, complex_dbl>::value)
205 stale = stale || Precision(probe) != prec;
206 if (stale)
207 {
208 probe.resize(size);
209 for (unsigned ii = 0; ii < size; ++ii)
210 probe(ii) = ComplexT(c_over_k_probe_as_drawn_(ii));
211 if constexpr (!std::is_same<ComplexT, complex_dbl>::value)
212 Precision(probe, prec);
213 }
214 return probe;
215 }

236 template<typename ComplexT>
237 auto ComputeCOverK(SampCont<ComplexT> const& samples) const -> typename Eigen::NumTraits<ComplexT>::Real
238 {
239 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
240 using std::abs;
241 using std::log;
242
243 assert(samples.size()>=3);
244 const Vec<ComplexT> & sample0 = samples[0];
245 const Vec<ComplexT> & sample1 = samples[1];
246 const Vec<ComplexT> & sample2 = samples[2];
247
248 const Vec<ComplexT> & rand_vector =
249 GetCOverKProbe<ComplexT>(static_cast<unsigned>(sample0.size()), Precision(sample0));
250
251 // //DO NOT USE Eigen .dot() it will do conjugate transpose which is not what we want.
252 // //Also, the .transpose*rand_vector returns an expression template that we do .norm of since abs is not available for that expression type.
253 RealT estimate = abs(log(abs((((sample2 - sample1).transpose()*rand_vector).template lpNorm<Eigen::Infinity>())/(((sample1 - sample0).transpose()*rand_vector).template lpNorm<Eigen::Infinity>()))));
254 estimate = abs(log(RealT(this->EndgameSettings().sample_factor)))/estimate;
255 if (estimate < 1)
256 return RealT(1);
257 else
258 return estimate;
259 }

275 template<typename T>
276 bool CheckForCOverKStabilization(std::deque<T> const& c_over_k_array) const
277 {
278 using RealT = typename Eigen::NumTraits<T>::Real;
279 using std::abs;
280
281 assert(c_over_k_array.size()>=this->EndgameSettings().num_needed_for_stabilization);
282 for(unsigned ii = 1; ii < this->EndgameSettings().num_needed_for_stabilization ; ++ii)
283 {
284 RealT a = abs(c_over_k_array[ii-1]);
285 RealT b = abs(c_over_k_array[ii]);
286
287 RealT divide = a;
288 if(a < b)
289 divide = a/b;
290 else
291 divide = b/a;
292
293 if(divide < this->EndgameSettings().minimum_for_c_over_k_stabilization)
294 return false;
295 }
296 return true;
297 }

338 SuccessCode Run(Vec<BCT> const& start_point)
339 {
340 using bertini::Precision;
341 // EVERY path enters here, so this is the per-run boundary an observer needs to tell
342 // one path's samples from the next -- without it, a collector attached to a solver's
343 // endgame concatenates every path into one undifferentiated heap. Initializing was
344 // declared, and handled by GoryDetailLogger, but emitted by nothing.
345 NotifyObservers(Initializing<EmitterType>(this->AsFlavor()));
346 // a run's c/k direction is drawn here, at its start, never inherited from an earlier run
347 RefreshCOverKProbe(static_cast<unsigned>(start_point.size()));
348 auto prec = Precision(start_point);
349 BCT t = start_time_; Precision(t, prec);
350 BCT t0 = target_time_; Precision(t0, prec);
351 // Fixed precision runs today's straight-line RunImpl<BCT> verbatim (untouched). Adaptive
352 // precision takes the separate double-first driver, RunImplAMP, which lives in the flavor.
353 // RunImplAMP is a member template (template<typename=void>) so the explicit fixed-precision
354 // class instantiations never force-compile it -- hence the .template disambiguator here.
355 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
356 return this->AsFlavor().template RunImplAMP<>(t, start_point, t0);
357 else
358 return this->AsFlavor().RunImpl(t, start_point, t0);
359 }

389 template<typename ComplexT>
390 SuccessCode RefineAllSamples(SampCont<ComplexT> & samples, TimeCont<ComplexT> & times)
391 {
392 for (size_t ii=0; ii<samples.size(); ++ii)
393 {
394 auto refine_success = this->RefineSample(samples[ii], samples[ii], times[ii],
395 this->FinalTolerance() * this->EndgameSettings().sample_point_refinement_factor,
396 this->EndgameSettings().max_num_refinements);
397 if (refine_success != SuccessCode::Success)
398 {
399 // BOOST_LOG_TRIVIAL(severity_level::trace) << "refining failed, code " << int(refine_success);
400 return refine_success;
401 }
402 NotifyObservers(SampleRefined<EmitterType>(AsFlavor()));
403 }
404
405 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec) // known at compile time
406 {
407 auto max_precision = this->EnsureAtUniformPrecision(times, samples);
408 }
409
410 return SuccessCode::Success;

445 template<typename ComplexT>
446 SuccessCode EndgameTrackPath(Vec<ComplexT> & result, ComplexT const& start_time, ComplexT const& end_time, Vec<ComplexT> const& start_point) const
447 {
448 if constexpr (std::is_same<ComplexT, BCT>::value)
449 {
450 return this->GetTracker().TrackPath(result, start_time, end_time, start_point);
451 }
452 else
453 {
454 using bertini::Precision;
455 Vec<BCT> mp_start(start_point.size());
456 for (Eigen::Index i = 0; i < start_point.size(); ++i) mp_start(i) = BCT(start_point(i));
457 Precision(mp_start, DoublePrecision());
458
459 BCT mp_t0(start_time); Precision(mp_t0, DoublePrecision());
460 BCT mp_t1(end_time); Precision(mp_t1, DoublePrecision());
461
462 Vec<BCT> mp_result(start_point.size());
463 auto code = this->GetTracker().TrackPath(mp_result, mp_t0, mp_t1, mp_start);
464 if (code != SuccessCode::Success)
465 return code;
466
467 result.resize(mp_result.size());
468 for (Eigen::Index i = 0; i < mp_result.size(); ++i) result(i) = complex_dbl(mp_result(i));
469 return code;
470 }
471 }

474 // EscalateAndMigrate bridges the AMP precision policy (NextEscalatedPrecision and the Cross* helpers,
475 // all in AMPEndgame) with the flavor's container list (its MigrateContainersToPrecision). It lives
476 // here, not in AMPEndgame, because it needs AsFlavor(), which the precision-policy base lacks. Member
477 // template so the explicit fixed-precision class instantiations never force-compile it. Raises the
478 // working precision and migrates every durable container up to it; bounded by a runaway guard.
480 template<typename Dummy = void>
481 SuccessCode EscalateAndMigrate(unsigned guard)
482 {
483 if (guard > 64)
484 return SuccessCode::HigherPrecisionNecessary;
485 unsigned newprec = this->NextEscalatedPrecision();
486 this->AsFlavor().template MigrateContainersToPrecision<>(newprec);
487 this->current_endgame_precision_ = newprec;
488 SetThreadPrecision(newprec);
489 return SuccessCode::Success;
490 }

655 template<typename ComplexT>
656 SuccessCode ComputeInitialSamples(const ComplexT & start_time,const ComplexT & target_time, const Vec<ComplexT> & x_endgame_start, TimeCont<ComplexT> & times, SampCont<ComplexT> & samples) // passed by reference to allow times to be filled as well.
657 {
658 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
659 assert(this->template Get<EndgameConfig>().num_sample_points>0 && "number of sample points must be positive");
660
661 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
662 {
663 assert(Precision(start_time)==Precision(x_endgame_start) && "Computing initial samples requires input time and space with uniform precision");
664 }
665
666 samples.clear();
667 times.clear();
668
669 samples.push_back(x_endgame_start);
670 times.push_back(start_time);
671 // The endgame's own input point, AT the boundary time: the coarsest sample of
672 // the approach, and the widest |t| any consumer will ever see. Without these
673 // the sequence starts at boundary * sample_factor^(num_sample_points-1) --
674 // measured, 0.1 -> 1.25e-2, throwing away most of the span before anyone
675 // looks at it.
676 EmitComputedSample(samples[0], times[0]);
677
678 auto num_vars = this->GetSystem().NumVariables();
679 //start at 1, because the input point is the 0th element.
680 for(unsigned ii=1; ii < this->template Get<EndgameConfig>().num_sample_points; ++ii)
681 {
682 times.emplace_back((times[ii-1] + target_time) * RealT(this->template Get<EndgameConfig>().sample_factor)); // next time is a point between the previous time and target time.
683 samples.emplace_back(Vec<ComplexT>(num_vars)); // sample_factor gives us some point between the two, usually the midpoint.
684
685 auto tracking_success = this->EndgameTrackPath(samples[ii],times[ii-1],times[ii],samples[ii-1]);
686 this->EnsureAtPrecision(times[ii],Precision(samples[ii]));
687
688 if (tracking_success!=SuccessCode::Success)
689 return tracking_success;
690
691 EmitComputedSample(samples[ii], times[ii]);
692 }
693
694 return SuccessCode::Success;
695 }
