59 inline void EmitSystem(std::ostream& out, System const& sys)
60 {
61 auto emit_groups = [&out](char const* keyword, auto const& groups){
62 for (auto const& grp : groups)
63 {
64 out << keyword << " ";
65 for (size_t i = 0; i < grp.size(); ++i)
66 out << (i ? ", " : "") << *grp[i];
67 out << ";\n";
68 }
69 };
70 emit_groups("variable_group", sys.VariableGroups());
71 emit_groups("hom_variable_group", sys.HomVariableGroups());
72
73 // a homotopy is a system too: its path variable must be declared, or the functions that
74 // use it do not parse back and the file does not round-trip (b2#366)
75 if (sys.HavePathVariable())
76 out << "pathvariable " << *sys.GetPathVariable() << ";\n";
77
78 auto functions = sys.NaturalFunctionsAsNodes();
79
80 // Named subexpressions are not stored separately; they are discovered inside the function
81 // trees (nested ones included). Emit each as `name = expr;` before the functions, the way
82 // the classic parser expects a subfunction to be defined ahead of its use.
83 {
84 std::vector<std::shared_ptr<const node::Node>> roots(functions.begin(), functions.end());
85 for (auto const& ne : node::Find<node::NamedExpression>(roots))
86 out << ne->name() << " = " << ne->EntryNode() << ";\n";
87 }
88
89 out << "function ";
90 for (size_t i = 0; i < functions.size(); ++i)
91 out << (i ? ", " : "") << "f" << i;
92 out << ";\n";
93 for (size_t i = 0; i < functions.size(); ++i)
94 out << "f" << i << " = " << functions[i] << ";\n";
95 }

137 inline int PredictorToClassic(tracking::Predictor p)
138 {
139 using P = tracking::Predictor;
140 switch (p)
141 {
142 case P::Constant: return -1;
143 case P::Euler: return 0;
144 case P::Heun: return 1;
145 case P::RK4: return 2;
146 case P::HeunEuler: return 1;
147 case P::RKNorsett34: return 4;
148 case P::RKF45: return 5;
149 case P::RKCashKarp45: return 6;
150 case P::RKDormandPrince56: return 7;
151 case P::RKVerner67: return 8;
152 }
153 return 5; // RKF45 -- the Bertini 2 default, a safe fallback
154 }

160 inline void EmitConfig(std::ostream& out, System const& sys, ClassicWriteOptions const& opt)
161 {
162 auto num = [](double v){ std::ostringstream s; s << std::setprecision(15) << v; return s.str(); };
163 out << "tracktype: " << opt.tracktype << ";\n";
164 out << "mptype: " << opt.mptype << ";\n";
165 out << "odepredictor: " << opt.odepredictor << ";\n";
166 out << "tracktolbeforeeg: " << num(opt.tracktolbeforeeg) << ";\n";
167 out << "tracktolduringeg: " << num(opt.tracktolduringeg) << ";\n";
168 out << "finaltol: " << num(opt.finaltol) << ";\n";
169 out << "maxstepsize: " << num(opt.maxstepsize) << ";\n";
170 out << "stepsuccessfactor: " << num(opt.stepsuccessfactor) << ";\n";
171 out << "stepfailfactor: " << num(opt.stepfailfactor) << ";\n";
172 out << "stepsforincrease: " << opt.stepsforincrease << ";\n";
173 out << "maxnumbersteps: " << opt.maxnumbersteps << ";\n";
174 out << "maxnewtonits: " << opt.maxnewtonits << ";\n";
175 out << "maxcrossedpathresolves: " << opt.maxcrossedpathresolves << ";\n";
176 out << "coefficientbound: " << num(static_cast<double>(sys.CoefficientBound<complex_dbl>())) << ";\n";
177 // A system that is not a polynomial one has no degree bound, and asking for one now
178 // throws. Write the file anyway -- "this is what Bertini 1 would eat, if it could
179 // eat it" is a useful thing to look at -- but say why the setting is absent instead
180 // of inventing a number. `%` is a comment in this format, and our own parser strips
181 // comments, so the file still reads back into b2.
182 if (sys.IsPolynomial())
183 out << "degreebound: " << sys.DegreeBound() << ";\n";
184 else
185 out << "% degreebound: omitted -- this system is not polynomial, so it has none\n";
186 }

189 inline void WriteClassicInput(std::ostream& out, System const& sys,
190 ClassicWriteOptions const& opt = ClassicWriteOptions{})
191 {
192 out << "CONFIG\n";
193 EmitConfig(out, sys, opt);
194 out << "END;\n\nINPUT\n";
195 EmitSystem(out, sys);
196 out << "END;\n";
197 }

200 inline std::string SystemToClassicFile(System const& sys,
201 ClassicWriteOptions const& opt = ClassicWriteOptions{})
202 {
203 std::ostringstream ss;
204 WriteClassicInput(ss, sys, opt);
205 return ss.str();
206 }
