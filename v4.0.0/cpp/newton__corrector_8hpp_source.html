186 template <typename ComplexT>
187 SuccessCode Correct(Vec<ComplexT> & next_space,
188 StepMetadata & meta,
189 System const& S,
190 Vec<ComplexT> const& current_space,
191 ComplexT const& current_time,
192 NumErrorT const& tracking_tolerance,
193 unsigned min_num_newton_iterations,
194 unsigned max_num_newton_iterations,
195 AdaptiveMultiplePrecisionConfig const* AMP_config = nullptr)
196 {
197 #ifndef BERTINI_DISABLE_ASSERTS
198 assert(max_num_newton_iterations >= min_num_newton_iterations && "max number newton iterations must be at least the min.");
199 #endif
200
201 Vec<ComplexT>& step_ref = std::get< Vec<ComplexT> >(step_temp_);
202
203 next_space = current_space;
204 for (unsigned ii = 0; ii < max_num_newton_iterations; ++ii)
205 {
206 //Update the newton iterate by one iteration
207 auto success_code = EvalIterationStep(step_ref, S, next_space, current_time);
208 if(success_code != SuccessCode::Success)
209 return success_code;
210
211 next_space -= step_ref; // step_ref = +J^{-1}f = -(Newton step); see EvalIterationStep
212
213 // Fixed precision: cheap convergence-only loop, no norms / probe / criteria.
214 if (AMP_config == nullptr)
215 {
216 if ( (step_ref.template lpNorm<Eigen::Infinity>() < tracking_tolerance) && (ii >= (min_num_newton_iterations-1)) )
217 return SuccessCode::Success;
218 continue;
219 }
220
221 // Adaptive precision: fill metadata + enforce AMP criteria B and C.
222 Mat<ComplexT>& J_temp_ref = std::get< Mat<ComplexT> >(J_temp_);
223 linalg::PartialPivLU<ComplexT>& LU_ref = std::get< linalg::PartialPivLU<ComplexT> >(LU_);
224
225 meta.norm_delta_z = NumErrorT(step_ref.template lpNorm<Eigen::Infinity>());
226 meta.norm_J = NumErrorT(J_temp_ref.norm());
227 {
228 // Reuse the FIXED probe vector generated once at setup (do NOT regenerate
229 // per call): a fresh random probe direction every Newton step occasionally
230 // produced an inflated ||J^{-1}|| estimate -> spurious HigherPrecisionNecessary
231 // -> precision escalation/grind. A single fixed direction also makes the
232 // condition estimates comparable across steps and keeps tracking deterministic.
233 Vec<ComplexT>& rand_ref = std::get< Vec<ComplexT> >(rand_temp_);
234 Vec<ComplexT>& solve_ref = std::get< Vec<ComplexT> >(solve_temp_);
235 LU_ref.Solve(rand_ref, solve_ref); // reuse the factorization from EvalIterationStep
236 meta.norm_J_inverse = NumErrorT(solve_ref.norm());
237 }
238 meta.condition_number_estimate = NumErrorT(meta.norm_J*meta.norm_J_inverse);
239
240 if ( (meta.norm_delta_z < tracking_tolerance) && (ii >= (min_num_newton_iterations-1)) )
241 return SuccessCode::Success;
242
243 if (!amp::CriterionB<ComplexT>(meta.norm_J, meta.norm_J_inverse, max_num_newton_iterations - ii, tracking_tolerance, meta.norm_delta_z, *AMP_config))
244 return SuccessCode::HigherPrecisionNecessary;
245
246 if (!amp::CriterionC<ComplexT>(meta.norm_J_inverse, next_space, tracking_tolerance, *AMP_config))
247 return SuccessCode::HigherPrecisionNecessary;
248 }
249
250 return SuccessCode::FailedToConverge;
251 }

273 template<typename ComplexT, typename Derived>
274 SuccessCode EvalIterationStep(Vec<ComplexT> & newton_step,
275 const System& S,
276 const Eigen::MatrixBase<Derived>& current_space, const ComplexT& current_time)
277 {
278 Vec<ComplexT>& f_temp_ref = std::get< Vec<ComplexT> >(f_temp_);
279 Mat<ComplexT>& J_temp_ref = std::get< Mat<ComplexT> >(J_temp_);
280
281 linalg::PartialPivLU<ComplexT>& LU_ref = std::get< linalg::PartialPivLU<ComplexT> >(LU_);
282
283 S.SetAndReset<ComplexT>(current_space, current_time);
284 S.EvalInPlace(f_temp_ref);
285 S.JacobianInPlace(J_temp_ref);
286 // Factor a copy of J (J_temp_ref is read again afterward for meta.norm_J). The health
287 // check is folded into Factor().
288 if (LU_ref.Factor(J_temp_ref)!=MatrixSuccessCode::Success)
289 return SuccessCode::MatrixSolveFailure;
290
291 // Solve J*newton_step = f (NOT -f): this lets us skip materializing the negated RHS
292 // temporary. newton_step therefore holds +J^{-1}f = -(true Newton step), so callers
293 // SUBTRACT it (next_space -= newton_step). The convergence test uses lpNorm, which is
294 // sign-insensitive, so it is unaffected.
295 LU_ref.Solve(f_temp_ref, newton_step);
296
297 return SuccessCode::Success;
298
299 }
