131 void Homogenize(VariableGroup const& /*group*/, std::shared_ptr<node::Variable> const& /*hom_var*/)
132 {
133 if (homogeneous_)
134 throw std::runtime_error("LinearFormsBlock::Homogenize: block is already homogenized "
135 "(multiple affine variable groups are not yet supported for linear-forms blocks)");
136 const auto& M = coefficients_highest_precision_;
137 Mat<complex_mp> Mh(M.rows(), M.cols()); // same #cols: n+1
138 Mh.col(0) = M.col(static_cast<Eigen::Index>(num_vars_)); // constant -> h column (front)
139 Mh.rightCols(static_cast<Eigen::Index>(num_vars_)) =
140 M.leftCols(static_cast<Eigen::Index>(num_vars_)); // original variable columns
141 coefficients_highest_precision_ = Mh;
142 num_vars_ += 1; // h is now a real variable; all n+1 columns are variable columns
143 homogeneous_ = true;
144 BuildWorking();
145 }

161 void Describe(std::ostream& out, size_t& row, VariableGroup const& vars, bool verbose) const
162 {
163 auto const& M = coefficients_highest_precision_;
164 const Eigen::Index n = M.rows();
165 if (n == 0)
166 return;
167 const Eigen::Index cap = static_cast<Eigen::Index>(describe_detail::kTerseRowCap);
168 const Eigen::Index shown = (verbose || n <= cap) ? n : cap;
169
170 // placeholder line per form: f_k = c.[x, y, 1]
171 for (Eigen::Index r = 0; r < shown; ++r)
172 {
173 out << " f_" << row++ << " = c.";
174 describe_detail::PrintAugmentedVars(out, vars, num_vars_, homogeneous_);
175 out << "\n";
176 }
177
178 // the actual coefficients below, as a named-expression-style legend (row r <-> f_k above)
179 const int sig = describe_detail::CoeffSig(verbose);
180 out << " c =\n";
181 for (Eigen::Index r = 0; r < shown; ++r)
182 {
183 out << " [ ";
184 for (Eigen::Index c = 0; c < M.cols(); ++c)
185 {
186 if (c) out << ", ";
187 describe_detail::PrintCoeff(out, M(r, c), sig);
188 }
189 out << " ]\n";
190 }
191
192 if (shown < n)
193 {
194 row += static_cast<size_t>(n - shown); // keep the global row index correct
195 out << " ... (" << (n - shown) << " more form" << (n - shown == 1 ? "" : "s")
196 << "; describe(verbose=True) for all)\n";
197 }
198 }

213 void Precision(unsigned new_precision) const
214 {
215 // short-circuit when already materialized here. Each holder of mp values keeps its
216 // own "materialized at" tag; System deliberately keeps none and simply fans out on
217 // every evaluation, which is cheap precisely because of this early return (ADR-0057).
218 if (precision_==new_precision)
219 return;
220 if (new_precision > DoublePrecision())
221 {
222 auto& wm = std::get<Mat<complex_mp>>(coefficients_working_);
223 for (Eigen::Index r = 0; r < wm.rows(); ++r)
224 for (Eigen::Index c = 0; c < wm.cols(); ++c)
225 {
226 wm(r, c).precision(new_precision);
227 if (new_precision > precision_)
228 wm(r, c) = coefficients_highest_precision_(r, c);
229 }
230 }
231 precision_ = new_precision;
232 }

242 template <typename T>
243 void EvalInPlace(Eigen::Ref<Vec<T>> result, Vec<T> const& vars, T const& /*path_value*/) const
244 {
245 SyncPrecision(vars);
246 // affine: f(x) = W * [x ; 1] (the trailing 1 carries each row's constant in the last
247 // column). homogeneous (post-Homogenize): every column is a variable column, so it is
248 // just W * vars (the old constant is now the homogenizing variable's coefficient).
249 if (homogeneous_)
250 result.noalias() = Working<T>() * vars;
251 else
252 result.noalias() = Working<T>() * Augment<T>(vars);
253 }

266 template <typename T>
267 void JacobianInPlace(Eigen::Ref<Mat<T>> J, Vec<T> const& vars, T const& /*path_value*/) const
268 {
269 SyncPrecision(vars);
270 // homogeneous: every column is d f / d x. affine: drop the trailing constant column.
271 if (homogeneous_)
272 J = Working<T>();
273 else
274 J = Working<T>().leftCols(static_cast<Eigen::Index>(num_vars_));
275 }

294 void SyncPrecision(Vec<T> const& vars) const
295 {
296 if constexpr (!std::is_same<T, complex_dbl>::value)
297 {
298 if (vars.size() > 0)

313 template <typename T>
314 Vec<T> Augment(Vec<T> const& vars) const
315 {
316 Vec<T> aug(static_cast<Eigen::Index>(num_vars_ + 1));
317 aug.head(static_cast<Eigen::Index>(num_vars_)) = vars;
318 T one(1);
319 if constexpr (!std::is_same<T, complex_dbl>::value)
320 one.precision(precision_);
321 aug(static_cast<Eigen::Index>(num_vars_)) = one;
322 return aug;
323 }

325 void BuildWorking() const
326 {
327 const auto& M = coefficients_highest_precision_;
328 auto& wd = std::get<Mat<complex_dbl>>(coefficients_working_);
329 auto& wm = std::get<Mat<complex_mp>>(coefficients_working_);
330 wd.resize(M.rows(), M.cols());
331 wm.resize(M.rows(), M.cols());
332 for (Eigen::Index r = 0; r < M.rows(); ++r)
333 for (Eigen::Index c = 0; c < M.cols(); ++c)
334 {
335 wd(r, c) = complex_dbl(M(r, c));
336 wm(r, c) = M(r, c);
337 }
338 }
