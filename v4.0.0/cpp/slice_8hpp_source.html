129 static Slice RandomReal(VariableGroup const& v, unsigned dim, bool homogeneous = false, bool orthogonal = true, Vec<complex_mp> const* through_point = nullptr)
130 {
131 typedef void (*funtype) (complex_mp&, unsigned); // the type for number generation
132 // bounded-modulus draw (away from 0 and infinity), matching patches and the start systems;
133 // kept REAL so a real slice stays real. The orthogonal path is real too (real=true below):
134 // it QR-factors a matrix of REAL units, yielding a real orthogonal coefficient block (issue
135 // #294 -- previously the orthogonal path hardcoded the complex orthonormal matrix, so a
136 // "real" slice came out complex).
137 funtype gen = bertini::multiprecision::RandomRealBoundedModulusAssign;
138 return Make(v, dim, homogeneous, orthogonal, /*real=*/true, gen, through_point);
139 }

173 static Slice Make(VariableGroup const& v, unsigned dim, bool homogeneous, bool orthogonal, bool real, std::function<void(complex_mp&, unsigned)> gen, Vec<complex_mp> const* through_point = nullptr)
174 {
175 const unsigned num_vars = static_cast<unsigned>(v.size());
176
177 // more forms than variables cannot be independent: the orthonormalization below would
178 // hand back dependent rows without a word, and the slice would cut nothing new (b2#380)
179 if (dim > num_vars)
180 throw std::invalid_argument("Slice: " + std::to_string(dim) + " linear forms on "
181 + std::to_string(num_vars) + " variables; a slice cannot have more forms than variables");
182 if (homogeneous && through_point && dim + 1 > num_vars)
183 throw std::invalid_argument("Slice: " + std::to_string(dim) + " homogeneous linear forms through a point on "
184 + std::to_string(num_vars) + " variables; the forms live in the point's orthogonal complement, which has "
185 + std::to_string(num_vars - 1) + " dimensions");
186
187 if (through_point && static_cast<unsigned>(through_point->size()) != num_vars)
188 throw std::runtime_error("Slice: through_point must have num_variables entries");
189
190 Mat<complex_mp> coeffs(dim, num_vars); // the variable coefficients (one row per form)
191
192 if (orthogonal)
193 {
194 // conjugate-orthonormal coefficient matrix (orthonormal linear forms), drawn the b1 way
195 // via RandomConjugateOrthonormalMatrix (ADR-0041) -- it generates square and truncates,
196 // so the old transpose dance is gone. Built at max precision, like the rest of the slice.
197 // A real slice QR-factors a matrix of REAL units (a real orthogonal block); a complex
198 // slice uses complex units. (Issue #294: the real path must stay real.)
199 auto prev_precision = DefaultPrecision();
200 DefaultPrecision(MaxPrecisionAllowed());
201 if (real)
202 coeffs = bertini::RandomConjugateOrthonormalMatrix<real_mp>(dim, num_vars)
203 .unaryExpr([](real_mp const& r){ return complex_mp(r); });
204 else
205 coeffs = bertini::RandomConjugateOrthonormalMatrix<complex_mp>(dim, num_vars);
206 DefaultPrecision(prev_precision);
207 }
208 else
209 {
210 for (unsigned ii(0); ii < dim; ++ii)
211 for (unsigned jj(0); jj < num_vars; ++jj)
212 gen(coeffs(ii, jj), MaxPrecisionAllowed());
213 }
214
215 assert(static_cast<unsigned>(coeffs.rows()) == dim);
216 assert(static_cast<unsigned>(coeffs.cols()) == num_vars);
217
218 // Through a point. Homogeneous: project the rows into p's orthogonal complement (a.p = 0),
219 // zero constant column. Affine: the constant column is -A*p, so every form vanishes at p.
220 if (through_point && homogeneous)
221 return FromCoefficients(v, AugmentHomogeneousThroughPoint(coeffs, *through_point, orthogonal), /*homogeneous=*/true);
222 if (through_point)
223 return FromCoefficients(v, AugmentAffineThroughPoint(coeffs, *through_point), /*homogeneous=*/false);
224
225 // Assemble the augmented matrix: [ coeffs | constants ]. A homogeneous slice's constant
226 // column is zero; otherwise it is freshly generated.
227 Mat<complex_mp> augmented(dim, num_vars + 1);
228 augmented.leftCols(num_vars) = coeffs;
229 if (homogeneous)
230 augmented.col(num_vars).setZero();
231 else
232 for (unsigned ii(0); ii < dim; ++ii)
233 gen(augmented(ii, num_vars), MaxPrecisionAllowed());
234
235 return FromCoefficients(v, augmented, homogeneous);
236 }

317 Slice Concatenate(Slice const& other) const
318 {
319 if (NumVariables() != other.NumVariables())
320 throw std::runtime_error("Slice::Concatenate requires both slices to be on the same number of variables");
321
322 Mat<complex_mp> const& A = Coefficients();
323 Mat<complex_mp> const& B = other.Coefficients();
324 Mat<complex_mp> stacked(A.rows() + B.rows(), A.cols());
325 stacked.topRows(A.rows()) = A;
326 stacked.bottomRows(B.rows()) = B;
327
328 return FromCoefficients(sliced_vars_, stacked, is_homogeneous_ && other.is_homogeneous_);
329 }

430 static Mat<complex_mp> AugmentAffineThroughPoint(Mat<complex_mp> const& coeffs, Vec<complex_mp> const& point)
431 {
432 const auto num_vars = coeffs.cols();
433
434 auto prev_precision = DefaultPrecision();
435 DefaultPrecision(MaxPrecisionAllowed());
436
437 Mat<complex_mp> A = coeffs; // copies, re-materialized at max precision below
438 Vec<complex_mp> p = point;
439 bertini::Precision(A, MaxPrecisionAllowed()); // qualify: class has member Precision overloads
440 bertini::Precision(p, MaxPrecisionAllowed());
441
442 Mat<complex_mp> augmented(coeffs.rows(), num_vars + 1);
443 augmented.leftCols(num_vars) = A;
444 augmented.col(num_vars) = -(A * p); // b = -A*p, so A*p + b = 0
445
446 DefaultPrecision(prev_precision);
447 return augmented;
448 }

464 static Mat<complex_mp> AugmentHomogeneousThroughPoint(Mat<complex_mp> const& coeffs, Vec<complex_mp> const& point, bool orthogonal)
465 {
466 const auto dim = coeffs.rows();
467 const auto num_vars = coeffs.cols();
468
469 auto prev_precision = DefaultPrecision();
470 DefaultPrecision(MaxPrecisionAllowed());
471
472 Mat<complex_mp> A = coeffs;
473 Vec<complex_mp> p = point;
474 bertini::Precision(A, MaxPrecisionAllowed()); // qualify: class has member Precision overloads
475 bertini::Precision(p, MaxPrecisionAllowed());
476
477 real_mp p_norm_sq = p.squaredNorm(); // pbar.p = ||p||^2, real-positive for p != 0
478 if (p_norm_sq == real_mp(0))
479 throw std::runtime_error("Slice: a homogeneous slice through a point needs a nonzero point");
480
481 // project every row into p's orthogonal complement (bilinear a.p = 0): subtract the
482 // rank-one correction (A*p) pbar^T / ||p||^2.
483 Vec<complex_mp> Ap = A * p; // bilinear a_i . p, one per row
484 A = A - (Ap / complex_mp(p_norm_sq)) * p.conjugate().transpose();
485
486 if (orthogonal)
487 {
488 // orthonormalize the projected rows via a QR of A^T: the columns of Q span the row
489 // space of A (which lies in p's complement), so Q^T's rows stay orthogonal to p.
490 Eigen::HouseholderQR<Mat<complex_mp>> qr(A.transpose());
491 Mat<complex_mp> Q = qr.householderQ() * Mat<complex_mp>::Identity(num_vars, num_vars);
492 A = Q.leftCols(dim).transpose();
493 }
494
495 Mat<complex_mp> augmented(dim, num_vars + 1);
496 augmented.leftCols(num_vars) = A;
497 augmented.col(num_vars).setZero(); // homogeneous: no constant term
498
499 DefaultPrecision(prev_precision);
500 return augmented;
501 }
