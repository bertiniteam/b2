244 unsigned int maximum_precision = 300;
245
246 // Note: a single setting -- Bertini 1's StepsForIncrease, i.e.
247 // SteppingConfig::consecutive_successful_steps_before_stepsize_increase -- gates BOTH stepsize
248 // increase AND precision decrease (the required number of consecutive successful steps).
249 // Precision decrease is additionally subject to B1's digits-margin hysteresis (see
250 // ExtraDigitsBeforePrecisionDecrease in amp_tracker.hpp). The old, duplicate AMP-config setting
251 // `consecutive_successful_steps_before_precision_decrease` (a B2 deviation) was removed.
252
253 unsigned max_num_precision_decreases = 10;

264 void SetBoundsFrom(System const& sys)
265 {
266 using std::pow;
267
268 // Refuse before DegreeBound() does, so the message is about what the caller was
269 // trying to do rather than about degrees. The two bounds are on the error of
270 // evaluating the Jacobian and the functions; the degree and coefficient bounds are
271 // merely the recipe for them that holds for polynomials. There is no accepted
272 // recipe for an analytic system -- see issue #439 -- so we do not invent one.
273 if (!sys.IsPolynomial())
274 throw std::runtime_error("adaptive precision derives its error bounds from the "
275 "degree of the system, and this system is not a polynomial one, so it has no "
276 "degree. Two ways on: track with a fixed-precision tracker, at double or at "
277 "multiple precision, which needs no such bound; or set this config's "
278 "jacobian_eval_error_bound and function_eval_error_bound yourself and hand it "
279 "to the tracker, choosing values you can defend for your system.");
280
281 linear_solve_error_bound = pow(NumErrorT(sys.NumVariables()),2);
282 degree_bound = sys.DegreeBound();
283 coefficient_bound = sys.CoefficientBound<complex_dbl>();
284 }
285
286
294 void SetErrorBoundsFromDegreeAndCoefficient()
295 {
296 jacobian_eval_error_bound = degree_bound*(degree_bound-NumErrorT(1))*coefficient_bound;
297 function_eval_error_bound = degree_bound*coefficient_bound;
298 }
299
302 void SetAMPConfigFrom(System const& sys)
303 {
304 SetBoundsFrom(sys);
305 SetErrorBoundsFromDegreeAndCoefficient();
306 }
307
317 AdaptiveMultiplePrecisionConfig() : coefficient_bound(1000), degree_bound(5), linear_solve_error_bound(1), safety_digits_1(1), safety_digits_2(1), maximum_precision(300)
318 {
319 SetErrorBoundsFromDegreeAndCoefficient();
320 }
