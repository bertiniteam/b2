197 template <typename OutputIterator, typename T>
198 static bool generate(OutputIterator sink, Vec<T> const& c)
199 {
200 auto s = c.size();
201 for (decltype(s) ii=0; ii<s; ++ii)
202 {
203 bool b = generate(sink, c(ii));
204 if (!b)
205 return b;
206 boost::spirit::karma::generate(sink, (boost::spirit::karma::char_), '\n');
207 }
208 return true;
209 }
