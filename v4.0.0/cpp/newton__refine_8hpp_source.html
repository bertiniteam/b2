85template <typename ComplexT>
86NewtonRefineResult<ComplexT> NewtonRefine(System const& S,
87 Vec<ComplexT> const& start,
88 NumErrorT tolerance,
89 unsigned max_iterations,
90 ComplexT time = ComplexT(0))
91{
92 const auto n_funcs = S.NumTotalFunctions();
93 const auto n_vars = S.NumVariables();
94 if (n_funcs != n_vars)
95 {
96 std::stringstream ss;
97 ss << "NewtonRefine requires a SQUARE system, but this one has "
98 << n_funcs << " total functions (user functions plus patches) over "
99 << n_vars << " variables. Square an overdetermined system by "
100 << "randomization (multiply by a generic full-rank matrix) before refining.";
101 throw std::runtime_error(ss.str());
102 }
103 if (start.size() != static_cast<Eigen::Index>(n_vars))
104 {
105 std::stringstream ss;
106 ss << "NewtonRefine: start point has " << start.size()
107 << " coordinates but the system has " << n_vars << " variables.";
108 throw std::runtime_error(ss.str());
109 }
110
111 NewtonRefineResult<ComplexT> result{SuccessCode::FailedToConverge,
112 start, static_cast<NumErrorT>(-1), 0};
113
114 Vec<ComplexT> f(n_funcs);
115 Mat<ComplexT> J(n_funcs, n_vars);
116 Vec<ComplexT> step(n_vars);
117 linalg::PartialPivLU<ComplexT> lu;
118 lu.ChangeSize(static_cast<unsigned>(n_vars)); // the LU workspace is stateful:
119 // without this, Factor/Solve run
120 // over a 0-dimensional system
121
122 for (unsigned it = 0; it < max_iterations; ++it)
123 {
124 if (S.HavePathVariable())
125 S.SetAndReset(result.point, time);
126 else
127 S.SetAndReset(result.point);
128 S.EvalInPlace(f);
129 S.JacobianInPlace(J);
130
131 if (lu.Factor(J) != MatrixSuccessCode::Success)
132 {
133 result.code = SuccessCode::MatrixSolveFailure;
134 return result;
135 }
136 lu.Solve(f, step); // step = +J^{-1} f = -(Newton step)
137 result.point -= step;
138 result.iterations = it + 1;
139 result.achieved = static_cast<NumErrorT>(
140 step.template lpNorm<Eigen::Infinity>());
141
142 if (result.achieved <= tolerance)
143 {
144 result.code = SuccessCode::Success;
145 return result;
146 }
147 }
148 return result;
149}
