54 // Reset on Initializing, never on TrackingStarted, and never unsubscribe: a reset at
55 // TrackingStarted comes after the refinement and misses its increases, and an observer
56 // that unsubscribes reports one track forever after (#378).
57 template<class TrackerT>
58 class FirstPrecisionRecorder
59 : public TypedObserver< FirstPrecisionRecorder<TrackerT>, TrackerT,
60 Initializing<typename TrackerTraits<TrackerT>::EventEmitterType,
61 typename TrackerTraits<TrackerT>::BaseComplexT>,
62 TrackingStarted<typename TrackerTraits<TrackerT>::EventEmitterType>,
63 PrecisionChanged<typename TrackerTraits<TrackerT>::EventEmitterType> >
64 { BOOST_TYPE_INDEX_REGISTER_CLASS
65
66 using EmitterT = typename TrackerTraits<TrackerT>::EventEmitterType;
67 using BCT = typename TrackerTraits<TrackerT>::BaseComplexT;
68
69 public:
70
72 ObserveResult OnEvent(Initializing<EmitterT, BCT> const&)
73 {
74 have_start_ = false;
75 precision_increased_ = false;
76 starting_precision_ = 0;
77 next_precision_ = 0;
78 time_of_first_increase_ = typename TrackerTraits<TrackerT>::BaseComplexT{};
79 return ObserveResult::KeepObserving;
80 }
81
83 ObserveResult OnEvent(TrackingStarted<EmitterT> const& e)
84 {
85 if (!have_start_)
86 {
87 starting_precision_ = e.Get().CurrentPrecision();
88 have_start_ = true;
89 }
90 return ObserveResult::KeepObserving;
91 }
92
94 ObserveResult OnEvent(PrecisionChanged<EmitterT> const& e)
95 {
96 if (!have_start_)
97 {
98 starting_precision_ = e.Previous();
99 have_start_ = true;
100 }
101 if (!precision_increased_ && e.Next() > e.Previous())
102 {
103 precision_increased_ = true;
104 next_precision_ = e.Next();
105 time_of_first_increase_ = e.Get().CurrentTime();
106 }
107 return ObserveResult::KeepObserving;
108 }
109
111 unsigned StartPrecision() const
112 {
113 return starting_precision_;
114 }
115
117 unsigned NextPrecision() const
118 {
119 return next_precision_;
120 }
121
123 bool DidPrecisionIncrease() const
124 {
125 return precision_increased_;
126 }
127
129 typename TrackerTraits<TrackerT>::BaseComplexT TimeOfIncrease() const
130 {
131 return time_of_first_increase_;
132 }
133
134 virtual ~FirstPrecisionRecorder() = default;
135
136 private:
137
138 bool have_start_ = false;
139 unsigned starting_precision_ = 0;
140 unsigned next_precision_ = 0;
141 bool precision_increased_ = false;
142 typename TrackerTraits<TrackerT>::BaseComplexT time_of_first_increase_{};
143 };

378 template<class TrackerT>
379 class GoryDetailLogger : public Observer<TrackerT>
380 { BOOST_TYPE_INDEX_REGISTER_CLASS
381 public:
382
383 using EmitterT = typename TrackerTraits<TrackerT>::EventEmitterType;
384
385 virtual ~GoryDetailLogger() = default;
386
388 virtual ObserveResult Observe(AnyEvent const& e) override
389 {
390
391
392 if (auto p = dynamic_cast<const Initializing<EmitterT,complex_dbl>*>(&e))
393 {
394 // digits to print come from the VALUES being printed -- not from the ambient
395 // default, which need not be the precision anything here was computed at, and
396 // not from the System, which no longer carries one (ADR-0057)
397 BOOST_LOG_TRIVIAL(severity_level::debug) << std::setprecision(static_cast<int>(DoublePrecision()))
398 << "initializing in double, tracking path\nfrom\tt = "
399 << p->StartTime() << "\nto\tt = " << p->EndTime()
400 << "\n from\tx = \n" << p->StartPoint()
401 << "\n tracking system " << p->Get().GetSystem() << "\n\n";
402 }
403 else if (auto p = dynamic_cast<const Initializing<EmitterT,complex_mp>*>(&e))
404 {
405 // ditto: ask the point being logged
406 BOOST_LOG_TRIVIAL(severity_level::debug) << std::setprecision(static_cast<int>(Precision(p->StartPoint())))
407 << "initializing in multiprecision, tracking path\nfrom\tt = " << p->StartTime() << "\nto\tt = " << p->EndTime() << "\n from\tx = \n" << p->StartPoint()
408 << "\n tracking system " << p->Get().GetSystem() << "\n\n";
409 }
410
411 else if(auto p = dynamic_cast<const TrackingEnded<EmitterT>*>(&e))
412 BOOST_LOG_TRIVIAL(severity_level::trace) << "tracking ended";
413
414 else if (auto p = dynamic_cast<const NewStep<EmitterT>*>(&e))
415 {
416 auto& t = p->Get();
417 BOOST_LOG_TRIVIAL(severity_level::trace) << "Tracker iteration " << t.NumTotalStepsTaken() << "\ncurrent precision: " << t.CurrentPrecision();
418
419
420 BOOST_LOG_TRIVIAL(severity_level::trace) << std::setprecision(static_cast<int>(t.CurrentPrecision()))
421 << "t = " << t.CurrentTime()
422 << "\ncurrent stepsize: " << t.CurrentStepsize()
423 << "\ndelta_t = " << t.DeltaT()
424 << "\ncurrent x size = " << t.CurrentPoint().size()
425 << "\ncurrent x = " << t.CurrentPoint();
426 }
427
428
429
430 else if (auto p = dynamic_cast<const SingularStartPoint<EmitterT>*>(&e))
431 BOOST_LOG_TRIVIAL(severity_level::trace) << "singular start point";
432 else if (auto p = dynamic_cast<const InfinitePathTruncation<EmitterT>*>(&e))
433 BOOST_LOG_TRIVIAL(severity_level::trace) << "tracker iteration indicated going to infinity, truncated path";
434
435
436
437
438 else if (auto p = dynamic_cast<const SuccessfulStep<EmitterT>*>(&e))
439 {
440 BOOST_LOG_TRIVIAL(severity_level::trace) << "tracker iteration successful\n\n\n";
441 }
442
443 else if (auto p = dynamic_cast<const FailedStep<EmitterT>*>(&e))
444 {
445 BOOST_LOG_TRIVIAL(severity_level::trace) << "tracker iteration unsuccessful\n\n\n";
446 }
447
448
449
450
451
452
453 else if (auto p = dynamic_cast<const SuccessfulPredict<EmitterT,complex_mp>*>(&e))
454 {
455 BOOST_LOG_TRIVIAL(severity_level::trace) << std::setprecision(static_cast<int>(Precision(p->ResultingPoint()))) << "prediction successful (complex_mp), result:\n" << p->ResultingPoint();
456 }
457 else if (auto p = dynamic_cast<const SuccessfulPredict<EmitterT,complex_dbl>*>(&e))
458 {
459 BOOST_LOG_TRIVIAL(severity_level::trace) << std::setprecision(static_cast<int>(Precision(p->ResultingPoint()))) << "prediction successful (complex_dbl), result:\n" << p->ResultingPoint();
460 }
461
462 else if (auto p = dynamic_cast<const SuccessfulCorrect<EmitterT,complex_mp>*>(&e))
463 {
464 BOOST_LOG_TRIVIAL(severity_level::trace) << std::setprecision(static_cast<int>(Precision(p->ResultingPoint()))) << "correction successful (complex_mp), result:\n" << p->ResultingPoint();
465 }
466 else if (auto p = dynamic_cast<const SuccessfulCorrect<EmitterT,complex_dbl>*>(&e))
467 {
468 BOOST_LOG_TRIVIAL(severity_level::trace) << std::setprecision(static_cast<int>(Precision(p->ResultingPoint()))) << "correction successful (complex_dbl), result:\n" << p->ResultingPoint();
469 }
470
471
472 else if (auto p = dynamic_cast<const PredictorHigherPrecisionNecessary<EmitterT>*>(&e))
473 BOOST_LOG_TRIVIAL(severity_level::trace) << "Predictor, higher precision necessary";
474 else if (auto p = dynamic_cast<const CorrectorHigherPrecisionNecessary<EmitterT>*>(&e))
475 BOOST_LOG_TRIVIAL(severity_level::trace) << "corrector, higher precision necessary";
476
477
478
479 else if (auto p = dynamic_cast<const CorrectorMatrixSolveFailure<EmitterT>*>(&e))
480 BOOST_LOG_TRIVIAL(severity_level::trace) << "corrector, matrix solve failure or failure to converge";
481 else if (auto p = dynamic_cast<const PredictorMatrixSolveFailure<EmitterT>*>(&e))
482 BOOST_LOG_TRIVIAL(severity_level::trace) << "predictor, matrix solve failure or failure to converge";
483 else if (auto p = dynamic_cast<const FirstStepPredictorMatrixSolveFailure<EmitterT>*>(&e))
484 BOOST_LOG_TRIVIAL(severity_level::trace) << "Predictor, matrix solve failure in initial solve of prediction";
485
486
487 else if (auto p = dynamic_cast<const PrecisionChanged<EmitterT>*>(&e))
488 BOOST_LOG_TRIVIAL(severity_level::debug) << "changing precision from " << p->Previous() << " to " << p->Next();
489
490 else
491 BOOST_LOG_TRIVIAL(severity_level::debug) << "unlogged event, of type: " << boost::typeindex::type_id_runtime(e).pretty_name();
492
493 return ObserveResult::KeepObserving;
494 }
