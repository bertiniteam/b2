893 template<typename T>
894 void SetVariables(const Vec<T> & new_values) const
895 {
896 if (new_values.size()!= static_cast<Eigen::Index>(NumVariables()))
897 {
898 throw std::runtime_error("variable vector of different length from system-owned variables in SetVariables");
899 }
900
901 // ALIGN to the incoming point rather than refusing it (#377). A System's
902 // precision is not an invariant the caller must maintain by hand: the caller's
903 // intent is always just "evaluate f at x", and x carries the precision that
904 // evaluation should happen at. Aligning here pushes the new precision down to
905 // every block (each block re-tags its SLP's memory and rebuilds its constants
906 // from their exact recipes), to the stored path value, and to the patch, so the
907 // whole System is consistent for the evaluation that follows.
908 //
909 // Refusing was not merely unergonomic, it was inescapable in one direction: a
910 // block's SLP takes its memory precision from the ambient DefaultPrecision() at
911 // lazy compile time while the System keeps whatever it was told, and once those
912 // diverge, System::precision(n) cannot repair it -- both it and
913 // StraightLineProgram::precision(n) short-circuit when handed the value they
914 // already hold. See the SLP-side note in SetVariableValues.
915 //
916 // Deliberately NOT behind BERTINI_DISABLE_PRECISION_CHECKS: this is required
917 // behaviour, not a debug assertion. A system with no variables (a constant) has
918 // an empty point and so carries no precision to read.
919 // NO fan-out. Every evaluable type self-aligns from the point it is handed --
920 // each block, the patch and the SLP all do it under SyncPrecision -- so there is
921 // nothing to push down from here and nothing to keep in sync (ADR-0057).
922
923 // Blocks are value-in: the polynomial block feeds this stored vector into its SLP, the
924 // structured blocks compute on it directly, and the patch reads it too (see
925 // EvalBlocksInPlace). The shared Variable nodes are no longer written during evaluation
926 // (ADR-0027), so the node DAG stays read-only across threads.
927 std::get<Vec<T> >(current_variable_values_) = new_values;
928 }

1442 template<typename T>
1443 auto InfinityNormOfDehomogenized(Vec<T> const& x) const
1444 {
1445 auto const d = DehomogenizePoint(x);
1446 if (!HaveAuxiliaryCoordinates())
1447 return d.template lpNorm<Eigen::Infinity>();
1448
1449 // the value type, not decltype(abs(...)): a multiprecision abs() is an expression
1450 // template, and a variable of that type cannot be reassigned in the loop
1451 using std::abs;
1452 using RealT = typename NumTraits<T>::Real;
1453 RealT biggest{0};
1454 for (Eigen::Index k{0}; k < d.size(); ++k)
1455 if (!CoordinateIsAuxiliary(static_cast<unsigned>(k)))
