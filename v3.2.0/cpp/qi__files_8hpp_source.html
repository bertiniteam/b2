72inline void StripUTF8BOM(std::string& s)
73{
74 if (s.size() >= 3 &&
75 static_cast<unsigned char>(s[0]) == 0xEF &&
76 static_cast<unsigned char>(s[1]) == 0xBB &&
77 static_cast<unsigned char>(s[2]) == 0xBF)
78 {
79 s.erase(0, 3);
80 }
81}

84inline std::string FormatParseError(
85 std::string::const_iterator begin,
86 std::string::const_iterator end,
87 std::string::const_iterator err_pos,
88 boost::spirit::info const& what,
89 std::string const& parser_name)
90{
91 int line = 1 + (int)std::count(begin, err_pos, '\n');
92 auto rev = std::find(std::make_reverse_iterator(err_pos),
93 std::make_reverse_iterator(begin), '\n');
94 int col = 1 + (int)std::distance(rev.base(), err_pos);
95
96 std::string remaining(err_pos, end);
97 if (remaining.size() > 60)
98 remaining = remaining.substr(0, 60) + "...";
99 if (remaining.empty())
100 remaining = "<end of input>";
101
102 std::ostringstream oss;
103 oss << "[" << parser_name << "] parse error at line " << line
104 << ", col " << col << ":\n"
105 << " expected: " << what << "\n"
106 << " found: \"" << remaining << "\"\n";
107 return oss.str();
108}
