137 struct CauchyConfig
138 {
139 using T = NumErrorT;
140
141 T cycle_cutoff_time = T(1)/T(100000000);
142 T ratio_cutoff_time = T(1)/T(100000000000000);
143 T minimum_for_c_over_k_stabilization = T(3)/T(4);
144 unsigned int num_needed_for_stabilization = 3;
145 T maximum_cauchy_ratio = T(1)/T(2);
146 unsigned int fail_safe_maximum_cycle_number = 250;
147
148 // Number of consecutive circle-tracked Cauchy approximations that must report the SAME cycle
149 // number before a converged approximation is trusted. Guards against an UNRELIABLE cycle
150 // number: when the working precision is too low to close the loop accurately (the circle of
151 // radius |t| has enough path variation that tracking error breaks closure -- NOT monodromy:
152 // at high precision a nonsingular endpoint closes at cycle 1 for every radius), the
153 // loop-closing count thrashes (e.g. 41, 14, 36). Requiring N consecutive identical cycle
154 // numbers refuses to accept convergence until the estimate has genuinely settled.
155 // See z_notes/20260629_endgame_stepsize_reset_rootcause.
156 unsigned int num_consecutive_same_cycle_number = 2;
157
158 };
