53inline std::string ExactDoubleText(double v)

99{
100 auto const digits = static_cast<unsigned>(triple.at(2).as_int64());
101 complex_mp z(real_mp(std::string(triple.at(0).as_string()), digits),

108template <typename ComplexT>
109boost::json::array EncodePoint(Vec<ComplexT> const& point)
110{
111 boost::json::array out;
112 for (Eigen::Index ii = 0; ii < point.size(); ++ii)
113 out.push_back(EncodeComplexScalar(point(ii)));
114 return out;
115}

118template <typename ComplexT>
119Vec<ComplexT> DecodePoint(boost::json::array const& coords)
120{
121 Vec<ComplexT> out(static_cast<Eigen::Index>(coords.size()));
122 for (std::size_t ii = 0; ii < coords.size(); ++ii)
123 out(static_cast<Eigen::Index>(ii)) = DecodeComplexScalar<ComplexT>(coords[ii].as_array());
124 return out;
125}

136inline std::string CanonicalName(SuccessCode code)
137{
138 switch (code)
139 {
140 case SuccessCode::NeverStarted: return "NeverStarted";
141 case SuccessCode::Success: return "Success";
142 case SuccessCode::HigherPrecisionNecessary: return "HigherPrecisionNecessary";
143 case SuccessCode::ReduceStepSize: return "ReduceStepSize";
144 case SuccessCode::GoingToInfinity: return "GoingToInfinity";
145 case SuccessCode::FailedToConverge: return "FailedToConverge";
146 case SuccessCode::MatrixSolveFailure: return "MatrixSolveFailure";
147 case SuccessCode::MatrixSolveFailureFirstPartOfPrediction: return "MatrixSolveFailureFirstPartOfPrediction";
148 case SuccessCode::MaxNumStepsTaken: return "MaxNumStepsTaken";
149 case SuccessCode::MaxPrecisionReached: return "MaxPrecisionReached";
150 case SuccessCode::MinStepSizeReached: return "MinStepSizeReached";
151 case SuccessCode::Failure: return "Failure";
152 case SuccessCode::SingularStartPoint: return "SingularStartPoint";
153 case SuccessCode::ExternallyTerminated: return "ExternallyTerminated";
154 case SuccessCode::MinTrackTimeReached: return "MinTrackTimeReached";
155 case SuccessCode::SecurityMaxNormReached: return "SecurityMaxNormReached";
156 case SuccessCode::CycleNumTooHigh: return "CycleNumTooHigh";
157 case SuccessCode::FailedToSelectPrecisionAndStepsize: return "FailedToSelectPrecisionAndStepsize";

174inline std::string CoarsePathStatus(SuccessCode code)
175{
176 if (code == SuccessCode::Success)
177 return "success";
178 if (code == SuccessCode::GoingToInfinity || code == SuccessCode::SecurityMaxNormReached)
179 return "diverged";

190template <typename ComplexT>
191boost::json::object EncodeFullPathResult(parallel::FullPathResult<ComplexT> const& r)
192{
193 boost::json::object out;
194 out["pre_endgame_success_code"] = static_cast<std::int64_t>(r.pre_endgame_success_code);
195 out["pre_endgame_success_code_name"] = CanonicalName(r.pre_endgame_success_code);
196 out["boundary_point"] = EncodePoint(r.boundary_point);
197 out["boundary_stepsize"] = EncodeRealScalar(r.boundary_stepsize);
198 out["boundary_precision"] = static_cast<std::int64_t>(r.boundary_precision);
199 out["endgame_success_code"] = static_cast<std::int64_t>(r.endgame_success_code);
200 out["endgame_success_code_name"] = CanonicalName(r.endgame_success_code);
201 out["endpoint"] = EncodePoint(r.solution);
202 out["function_residual"] = ExactDoubleText(r.function_residual);
203 out["condition_number"] = ExactDoubleText(r.condition_number);
204 out["newton_residual"] = ExactDoubleText(r.newton_residual);
205 out["final_time_used"] = EncodeComplexScalar(r.final_time_used);
206 out["accuracy_estimate"] = ExactDoubleText(r.accuracy_estimate);
207 out["accuracy_estimate_user_coords"] = ExactDoubleText(r.accuracy_estimate_user_coords);
208 out["cycle_num"] = static_cast<std::int64_t>(r.cycle_num);
209 out["precision_digits"] = static_cast<std::int64_t>(r.precision_digits);
210 out["accuracy_digits"] = static_cast<std::int64_t>(r.accuracy_digits);
211 out["precision_changed"] = r.precision_changed;
212 out["time_of_first_prec_increase"] = EncodeComplexScalar(r.time_of_first_prec_increase);
213 out["max_precision_used"] = static_cast<std::int64_t>(r.max_precision_used);
214 out["path_time_seconds"] = ExactDoubleText(r.path_time_seconds);
215 return out;
216}

219template <typename ComplexT>
220parallel::FullPathResult<ComplexT> DecodeFullPathResult(boost::json::object const& rec,
221 std::size_t path_index)
222{
223 using RealT = typename parallel::FullPathResult<ComplexT>::RealT;
224 parallel::FullPathResult<ComplexT> r;
225 r.path_index = path_index;
226 r.pre_endgame_success_code =
227 static_cast<SuccessCode>(rec.at("pre_endgame_success_code").as_int64());
228 r.boundary_point = DecodePoint<ComplexT>(rec.at("boundary_point").as_array());
229 if constexpr (std::is_same<RealT, double>::value)
230 r.boundary_stepsize = DoubleFromText(std::string(rec.at("boundary_stepsize").as_string()));
231 else
232 r.boundary_stepsize = RealT(std::string(rec.at("boundary_stepsize").as_string()));
233 r.boundary_precision = static_cast<unsigned>(rec.at("boundary_precision").as_int64());
234 r.endgame_success_code = static_cast<SuccessCode>(rec.at("endgame_success_code").as_int64());
235 r.solution = DecodePoint<ComplexT>(rec.at("endpoint").as_array());
236 r.function_residual = DoubleFromText(std::string(rec.at("function_residual").as_string()));
237 r.condition_number = DoubleFromText(std::string(rec.at("condition_number").as_string()));
238 r.newton_residual = DoubleFromText(std::string(rec.at("newton_residual").as_string()));
239 r.final_time_used = DecodeComplexScalar<ComplexT>(rec.at("final_time_used").as_array());
240 r.accuracy_estimate = DoubleFromText(std::string(rec.at("accuracy_estimate").as_string()));
241 r.accuracy_estimate_user_coords =
242 DoubleFromText(std::string(rec.at("accuracy_estimate_user_coords").as_string()));
243 r.cycle_num = static_cast<unsigned>(rec.at("cycle_num").as_int64());
244 r.precision_digits = static_cast<unsigned>(rec.at("precision_digits").as_int64());
245 r.accuracy_digits = static_cast<unsigned>(rec.at("accuracy_digits").as_int64());
246 r.precision_changed = rec.at("precision_changed").as_bool();
247 r.time_of_first_prec_increase =
248 DecodeComplexScalar<ComplexT>(rec.at("time_of_first_prec_increase").as_array());
249 r.max_precision_used = static_cast<unsigned>(rec.at("max_precision_used").as_int64());
250 r.path_time_seconds = DoubleFromText(std::string(rec.at("path_time_seconds").as_string()));
251 return r;
252}
