177 template <typename ComplexT>
178 SuccessCode Correct(Vec<ComplexT> & next_space,
179 StepMetadata & meta,
180 System const& S,
181 Vec<ComplexT> const& current_space,
182 ComplexT const& current_time,
183 NumErrorT const& tracking_tolerance,
184 unsigned min_num_newton_iterations,
185 unsigned max_num_newton_iterations,
186 AdaptiveMultiplePrecisionConfig const* AMP_config = nullptr)
187 {
188 #ifndef BERTINI_DISABLE_ASSERTS
189 assert(max_num_newton_iterations >= min_num_newton_iterations && "max number newton iterations must be at least the min.");
190 #endif
191
192 Vec<ComplexT>& step_ref = std::get< Vec<ComplexT> >(step_temp_);
193
194 next_space = current_space;
195 for (unsigned ii = 0; ii < max_num_newton_iterations; ++ii)
196 {
197 //Update the newton iterate by one iteration
198 auto success_code = EvalIterationStep(step_ref, S, next_space, current_time);
199 if(success_code != SuccessCode::Success)
200 return success_code;
201
202 next_space -= step_ref; // step_ref = +J^{-1}f = -(Newton step); see EvalIterationStep
203
204 // Fixed precision: cheap convergence-only loop, no norms / probe / criteria.
205 if (AMP_config == nullptr)
206 {
207 if ( (step_ref.template lpNorm<Eigen::Infinity>() < tracking_tolerance) && (ii >= (min_num_newton_iterations-1)) )
208 return SuccessCode::Success;
209 continue;
210 }
211
212 // Adaptive precision: fill metadata + enforce AMP criteria B and C.
213 Mat<ComplexT>& J_temp_ref = std::get< Mat<ComplexT> >(J_temp_);
214 linalg::PartialPivLU<ComplexT>& LU_ref = std::get< linalg::PartialPivLU<ComplexT> >(LU_);
215
216 meta.norm_delta_z = NumErrorT(step_ref.template lpNorm<Eigen::Infinity>());
217 meta.norm_J = NumErrorT(J_temp_ref.norm());
218 {
219 // Reuse the FIXED probe vector generated once at setup (do NOT regenerate
220 // per call): a fresh random probe direction every Newton step occasionally
221 // produced an inflated ||J^{-1}|| estimate -> spurious HigherPrecisionNecessary
222 // -> precision escalation/grind. A single fixed direction also makes the
223 // condition estimates comparable across steps and keeps tracking deterministic.
224 Vec<ComplexT>& rand_ref = std::get< Vec<ComplexT> >(rand_temp_);
225 Vec<ComplexT>& solve_ref = std::get< Vec<ComplexT> >(solve_temp_);
226 LU_ref.Solve(rand_ref, solve_ref); // reuse the factorization from EvalIterationStep
227 meta.norm_J_inverse = NumErrorT(solve_ref.norm());
228 }
229 meta.condition_number_estimate = NumErrorT(meta.norm_J*meta.norm_J_inverse);
230
231 if ( (meta.norm_delta_z < tracking_tolerance) && (ii >= (min_num_newton_iterations-1)) )
232 return SuccessCode::Success;
233
234 if (!amp::CriterionB<ComplexT>(meta.norm_J, meta.norm_J_inverse, max_num_newton_iterations - ii, tracking_tolerance, meta.norm_delta_z, *AMP_config))
235 return SuccessCode::HigherPrecisionNecessary;
236
237 if (!amp::CriterionC<ComplexT>(meta.norm_J_inverse, next_space, tracking_tolerance, *AMP_config))
238 return SuccessCode::HigherPrecisionNecessary;
239 }
240
241 return SuccessCode::FailedToConverge;
242 }

264 template<typename ComplexT, typename Derived>
265 SuccessCode EvalIterationStep(Vec<ComplexT> & newton_step,
266 const System& S,
267 const Eigen::MatrixBase<Derived>& current_space, const ComplexT& current_time)
268 {
269 Vec<ComplexT>& f_temp_ref = std::get< Vec<ComplexT> >(f_temp_);
270 Mat<ComplexT>& J_temp_ref = std::get< Mat<ComplexT> >(J_temp_);
271
272 linalg::PartialPivLU<ComplexT>& LU_ref = std::get< linalg::PartialPivLU<ComplexT> >(LU_);
273
274 S.SetAndReset<ComplexT>(current_space, current_time);
275 S.EvalInPlace(f_temp_ref);
276 S.JacobianInPlace(J_temp_ref);
277 // Factor a copy of J (J_temp_ref is read again afterward for meta.norm_J). The health
278 // check is folded into Factor().
279 if (LU_ref.Factor(J_temp_ref)!=MatrixSuccessCode::Success)
280 return SuccessCode::MatrixSolveFailure;
281
282 // Solve J*newton_step = f (NOT -f): this lets us skip materializing the negated RHS
283 // temporary. newton_step therefore holds +J^{-1}f = -(true Newton step), so callers
284 // SUBTRACT it (next_space -= newton_step). The convergence test uses lpNorm, which is
285 // sign-insensitive, so it is unaffected.
286 LU_ref.Solve(f_temp_ref, newton_step);
287
288 return SuccessCode::Success;
289
290 }
