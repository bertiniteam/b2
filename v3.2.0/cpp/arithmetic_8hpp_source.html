116 // Build a complete sum from a full (term, sign) list. The node is fully constructed
117 // before Make() interns it -- so callers never AddOperand AFTER Make (which, with interning,
118 // could mutate a shared interned node).

120 {
121 for (auto const& t : terms)
122 AddOperand(t.first, t.second);
123 CanonicalizeNaryOperands(operands_, signs_, false);
124 }
