141 template<class ObservedT, template<class> class... ObserverTypes>
142 class MultiObserver : public Observer<ObservedT>
143 { BOOST_TYPE_INDEX_REGISTER_CLASS
144 public:
145
155 ObserveResult Observe(AnyEvent const& e) override
156 {
157 using namespace boost::fusion;
158 bool keep = false;
159 // Route through the AnyObserver interface: a glued-in observer may be a
160 // TypedObserver, whose typed Observe(ConcreteEvent const&) overloads hide
161 // the untyped Observe(AnyEvent const&) for ordinary name lookup. The
162 // virtual call reaches the right dispatcher in every case.
163 auto f = [&e,&keep](auto &obs) {
164 if (static_cast<AnyObserver&>(obs).Observe(e) == ObserveResult::KeepObserving)
165 keep = true;
166 };
167 for_each(observers_, f);
168 return keep ? ObserveResult::KeepObserving : ObserveResult::Unsubscribe;
169 }
170
171 std::tuple<ObserverTypes<ObservedT>...> observers_;
172 virtual ~MultiObserver() = default;
173 };

203 template<class Derived, class ObservedT, class... ConcreteEvents>
204 class TypedObserver : public Observer<ObservedT>
205 { BOOST_TYPE_INDEX_REGISTER_CLASS
206 public:
207 std::vector<std::type_index> SubscribedEventTypes() const override
208 {
209 return { std::type_index(typeid(ConcreteEvents))... };
210 }
211
212 ObserveResult Observe(AnyEvent const& e) override
213 {
214 ObserveResult result = ObserveResult::KeepObserving;
215 bool matched = false;
216 // try each subscribed type; the first whose dynamic type matches wins.
217 (TryDispatch<ConcreteEvents>(e, result, matched), ...);
218 return result;
219 }
220
221 virtual ~TypedObserver() = default;
222
223 private:
224 template<class Ev>
225 void TryDispatch(AnyEvent const& e, ObserveResult& result, bool& matched)
226 {
227 if (matched)
228 return;
229 if (auto p = dynamic_cast<const Ev*>(&e))
230 {
231 matched = true;
232 result = static_cast<Derived*>(this)->OnEvent(*p);
233 }
234 }
235 };
