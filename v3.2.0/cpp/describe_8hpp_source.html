92inline void PrintLinearFormVerbose(std::ostream& out, Mat<complex_mp> const& M, Eigen::Index r,
93 VariableGroup const& vars, size_t num_vars, bool homogeneous)
94{
95 bool first = true;
96 const size_t ncol = homogeneous ? num_vars : num_vars + 1;
97 for (size_t c = 0; c < ncol; ++c)
98 {
99 complex_mp const& coeff = M(r, static_cast<Eigen::Index>(c));
100 if (coeff.real() == 0 && coeff.imag() == 0)
101 continue;
102 if (!first) out << " + ";
103 first = false;
104 out << "(";
105 PrintCoeff(out, coeff);
106 out << ")";
107 if (c < num_vars && c < vars.size()) // a variable column (the last affine column is the constant)
108 out << "*" << *vars[c];
109 }
110 if (first)
111 out << "0";
112}
