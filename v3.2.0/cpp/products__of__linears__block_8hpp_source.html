183 template <typename T>
184 void EvalInPlace(Eigen::Ref<Vec<T>> result, Vec<T> const& vars, T const& /*path_value*/) const
185 {
186 const auto& W = Working<T>();
187 // aug = [vars ; 1]. The trailing 1 lets each coefficient row carry its constant
188 // term in its last column, so a linear factor is just the dot product (row . aug).
189 const Vec<T> aug = Augment<T>(vars);
190
191 // Function i is a product of linear factors. W[i] is its k x (n+1) coefficient
192 // matrix, one row per factor, so W[i] * aug evaluates all k factors at once and the
193 // function value is their product.
194 for (size_t i = 0; i < W.size(); ++i)
195 {
196 const Vec<T> fvals = W[i] * aug; // the k factor values L_0 .. L_{k-1}
197 T val(1);
198 for (Eigen::Index r = 0; r < fvals.size(); ++r)
199 val *= fvals(r); // f_i = prod_r L_r
200 result(static_cast<Eigen::Index>(i)) = val;
201 }
202 }

212 template <typename T>
213 void JacobianInPlace(Eigen::Ref<Mat<T>> J, Vec<T> const& vars, T const& /*path_value*/) const
214 {
215 const auto& W = Working<T>();
216 const Vec<T> aug = Augment<T>(vars); // [vars ; 1]; see EvalInPlace
217
218 // Function i is f_i = prod_r L_r, where L_r = (row r of M) . aug is the r-th linear
219 // factor's value. By the product rule, the partial derivative w.r.t. variable c is
220 //
221 // d f_i / d x_c = sum_r (d L_r / d x_c) * prod_{s != r} L_s
222 // = sum_r M(r,c) * weight_r,
223 //
224 // since d L_r / d x_c is just the coefficient M(r,c) of x_c in factor r, and
225 // weight_r := prod_{s != r} L_s is the product of every factor value except r's.
226 for (size_t i = 0; i < W.size(); ++i)
227 {
228 const Mat<T>& M = W[i]; // k x (n+1): rows are factors, last col is constant
229 const Eigen::Index k = M.rows();
230
231 if (k == 0) // a 0-factor product is the constant 1; derivative 0
232 {
233 J.row(static_cast<Eigen::Index>(i)).setZero();
234 continue;
235 }
236
237 const Vec<T> fvals = M * aug; // the factor values L_0 .. L_{k-1}
238
239 // weight_r = prod_{s != r} L_s, computed in O(k) and WITHOUT division -- so it
240 // stays correct even when some factor value L_s is zero (dividing the total
241 // product by L_r would not). We split the "all but r" product into the factors
242 // before r and the factors after r, each built by a running accumulator:
243 // forward pass: weight_r <- prod_{s < r} L_s (the prefix product)
244 // backward pass: weight_r <- weight_r * prod_{s > r} L_s (times the suffix product)
245 Vec<T> weight(k);
246 {
247 T acc(1);
248 for (Eigen::Index r = 0; r < k; ++r) { weight(r) = acc; acc *= fvals(r); } // prefix
249 acc = T(1);
250 for (Eigen::Index r = k - 1; r >= 0; --r) { weight(r) *= acc; acc *= fvals(r); } // suffix
251 }
252
253 // Row i of the Jacobian is (sum_r M(r,c) * weight_r) over the variable columns
254 // c only; M.leftCols(num_vars_) drops the trailing constant column (not a variable).
255 J.row(static_cast<Eigen::Index>(i)) = weight.transpose() * M.leftCols(static_cast<Eigen::Index>(num_vars_));
256 }
257 }

276 template <typename T>
277 Vec<T> Augment(Vec<T> const& vars) const
278 {
279 Vec<T> aug(static_cast<Eigen::Index>(num_vars_ + 1));
280 aug.head(static_cast<Eigen::Index>(num_vars_)) = vars;
281 T one(1);
282 if constexpr (!std::is_same<T, complex_dbl>::value)
283 one.precision(precision_);
284 aug(static_cast<Eigen::Index>(num_vars_)) = one;
285 return aug;
286 }

288 void BuildWorking() const
289 {
290 auto& wd = std::get<std::vector<Mat<complex_dbl>>>(factors_working_);
291 auto& wm = std::get<std::vector<Mat<complex_mp>>>(factors_working_);
292 const size_t n = factors_highest_precision_.size();
293 wd.resize(n);
294 wm.resize(n);
295 for (size_t i = 0; i < n; ++i)
296 {
297 const auto& M = factors_highest_precision_[i];
298 wd[i].resize(M.rows(), M.cols());
299 wm[i].resize(M.rows(), M.cols());
300 for (Eigen::Index r = 0; r < M.rows(); ++r)
301 for (Eigen::Index c = 0; c < M.cols(); ++c)
302 {
303 wd[i](r, c) = complex_dbl(M(r, c));
304 wm[i](r, c) = M(r, c);
305 }
306 }
307 }
