100 SuccessCode RefineSampleImpl(Vec<complex_mp> & result, Vec<complex_mp> const& current_sample, complex_mp const& current_time, NumErrorT tol, unsigned max_iterations) const
101 {
102
103 using bertini::Precision;
104 assert(Precision(current_time)==Precision(current_sample) && "precision of sample and time to be refined in AMP endgame must match");
105
106 using std::max;
107 auto& TR = this->GetTracker();
108 TR.ChangePrecision(Precision(current_time));
109
110 auto refinement_success = this->GetTracker().Refine(result,current_sample,current_time,
111 tol,
112 max_iterations);
113
114
115 if (refinement_success==SuccessCode::HigherPrecisionNecessary ||
116 refinement_success==SuccessCode::FailedToConverge)
117 {
118
119 using bertini::Precision;
120
121 // thread-local precision: endgames may run on std::thread workers, so
122 // neither read nor write the global default precision.
123 auto prev_precision = ThreadPrecision();
124 auto higher_precision = max(prev_precision,LowestMultiplePrecision())+ PrecisionIncrement();
125 SetThreadPrecision(higher_precision);
126 this->GetTracker().ChangePrecision(higher_precision);
127
128 NotifyObservers(PrecisionChanged<EmitterType>(*this, prev_precision, higher_precision));
129
130 auto next_sample_higher_prec = current_sample;
131 Precision(next_sample_higher_prec, higher_precision);
132
133 auto result_higher_prec = Vec<complex_mp>(current_sample.size());
134
135 auto time_higher_precision = current_time;
136 Precision(time_higher_precision,higher_precision);
137
138 assert(time_higher_precision.precision()==ThreadPrecision());
139 refinement_success = this->GetTracker().Refine(result_higher_prec,
140 next_sample_higher_prec,
141 time_higher_precision,
142 tol,
143 max_iterations);
144
145 Precision(result, higher_precision);
146 result = result_higher_prec;
147
148 assert(Precision(result)==ThreadPrecision());
149 }
150 return refinement_success;
151 }

170 // ================================================================================================
171 // Adaptive-numeric-type endgame state + the precision/numeric-type helpers shared by both AMP
172 // endgame flavors (Cauchy and PowerSeries). These live here, in the AMP precision policy, because
173 // they are entirely about the complex_dbl <-> complex_mp crossing and the AMP tracker's authority --
174 // not generic endgame concerns. The flavors inherit them (AMPEndgame is the precision-policy base
175 // of EndgameBase) and call them through this->.
176 // ================================================================================================
177
178 // current_endgame_precision_ is the precision the endgame is *currently computing in*: DoublePrecision()
179 // is the hardware-complex_dbl fast lane, higher is the mpfr slot. adaptive_numeric_type_active_ gates
180 // the escalation hooks in the flavors' shared phase methods so they fire only while the double-first
181 // driver is orchestrating (never for the AMP-PowerSeries forwarding path, were one to exist).
182 mutable unsigned current_endgame_precision_ = DoublePrecision();
183 mutable bool adaptive_numeric_type_active_ = false;
184
185 // Choose the next (higher) working precision: at least LowestMultiplePrecision(), following the
186 // tracker's authority if it climbed higher, and strictly above the current precision.
188 unsigned NextEscalatedPrecision() const
189 {
190 using std::max;
191 unsigned from_tracker = this->GetTracker().GetCurrentPrecision();
192 unsigned newprec = max(static_cast<unsigned>(LowestMultiplePrecision()), from_tracker);
193 if (newprec <= current_endgame_precision_)
194 newprec = current_endgame_precision_ + PrecisionIncrement();
195 return newprec;
196 }
197
198 // Cross a deque of times / samples, or a single vector, from the complex_dbl slot to the complex_mp
199 // slot, then set the mpfr precision via the existing SetPrecision helper. Templated on the tuple type
200 // so this header need not name the endgames' TupleOfTimes / TupleOfSamps / TupOfVec aliases. The
201 // element crossing is the same one the AMP tracker performs; no library Vec-cast exists for these types.
203 template<typename TimesTuple>
204 void CrossTimesUp(TimesTuple& times, unsigned newprec) const
205 {
206 auto& from = std::get<TimeCont<complex_dbl> >(times);
207 auto& to = std::get<TimeCont<complex_mp> >(times);
208 to.clear();
209 for (auto const& t : from) to.push_back(complex_mp(t));
210 from.clear();
211 tracking::adaptive::SetPrecision(to, newprec);
212 }
213
215 template<typename SampsTuple>
216 void CrossSampsUp(SampsTuple& samps, unsigned newprec) const
217 {
218 auto& from = std::get<SampCont<complex_dbl> >(samps);
219 auto& to = std::get<SampCont<complex_mp> >(samps);
220 to.clear();
221 for (auto const& v : from)
222 {
223 Vec<complex_mp> w(v.size());
224 for (Eigen::Index i = 0; i < v.size(); ++i) w(i) = complex_mp(v(i));
225 to.push_back(std::move(w));
226 }
227 from.clear();
228 tracking::adaptive::SetPrecision(to, newprec);
229 }
230
232 template<typename VecTuple>
233 void CrossVecUp(VecTuple& vec, unsigned newprec) const
234 {
235 using bertini::Precision;
236 auto& from = std::get<Vec<complex_dbl> >(vec);
237 auto& to = std::get<Vec<complex_mp> >(vec);
238 to.resize(from.size());
239 for (Eigen::Index i = 0; i < from.size(); ++i) to(i) = complex_mp(from(i));
240 if (to.size() > 0) Precision(to, newprec);
241 from.resize(0);
242 }

248 Vec<complex_dbl> out(v.size());

262 // Widen an active-type vector (a fast-lane extrapolation result) to complex_mp at the given precision:
263 // the single boundary conversion of an approximation back to the ambient type.
265 template<typename ComplexT>
266 Vec<complex_mp> ToBCT(Vec<ComplexT> const& v, unsigned prec) const
267 {
268 using bertini::Precision;
269 Vec<complex_mp> out(v.size());
270 for (Eigen::Index i = 0; i < v.size(); ++i) out(i) = complex_mp(v(i));
271 if (out.size() > 0) Precision(out, prec);
272 return out;
273 }
