221 // Compile-time bank selectors (ADR-0034). After NumType inference, each instruction's opcode word
222 // gets these high bits set to record which bank (real or complex) each operand and the result live
223 // in, so the hot eval loop reads the banks inline from the instruction it has already loaded instead
224 // of looking up slot_numtype_[slot] per operand. The base Operation occupies bits 0..16, so these
225 // sit well clear of it; masking with kOpcodeMask recovers the base op for the switch and for IsUnary.
226 constexpr size_t kOpcodeMask = (static_cast<size_t>(1) << 17) - 1;
227 constexpr size_t kArg0Real = static_cast<size_t>(1) << 20;
228 constexpr size_t kArg1Real = static_cast<size_t>(1) << 21;
229 constexpr size_t kOutReal = static_cast<size_t>(1) << 22;

683 // Read a slot's value as NumT (complex), pulling from the real or complex bank per its NumType
684 // (ADR-0034). Used to copy outputs out, since a function/derivative slot could be Real-typed.

697 template<typename NumT>
698 void GetFuncValsInPlace(Eigen::Ref<Vec<NumT>> result) const{
699 if (!memory_.is_evaluated_)
700 program_->Eval<NumT>(memory_);
701
702 // copy content (an output slot may be Real-typed; read from the bank its NumType selects)
703 for (size_t ii = 0; ii < program_->number_of_.Functions; ++ii) {
704 result(static_cast<Eigen::Index>(ii)) = ReadSlotAsComplex<NumT>(ii + program_->output_locations_.Functions);
705 }

719 template<typename NumT>
720 void GetJacobianInPlace(Eigen::Ref<Mat<NumT>> result) const{
721 if (!memory_.is_evaluated_)
722 program_->Eval<NumT>(memory_);
723
724 // copy content (a derivative slot may be Real-typed; read from the bank its NumType selects)
725 for (size_t jj =0; jj < program_->number_of_.Variables; ++jj) {
726 for (size_t ii = 0; ii < program_->number_of_.Functions; ++ii) {
727 result(static_cast<Eigen::Index>(ii), static_cast<Eigen::Index>(jj)) = ReadSlotAsComplex<NumT>(ii+jj*program_->number_of_.Functions + program_->output_locations_.Jacobian);
728 }
729 }

851 template<typename Derived>
852 void SetVariableValues(Eigen::MatrixBase<Derived> const& variable_values) const{
853 using NumT = typename Derived::Scalar;
854
855#if !defined(BERTINI_DISABLE_PRECISION_CHECKS)
856// && _WIN32
857 // An empty variable vector (a constant program with no variables) has no
858 // precision to read or check.
859 if (!std::is_same<NumT,complex_dbl>::value && variable_values.size() > 0 && Precision(variable_values)!=memory_.precision_){
860 std::stringstream err_msg;
861 err_msg << "variable_values and SLP must be of same precision. respective precisions: " << Precision(variable_values) << " " << memory_.precision_ << std::endl;
862 throw std::runtime_error(err_msg.str());
863 }
864#endif
865
866 auto& memory = memory_.Get<NumT>(); // unpack for local reference
867
868 for (size_t ii = 0; ii < program_->number_of_.Variables; ++ii) {
869 //assign to memory
870 memory[ii + program_->input_locations_.Variables] = variable_values(static_cast<Eigen::Index>(ii));
871 }
872 memory_.is_evaluated_ = false;
873 }

883 template<typename ComplexT>
884 void SetPathVariable(ComplexT const& time) const{
885
886#if !defined(BERTINI_DISABLE_PRECISION_CHECKS)
887// && _WIN32
888 if (Precision(time)!= DoublePrecision() && Precision(time)!=memory_.precision_){
889 std::stringstream err_msg;
890 err_msg << "time value and SLP must be of same precision. respective precisions: " << Precision(time) << " " << memory_.precision_ << std::endl;
891 throw std::runtime_error(err_msg.str());
892 }
893#endif
894
895 if (!this->HavePathVariable())
896 throw std::runtime_error("calling Eval with path variable, but this StraightLineProgram doesn't have one.");
897 // then actually copy the path variable into where it goes in memory
898
899 auto& memory = memory_.Get<ComplexT>(); // unpack for local reference
900
901 memory[program_->input_locations_.Time] = time;
902 memory_.is_evaluated_ = false;
903 }
