410 template<typename ComplexT>
411 SuccessCode CircleTrack(ComplexT const& target_time)
412 {
413 using bertini::Precision;
414 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
415 using std::acos;
416
417 ValidateConfigs();
418
419 auto& circle_times = std::get<TimeCont<ComplexT> >(cauchy_times_);
420 auto& circle_samples = std::get<SampCont<ComplexT> >(cauchy_samples_);
421
422 ComplexT starting_time = circle_times.back(); // take a COPY here, so won't invalidate it later
423
424 // the initial sample has already been added to the sample repo... so don't do that here, please
425
426 const auto num_vars = this->GetSystem().NumVariables();
427
428 for (unsigned ii = 0; ii < this->EndgameSettings().num_sample_points; ++ii)
429 {
430 const Vec<ComplexT>& current_sample = circle_samples.back();
431 const ComplexT& current_time = circle_times.back();
432
433#ifndef BERTINI_DISABLE_PRECISION_CHECKS
434 if (Precision(current_time)!=Precision(current_sample)){
435 std::stringstream err_msg;
436 err_msg << "current time and sample for circle track must be of same precision. respective precisions: " << Precision(current_time) << " " << Precision(current_sample) << std::endl;
437 throw std::runtime_error(err_msg.str());
438 }
439#endif
440
441 //set up the time value for the next sample.
442 using std::polar;
443
444#ifndef USE_BMP_COMPLEX
445 using bertini::polar;
446#endif
447
448 //Generalized since we could have a nonzero target time.
449 using std::arg;
450 RealT radius = abs(starting_time - target_time), angle = arg(starting_time - target_time); // generalized for nonzero target_time.
451
452 auto next_sample = Vec<ComplexT>(num_vars);
453 ComplexT next_time = (ii==this->EndgameSettings().num_sample_points-1)
454 ?
455 starting_time
456 :
457 polar(radius, (ii+1)*2*acos(static_cast<RealT>(-1)) / (this->EndgameSettings().num_sample_points) + angle) + target_time;
458 // If we are tracking to a nonzero target time we need to shift our values to track to. This is a step that may not be needed if target_time = 0
459 ;
460
461
462 auto tracking_success = this->EndgameTrackPath(next_sample, current_time, next_time, current_sample);
463 if (tracking_success != SuccessCode::Success)
464 {
465 return tracking_success;
466 }
467
468 // Pure-(i) numeric-type escalation: if the tracker's authoritative precision climbed above
469 // the precision this endgame is computing in, double no longer suffices for this circle.
470 // Bail to the migrate-and-retry driver (RunImplAMP), which discards this partial circle,
471 // migrates the durable state to mpfr, and re-tracks the circle in mpfr. Compile-time elided
472 // for fixed precision (and shields GetCurrentPrecision(), which fixed trackers lack).
473 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
474 if (this->adaptive_numeric_type_active_ &&
475 this->GetTracker().GetCurrentPrecision() > this->current_endgame_precision_)
476 return SuccessCode::HigherPrecisionNecessary;
477
478 // CircleAdvanced carries the new point/time at BaseComplexT. The fixed/mpfr lane emits exactly
479 // as before. The complex_dbl fast lane would have to convert the point to mpfr for the event,
480 // so we only pay that when something is actually observing (temporaries live through the
481 // synchronous NotifyObservers).
482 if constexpr (std::is_same<ComplexT, BCT>::value)
483 NotifyObservers(CircleAdvanced<EmitterType>(*this, next_sample, next_time));
484 else if (this->HasObservers())
485 {
486 Vec<BCT> ev_pt(next_sample.size());
487 for (Eigen::Index i = 0; i < next_sample.size(); ++i) ev_pt(i) = BCT(next_sample(i));
488 BCT ev_t(next_time);
489 NotifyObservers(CircleAdvanced<EmitterType>(*this, ev_pt, ev_t));
490 }
491
492
493 this->EnsureAtPrecision(next_time,Precision(next_sample)); assert(Precision(next_time)==Precision(next_sample));
494
495 // auto refinement_success = this->RefineSample(next_sample, next_sample, next_time,
496 // this->FinalTolerance() * this->EndgameSettings().sample_point_refinement_factor,
497 // this->EndgameSettings().max_num_refinements);
498 // if (refinement_success != SuccessCode::Success)
499 // {
500 // return refinement_success;
501 // }
502
503 // this->EnsureAtPrecision(next_time,Precision(next_sample));
504 // assert(Precision(next_time)==Precision(next_sample));
505
506 AddToCauchyData(next_time, next_sample);
507
508 // NotifyObservers(SampleRefined<EmitterType>(*this));
509 }
510
511 return SuccessCode::Success;
512
513 }//end CircleTrack

570 template<typename ComplexT>
571 auto ComputeCOverK() const -> typename Eigen::NumTraits<ComplexT>::Real
572 {//Obtain samples for computing C over K.
573 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
574 using std::abs;
575 using std::log;
576
577 const auto& pseg_samples = std::get<SampCont<ComplexT> >(pseg_samples_);
578
579 assert(pseg_samples.size()>=3);
580 const Vec<ComplexT> & sample0 = pseg_samples[0];
581 const Vec<ComplexT> & sample1 = pseg_samples[1];
582 const Vec<ComplexT> & sample2 = pseg_samples[2];
583
584 // Use a fixed random probe vector, generated once and reused across the whole endgame, so this
585 // estimate is deterministic. A fresh random vector per call made consecutive c/k estimates
586 // disagree by probe noise alone, which could trip (or stall) CheckForCOverKStabilization.
587 const Vec<ComplexT> & rand_vector = GetCOverKProbe<ComplexT>(static_cast<unsigned>(sample0.size()), Precision(sample0));
588
589 // //DO NOT USE Eigen .dot() it will do conjugate transpose which is not what we want.
590 // //Also, the .transpose*rand_vector returns an expression template that we do .norm of since abs is not available for that expression type.
591 RealT estimate = abs(log(abs((((sample2 - sample1).transpose()*rand_vector).template lpNorm<Eigen::Infinity>())/(((sample1 - sample0).transpose()*rand_vector).template lpNorm<Eigen::Infinity>()))));
592 estimate = abs(log(RealT(this->EndgameSettings().sample_factor)))/estimate;
593 if (estimate < 1)
594 return RealT(1);
595 else
596 return estimate;
597
598 }//end ComputeCOverK

617 {

622 for(unsigned ii = 1; ii < GetCauchySettings().num_needed_for_stabilization ; ++ii)
623 {
624 RealT a = abs(c_over_k_array[ii-1]);
625 RealT b = abs(c_over_k_array[ii]);

628

642 /*
643 Input: A time value and the space value above that time.
644
645 Output: An real_mp representing a tolerance threshold for declaring a loop to be closed.
646 Details: Used in Bertini 1 as a heuristic for computing separatedness of roots. Decided to not be used since assumptions for this tolerance are not usually met.
647 template<typename ComplexT>
648 real_mp FindToleranceForClosedLoop(ComplexT x_time, Vec<ComplexT> x_sample)
649 {
650 auto degree_max = std::max(this->GetTracker().AMP_config_.degree_bound,real_mp("2.0"));
651 auto K = this->GetTracker().AMP_config_.coefficient_bound;
652 real_mp N;
653 real_mp M;
654 real_mp L;
655 if(max_closed_loop_tolerance_ < min_closed_loop_tolerance_)
656 {
657 max_closed_loop_tolerance_ = min_closed_loop_tolerance_;
658 }
659 auto error_tolerance = real_mp("1e-13");
660 if(x_sample.size() <= 1)
661 {
662 N = degree_max;
663 }
664 else
665 {
666 N = ComputeCombination(degree_max + x_sample[0].precision() - 1, x_sample[0].precision() - 1);
667 }
668 M = degree_max * (degree_max - 1) * N;
669 auto jacobian_at_current_time = this->GetSystem().Jacobian(x_sample,x_time);
670 auto minimum_singular_value = Eigen::JacobiSVD< Mat<ComplexT> >(jacobian_at_current_time).singularValues()(this->GetSystem().NumVariables() - 1 );
671 auto norm_of_sample = x_sample.norm();
672 L = pow(norm_of_sample,degree_max - 2);
673 auto tol = K * L * M;
674 if (tol == 0) // fail-safe
675 tol = minimum_singular_value;
676 else
677 {
678 tol = real_mp("2.0") / tol;
679 tol = tol * minimum_singular_value;
680 }
681 // make sure that tol is between min_tol & max_tol
682 if (tol > max_closed_loop_tolerance_) // tol needs to be <= max_tol
683 tol = max_closed_loop_tolerance_;
684 if (tol < min_closed_loop_tolerance_) // tol needs to be >= min_tol
685 tol = min_closed_loop_tolerance_;
686 return tol;
687 }// end FindToleranceForClosedLoop
688 */

756 template<typename ComplexT>
757 bool RatioEGOperatingZoneTest(ComplexT const& target_time) const
758 {
759 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
760 RealT min(1e300);
761 RealT max(0);
762 auto& times = std::get<TimeCont<ComplexT> >(cauchy_times_);
763 auto& samples = std::get<SampCont<ComplexT> >(cauchy_samples_);
764 if(norm(times.front() - target_time) < GetCauchySettings().ratio_cutoff_time)
765 {
766 return true;
767 }
768 else
769 {
770 RealT norm;
771 for(unsigned int ii=0; ii < this->EndgameSettings().num_sample_points; ++ii)
772 {
773 norm = samples[ii].template lpNorm<Eigen::Infinity>();
774 if(norm > max)
775 {
776 max = norm;
777 }
778 if(norm < min)
779 {
780 min = norm;
781 }
782 }
783
784 if(min > this->FinalTolerance() && max > this->FinalTolerance())
785 {
786 norm = min / max;
787 if(norm < GetCauchySettings().maximum_cauchy_ratio && (max - min) > this->FinalTolerance())
788 {
789 return false; // bad ratio and too far apart.
790 }
791 }
792 }
793 return true;
794
795 }//end RatioEGOperatingZoneTest

822 template<typename ComplexT>
823 SuccessCode InitialCauchyLoops(ComplexT const& target_time)
824 {
825 using std::max;
826 // auto fail_safe_max_cycle_number = max(GetCauchySettings().fail_safe_maximum_cycle_number,this->CycleNumber());
827 auto fail_safe_max_cycle_number = GetCauchySettings().fail_safe_maximum_cycle_number;
828
829 auto initial_cauchy_loop_success = SuccessCode::Success;
830
831 bool loop_hasnt_closed = true;
832 while (loop_hasnt_closed)
833 {
834 this->CycleNumber(0);
835 ClearAndSeedCauchyData<ComplexT>();
836
837 // track around a circle once. we'll use it to measure whether we believe we are in the eg operating zone, based on the ratio of norms of sample points around the circle
838 auto tracking_success = CircleTrack(target_time);
839 this->IncrementCycleNumber(1);
840 if (tracking_success != SuccessCode::Success)
841 return tracking_success;
842
843 // find the ratio of the maximum and minimum coordinate wise for the loop.
844 if (RatioEGOperatingZoneTest<ComplexT>(target_time))
845 { // then we believe we are in the EG operating zone, since the path is relatively flat. i still disbelieve this is a good test (dab 20160310)
846 while (true)
847 {
848 if (CheckClosedLoop<ComplexT>())
849 {//error is small enough, exit the loop with success.
850 NotifyObservers(ClosedLoop<EmitterType>(*this));
851 initial_cauchy_loop_success = SuccessCode::Success;
852 loop_hasnt_closed = false;
853 break;
854 }
855 else if(this->CycleNumber() > fail_safe_max_cycle_number)
856 {// too many iterations
857 initial_cauchy_loop_success = SuccessCode::CycleNumTooHigh;
858 loop_hasnt_closed = false;
859 break;
860 }
861
862 //compute next loop, the last sample in times and samples is the sample our loop ended on. Either where we started or on another sheet at the same time value.
863 tracking_success = CircleTrack(target_time);
864 this->IncrementCycleNumber(1);
865 if(tracking_success != SuccessCode::Success)
866 return tracking_success;
867 }
868 }//end if (RatioEGOperatingZoneTest())
869 else
870 {
871 auto advance_success = AdvanceTime<ComplexT>(target_time);
872 if (advance_success!=SuccessCode::Success)
873 return advance_success;
874 }
875 } //end while(loop_hasnt_closed)
876
877 return initial_cauchy_loop_success;
878 }//end InitialCauchyLoops

1182 template<typename ComplexT>
1183 NumErrorT PoleComponentMass() const
1184 {
1185 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
1186 auto const& samples = std::get<SampCont<ComplexT>>(cauchy_samples_);
1187 auto const& times = std::get<TimeCont<ComplexT>>(cauchy_times_);
1188 auto const c = this->CycleNumber();
1189 auto const N = this->EndgameSettings().num_sample_points;
1190 auto const M = c * N;
1191 if (c == 0 || samples.size() < M + 1 || times.size() < 2)
1192 return NumErrorT(0);
1193
1194 // orientation of the loop (which way theta advances); the frequency sign follows it
1195 RealT const orientation_test = imag(times[1] * conj(times[0]));
1196 RealT const orientation = (orientation_test > RealT(0)) ? RealT(1) : RealT(-1);
1197
1198 RealT const two_pi = RealT(2) * acos(RealT(-1));
1199 NumErrorT mass(0);
1200 for (unsigned m = 1; m <= c; ++m)
1201 {
1202 Vec<ComplexT> twisted = Vec<ComplexT>::Zero(samples[0].size());
1203 for (unsigned j = 0; j < M; ++j)
1204 {
1205 // weight e^{+i m theta_j / c} with theta_j = orientation * 2 pi j / N:
1206 // picks out exactly the k = -m Puiseux mode
1207 RealT const angle = orientation * two_pi * RealT(j * m) / RealT(N * c);
1208 twisted += samples[j] * ComplexT(cos(angle), sin(angle));
1209 }
1210 twisted /= RealT(M);
1211 auto const twisted_norm = static_cast<NumErrorT>(twisted.template lpNorm<Eigen::Infinity>());
1212 if (twisted_norm > mass)
1213 mass = twisted_norm;
1214 }
1215 return mass;
1216 }

1262 template<typename ComplexT>
1263 SuccessCode RunImpl(ComplexT const& start_time, Vec<ComplexT> const& start_point, ComplexT const& target_time)
1264 {
1265 if (start_point.size()!=static_cast<Eigen::Index>(this->GetSystem().NumVariables()))
1266 {
1267 std::stringstream err_msg;
1268 err_msg << "number of variables in start point for CauchyEG, " << start_point.size() << ", must match the number of variables in the system, " << this->GetSystem().NumVariables();
1269 throw std::runtime_error(err_msg.str());
1270 }
1271
1272 if (Precision(start_time)!=Precision(start_point))
1273 {
1274 std::stringstream ss;
1275 ss << "CauchyEG Run time and point must be of matching precision. (" << Precision(start_time) << "!=" << Precision(start_point) << ")";
1276 throw std::runtime_error(ss.str());
1277 }
1278
1279 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
1280
1281 Vec<ComplexT>& latest_approx = this->final_approximation_;
1282 Vec<ComplexT>& prev_approx = this->previous_approximation_;
1283 NumErrorT& approx_error = this->approximate_error_;
1284
1285
1286 ClearTimesAndSamples<ComplexT>(); //clear times and samples before we begin.
1287 this->CycleNumber(0);
1288 prev_approx = start_point;
1289
1290 auto init_success = GetIntoEGZone(start_time, start_point, target_time);
1291 if (init_success!= SuccessCode::Success)
1292 return init_success;
1293
1294 auto cauchy_loop_success = InitialCauchyLoops<ComplexT>(target_time);
1295 if (cauchy_loop_success != SuccessCode::Success)
1296 return cauchy_loop_success;
1297
1298
1299 // The security check watches the extrapolated ENDPOINT for divergence to
1300 // infinity. A genuinely-infinite endpoint -- a patched projective path leaving
1301 // the affine chart -- has an approximation whose dehomogenized norm grows to
1302 // infinity, and it grows FASTER than any finite loop sample along the way (the
1303 // endpoint IS the infinity; the samples are all finite), so the endpoint is the
1304 // earliest, strongest divergence signal. Watching the loop samples instead
1305 // over-truncates finite paths whose samples transiently spike above max_norm
1306 // before settling (found via cyclic-6: 156 -> 153 lost, all nonsingular). The
1307 // pole-annihilation case -- a Laurent pole whose Cauchy mean is a STATIONARY
1308 // FINITE number, which no norm check can catch -- is handled separately by the
1309 // pole-component operating-zone check below, so this check need only see honest
1310 // infinity. Tracked across consecutive IN-ZONE rounds only (see below);
1311 // initialized to 0 so the check never reads indeterminate values.
1312 RealT norm_of_dehom_prev(0), norm_of_dehom_latest(0);
1313
1314 // Cycle-number consistency: refuse to accept a converged approximation until the cycle number
1315 // has reported the SAME value for num_consecutive_same_cycle_number consecutive approximations.
1316 // When the working precision is too low to close the loop accurately (NOT monodromy -- a
1317 // nonsingular endpoint is cycle 1 at every radius in high precision), the cycle number thrashes
1318 // (41, 14, 36, ...) and a coincidental approx_error dip must not be mistaken for convergence.
1319 // prev_cycle == 0 means "no prior measurement".
1320 unsigned prev_cycle = 0, same_cycle_count = 0;
1321
1322 do
1323 {
1324 //Compute a cauchy approximation. Uses the previously computed samples,
1325 //either from InitialCauchyLoops, or ComputeCauchySamples
1326 auto extrapolation_success = ComputeCauchyApproximationOfXAtT0<ComplexT>(latest_approx);
1327 if (extrapolation_success!=SuccessCode::Success)
1328 return extrapolation_success;
1329
1330 unsigned cur_cycle = this->CycleNumber();
1331 if (cur_cycle == prev_cycle)
1332 ++same_cycle_count;
1333 else
1334 same_cycle_count = 1;
1335 prev_cycle = cur_cycle;
1336
1337 approx_error = static_cast<NumErrorT>((latest_approx - prev_approx).template lpNorm<Eigen::Infinity>());
1338 NotifyObservers(ApproximatedRoot<EmitterType>(*this));
1339
1340 // the operating-zone ACCEPTANCE GATE: negative-mode (pole) mass in the loop.
1341 // Nonzero mass means the disk between here and the target time is not clean
1342 // -- a pole at the target, or another branch point inside the loop -- and
1343 // the mean is not to be trusted, however stationary it looks. So convergence
1344 // is refused until the zone is reached (pole mass insignificant). This alone
1345 // cures the junk-success bug: an affine Laurent pole has a stationary FINITE
1346 // Cauchy mean (roots-of-unity annihilation), so no norm check can catch it,
1347 // but its pole mass never vanishes, so this gate never accepts it -- the path
1348 // instead runs to the endgame's natural terminal condition (min track time /
1349 // fail-safe max cycle) and returns non-Success. There is deliberately NO
1350 // active pole-growth truncation: distinguishing a genuine pole (mass grows
1351 // unboundedly) from a branch point merely inside the loop (mass grows a round
1352 // or two, then dies as the radius shrinks past it) by counting growing rounds
1353 // produced false positives that truncated CLEAN convergent paths -- cyclic-6
1354 // lost genuine nonsingular solutions (156 -> 153). The gate suffices.
1355 auto const pole_mass = PoleComponentMass<ComplexT>();
1356 bool const in_operating_zone = !PoleMassSignificant(pole_mass,
1357 static_cast<NumErrorT>(latest_approx.template lpNorm<Eigen::Infinity>()));
1358
1359 if (in_operating_zone
1360 && approx_error < this->FinalTolerance()
1361 && same_cycle_count >= GetCauchySettings().num_consecutive_same_cycle_number)
1362 {
1363 NotifyObservers(Converged<EmitterType>(*this));
1364 return SuccessCode::Success;
1365 }
1366
1367 // Security truncation happens ONLY in the operating zone. Outside it the
1368 // loop may encircle a pole or branch point and the Cauchy mean is garbage --
1369 // its norm means nothing, and truncating on it killed clean convergent paths
1370 // (the cyclic-6 residual, 156 -> 155). Inside the zone the mean is
1371 // trustworthy: a genuinely-infinite endpoint is in-zone AND large (clean
1372 // homogeneous convergence, the endpoint growing to infinity faster than any
1373 // sample), so two consecutive IN-ZONE rounds above max_norm truncate honestly.
1374 // An out-of-zone round restarts the two-consecutive count.
1375 if (this->SecuritySettings().level <= 0)
1376 {
1377 if (in_operating_zone)
1378 {
1379 norm_of_dehom_latest = this->GetSystem().InfinityNormOfDehomogenized(latest_approx);
1380 if (norm_of_dehom_prev > this->SecuritySettings().max_norm &&
1381 norm_of_dehom_latest > this->SecuritySettings().max_norm )
1382 {
1383 NotifyObservers(SecurityMaxNormReached<EmitterType>(*this));
1384 return SuccessCode::SecurityMaxNormReached;
1385 }
1386 norm_of_dehom_prev = norm_of_dehom_latest;
1387 }
1388 else
1389 norm_of_dehom_prev = RealT(0); // out of zone: restart the count
1390 }
1391
1392 prev_approx = latest_approx;
1393
1394 auto advance_success = AdvanceTime<ComplexT>(target_time);
1395 if (advance_success != SuccessCode::Success)
1396 return advance_success;
1397
1398 // then compute the next set of cauchy samples used for extrapolating the point at target time
1399 auto cauchy_samples_success = ComputeCauchySamples(target_time);
1400 if (cauchy_samples_success != SuccessCode::Success)
1401 return cauchy_samples_success;
1402
1403 } while (true);
1404
1405 return SuccessCode::Success;
1406 } //end main CauchyEG function

1422 template<typename Dummy = void>
1423 SuccessCode RunImplAMP(BCT const& start_time, Vec<BCT> const& start_point, BCT const& target_time)
1424 {
1425 using bertini::Precision;
1426 using RealT = typename Eigen::NumTraits<BCT>::Real;
1427
1428 if (start_point.size()!=static_cast<Eigen::Index>(this->GetSystem().NumVariables()))
1429 {
1430 std::stringstream err_msg;
1431 err_msg << "number of variables in start point for CauchyEG, " << start_point.size() << ", must match the number of variables in the system, " << this->GetSystem().NumVariables();
1432 throw std::runtime_error(err_msg.str());
1433 }
1434
1435 // Arm the escalation hooks in the shared phase methods (CircleTrack / AdvanceTime / refine);
1436 // disarm on every exit path.
1437 this->adaptive_numeric_type_active_ = true;
1438 struct Disarmer { bool& flag; ~Disarmer(){ flag = false; } } disarm{this->adaptive_numeric_type_active_};
1439
1440 // Start in the precision the tracker handed us at the endgame boundary: double for the easy
1441 // majority, already-mpfr for the few paths that escalated before the endgame. Migrate UP only.
1442 this->current_endgame_precision_ = std::max(DoublePrecision(), Precision(start_point));
1443 this->CycleNumber(0);
1444
1445 // ---- INITIALIZATION (GetIntoEGZone + InitialCauchyLoops). On escalation, restart init from the
1446 // boundary at the higher precision: init lives at the well-conditioned large-|t| end where
1447 // escalation is rare, and a fresh init clears its own containers so it is self-consistent. ----
1448 while (true)
1449 {
1450 SuccessCode init_code;
1451 if (this->current_endgame_precision_ == DoublePrecision())
1452 init_code = RunInitSegmentT<complex_dbl>(complex_dbl(start_time), this->DowncastToDouble(start_point), complex_dbl(target_time));
1453 else

1455
1456 if (init_code == SuccessCode::HigherPrecisionNecessary)

1458 this->current_endgame_precision_ = this->NextEscalatedPrecision();
1459 SetThreadPrecision(this->current_endgame_precision_);

1464 return init_code;
1465 break;

1469 // up to mpfr (widening the retained PSEG window, never re-tracking it) and retries in mpfr.
1470 // The approximations live in BCT, so the per-iteration bookkeeping arithmetic is mpfr -- one
1471 // small vector op next to the tracking, which itself stays in the fast lane. ----
1472 // the security check watches the extrapolated ENDPOINT for honest divergence to
1473 // infinity (which grows faster than any finite loop sample), but ONLY in the
1474 // operating zone (a trustworthy mean) -- see RunImpl for the full reasoning.
1475 // Acceptance is gated on the same operating-zone measurement (the finite-mean
1476 // pole case). Tracked across consecutive in-zone rounds; starts at 0.
1477 RealT norm_of_dehom_prev(0), norm_of_dehom_latest(0);
1478
1479 unsigned prev_cycle = 0, same_cycle_count = 0;

1493 NotifyObservers(ApproximatedRoot<EmitterType>(*this));

1496 // convergence is refused while pole mass is significant, which alone cures

1498 // false-positived on clean convergent paths -- cyclic-6, 156 -> 153).
1499 auto const pole_mass = PoleComponentMassAMP();
1500 bool const in_operating_zone = !PoleMassSignificant(pole_mass,
1501 static_cast<NumErrorT>(this->final_approximation_.template lpNorm<Eigen::Infinity>()));
1502
1503 if (in_operating_zone
1504 && this->approximate_error_ < this->FinalTolerance()
1505 && same_cycle_count >= GetCauchySettings().num_consecutive_same_cycle_number)
1506 {
1507 NotifyObservers(Converged<EmitterType>(*this));
1508 return SuccessCode::Success;
1509 }

1512 // out-of-zone round restarts the two-consecutive count. See RunImpl.
1513 if (this->SecuritySettings().level <= 0)
1514 {
1515 if (in_operating_zone)
1516 {
1517 norm_of_dehom_latest = this->GetSystem().InfinityNormOfDehomogenized(this->final_approximation_);
1518 if (norm_of_dehom_prev > this->SecuritySettings().max_norm &&
1519 norm_of_dehom_latest > this->SecuritySettings().max_norm)
1520 {
1521 NotifyObservers(SecurityMaxNormReached<EmitterType>(*this));
1522 return SuccessCode::SecurityMaxNormReached;
1523 }
1524 norm_of_dehom_prev = norm_of_dehom_latest;
1525 }
1526 else
1527 norm_of_dehom_prev = RealT(0); // out of zone: restart the count
1528 }
1529
1530 this->previous_approximation_ = this->final_approximation_;

1535

1537 if (samples_code != SuccessCode::Success)

1539 }
