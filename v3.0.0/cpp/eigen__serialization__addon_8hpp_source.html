30friend class boost::serialization::access;
31template<class Archive>
32void save(Archive & ar, const unsigned int /*version*/) const {
33 derived().eval();
34 const Eigen::Index rows = derived().rows(), cols = derived().cols();
35 ar & rows;
36 ar & cols;
37 for (Index j = 0; j < cols; ++j )
38 for (Index i = 0; i < rows; ++i )
39 ar & derived().coeff(i, j);
40}

42template<class Archive>
43void load(Archive & ar, const unsigned int /*version*/) {
44 Eigen::Index rows, cols;
45 ar & rows;
46 ar & cols;
47 if (rows != derived().rows() || cols != derived().cols() )
48 derived().resize(rows, cols);
49 ar & boost::serialization::make_array(derived().data(), derived().size());
50}

52template<class Archive>
53void serialize(Archive & ar, const unsigned int file_version) {
54 boost::serialization::split_member(ar, *this, file_version);
55}
