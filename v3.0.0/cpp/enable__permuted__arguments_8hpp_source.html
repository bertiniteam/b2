48 // find the index of a type in a list of types,
49 // return sizeof...(Remaining) if T is not found
50 template< typename T, typename... Remaining >
51 struct IndexByType : std::integral_constant< std::size_t, 0 >
52 {};

54 // template specialization, enforcing the equality of the number of variadic arguments, and the corresponding index.
55 // if two types are the same, this template will be preferred since it's a closer match, and the static_assert will fail, preventing compilation.
56 template< typename T, typename... RemainingT >
57 struct IndexByType< T, T, RemainingT... > : std::integral_constant< std::size_t, 0 >
58 {
59 static_assert( IndexByType< T, RemainingT... >::value == sizeof...( RemainingT ), "No duplicates are allowed when enabling permuted arguments" );
60 };
61
62 // template specialization on two types. looks for the index of T in the latter arguments. If T1==T2, this template will not be used, but instead the above will.
63 template< typename T1, typename T2, typename... RemainingT >
64 struct IndexByType< T1, T2, RemainingT... > : std::integral_constant< std::size_t, IndexByType< T1, RemainingT... >::value + 1 >
65 {};

122 template< typename... UnpermutedT, // these must be declared when using the function
123 typename... PermutedPresentT > // these are inferred, cannot be declared
124 std::tuple< const UnpermutedT&... > Unpermute( const PermutedPresentT&... present_permuted_args )
125 {
126 using namespace detail;
127 auto present_args = std::tie( present_permuted_args... );
128 auto absent_args = std::tie( DefaultConstruct< UnpermutedT >::value... );
129
130 //next we validate that the input arguments are valid for unpermuting. must have all distinct types.
131 ValidateTypes< AreValidArguments< IndexByType< const PermutedPresentT&,
132 const UnpermutedT&... >::value != sizeof...( UnpermutedT )
133 >::value...
134 >(); // this will fail if AreValidArguments are not valid ('true')
135
136 // finally, we return a tie of the unpermuted types, default constructed if necessary
137 return std::tie( GetByIndex< IndexByType< const UnpermutedT&, const PermutedPresentT&... >::value,
138 IndexByType< const UnpermutedT&, const UnpermutedT&... >::value, UnpermutedT >
139 ( present_args, absent_args )... );
140 }
