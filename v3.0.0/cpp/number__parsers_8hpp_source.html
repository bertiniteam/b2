52 using qi::_val;
53
54 root_rule_.name("real_mp");
55 root_rule_ = mpfr_rules_.number_string_
56 [ phx::bind(

60 using std::max;

62 auto asdf = max(prev_prec,LowestMultiplePrecision());

67 },
68 _val,_1
69 )
70 ];

73 root_rule_,
74 phx::bind(&ReportParseError, _1, _2, _3, _4, std::string("MpfrFloatParser"))
