80 template <typename OutT>
81 static
82 void MainData(OutT & out, ZDT const& zd)
83 {
84 NumVariables(out, zd);
85 Variables(out, zd,"\n\n");
86
87 const auto& s = zd.SolutionsInternalCoords();
88 const auto n = s.size();
89 for (decltype(s.size()) ii{0}; ii<n; ++ii)
90 {
91 // only successful endgames have a final approximation to report
92 if (zd.SolutionMetadata()[ii].endgame_success_code != SuccessCode::Success)
93 continue;
94
95 EndPointMDFull(ii, out, zd);
96 EndPoint(ii, out, zd,"\n\n");
97 }
98
99 out << "\n\n";
100 TargetSystem(out,zd,"\n\n");
101
102 StartSystem(out,zd,"\n\n");
103
104 Homotopy(out,zd,"\n\n");
105 }

109 static
110 void RawData(OutT & out, ZDT const& zd)
111 {
112 const auto n = zd.SolutionsInternalCoords().size();
113 NumVariables(out, zd,"\n\n");
114 for (decltype(zd.SolutionsInternalCoords().size()) ii{0}; ii<n; ++ii)
115 {
116 // only successful endgames have a final approximation to report
117 if (zd.SolutionMetadata()[ii].endgame_success_code != SuccessCode::Success)
118 continue;
119 EndPointMDRaw(ii,out,zd,"\n\n");
120 }
121
122 out << "\n\n";
123 TargetSystem(out,zd,"\n\n");
124 }

127 // --- Bertini 1.7 machine-readable solution files -------------------------------------
128 //
129 // These mirror the classic solution-file format so that tooling written against Bertini 1.7
130 // reads Bertini 2's output unchanged: the first line is the solution count, then each
131 // solution is a block of NumVariables coordinate lines ("re im", scientific), blocks
132 // separated by a blank line. Byte-for-byte equality with Bertini 1 is neither sought nor
133 // possible (different implementations and RNG); the contract is *machine-readable* parity.
134 // Coordinates are in user (dehomogenized) coordinates, as Bertini 1 reports them.
135
136 // Write a count-led list of solution points (each a Vec): "<count>\n\n" then, per point,
137 // the coordinate lines followed by a blank-line separator.
139 template <typename OutT, typename SolListT>
140 static void SolutionList(OutT & out, SolListT const& sols)
141 {
142 out << sols.size() << "\n\n";
143 for (auto const& v : sols)
144 {
145 generators::Classic::generate(boost::spirit::ostream_iterator(out), v);
146 out << "\n";
147 }
148 }
149
151 template <typename OutT>
152 static void FiniteSolutions(OutT & out, ZDT const& zd) { SolutionList(out, zd.FiniteSolutions()); }
153
155 template <typename OutT>
156 static void RealFiniteSolutions(OutT & out, ZDT const& zd) { SolutionList(out, zd.RealSolutions()); }

159 template <typename OutT>
160 static void NonsingularSolutions(OutT & out, ZDT const& zd) { SolutionList(out, zd.NonsingularSolutions()); }

163 template <typename OutT>
164 static void SingularSolutions(OutT & out, ZDT const& zd) { SolutionList(out, zd.SingularSolutions()); }

166 // raw_solutions: every successful endpoint, each preceded by its path number (Bertini 1 lists
167 // the raw endpoints before finite/infinite classification, tagged by path).
169 template <typename OutT>
170 static void RawSolutions(OutT & out, ZDT const& zd)
171 {
172 auto const& md = zd.SolutionMetadata();
173 auto const& sols = zd.SolutionsUserCoords();
174 const auto m = std::min(md.size(), sols.size());
175
176 std::size_t count{0};
177 for (std::size_t ii{0}; ii<m; ++ii)
178 if (md[ii].endgame_success_code == SuccessCode::Success)
179 ++count;
180
181 out << count << "\n\n";
182 for (std::size_t ii{0}; ii<m; ++ii)
183 {
184 if (md[ii].endgame_success_code != SuccessCode::Success)
185 continue;
186 out << md[ii].path_index << "\n";
187 generators::Classic::generate(boost::spirit::ostream_iterator(out), sols[ii]);
188 out << "\n";
189 }
190 }

261 template <typename IndexT, typename OutT>
262 static
263 void EndPointMDFull(IndexT const& ind, OutT & out, ZDT const& zd, std::string const& additional = "\n")
264 {
265 const auto& data = zd.SolutionMetadata()[ind];
266 out << data.path_index << '\n'
267 << data.solution_index << '\n'
268 << data.condition_number << '\n'
269 << data.function_residual << '\n'
270 << data.newton_residual << '\n';
271 generators::Classic::generate(boost::spirit::ostream_iterator(out), data.final_time_used);
272 out << '\n' << data.max_precision_used << '\n';
273
274 generators::Classic::generate(boost::spirit::ostream_iterator(out), data.time_of_first_prec_increase);
275 out << '\n' << data.accuracy_estimate << '\n'
276 << data.accuracy_estimate_user_coords << '\n'
277 << data.cycle_num << '\n'
278 << data.multiplicity << '\n'
279 << data.pre_endgame_success_code << ' ' << data.endgame_success_code << '\n';
280 out << additional;
281 }

320
322 template<typename AlgoT>
323 static
324 auto Extract(AlgoT const& alg)
325 {
326 using BCT = typename AlgoTraits<AlgoT>::BaseComplexT;
327
328 SampCont<BCT> solns;
329
330 const auto& s = alg.SolutionsUserCoords();
331 const auto& m = alg.SolutionMetadata();
332 const auto n = s.size();
333 for (decltype(s.size()) ii{0}; ii<n; ++ii)
334 {
335 const auto& d = m[ii];
336 if (d.endgame_success_code == SuccessCode::Success &&
337 d.multiplicity==1)
338 {
339 solns.push_back(s[ii]);
340 }
341 }
342
343 return solns;
344 }
