140 inline RealT StepsizeSatisfyingCriterionB(unsigned precision,
141 unsigned digits_B,
142 unsigned num_newton_iterations,
143 unsigned predictor_order = 0)
144 {
145 return pow(RealT(10), -(static_cast<int>(digits_B) - static_cast<int>(precision))
146 * static_cast<int>(num_newton_iterations) / (predictor_order+1.0));
147 }

240 template<typename RealT>
241 void MinimizeTrackingCost(unsigned & new_precision, RealT & new_stepsize,
242 unsigned min_precision, RealT const& min_stepsize,
243 unsigned max_precision, RealT const& max_stepsize,
244 unsigned digits_B,
245 unsigned num_newton_iterations,
246 unsigned predictor_order = 0)
247 {
248 double min_cost = Eigen::NumTraits<double>::highest();
249
250 unsigned minimizing_precision = 0; // initialize to an impossible value.
251
252 // a few casts so that we can work in double precision, because doing this in multiprec sucks
253 double min_stepsize_lowprec = static_cast<double>(min_stepsize);
254 double max_stepsize_lowprec = static_cast<double>(max_stepsize);
255
256 auto minimizer_routine =
257 [&min_cost, &minimizing_precision, digits_B, num_newton_iterations, predictor_order, min_stepsize_lowprec, max_stepsize_lowprec](unsigned p)
258 {
259 double criterion_b_stepsize = StepsizeSatisfyingCriterionB<double>(p, digits_B, num_newton_iterations, predictor_order);
260 if (criterion_b_stepsize < min_stepsize_lowprec)
261 return; // precision too low to satisfy min_stepsize constraint
262 double candidate_stepsize = min(criterion_b_stepsize, max_stepsize_lowprec);
263 using std::abs;
264 double current_cost = ArithmeticCost(p) / abs(candidate_stepsize);
265
266 if (current_cost < min_cost)
267 {
268 min_cost = current_cost;
269 minimizing_precision = p;
270 }
271 };
272
273 unsigned lowest_mp_precision_to_test = min_precision;
274
275 if (min_precision<=DoublePrecision())
276 minimizer_routine(DoublePrecision());
277
278
279 if (lowest_mp_precision_to_test < LowestMultiplePrecision())
280 lowest_mp_precision_to_test = LowestMultiplePrecision();
281 else
282 lowest_mp_precision_to_test = (lowest_mp_precision_to_test/PrecisionIncrement()) * PrecisionIncrement(); // use integer arithmetic to round.
283
284
285 if (max_precision < LowestMultiplePrecision())
286 max_precision = LowestMultiplePrecision();
287 else
288 max_precision = (max_precision/PrecisionIncrement()) * PrecisionIncrement(); // use integer arithmetic to round.
289
290 for (unsigned p = lowest_mp_precision_to_test; p <= max_precision; p+=PrecisionIncrement())
291 minimizer_routine(p);
292
293 if (minimizing_precision==0){
294 throw std::runtime_error("MinimizeTrackingCost failed to find a suitable stepsize and precision");
295 }
296
297 new_precision = minimizing_precision; // copy the computed value
298 // next, because the above computed the new stepsize in double precision, which may be lowprec, we compute in full precision
299 new_stepsize = max(
300 min(
301 StepsizeSatisfyingCriterionB<RealT>(new_precision, digits_B, num_newton_iterations, predictor_order),
302 max_stepsize
303 ),
304 min_stepsize
305 );
306 }

508 SuccessCode TrackerLoopInitialization(complex_mp const& start_time,
509 complex_mp const& end_time,
510 Vec<complex_mp> const& start_point) const override
511 {
512 initial_precision_ = override_start_precision_.value_or(Precision(start_point(0)));
513
514 #ifndef BERTINI_DISABLE_ASSERTS
515 assert(
516 (!preserve_precision_
517 ||
518 -log10(tracking_tolerance_) <= initial_precision_)
519 && "when tracking a path, either preservation of precision must be turned off (so precision can be higher at the end of tracking), or the initial precision must be high enough to support the resulting points to the desired tolerance"
520 );
521 #endif
522
523 NotifyObservers(Initializing<AMPTracker,complex_mp>(*this,start_time, end_time, start_point));
524 SetThreadPrecision(initial_precision_);
525 // set up the master current time and the current step size
526
527 current_time_ = start_time;
528 current_time_.precision(initial_precision_);
529
530
531 endtime_highest_precision_ = end_time;
532 endtime_highest_precision_.precision(initial_precision_);
533
534 endtime_ = end_time;
535 endtime_.precision(initial_precision_);
536
537 current_stepsize_.precision(initial_precision_);
538 if (reinitialize_stepsize_)
539 {
540 real_mp segment_length = abs(start_time-end_time)/Get<Stepping>().min_num_steps;
541 SetStepSize(min(real_mp(Get<Stepping>().initial_step_size, current_precision_),segment_length));
542 }
543
544 // populate the current space value with the start point, in appropriate precision
545 if (initial_precision_==DoublePrecision())
546 MultipleToDouble(start_point);
547 else
548 MultipleToMultiple(initial_precision_, start_point);
549
550 ChangePrecision<upsample_refine_off>(initial_precision_);
551
552 ResetCounters();
553
554 auto initial_refinement_code = InitialRefinement();
555
556 #ifndef BERTINI_DISABLE_ASSERTS
557 if (initial_precision_==DoublePrecision()){
558 PrecisionSanityCheck<complex_dbl>();
559 }
560 else{
561 PrecisionSanityCheck<complex_mp>();
562 }
563 #endif
564
565 return initial_refinement_code;

698 template <typename ComplexT>
699 SuccessCode TrackerIteration() const // not an override, because it is templated
700 {
701
702 using RealT = typename Eigen::NumTraits<ComplexT>::Real;
703
704 #ifndef BERTINI_DISABLE_ASSERTS
705 assert(PrecisionSanityCheck<ComplexT>() && "precision sanity check failed. some internal variable is not in correct precision");
706 #endif
707
708 NotifyObservers(NewStep<EmitterType>(*this));
709
710 Vec<ComplexT>& predicted_space = std::get<Vec<ComplexT> >(temporary_space_); // this will be populated in the Predict step
711 Vec<ComplexT>& current_space = std::get<Vec<ComplexT> >(current_space_); // the thing we ultimately wish to update
712 ComplexT current_time = ComplexT(current_time_);
713 ComplexT delta_t = ComplexT(delta_t_);
714
715 #ifndef BERTINI_DISABLE_ASSERTS
716 PrecisionSanityCheck<ComplexT>();
717 #endif
718
719 SuccessCode predictor_code = Predict<ComplexT, RealT>(predicted_space, current_space, current_time, delta_t);
720 if (predictor_code==SuccessCode::MatrixSolveFailureFirstPartOfPrediction)
721 {
722 NotifyObservers(FirstStepPredictorMatrixSolveFailure<EmitterType>(*this));
723 InitialMatrixSolveError();
724 return predictor_code;
725 }
726 else if (predictor_code==SuccessCode::MatrixSolveFailure)
727 {
728 NotifyObservers(PredictorMatrixSolveFailure<EmitterType>(*this));
729 NewtonConvergenceError();// decrease stepsize, and adjust precision as necessary
730 return predictor_code;
731 }
732 else if (predictor_code==SuccessCode::HigherPrecisionNecessary)
733 {
734 NotifyObservers(PredictorHigherPrecisionNecessary<EmitterType>(*this));
735 auto adj_code = AMPCriterionError<ComplexT>();
736 if (adj_code != SuccessCode::Success)
737 return adj_code;
738 return predictor_code;
739 }
740
741
742 NotifyObservers(SuccessfulPredict<AMPTracker, ComplexT>(*this, predicted_space));
743
744 Vec<ComplexT>& tentative_next_space = std::get<Vec<ComplexT> >(tentative_space_); // this will be populated in the Correct step
745
746 ComplexT tentative_next_time = current_time + delta_t;
747
748 SuccessCode corrector_code = Correct<ComplexT, RealT>(tentative_next_space,
749 predicted_space,
750 tentative_next_time);
751
752 #ifndef BERTINI_DISABLE_ASSERTS
753 assert(PrecisionSanityCheck<ComplexT>() && "precision sanity check failed. some internal variable is not in correct precision");
754 #endif
755
756 if (corrector_code==SuccessCode::MatrixSolveFailure || corrector_code==SuccessCode::FailedToConverge)
757 {
758 NotifyObservers(CorrectorMatrixSolveFailure<EmitterType>(*this));
759 NewtonConvergenceError();
760 return corrector_code;
761 }
762 else if (corrector_code == SuccessCode::HigherPrecisionNecessary)
763 {
764 NotifyObservers(CorrectorHigherPrecisionNecessary<EmitterType>(*this));
765 auto adj_code = AMPCriterionError<ComplexT>();
766 if (adj_code != SuccessCode::Success)
767 return adj_code;
768 return corrector_code;
769 }
770 else if (corrector_code == SuccessCode::GoingToInfinity)
771 {
772 // there is no corrective action possible...
773 return corrector_code;
774 }
775
776 NotifyObservers(SuccessfulCorrect<AMPTracker, ComplexT>(*this, tentative_next_space));
777
778 // copy the tentative vector into the current space vector;
779 current_space = tentative_next_space;
780 return AdjustAMPStepSuccess<ComplexT>();
781 }

849 template <typename ComplexT>
850 SuccessCode AdjustAMPStepSuccess() const
851 {
852 // This mirrors Bertini 1's AMP2_update: pick the cost-minimizing (precision, stepsize)
853 // after a successful step. Faithful-to-B1 points, two of which were previously wrong:
854 // (1) The rule-B precision requirement CREDITS THE STEP. digits_B is the right-hand side
855 // of P + (p+1)*xi/N > digits_B (xi = -log10(stepsize)); the floor is therefore
856 // digits_B - (p+1)*(-log10(min_stepsize))/N, crediting the smallest allowed step, NOT
857 // the raw digits_B. Without this credit a tiny step (large size_proportion in the
858 // roundoff regime) spuriously forced precision up. B1: digits_B2 = ceil(P0 - eta_minSS/N).
859 // (2) Precision decrease (and stepsize increase) is gated by BOTH B1's StepsForIncrease
860 // consecutive-successful-steps counter AND the digits-margin hysteresis
861 // (ExtraDigitsBeforePrecisionDecrease). The single StepsForIncrease setting
862 // (SteppingConfig::consecutive_successful_steps_before_stepsize_increase) governs both
863 // gates -- the old, duplicate AMP-config precision-decrease setting was removed.
864 real_mp min_stepsize = current_stepsize_ * real_mp(Get<Stepping>().step_size_fail_factor, current_precision_);
865 real_mp max_stepsize = min( current_stepsize_ * real_mp(Get<Stepping>().step_size_success_factor, current_precision_), real_mp(Get<Stepping>().max_step_size, current_precision_));
866
867 const unsigned steps_for_increase = Get<Stepping>().consecutive_successful_steps_before_stepsize_increase; // B1 StepsForIncrease
868
869 if (num_successful_steps_since_stepsize_increase_ < steps_for_increase)
870 max_stepsize = current_stepsize_; // not enough successes in a row: disallow stepsize increase
871
872 const unsigned N = Get<NewtonConfig>().max_num_newton_iterations;
873 const unsigned curr_digits = current_precision_;
874 const unsigned extra = ExtraDigitsBeforePrecisionDecrease(curr_digits);
875
876 // rule B, step-credited (credit the smallest allowed step = the most credit)
877 int digits_B = static_cast<int>(ceil( double(B_RHS<ComplexT>())
878 - (predictor_order_ + 1.0) * double(-log10(min_stepsize)) / N ));
879 int digits_C = static_cast<int>(DigitsC<ComplexT>());
880 int digits_stepsize = static_cast<int>(MinDigitsForStepsizeInterval(min_stepsize, max_stepsize, abs(current_time_ - endtime_)));
881
882 // each requirement may only lower precision if it clears the hysteresis margin
883 ApplyPrecisionDecreaseMargin(digits_B, curr_digits, extra);
884 ApplyPrecisionDecreaseMargin(digits_C, curr_digits, extra);
885 ApplyPrecisionDecreaseMargin(digits_stepsize, curr_digits, extra);
886
887 int min_digits = std::max({ static_cast<int>(digits_tracking_tolerance_),
888 static_cast<int>(digits_final_),
889 digits_B, digits_C, digits_stepsize,
890 static_cast<int>(DoublePrecision()) });
891 unsigned min_precision = static_cast<unsigned>(min_digits); // >= DoublePrecision(), so non-negative
892
893 // StepsForIncrease gate + safety cap: precision may only be lowered after enough successful
894 // steps in a row and while under the decrease budget; otherwise hold the current precision.
895 const bool decrease_allowed =
896 (num_successful_steps_since_precision_decrease_ >= steps_for_increase)
897 &&
898 (num_precision_decreases_ < Get<PrecConf>().max_num_precision_decreases);
899 if (!decrease_allowed)
900 min_precision = max(min_precision, current_precision_);
901
902 unsigned max_precision = Get<PrecConf>().maximum_precision;
903
904 try {
905 MinimizeTrackingCost(next_precision_, next_stepsize_,
906 min_precision, min_stepsize,
907 max_precision, max_stepsize,
908 DigitsB<ComplexT>(), // RAW digits_B (=P0) for the stepsize relation, per B1 minimize_cost
909 N,
910 predictor_order_);
911 } catch (std::runtime_error const&) {
912 return SuccessCode::FailedToSelectPrecisionAndStepsize;
913 }
914
915 if ( (next_stepsize_ > current_stepsize_) || (next_precision_ < current_precision_) )
916 num_successful_steps_since_stepsize_increase_ = 0;
917 else
918 ++num_successful_steps_since_stepsize_increase_;
919
920 if (next_precision_ < current_precision_)
921 {
922 ++num_precision_decreases_;
923 num_successful_steps_since_precision_decrease_ = 0;
924 }
925 else
926 ++num_successful_steps_since_precision_decrease_;
927
928 return UpdatePrecisionAndStepsize();
929 }

987 template<typename ComplexT>
988 SuccessCode AMPCriterionError() const
989 {
990
991 unsigned min_next_precision; // sure, i could use a trigraph here, but it'd be terrible
992 if (current_precision_==DoublePrecision())
993 min_next_precision = LowestMultiplePrecision(); // precision increases
994 else
995 min_next_precision = current_precision_ + (1+num_consecutive_failed_steps_)*PrecisionIncrement(); // precision increases
996
997
998 real_mp min_stepsize = MinStepSizeForPrecision(current_precision_, abs(current_time_ - endtime_));
999 real_mp max_stepsize = current_stepsize_ * real_mp(Get<Stepping>().step_size_fail_factor, current_precision_); // Stepsize decreases.
1000
1001 if (min_stepsize > max_stepsize)
1002 {
1003 // stepsizes are incompatible, must increase precision
1004 next_precision_ = min_next_precision;
1005 // decrease stepsize somewhat less than the fail factor
1006 next_stepsize_ = max(current_stepsize_ * (1+real_mp(Get<Stepping>().step_size_fail_factor, current_precision_))/2, min_stepsize);
1007 }
1008 else
1009 {
1010 unsigned digits_B = DigitsB<ComplexT>(); // RAW digits_B (=P0); the stepsize relation in MinimizeTrackingCost needs it raw
1011
1012 // rule-B precision floor CREDITS THE STEP (B1 AMP2_update: digits_B2 = ceil(P0 - eta_minSS/N)):
1013 // floor = digits_B - (p+1)*(-log10(min_stepsize))/N, crediting the smallest allowed step.
1014 // This keeps a small step (large size_proportion in the roundoff regime) from forcing
1015 // precision up by itself; genuine ill-conditioning still escalates via D / rules A,C.
1016 const unsigned N = Get<NewtonConfig>().max_num_newton_iterations;
1017 int digits_B_credited = static_cast<int>(ceil( double(digits_B)
1018 - (predictor_order_ + 1.0) * double(-log10(min_stepsize)) / N ));
1019 if (digits_B_credited < 0)
1020 digits_B_credited = 0;
1021
1022 unsigned min_precision = max(min_next_precision,

1403 template <UpsampleRefinementOption refine_if_necessary = upsample_refine_off>
1404 SuccessCode ChangePrecision(unsigned new_precision) const
1405 {
1406 if (new_precision==current_precision_) // no op
1407 return SuccessCode::Success;
1408
1409
1410 NotifyObservers(PrecisionChanged<EmitterType>(*this,current_precision_,new_precision));
1411
1412
1413 bool upsampling_needed = new_precision > current_precision_;
1414 // reset the counter for estimating the condition number.
1415 num_steps_since_last_condition_number_computation_ = this->Get<Stepping>().frequency_of_CN_estimation;
1416
1417 if (new_precision==DoublePrecision() && current_precision_>DoublePrecision())
1418 {
1419 // convert from multiple precision to double precision
1420 MultipleToDouble();
1421 }
1422 else if(new_precision > DoublePrecision() && current_precision_ == DoublePrecision())
1423 {
1424 // convert from double to multiple precision
1425 DoubleToMultiple(new_precision);
1426 #ifndef BERTINI_DISABLE_ASSERTS
1427 assert(PrecisionSanityCheck<complex_mp>() && "precision sanity check failed. some internal variable is not in correct precision");
1428 #endif

1435 #endif
1436 }

1439 return RefineStoredPoint();

1441 return SuccessCode::Success;
