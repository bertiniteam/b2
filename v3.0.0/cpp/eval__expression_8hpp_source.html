67 template<typename T>
68 T EvalExpression(std::shared_ptr<node::Node> const& expr,
69 std::map<std::string, T> const& variable_values)
70 {
71 auto vars = node::GatherVariables(expr); // distinct, sorted by name
72
73 // Guard against typos: every supplied name must be a variable of the expression.
74 for (auto const& named_value : variable_values)
75 {
76 bool appears = false;
77 for (auto const& v : vars)
78 if (v->name() == named_value.first) { appears = true; break; }
79 if (!appears)
80 throw std::runtime_error("value supplied for '" + named_value.first +
81 "', which is not a variable of the expression being evaluated");
82 }
83
84 // Bind values into the System's variable order (which is the name order of `vars`).
85 Vec<T> point(static_cast<Eigen::Index>(vars.size()));
86 for (size_t i = 0; i < vars.size(); ++i)
87 {
88 auto it = variable_values.find(vars[i]->name());
89 if (it == variable_values.end())
90 throw std::runtime_error("no value supplied for variable '" + vars[i]->name() +
91 "' when evaluating an expression");
92 point(static_cast<Eigen::Index>(i)) = it->second;
93 }
94
95 // Compile the expression's evaluator once and memoize it on the (immutable, hash-consed)
96 // node, so repeated evaluation of the same expression reuses the compiled program rather
97 // than rebuilding a throwaway System and recompiling an SLP every call (ADR-0027).
98 // The compiled program holds no node pointers (its constants are value recipes), so caching
99 // it on the node creates no reference cycle.
100 auto cached = std::static_pointer_cast<const StraightLineProgram>(expr->EvalProgram());
101 if (!cached)
102 {
103 System sys;
104 sys.AddFunction(expr);
105 sys.AddVariableGroup(vars);
106 cached = std::make_shared<const StraightLineProgram>(sys);
107 expr->SetEvalProgram(cached);
108 }
109
110 // Per-call working copy: it shares the immutable compiled Program but gets its own
111 // evaluation Memory, so concurrent evaluations of the same cached expression do not race.
112 StraightLineProgram slp = *cached;
113
114 if constexpr (!std::is_same<T, complex_dbl>::value)
115 if (vars.size() > 0)
116 slp.precision(Precision(point));
117
118 slp.Eval(point);
119 return slp.template GetFuncVals<T>()(0);
120 }
