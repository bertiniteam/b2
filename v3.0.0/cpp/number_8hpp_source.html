408 void print(std::ostream & target) const override;
409
410 // real-valued rationals print bare (no complex pair). the bare form is
411 // textually an expression: a leading '-' parenthesizes like a Negate, and
412 // 'p/q' contains a division, so it binds like a Mult (x/(1/3), not x/1/3).
413 // complex pairs self-delimit.
414 unsigned Precedence() const override
415 {
416 if (true_value_imag_ == 0)
417 {
418 if (true_value_real_ < 0)
419 return PrecNegate;
420 if (denominator(true_value_real_) != 1)
421 return PrecMult;
422 }
423 return PrecAtom;
424 }

450 template<typename NumT>
451 NumT Value() const
452 {
453 if constexpr (std::is_same<NumT, complex_dbl>::value)
454 return complex_dbl(double(true_value_real_), double(true_value_imag_));

456 return NumT(boost::multiprecision::mpfr_float(true_value_real_, ThreadPrecision()),
457 boost::multiprecision::mpfr_float(true_value_imag_, ThreadPrecision()));
458 }
