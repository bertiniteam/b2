392 unsigned long long num_finite_solutions = 0;

394 unsigned long long num_diverged = 0;
395 unsigned long long num_failed = 0;
396 unsigned long long num_singular = 0;
397 unsigned long long num_real = 0;
398 unsigned long long num_nonsolutions = 0;
399 std::map<SuccessCode, unsigned long long> failures_by_reason;
400 double max_condition_number = 0;
401 unsigned max_precision_used = 0;
402 MidpathCheckReport midpath;
403 bool all_paths_resolved = false;
404};

410template<typename ComplexT>
411SolveReport SummarizeSolve(std::vector<SolutionMetaData<ComplexT>> const& metadata,
412 MidpathCheckReport const& midpath)
413{
414 SolveReport r;
415 r.num_paths_tracked = metadata.size();
416 r.midpath = midpath;
417 double finite_distinct = 0;
418 for (auto const& m : metadata)
419 {
420 if (m.max_precision_used > r.max_precision_used)
421 r.max_precision_used = m.max_precision_used;
422
423 // A path the endgame's security check truncated near infinity (SecurityMaxNormReached) is
424 // diverging, not failing -- it is the security/max-norm heuristic catching a path on its way
425 // to infinity, exactly like a clean GoingToInfinity. (The endgame tests classify the two
426 // together too.) With infinite-path truncation on by default, these are expected, so they
427 // must not inflate num_failed / clear all_paths_resolved.
428 bool diverged = (m.endgame_success_code == SuccessCode::GoingToInfinity
429 || m.endgame_success_code == SuccessCode::SecurityMaxNormReached);
430 if (m.endgame_success_code == SuccessCode::Success)
431 {
432 if (m.is_nonsolution)
433 ++r.num_nonsolutions; // a nonsolution: finite, but not a solution of the target
434 else if (m.is_finite)
435 {
436 ++r.num_finite_endpoints;
437 finite_distinct += 1.0 / m.multiplicity;
438 if (m.is_singular) ++r.num_singular;
439 if (m.is_real) ++r.num_real;
440 if (static_cast<double>(m.condition_number) > r.max_condition_number)
441 r.max_condition_number = static_cast<double>(m.condition_number);
442 }
443 else
444 diverged = true;
445 }
446
447 if (diverged)
448 ++r.num_diverged;
449 else if (m.endgame_success_code != SuccessCode::Success) // neither a solution nor a clean divergence
450 {
451 ++r.num_failed;
452 ++r.failures_by_reason[m.endgame_success_code];
453 }
454 }
455 r.num_finite_solutions = static_cast<unsigned long long>(finite_distinct + 0.5);
456 r.all_paths_resolved = (r.num_failed == 0) && midpath.passed;
457 return r;
458}

663#ifdef BERTINI2_HAVE_MPI
676 void RunParallel(MPI_Comm comm)
677 {
678 using Result = parallel::FullPathResult<BaseComplexT>;
679 using Task = parallel::StartPointTask<BaseComplexT>;
680
681 ScopedSessionState const session_state_guard{}; // a solve leaves the session as it found it
682
683 mpfr_free_cache(); // reproducibility: clear this rank's mpfr constant cache (see Solve())
684
685 solutions_user_coords_fresh_ = false;
686
687 // Each rank built its homotopy independently in the constructor -- with its OWN
688 // random patch, start-system coefficients, and gamma -- so a given path index would
689 // mean a different path on each worker, and the manager would gather a scrambled,
690 // mostly-wrong solution set. Make rank 0 the single authoritative source: broadcast
691 // its actual target system, start system, and homotopy to every rank. These are all
692 // exact (rational coefficients + integer/rational patch), so serialization carries
693 // them bit-for-bit -- there is no per-rank re-derivation that could drift, and the
694 // distributed homotopy is identical to rank 0's serial one. We also broadcast rank
695 // 0's RNG seed so any randomness drawn later (e.g. in the endgame) is consistent; the
696 // per-path tracking RNG is already deterministic in the path index (ReseedThisThread).
697 {
698 unsigned long seed = parallel::IsManager() ? GetGlobalSeed() : 0ul;
699 MPI_Bcast(&seed, 1, MPI_UNSIGNED_LONG, 0, comm);
700 SetGlobalSeed(seed);
701
702 // Install rank 0's authoritative systems on every rank. The plain engine over a
703 // user-supplied homotopy does nothing here (the user owns cross-rank consistency);
704 // ZeroDimSolver, which owns its systems, overrides DistributeSystems to broadcast
705 // rank 0's target / start / homotopy and re-seat the engine's references.
706 DistributeSystems(comm);
707
708 num_start_points_ = StartSystem().NumStartPoints();
709 GetTracker().SetSystem(Homotopy());
710 }
711
712 PreSolveChecks();
713 PreSolveSetup();
714
715 const int n_threads = parallel::WorkerThreadCount();
716
717 // One round = dispatch a set of path indices, each executed as a WHOLE path
718 // (pre-endgame + endgame) by a worker. Round 0 is every path; later rounds re-run
719 // only the crossed paths with escalated settings -- the re-track is path-independent,
720 // so it rides the same demand-driven worker pool (no idle-manager bottleneck).
721 if (parallel::IsManager())
722 {
723 // Rank 0 computes every start point once, authoritatively, and ships each to the
724 // worker that tracks it. Cached so re-track rounds reuse the identical points.
725 std::vector<Vec<BaseComplexT>> start_points(num_start_points_);
726 for (decltype(num_start_points_) ii{0}; ii < num_start_points_; ++ii)
727 start_points[ii] = ComputeStartPoint(static_cast<SolnIndT>(ii));
728
729 auto run_round = [&](std::queue<Task>& q)
730 {
731 parallel::RunManagerLoop<Task, Result>(comm, q,
732 [this](Result const& r){ StoreFullPathResult(r); });
733 };
734
735 // the records seam (ADR-0046): the manager is the sole writer; recall what
736 // is already recorded and dispatch only the rest
737 MaybeAttachAmbientRecords();
738 std::vector<SolnIndT> indices_to_run;
739 for (decltype(num_start_points_) ii{0}; ii < num_start_points_; ++ii)
740 indices_to_run.push_back(static_cast<SolnIndT>(ii));
741 if (records_)
742 {
743 EnsureRunRecorded();
744 indices_to_run = RecallRecordedPaths(indices_to_run);
745 }
746
747 std::queue<Task> queue;
748 for (auto idx : indices_to_run)
749 queue.push(Task{ idx, start_points[idx] });
750 run_round(queue);
751
752 // Midpath/crossing check on the collected boundary points, then bounded parallel
753 // resolve rounds. A continue/stop flag is broadcast to the workers each round so
754 // they know whether to run again (and escalate their settings first).
755 auto passed = midpath_.Check(solutions_at_endgame_boundary_, StartSystem());
756 midpath_report_ = MidpathCheckReport{};
757 midpath_report_.passed = passed;
758 for (auto const& v : midpath_.GetCrossedPaths())
759 midpath_report_.crossed_path_indices.push_back(v.index());
760 midpath_report_.num_crossings_detected =
761 static_cast<unsigned>(midpath_report_.crossed_path_indices.size());
762
763 const auto max_attempts = this->template Get<ZeroDimConf>().max_num_crossed_path_resolve_attempts;
764 unsigned num_resolve_attempts = 0;
765 while (true)
766 {
767 int keep_going = (!passed && num_resolve_attempts < max_attempts) ? 1 : 0;
768 MPI_Bcast(&keep_going, 1, MPI_INT, 0, comm);
769 if (!keep_going)
770 break;
771
772 std::queue<Task> redo;
773 for (auto const& v : midpath_.GetCrossedPaths())
774 if (v.rerun())
775 {
776 auto idx = static_cast<SolnIndT>(v.index());
777 redo.push(Task{ idx, start_points[idx] });
778 }
779 run_round(redo);
780
781 passed = midpath_.Check(solutions_at_endgame_boundary_, StartSystem());
782 ++num_resolve_attempts;
783 }
784 midpath_report_.num_resolve_attempts = num_resolve_attempts;
785 midpath_report_.passed = passed;
786 if (!passed)
787 std::cerr << "warning: " << midpath_report_.num_crossings_detected
788 << " path crossing(s) detected at the endgame boundary remained unresolved "
789 << "after " << num_resolve_attempts << " re-track attempt(s); "
790 << "the affected solutions may be wrong. Consider a higher-order predictor "
791 << "or a tighter tracking tolerance. See EndgameBoundaryMetadata()." << std::endl;
792
793 PostEGAction();
794
795 DeclareFiniteSolutionsResult();
796 }
797 else // worker
798 {
799 auto run_worker_round = [&]()
800 {
801 if (n_threads <= 1)
802 {
803 // Single-threaded worker: execute whole paths on the rank's member
804 // tracker/endgame directly, from rank 0's authoritative start point.
805 parallel::RunWorkerLoop<Task, Result>(comm,
806 [this](Task const& task){ ExecuteOnePath(MemberDuringEGContext(), static_cast<SolnIndT>(task.path_index), task.start_point); },
807 [this](Task const& task) -> Result { return PackFullPathResult(static_cast<SolnIndT>(task.path_index)); });
808 }
809 else
810 {
811 // Threaded worker: each thread owns a self-contained PathThreadState clone
812 // so concurrent whole-path execution shares no mutable state. Cloned per
813 // round so it inherits any escalation (predictor bump) applied to the rank's
814 // member tracker between rounds.
815 auto state_factory = [this]() -> std::unique_ptr<PathThreadState>
816 {
817 // Clone(), not copy: System copies are SHALLOW (shared expression-tree
818 // nodes whose value caches mutate on Eval); Clone() deep-copies the tree.
819 // Trackers/Endgames have no default ctor, so aggregate-init via new.
820 std::unique_ptr<PathThreadState> s(
821 new PathThreadState{ Clone(GetTracker().GetSystem()), Clone(TargetSystem()),
822 GetTracker(), GetEndgame(), {}, {} });
823 s->tracker.SetSystem(s->sys);
824 s->endgame.SetTracker(s->tracker);
825 SetThreadPrecision(this->template Get<ZeroDimConf>().initial_ambient_precision);
826 return s;
827 };
828
829 auto track_fn = [this](std::unique_ptr<PathThreadState>& state, Task const& task) -> Result
830 {
831 auto idx = static_cast<SolnIndT>(task.path_index);
832 ExecuteOnePath(state->Context(), idx, task.start_point);
833 return PackFullPathResult(idx);
834 };
835
836 parallel::RunWorkerLoopThreaded<Task, Result>(
837 comm, state_factory, track_fn, n_threads);
838 }
839 }; // run_worker_round
840
841 while (true)
842 {
843 run_worker_round();
844
845 int keep_going = 0;
846 MPI_Bcast(&keep_going, 1, MPI_INT, 0, comm);
847 if (!keep_going)
848 break;
849
850 // Match the escalation the serial RunMidpathResolution applies between
851 // re-track passes: tighten midpath_retrack_tolerance_ (read by ExecuteOnePath)
852 // and bump a low-order predictor on this rank's member tracker, so the next
853 // round's thread clones inherit it.
854 EscalateRetrackSettings();
855 }
856 }
857 }

1084 void Solve()
1085 {
1086 // Reproducibility: clear this thread's mpfr constant cache (pi, etc.) at the solve
1087 // boundary. mpfr caches transcendental constants at the precision last requested; the
1088 // Cauchy endgame needs pi (roots of unity), so a high-precision solve leaves pi cached
1089 // at high precision. A subsequent lower-precision solve would otherwise reuse that
1090 // cached value rounded down -- differing by ~1 ULP from a freshly-computed
1091 // low-precision pi -- making a solve's behaviour depend on what ran before it in the
1092 // process (non-reproducible). Clearing here makes each solve independent of history.
1093 // Cheap (one constant recompute per solve); thread-local (each worker thread clears its
1094 // own when it starts a solve).
1095 mpfr_free_cache();
1096
1097 ScopedSessionState const session_state_guard{}; // a solve leaves the session as it found it
1098
1099 solutions_user_coords_fresh_ = false;
1100
1101 PreSolveChecks();
1102
1103 PreSolveSetup();
1104
1105 this->NotifyObservers(AlgorithmStarted<AnyZeroDim>(*this));
1106
1107 // Speculative-full-path model: carry every path all the way through (pre-endgame +
1108 // endgame) as one unit, then detect crossings from the collected boundary points and
1109 // re-run any crossed path in full. This is the same shape the distributed solve uses
1110 // (one worker per whole path), so serial and distributed share the per-path primitive
1111 // -- including computing the start point via the same ComputeStartPoint.
1112
1113 // Compute every start point once, up front (deterministic per index, as the MPI
1114 // manager does). The threaded dispatch ships each worker its start point, and the
1115 // serial path / crossed-path re-tracks reuse the identical points from here.
1116 std::vector<Vec<BaseComplexT>> start_points(num_start_points_);
1117 std::vector<SolnIndT> all_indices(num_start_points_);
1118 for (decltype(num_start_points_) ii{0}; ii < num_start_points_; ++ii)
1119 {
1120 auto idx = static_cast<SolnIndT>(ii);
1121 start_points[ii] = ComputeStartPoint(idx);
1122 all_indices[ii] = idx;
1123 }
1124
1125 // the records seam (ADR-0046): ensure the run is on record, recall any paths
1126 // already recorded (solve() is ensure-answered), and compute only the rest
1127 MaybeAttachAmbientRecords();
1128 std::vector<SolnIndT> indices_to_run = all_indices;
1129 if (records_)
1130 {
1131 EnsureRunRecorded();
1132 indices_to_run = RecallRecordedPaths(all_indices);
1133 }
1134
1135 // num_threads: 0 = auto (hardware_concurrency), 1 = serial, N = N threads;
1136 // OMP_NUM_THREADS overrides. n_threads <= 1 takes the pool-free serial path.
1137 const unsigned n_threads =
1138 parallel::EffectiveThreadCount(this->template Get<ZeroDimConf>().num_threads);
1139
1140 if (n_threads <= 1)
1141 {
1142 for (auto idx : indices_to_run)
1143 {
1144 ExecuteOnePath(MemberDuringEGContext(), idx, start_points[idx]);
1145 // the serial path installs results directly (no StoreFullPathResult),
1146 // so emit its record here
1147 RecordCompletedPath(PackFullPathResult(idx));
1148 }
1149 }
1150 else
1151 {
1152 RunPathsThreaded(indices_to_run, start_points, n_threads);
1153 }
1154
1155 // Crossed-path re-tracks run on the main thread against the (escalated) member
1156 // tracker. They are typically rare (often none); running them in parallel too is a
1157 // deferred optimization.
1158 RunMidpathResolution([this, &start_points](SolnIndT idx){
1159 ExecuteOnePath(MemberDuringEGContext(), idx, start_points[idx]);
1160 RecordCompletedPath(PackFullPathResult(idx)); // a re-track appends a fresh record (last wins)
1161 });
1162
1163 PostEGAction();
1164
1165 DeclareFiniteSolutionsResult();
1166
1167 this->NotifyObservers(AlgorithmComplete<AnyZeroDim>(*this));
1168 }

1353 void PreSolveSetup()
1354 {
1355 // Fixed-multiple precision: FixedPrecisionConfig.precision (on the tracker) is the
1356 // authoritative precision for the whole solve. Lift the tracker, the ambient/thread
1357 // precision, the start-point precision (via initial_ambient_precision), and the systems
1358 // all to that one value -- TrackerLoopInitialization requires them to agree. (Double is
1359 // fixed at 16 and adaptive manages its own precision, so neither enters here.)
1360 if constexpr (!tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1361 {
1362 auto const& fp = GetTracker().template Get<PrecisionConfig>();
1363 // Double validates (its PrecisionSetup throws on any precision other than 16);
1364 // fixed-multiple adopts the value into precision_.
1365 GetTracker().PrecisionSetup(fp);
1366 if constexpr (!std::is_same<BaseComplexT, complex_dbl>::value)
1367 if (fp.precision != 0)
1368 {
1369 ZeroDimConf zdc = this->template Get<ZeroDimConf>();
1370 zdc.initial_ambient_precision = fp.precision; // -> thread + start-point precision
1371 this->template Set<ZeroDimConf>(zdc);
1372 TargetSystem().precision(fp.precision);
1373 Homotopy().precision(fp.precision);
1374 }
1375 }
1376
1377 auto num_as_size_t = static_cast<SolnIndT>(num_start_points_);
1378
1379 solution_final_metadata_.resize(num_as_size_t);
1380 solutions_at_endgame_boundary_.resize(num_as_size_t);
1381 solutions_post_endgame_.resize(num_as_size_t);
1382
1383 SetMidpathRetrackTol(this->template Get<Tolerances>().newton_before_endgame);
1384
1385 // The solver's Tolerances.final_tolerance is the single source of truth for how tightly
1386 // a solve converges its endpoints. Flow it into the endgame here (the endgame keeps its
1387 // own EndgameConfig.final_tolerance for standalone, solver-independent use, but under a
1388 // solve the solver wins). Without this the endgame would silently ignore the solver's
1389 // final_tolerance -- e.g. a classic-input FinalTol -- and use its own default instead.
1390 endgame_.SetFinalTolerance(this->template Get<Tolerances>().final_tolerance);
1391
1392 // Default the start-point precision to the initial ambient precision. A caller can
1393 // override it via SetStartPointPrecision (e.g. to carry a higher precision forward
1394 // from a previous solve whose output feeds this one).
1395 if (!start_point_precision_set_by_user_)
1396 start_point_precision_ = this->template Get<ZeroDimConf>().initial_ambient_precision;
1397 }

1458 void ExecuteBeforeEG(BeforeEGContext ctx, SolnIndT soln_ind, Vec<BaseComplexT> const& start_point)
1459 {
1460 ReseedThisThread(static_cast<uint64_t>(soln_ind));
1461
1462 if (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1463 {
1464 ctx.tracker.AddObserver(ctx.first_prec_rec);
1465 ctx.tracker.AddObserver(ctx.min_max_prec);
1466 }
1467
1468 auto& smd = solution_final_metadata_[soln_ind];
1469 smd.path_index = soln_ind;
1470 smd.solution_index = soln_ind;
1471
1472 auto initial_prec = this->template Get<ZeroDimConf>().initial_ambient_precision;
1473 // SetThreadPrecision: writes thread-local only, safe from concurrent threads.
1474 SetThreadPrecision(initial_prec);
1475
1476 // Draw the condition-number probe direction ONCE here, for the whole track of this
1477 // point (pre-endgame AND endgame). We have just reseeded this thread's RNG to a value
1478 // determined solely by the path index, so the probe is deterministic per path and
1479 // identical whether the path is tracked in a serial loop or on a distributed worker --
1480 // regardless of how the paths were split across ranks. It is NOT refreshed for the
1481 // endgame (which would churn the RNG across its sample-circle sub-tracks), so the same
1482 // direction persists start to finish.
1483 ctx.tracker.RefreshConditionDirection();
1484
1485 // The times are stored precision-free (mpq_rational); materialize them as the tracking
1486 // complex type at the current working precision.
1487 BaseComplexT t_start ( BaseRealT(this->template Get<ZeroDimConf>().start_time) );
1488 BaseComplexT t_endgame_boundary( BaseRealT(this->template Get<ZeroDimConf>().endgame_boundary) );
1489
1490 // The start point is supplied by the caller (computed once, authoritatively, via
1491 // ComputeStartPoint) rather than regenerated here. In a distributed solve rank 0
1492 // computes it and sends it with the task, so a worker never derives its own -- that,
1493 // plus authoritative pi (issue #156), is what makes the start point identical across
1494 // serial and distributed runs.
1495
1496 // Reset to the configured initial step size for this fresh path. In the
1497 // speculative-full-path model a path's endgame runs immediately before the next
1498 // path's pre-endgame tracking on the same tracker, and the endgame leaves
1499 // ReinitializeInitialStepSize(false) with a tiny step; without restoring it here the
1500 // next path would crawl from the start time with that tiny step (effectively a hang).
1501 ctx.tracker.ReinitializeInitialStepSize(true);
1502
1503 // Begin every AMP path in hardware double precision; the AMP step criteria escalate to
1504 // multiprecision only where a path actually needs it (and the endgame, which clears
1505 // this override below, manages its own precision). Total-degree start points are born
1506 // at LowestMultiplePrecision (the generator's arithmetic widens past the requested
1507 // digits, see issue #308), and initial_ambient_precision is itself a multiprecision
1508 // value, so without forcing double here the tracker starts -- and, in practice, stays
1509 // -- in multiprecision on every path, even trivially well-conditioned ones.
1510 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1511 ctx.tracker.SetStartPrecision(DoublePrecision());
1512
1513 Vec<BaseComplexT> result;
1514 auto tracking_success = ctx.tracker.TrackPath(result, t_start, t_endgame_boundary, start_point);
1515
1516 solutions_at_endgame_boundary_[soln_ind] =
1517 EGBoundaryMetaDataT({ result, tracking_success, ctx.tracker.CurrentStepsize(), ctx.tracker.CurrentPrecision() });
1518
1519 // Clear the start-precision override so it does not leak into the endgame, which
1520 // shares this tracker instance for its sample circles.
1521 if constexpr (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1522 ctx.tracker.SetStartPrecision(std::nullopt);
1523
1524 smd.pre_endgame_success_code = tracking_success;
1525
1526 if (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1527 {
1528 if (ctx.first_prec_rec.DidPrecisionIncrease())
1529 {
1530 smd.precision_changed = true;
1531 smd.time_of_first_prec_increase = ctx.first_prec_rec.TimeOfIncrease();
1532 }
1533 else
1534 ctx.tracker.RemoveObserver(ctx.first_prec_rec);
1535 ctx.tracker.RemoveObserver(ctx.min_max_prec);

1537 smd.max_precision_used =

1681 void ExecuteOnePath(DuringEGContext ctx, SolnIndT soln_ind, Vec<BaseComplexT> const& start_point)
1682 {
1683 // Carry the executing tracker on the path events: &ctx.tracker is the member tracker
1684 // in a serial solve and the thread-local clone in a threaded one, so a meta-observer
1685 // attaches its per-path sub-observer to the tracker that actually runs this path.
1686 Observable const* exec_tracker = &ctx.tracker;
1687
1688 // wall-clock the whole path (pre-endgame + endgame). steady_clock: monotonic, safe on
1689 // the worker thread; written to this path's own metadata slot, so no cross-path race.
1690 auto const path_start_clock = std::chrono::steady_clock::now();
1691 auto stamp_path_time = [&]{
1692 solution_final_metadata_[soln_ind].path_time_seconds =
1693 std::chrono::duration<double>(std::chrono::steady_clock::now() - path_start_clock).count();
1694 };
1695
1696 this->NotifyObservers(PathStarted<AnyZeroDim>(*this, static_cast<std::size_t>(soln_ind), exec_tracker));
1697
1698 ctx.tracker.SetTrackingTolerance(midpath_retrack_tolerance_);
1699 ExecuteBeforeEG(BeforeEGContext{ ctx.tracker, ctx.first_prec_rec, ctx.min_max_prec }, soln_ind, start_point);
1700
1701 if (solution_final_metadata_[soln_ind].pre_endgame_success_code != SuccessCode::Success)
1702 {
1703 stamp_path_time();
1704 this->NotifyObservers(PathComplete<AnyZeroDim>(*this, static_cast<std::size_t>(soln_ind), exec_tracker));
1705 return;
1706 }
1707
1708 ctx.tracker.SetTrackingTolerance(this->template Get<Tolerances>().newton_during_endgame);
1709 ExecuteDuringEG(ctx, soln_ind);
1710
1711 stamp_path_time();
1712 this->NotifyObservers(PathComplete<AnyZeroDim>(*this, static_cast<std::size_t>(soln_ind), exec_tracker));
1713 }

1769 void ExecuteDuringEG(DuringEGContext ctx, SolnIndT soln_ind)
1770 {
1771 ReseedThisThread(static_cast<uint64_t>(soln_ind) + static_cast<uint64_t>(num_start_points_));
1772
1773 auto& smd = solution_final_metadata_[soln_ind];
1774 if (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1775 {
1776 if (!smd.precision_changed)
1777 ctx.tracker.AddObserver(ctx.first_prec_rec);
1778 ctx.tracker.AddObserver(ctx.min_max_prec);
1779 }
1780
1781 const auto& bdry_point = solutions_at_endgame_boundary_[soln_ind].path_point;
1782
1783 ctx.tracker.SetStepSize(solutions_at_endgame_boundary_[soln_ind].last_used_stepsize);
1784 ctx.tracker.ReinitializeInitialStepSize(false);
1785
1786 // Resume the endgame at the precision the path was actually using at the boundary,
1787 // rather than inferring it from the (always-multiprecision) point's mantissa, which
1788 // is unreliable for paths that tracked in double.
1789 auto start_prec = solutions_at_endgame_boundary_[soln_ind].precision;
1790 SetThreadPrecision(start_prec);
1791
1792 ctx.endgame.SetBoundaryTime(BaseComplexT(BaseRealT(this->template Get<ZeroDimConf>().endgame_boundary)));
1793 ctx.endgame.SetTargetTime (BaseComplexT(BaseRealT(this->template Get<ZeroDimConf>().target_time)));
1794
1795 auto eg_success = ctx.endgame.Run(bdry_point);
1796
1797 solutions_post_endgame_[soln_ind] = ctx.endgame.template FinalApproximation<BaseComplexT>();
1798
1799 smd.endgame_success_code = eg_success;
1800
1801 // an unsuccessful endgame has no final approximation, so the final-point-dependent
1802 // metadata cannot be computed.
1803 if (eg_success != SuccessCode::Success)
1804 {
1805 if (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1806 {
1807 ctx.tracker.RemoveObserver(ctx.first_prec_rec);
1808 ctx.tracker.RemoveObserver(ctx.min_max_prec);
1809 }
1810 return;
1811 }
1812 if (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1813 {
1814 if (!smd.precision_changed)
1815 {
1816 if (ctx.first_prec_rec.DidPrecisionIncrease())
1817 {
1818 smd.precision_changed = true;
1819 smd.time_of_first_prec_increase = ctx.first_prec_rec.TimeOfIncrease();
1820 }
1821 ctx.tracker.RemoveObserver(ctx.first_prec_rec);
1822 }
1823 ctx.tracker.RemoveObserver(ctx.min_max_prec);
1824 using std::max;
1825 smd.max_precision_used =
1826 max(smd.max_precision_used, ctx.min_max_prec.MaxPrecision());
1827 }
1828 if (tracking::TrackerTraits<TrackerType>::IsAdaptivePrec)
1829 {
1830 assert(Precision(solutions_post_endgame_[soln_ind])==Precision(ctx.endgame.template FinalApproximation<BaseComplexT>()));
1831 SetThreadPrecision(Precision(solutions_post_endgame_[soln_ind]));
1832 ctx.target_sys.precision(Precision(solutions_post_endgame_[soln_ind]));
1833 }
1834 smd.function_residual = static_cast<NumErrorT>(ctx.target_sys.Eval(solutions_post_endgame_[soln_ind]).template lpNorm<Eigen::Infinity>());
1835 smd.final_time_used = ctx.endgame.LatestTime();
1836 smd.condition_number = ctx.tracker.LatestConditionNumber();
1837 smd.newton_residual = ctx.tracker.LatestNormOfStep();
1838
1839 smd.accuracy_estimate = ctx.endgame.ApproximateError();
1840 smd.accuracy_estimate_user_coords =
1841 static_cast<NumErrorT>( (ctx.target_sys.DehomogenizePoint(solutions_post_endgame_[soln_ind]) -
1842 ctx.target_sys.DehomogenizePoint(ctx.endgame.template PreviousApproximation<BaseComplexT>())).template lpNorm<Eigen::Infinity>() );
1843 smd.cycle_num = ctx.endgame.CycleNumber();
1844
1845 // Honest precision/accuracy of this solution, as digit counts. precision_digits is the
1846 // precision the endgame actually finished in -- which is exactly the precision of the
1847 // stored solution point (DoublePrecision() for a fast-lane path, the mpfr precision for an
1848 // escalated one). accuracy_digits is how many of those digits the convergence agreement
1849 // supports: floor(-log10(accuracy_estimate)), clamped to [0, precision_digits].
1850 smd.precision_digits = static_cast<unsigned>(Precision(solutions_post_endgame_[soln_ind]));
1851 {
1852 using std::log10; using std::floor; using std::min; using std::max;
1853 double err = static_cast<double>(smd.accuracy_estimate);
1854 unsigned acc = (err > 0.0)
1855 ? static_cast<unsigned>(max(0.0, floor(-log10(err))))
1856 : smd.precision_digits;
1857 smd.accuracy_digits = min(acc, smd.precision_digits);
1858 }
1859 }

1899 void ClassifyFiniteAndReal()
1900 {
1901 using std::abs; using std::imag;
1902 const auto& post = this->template Get<PostProcessing>();
1903
1904 for (decltype(num_start_points_) ii{0}; ii < num_start_points_; ++ii)
1905 {
1906 auto& smd = solution_final_metadata_[ii];
1907
1908 if (smd.endgame_success_code==SuccessCode::GoingToInfinity ||
1909 smd.endgame_success_code==SuccessCode::SecurityMaxNormReached)
1910 {
1911 smd.is_finite = false; // the endgame already decided this path diverges
1912 continue;
1913 }
1914 if (smd.endgame_success_code!=SuccessCode::Success)
1915 continue; // failed otherwise: leave defaults (not finite/real/singular)
1916
1917 auto user_pt = this->TargetSystem().DehomogenizePoint(solutions_post_endgame_[ii]);
1918
1919 smd.is_finite =
1920 static_cast<NumErrorT>(user_pt.template lpNorm<Eigen::Infinity>()) <= post.endpoint_finite_threshold;
1921
1922 if (smd.is_finite)
1923 {
1924 NumErrorT max_imag{0};
1925 for (Eigen::Index k{0}; k < user_pt.size(); ++k)
1926 {
1927 NumErrorT a = static_cast<NumErrorT>(abs(imag(user_pt(k))));
1928 if (a > max_imag) max_imag = a;
1929 }
1930 smd.is_real = max_imag < post.real_threshold;
1931 }

1946 void ComputeMultiplicities()
1947 {
1948 const NumErrorT same_tol =
1949 this->template Get<Tolerances>().final_tolerance *
1950 this->template Get<PostProcessing>().same_point_tolerance_multiplier;
1951

1975 // representative; the lowest-index member of a cluster is never marked
1976 // here, so it remains the single representative.

1978 }

2537 template<typename SystemType>
2538 struct OwnedHomotopy
2539 {
2540 using StartSystemBaseT = bertini::start_system::StartSystem;
2541 using FactoryT = bertini::start_system::StartSystemFactory<SystemType>;
2542
2544 OwnedHomotopy(SystemType const& target, FactoryT factory, std::string const& path_variable_name)
2545 : user_target_(Clone(target)), owned_target_(Clone(target)), owned_factory_(std::move(factory))
2546 {
2547 ConsistencyCheck(); // feasibility (no path var; not under-constrained; polynomial)
2548 SquareUp(); // randomize an over-determined system down to square
2549 RankCheck(); // square, but can isolated solutions even exist?
2550 PrepareTarget(owned_target_); // homogenize + auto-patch the (now square) target
2551 owned_start_ = owned_factory_(owned_target_); // start system over the prepared target
2552 // Treat the configured name as a BASE: mangle it only if it would collide with a
2553 // user variable, so the injected path variable can never overlap a user symbol.
2554 owned_homotopy_ = MakeHomotopy(owned_target_, *owned_start_,
2555 UniquePathVariableName(owned_target_, path_variable_name));
2556 }

2559 SystemType const& BuiltTarget() const { return owned_target_; }
2564 SystemType const& UserTarget() const { return user_target_; }
2566 StartSystemBaseT const& BuiltStart() const { return *owned_start_; }

2569
2570 protected:
2571 SystemType user_target_;
2572 SystemType owned_target_;
2573 std::shared_ptr<StartSystemBaseT> owned_start_;
2574 SystemType owned_homotopy_;
2575 FactoryT owned_factory_;
2576
2577 // When the user's system was over-determined, SquareUp randomizes it down to square for
2578 // tracking and keeps the ORIGINAL (natural, un-homogenized) system here so ZeroDimSolver
2579 // can discard the extraneous solutions the squaring introduces. was_randomized_ gates
2580 // that filter; randomization_matrix_ records the (exact) coefficient matrix used.
2581 bool was_randomized_ = false;
2582 SystemType original_natural_target_;
2583 Mat<complex_mp> randomization_matrix_;
2584
2588 void ConsistencyCheck() const
2589 {
2590 if (owned_target_.HavePathVariable())
2591 throw std::runtime_error("unable to perform zero dim solve on target system -- has path variable, use a HomotopySolver instead.");
2592
2593 // A square zero-dim system needs one equation per dimension. Each projective
2594 // (homogeneous) variable group of size k spans P^{k-1}: its k coordinates carry only
2595 // k-1 dimensions because scale is free, so it needs one fewer equation than it has
2596 // variables. Subtract that free scale per projective group before comparing. An
2597 // UNDER-determined system has a positive-dimensional solution set, so no isolated
2598 // solutions to compute -- raise a helpful error rather than tracking garbage.
2599 if (owned_target_.NumVariables() - owned_target_.NumHomVariableGroups() > owned_target_.NumTotalFunctions())
2600 throw std::runtime_error("unable to perform zero dim solve on target system -- it is under-determined (fewer equations than variables), so its solution set is positive-dimensional, not zero-dimensional. ZeroDimSolver computes isolated solutions only; add equations, or use a positive-dimensional method.");
2601
2602 if (!owned_target_.IsPolynomial())
2603 throw std::runtime_error("unable to perform zero dim solve on target system -- system is non-polynomial, use a HomotopySolver instead.");
2604 }
2605
2612 void SquareUp()
2613 {
2614 auto const dimension = owned_target_.NumVariables() - owned_target_.NumHomVariableGroups();
2615 if (owned_target_.NumTotalFunctions() <= dimension)
2616 return; // already square (or under-determined, which ConsistencyCheck already rejected)
2617
2618 original_natural_target_ = Clone(owned_target_); // keep the original N-function system
2619 owned_target_ = owned_target_.Randomize();
2620 randomization_matrix_ = owned_target_.RandomizationMatrix();
2621 was_randomized_ = true;
2622 }
2623
2633 void RankCheck() const
2634 {
2635 if (owned_target_.HavePathVariable())
2636 return;
2637 auto const n = static_cast<Eigen::Index>(owned_target_.NumVariables());
2638 if (static_cast<Eigen::Index>(owned_target_.NumTotalFunctions()) != n || n == 0)
2639 return; // not a square map: ConsistencyCheck governs feasibility here
2640
2641 // a generic complex sample point: a zero-dimensional variety misses it almost surely,
2642 // where the Jacobian attains its generic (full) rank.
2643 Vec<complex_dbl> pt = Vec<complex_dbl>::Random(n);
2644 Mat<complex_dbl> J = owned_target_.template Jacobian<complex_dbl>(pt);
2645 Eigen::FullPivLU<Mat<complex_dbl>> lu(J);
2646 lu.setThreshold(1e-10);
2647 if (lu.rank() < n)
2648 throw std::runtime_error("unable to perform zero dim solve on target system -- the Jacobian is rank-deficient at a generic point, so the solution set is positive-dimensional, not zero-dimensional. ZeroDimSolver computes isolated solutions only.");
2649 }

2655 target.Homogenize(); // work over projective coordinates
2656 target.AutoPatch(); // then patch if needed
2657 }

2688 FactoryT factory = bertini::start_system::MakeStartFactory<bertini::start_system::TotalDegreeBinomial, SystemType>())

2700 Mat<complex_mp> const& RandomizationMatrix() const { return this->randomization_matrix_; }
2701

2715 void PostEGAction() override
2716 {
2717 EngineT::PostEGAction(); // the engine's finite/real/multiplicity/singular classification
2718
2719 if (!this->was_randomized_)
2720 return;
2721
2722 // genuine roots satisfy the original system to ~solve accuracy; spurious ones miss it
2723 // by O(1). A generous multiple of the endgame tolerance separates the two cleanly.
2724 const double threshold =
2725 1e3 * static_cast<double>(this->template Get<TolerancesConfig>().newton_during_endgame);
2726
2727 for (decltype(this->num_start_points_) ii{0}; ii < this->num_start_points_; ++ii)
2728 {
2729 auto& smd = this->solution_final_metadata_[ii];
2730 if (smd.endgame_success_code != SuccessCode::Success || !smd.is_finite)
2731 continue;
2732
2733 auto user_pt = this->TargetSystem().DehomogenizePoint(this->solutions_post_endgame_[ii]);
2734 // Evaluate the original system in DOUBLE precision: the filter only needs to tell a
2735 // genuine root (tiny residual) from an extraneous one (O(1)), and a double residual
2736 // does that robustly without a precision mismatch between the (possibly AMP/high
2737 // precision) solution point and the original system's working precision.
2738 Vec<complex_dbl> pt_d(user_pt.size());
2739 for (Eigen::Index k = 0; k < user_pt.size(); ++k)
2740 pt_d(k) = complex_dbl(user_pt(k));
2741 auto residual = static_cast<NumErrorT>(
2742 this->original_natural_target_.template Eval<complex_dbl>(pt_d).template lpNorm<Eigen::Infinity>());
2743 smd.function_residual = residual; // report residual against the ORIGINAL system
2744 if (static_cast<double>(residual) > threshold)
2745 smd.is_nonsolution = true; // extraneous: finite, but not a solution of the original system
2746 }
2747 }
